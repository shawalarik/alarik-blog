这一模块用原生 JavaScript 拆解响应式系统最小闭环，理解依赖追踪、触发更新与调度策略。

## 手动依赖收集

在深入 API 之前，先手动模拟一次完整的"数据变化→视图更新"过程：

```js
// ===== 原始需求：count 变化时自动更新 UI =====
let count = 0;

const render = () => {
  document.getElementById('app').textContent = `Count: ${count}`;
};

document.getElementById('btn').addEventListener('click', () => {
  count++;
  render();  // 每次修改后手动调用
});

render();  // 初始渲染
```

问题：**每次修改 count 都要记得调用 render**，一旦遗漏，UI 和状态就不一致了。

### 引入依赖追踪

核心思路：让系统自动知道"谁用了哪些数据"，数据变化时自动通知对应的使用者。

```js
// ===== 最小响应式实现 v1 =====
let activeEffect = null;  // 当前正在执行的副作用函数

function effect(fn) {
  activeEffect = fn;
  fn();
  activeEffect = null;
}

const targetMap = new WeakMap();

function track(target, key) {
  if (!activeEffect) return;

  let depsMap = targetMap.get(target);
  if (!depsMap) {
    targetMap.set(target, (depsMap = new Map()));
  }

  let dep = depsMap.get(key);
  if (!dep) {
    depsMap.set(key, (dep = new Set()));
  }

  dep.add(activeEffect);  // 记录：target.key 被 activeEffect 使用
}

function trigger(target, key) {
  const depsMap = targetMap.get(target);
  if (!depsMap) return;

  const dep = depsMap.get(key);
  if (dep) {
    for (const effect of dep) {
      effect();  // 重新执行所有依赖该属性的副作用
    }
  }
}
```

```js
// ===== 使用 =====
const data = { count: 0 };

effect(() => {                    // 注册副作用
  track(data, 'count');           // 声明：我依赖 data.count
  console.log(`Count: ${data.count}`);  // 执行副作用（渲染）
});                               // 输出: Count: 0

data.count = 1;
trigger(data, 'count');           // 输出: Count: 1

data.count = 5;
trigger(data, 'count');           // 输出: Count: 5
```

## Object.defineProperty

Vue 2 的响应式基于 `Object.defineProperty`，它能在属性被读写时拦截操作。

### 基础用法

```js
const data = { _count: 0 };

Object.defineProperty(data, 'count', {
  get() {
    console.log('读取 count');
    return this._count;
  },
  set(newValue) {
    console.log('设置 count 为:', newValue);
    this._count = newValue;
  },
});

data.count;          // "读取 count"
data.count = 3;      // "设置 count 为: 3"
```

### 结合依赖追踪与 defineProperty

```js
function reactive_defineProperty(obj) {
  const handler = {};

  for (const key in obj) {
    let value = obj[key];

    Object.defineProperty(handler, key, {
      get() {
        track(obj, key);     // 读取时收集依赖
        return value;
      },
      set(newValue) {
        if (value === newValue) return;  // 相同值不触发更新
        value = newValue;
        trigger(obj, key);   // 写入时触发通知
      },
    });
  }

  return handler;
}

// ===== 完整示例 =====
const raw = { count: 0, name: 'hello' };
const state = reactive_defineProperty(raw);

effect(() => {
  track(state, 'count');
  console.log(`count = ${state.count}, name = ${state.name}`);
});
// 输出: count = 0, name = hello

state.count = 1;   // 自动输出: count = 1, name = hello
state.name = 'world';  // 自动输出: count = 1, name = world
```

### defineProperty 的局限

```js
// 局限 1：无法检测新增/删除属性
state.newProp = 123;         // 不触发！
delete state.count;          // 不触发！

// 局限 2：无法监听数组下标变化
const arrData = reactive_defineProperty({ items: [1, 2, 3] });
arrData.items[0] = 99;       // 不触发！数组内部变化无法感知

// 局限 3：需要递归遍历所有嵌套属性（性能开销大）
const deep = reactive_defineProperty({
  nested: { a: { b: 1 } }   // 需要深层递归定义
});
```

## Proxy

ES6 的 `Proxy` 是更强大的元编程工具，可以解决 `defineProperty` 的诸多限制。

### 基础用法

```js
const target = { count: 0, name: 'hello' };

const proxy = new Proxy(target, {
  get(target, key, receiver) {
    console.log(`get ${key}`);
    return Reflect.get(target, key, receiver);
  },

  set(target, key, value, receiver) {
    console.log(`set ${key} = ${value}`);
    return Reflect.set(target, key, value, receiver);
  },

  has(target, key) {
    console.log(`has ${key}`);
    return Reflect.has(target, key);
  },

  deleteProperty(target, key) {
    console.log(`delete ${key}`);
    return Reflect.deleteProperty(target, key);
  },

  ownKeys(target) {
    console.log('ownKeys');
    return Reflect.ownKeys(target);
  },
});

proxy.count;          // get count
proxy.count = 5;      // set count = 5
proxy.newKey = 'hi';  // set newKey = hi ✅ 新增属性可拦截！
'count' in proxy;     // has count
delete proxy.count;   // delete count ✅ 删除属性可拦截！
```

### 为什么使用 Reflect

`Reflect` 与 `Proxy` 一一对应，确保代理行为与原始对象行为一致：

```js
// ❌ 错误：直接用 target[key] 会绕过 receiver
const proxy = new Proxy({}, {
  get(target, key, receiver) {
    return target[key];  // this 指向丢失
  },
});

// ✅ 正确：使用 Reflect 保持 receiver 链
const proxy = new Proxy({}, {
  get(target, key, receiver) {
    return Reflect.get(target, key, receiver);
  },
});
```

### 基于 Proxy 的响应式

```js
function reactive_Proxy(target) {
  return new Proxy(target, {
    get(target, key, receiver) {
      // 依赖追踪
      track(target, key);

      const result = Reflect.get(target, key, receiver);

      // 自动深层代理对象
      if (typeof result === 'object' && result !== null) {
        return reactive_Proxy(result);
      }

      return result;
    },

    set(target, key, value, receiver) {
      const oldValue = target[key];
      const result = Reflect.set(target, key, value, receiver);

      // 只有值真正变化才触发
      if (oldValue !== value) {
        trigger(target, key);
      }

      return result;
    },

    has(target, key) {
      track(target, key);  // in 操作符也触发依赖收集
      return Reflect.has(target, key);
    },

    ownKeys(target) {
      // 迭代时追踪（用于 for...in、Object.keys 等）
      track(target, ITERATE_KEY);
      return Reflect.ownKeys(target);
    },

    deleteProperty(target, key) {
      const hadKey = target.hasOwnProperty(key);
      const result = Reflect.deleteProperty(target, key);
      if (hadKey) {
        trigger(target, key);       // 触发具体属性
        trigger(target, ITERATE_KEY);  // 触发迭代
      }
      return result;
    },
  });
}

// 特殊的迭代 key（避免和真实属性冲突）
const ITERATE_KEY = Symbol('iterate');
```

### Proxy 解决了 defineProperty 的缺陷

```js
const state = reactive_Proxy({
  count: 0,
  user: { name: 'Alice' },
  items: [1, 2, 3],
});

// ✅ 新增属性 — 可以检测到
state.newProp = 42;  // 触发更新

// ✅ 删除属性 — 可以检测到
delete state.count;  // 触发更新

// ✅ 数组索引修改 — 可以检测到
state.items[0] = 99;  // 触发更新

// ✅ 数组长度修改 — 可以检测到
state.items.length = 0;  // 触发更新

// ✅ has/in 检测 — 可以追踪
'count' in state;  // 收集依赖
```

## 手写迷你响应式系统

将上述概念整合为一个完整的最小响应式系统：

```js
/**
 * Mini Reactivity System (~100 lines)
 * 包含：reactive、ref、computed、watch、watchEffect
 */

// ========== 核心：依赖收集与触发 ==========

let activeEffect = null;
const effectStack = [];  // 支持嵌套 effect

class Dep {
  subs = new Set();

  depend() {
    if (activeEffect) {
      this.subs.add(activeEffect);
    }
  }

  notify() {
    for (const sub of [...this.subs]) {
      sub.run();
    }
  }
}

const targetMap = new WeakMap();

function getDep(target, key) {
  let depsMap = targetMap.get(target);
  if (!depsMap) {
    targetMap.set(target, (depsMap = new Map()));
  }
  let dep = depsMap.get(key);
  if (!dep) {
    depsMap.set(key, (dep = new Dep()));
  }
  return dep;
}

// ========== Effect ==========

class ReactiveEffect {
  constructor(fn, scheduler = null) {
    this.fn = fn;
    this.scheduler = scheduler;
    this.active = true;
    this.parent = null;
  }

  run() {
    if (!this.active) return this.fn();

    try {
      this.parent = activeEffect;
      activeEffect = this;
      effectStack.push(this);
      cleanup(this);  // 清理旧依赖
      this.deps = new Set();  // 当前轮的新依赖集合

      return this.fn();
    } finally {
      effectStack.pop();
      activeEffect = this.parent;
    }
  }

  stop() {
    if (this.active) {
      cleanup(this);
      this.active = false;
    }
  }
}

function cleanup(effect) {
  if (effect.deps) {
    for (const dep of effect.deps) {
      dep.subs.delete(effect);
    }
    effect.deps.clear();
  }
}

function effect(fn, options = {}) {
  const _effect = new ReactiveEffect(
    fn,
    options.scheduler
  );

  _effect.run = function() {
    return ReactiveEffect.prototype.run.call(this);
  }.bind(_effect);

  if (!options.lazy) {
    _effect.run();
  }

  // 返回 runner 以便手动执行或停止
  const runner = _effect.run.bind(_effect);
  runner.effect = _effect;
  return runner;
}

// ========== reactive & ref ==========

function reactive(target) {
  return new Proxy(target, {
    get(target, key, receiver) {
      const res = Reflect.get(target, key, receiver);

      // 依赖收集
      if (typeof key === 'symbol' || key === '__v_isReactive') {
        return res;
      }

      getDep(target, key).depend();

      // 深层代理
      if (res !== null && typeof res === 'object') {
        return reactive(res);
      }
      return res;
    },

    set(target, key, value, receiver) {
      const oldValue = target[key];
      const result = Reflect.set(target, key, value, receiver);

      if (oldValue !== value && (oldValue === oldValue || value === value)) {
        getDep(target, key).notify();
      }

      return result;
    },
  });
}

function ref(value) {
  // ref 本身是一个包含 .value 的响应式对象
  const wrapper = {
    __v_isRef: true,
    get value() {
      getDep(wrapper, 'value').depend();
      return value;
    },
    set value(newValue) {
      if (value !== newValue) {
        value = newValue;
        getDep(wrapper, 'value').notify();
      }
    },
  };
  return wrapper;
}

// ========== computed ==========

functionComputed(getter) {
  let value;
  let dirty = true;  // 脏标记：是否需要重新计算

  const runner = effect(getter, {
    lazy: true,  // 不立即执行
    scheduler: () => {
      dirty = true;  // 依赖变化时只标记为脏，不立即重算
      trigger(wrapper, 'value');
    },
  });

  const wrapper = {
    __v_isRef: true,
    __v_isComputed: true,
    get value() {
      if (dirty) {
        value = runner();  // 按需计算
        dirty = false;
      }
      track(wrapper, 'value');
      return value;
    },
  };

  return wrapper;
}

// ========== watch / watchEffect ==========

function watchEffect(fn) {
  return effect(fn);
}

function watch(source, cb, options = {}) {
  let getter;

  if (typeof source === 'function') {
    getter = source;
  } else if ('__v_isRef' in source) {
    getter = () => source.value;
  } else {
    getter = () => traverse(source);
  }

  let oldValue;

  const job = () => {
    const newValue = getter();
    cb(newValue, oldValue);
    oldValue = newValue;
  };

  const runner = effect(getter, {
    lazy: true,
    scheduler: job,
  });

  if (options.immediate) {
    job();
  } else {
    oldValue = runner();
  }

  return () => runner.effect.stop();
}

function traverse(value, seen = new Set()) {
  if (typeof value !== 'object' || value === null || seen.has(value)) {
    return value;
  }
  seen.add(value);
  for (const key of Object.keys(value)) {
    traverse(value[key], seen);
  }
  return value;
}

// ========== 使用示例 ==========

const state = reactive({ count: 0, name: 'hello' });

// 基础 effect
watchEffect(() => {
  console.log(`state changed: count=${state.count}, name=${state.name}`);
});
// → state changed: count=0, name=hello

state.count++;   // → state changed: count=1, name=hello
state.name = 'world';  // → state changed: count=1, name=world

// computed
const doubled = computed(() => state.count * 2);
console.log(doubled.value);  // 2
state.count = 5;
console.log(doubled.value);  // 10

// watch
watch(
  () => state.count,
  (newVal, oldVal) => console.log(`count: ${oldVal} -> ${newVal}`)
);
state.count++;  // count: 5 -> 6
```

## 小结

| 技术 | 能力 | 局限 |
|------|------|------|
| **手动依赖管理** | 理解原理 | 不可维护 |
| **Object.defineProperty** | Vue 2 响应式基础 | 无法检测属性增删、数组索引变更、需递归遍历 |
| **Proxy** | Vue 3 响应式核心 | IE11 不支持、需配合 Reflect |

从手写迷你系统中可以看到，一个完整的响应式引擎只需要几个关键组件：

1. **Dep**：依赖桶（Set 存储订阅者）
2. **ReactiveEffect**：副作用包装器
3. **track() / trigger()**：依赖收集与触发通知
4. **dirty 标记 + 懒求值**：computed 的缓存机制
5. **scheduler**：异步调度策略

接下来我们将看到这些抽象如何在 Vue 2 和 Vue 3 中落地。
