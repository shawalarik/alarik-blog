这一模块聚焦 Vue 2 的经典响应式实现，理解它的工程价值、设计取舍与历史局限。

## 整体架构

Vue 2 的响应式系统由三个核心组件构成：

```
┌─────────────────────────────────────────────┐
│                 Observer                     │
│  (通过 Object.defineProperty 劫持对象/数组)    │
│                                              │
│  ┌──────────┐     ┌───────────┐             │
│  │   Dep    │◄───│  Watcher   │             │
│  │ (依赖收集) │    │ (观察者)   │             │
│  └──────────┘     └───────────┘             │
│         ▲               │                    │
│         │  notify()     │ get() 时 collect   │
└─────────│───────────────│────────────────────┘
          │               │
          ▼               ▼
    数据变化触发      重新执行回调/渲染
```

- **Observer**：将普通对象转为响应式（defineProperty 劫持）
- **Dep**：依赖收集器，每个属性一个 Dep 实例
- **Watcher**：观察者，组件渲染函数 / 用户 watch 回调都是 Watcher 实例

## Object.defineProperty 劫持

### 入口：initData

```js
// src/core/instance/state.js — 简化版
function initData(vm) {
  let data = vm.$options.data;
  data = vm._data = typeof data === 'function' ? data.call(vm) : data;

  const keys = Object.keys(data);

  // 代理：this.xxx → this._data.xxx
  keys.forEach(key => {
    proxy(vm, '_data', key);
  });

  // 将数据变为响应式
  observe(data);
}
```

### observe 函数

```js
// src/core/observer/index.js — 简化版
function observe(value) {
  if (typeof value !== 'object' || value === null) return;

  let ob;
  if (value.__ob__) {
    // 已被观察过
    ob = value.__ob__;
  } else {
    ob = new Observer(value);
  }
  return ob;
}

class Observer {
  constructor(value) {
    this.value = value;
    this.dep = new Dep();  // 自身依赖（用于 $set/$delete）

    def(value, '__ob__', this, false);  // 附加引用

    if (Array.isArray(value)) {
      // 数组特殊处理
      protoAugment(value, arrayMethods);
      this.observeArray(value);
    } else {
      this.walk(value);
    }
  }

  walk(obj) {
    const keys = Object.keys(obj);
    keys.forEach(key => defineReactive(obj, key));
  }

  observeArray(items) {
    items.forEach(item => observe(item));
  }
}
```

### defineReactive — 核心中的核心

```js
// src/core/observer/index.js
function defineReactive(obj, key, val, shallow) {
  const dep = new Dep();  // 每个属性一个 Dep

  const property = Object.getOwnPropertyDescriptor(obj, key);

  // 如果属性已有自定义的 getter/setter，需要保留
  const getter = property && property.get;
  const setter = property && property.set;

  if ((!getter || setter) && arguments.length === 2) {
    val = obj[key];
  }

  // 对值进行深度观察
  let childOb = !shallow && observe(val);

  Object.defineProperty(obj, key, {
    enumerable: true,
    configurable: true,

    get() {
      const value = getter ? getter.call(obj) : val;

      // 依赖收集 —— 关键！
      if (Dep.target) {  // 当前的 Watcher
        dep.depend();            // 收集当前属性依赖
        if (childOb) {
          childOb.dep.depend();  // 深层对象的依赖（$set 需要）
        }
      }
      return value;
    },

    set(newVal) {
      const value = getter ? getter.call(obj) : val;

      if (newVal === value || (newVal !== newVal && value !== value)) {
        return;  // NaN !== NaN 但语义上相同
      }

      if (setter) {
        setter.call(obj, newVal);
      } else if (getter) {
        // 有 getter 没有 setter —— 只读属性
        return;
      } else {
        val = newVal;
      }

      // 新值也需要变成响应式
      childOb = !shallow && observe(newVal);

      // 触发通知 —— 关键！
      dep.notify();
    },
  });
}
```

### 数组的特殊处理

由于 `Object.defineProperty` 无法天然拦截数组索引操作，Vue 2 通过**覆写数组原型方法**来变通：

```js
// src/core/observer/array.js
const arrayProto = Array.prototype;
const arrayMethods = Object.create(arrayProto);

const methodsToPatch = [
  'push', 'pop', 'shift', 'unshift',
  'splice', 'sort', 'reverse'
];

methodsToPatch.forEach(method => {
  def(arrayMethods, method, function(...args) {
    const original = arrayProto[method];
    const result = original.apply(this, args);

    const ob = this.__ob__;
    let inserted;

    switch (method) {
      case 'push':
      case 'unshift':
        inserted = args;
        break;
      case 'splice':
        inserted = args.slice(2);
        break;
    }

    // 对新插入的元素进行响应式处理
    if (inserted) {
      ob.observeArray(inserted);
    }

    // 触发更新
    ob.dep.notify();

    return result;
  });
});

// 替换数组实例的原型
function protoAugment(target, src) {
  target.__proto__ = src;
}
```

这就是为什么 Vue 2 中只能通过 `push/pop/splice` 等 7 个变异方法来触发数组的响应式更新。

## 依赖收集（Dep）

```js
// src/core/observer/dep.js
let uid = 0;

export default class Dep {
  static target = null;  // 全局唯一的当前 Watcher

  id = ++uid;
  subs = [];              // 订阅者数组

  addSub(sub) {
    this.subs.push(sub);
  }

  removeSub(sub) {
    remove(this.subs, sub);
  }

  depend() {
    if (Dep.target) {
      Dep.target.addDep(this);  // Watcher 把自己加入这个 Dep
    }
  }

  notify() {
    // 浅拷贝防止循环中修改
    const subs = this.subs.slice();

    for (let i = 0, l = subs.length; i < l; i++) {
      subs[i].update();  // 通知每个 Watcher 更新
    }
  }
}
```

`Dep.target` 的设计是整个依赖收集的关键——它利用 JavaScript 单线程特性，用一个全局变量存储"当前正在执行的 Watcher"。

## 观察者（Watcher）

Watcher 是连接数据与副作用的桥梁。Vue 2 中有三类 Watcher：

```js
// src/core/observer/watcher.js — 简化版
class Watcher {
  constructor(vm, expOrFn, cb, options) {
    this.vm = vm;
    this.cb = cb;
    this.id = ++uid;
    this.deps = [];
    this.newDeps = [];
    this.depIds = new Set();
    this.newDepIds = new Set();

    if (typeof expOrFn === 'function') {
      this.getter = expOrFn;  // 渲染 watcher 或 computed
    } else {
      this.getter = parsePath(expOrFn);  // watch 表达式如 'a.b.c'
    }

    this.value = this.lazy ? undefined : this.get();  // 非 lazy 立即求值
  }

  get() {
    pushTarget(this);  // 设置 Dep.target = this
    try {
      // 执行 getter（渲染函数 / watch 表达式）
      // 执行过程中访问的响应式属性会触发它们的 get，
      // 进而调用 Dep.target.addDep(this)
      value = this.getter.call(this.vm, this.vm);
    } catch (e) {
      throw e;
    } finally {
      popTarget();  // 恢复 Dep.target
    }
    this.cleanupDeps();
    return value;
  }

  addDep(dep) {
    const id = dep.id;
    if (!this.newDepIds.has(id)) {
      this.newDepIds.add(id);
      this.newDeps.push(dep);
      if (!this.depIds.has(id)) {
        dep.addSub(this);  // 双向记录
      }
    }
  }

  update() {
    if (this.lazy) {
      this.dirty = true;  // computed：仅标记脏
    } else if (this.sync) {
      this.run();         // 同步执行
    } else {
      queueWatcher(this);  // 异步队列
    }
  }

  run() {
    const value = this.get();
    if (value !== this.value || isObject(value) || this.deep) {
      const oldValue = this.value;
      this.value = value;
      this.cb.call(this.vm, value, oldValue);
    }
  }
}
```

### 三类 Watcher

| 类型 | 创建位置 | getter | 用途 |
|------|---------|--------|------|
| **渲染 Watcher** | mountComponent | updateComponent | 组件重新渲染 |
| **用户 Watcher** | `$watch()` | 用户表达式 | watch 回调 |
| **Computed Watcher** | initComputed | computed getter | 缓存派生值 |

### 依赖收集流程图

```
1. 组件首次渲染：
   mountComponent()
   └→ new Watcher(vm, updateComponent)
      └→ watcher.get()
         ├→ pushTarget(watcher)   // Dep.target = watcher
         ├→ updateComponent()     // 执行渲染函数
         │  └→ 访问 state.msg
         │     └→ msg 的 defineProperty.get()
         │        └→ dep.depend()
         │           └→ Dep.target.addDep(dep)  ← watcher 记录依赖
         └→ popTarget()         // Dep.target = null

2. 数据变化：
   state.msg = 'hi'
   └→ msg 的 defineProperty.set()
      └→ dep.notify()
         └→ watcher.update()
            └→ queueWatcher(watcher)  // 加入异步队列
               └→ nextTick 后 flushSchedulerQueue
                  └→ watcher.run()
                     └→ 重新执行 updateComponent（重新渲染）
```

## 异步更新队列（nextTick）

### 为什么需要异步批量更新？

```html
<template>
  <div>
    {{ count }}  <!-- Watcher 依赖 count -->
  </div>
</template>
```

```js
this.count++;
this.name = 'new';
this.flag = true;
```

如果每次 `set` 都同步触发渲染，上面三行代码会导致 **3 次 DOM diff + patch**。nextTick 将它们合并为 **1 次**。

### nextTick 实现

```js
// src/core/util/next-tick.js
const callbacks = [];
let pending = false;

function flushCallbacks() {
  pending = false;
  const copies = callbacks.slice(0);
  callbacks.length = 0;
  copies.forEach(cb => cb());
}

let timerFunc;

// 按优先级选择异步方法
if (typeof Promise !== 'undefined' && isNative(Promise)) {
  const p = Promise.resolve();
  timerFunc = () => {
    p.then(flushCallbacks);
  };
} else if (typeof MutationObserver !== 'undefined') {
  // MutationObserver 作为降级方案
  let counter = 1;
  const observer = new MutationObserver(flushCallbacks);
  const textNode = document.createTextNode(String(counter));
  observer.observe(textNode, { characterData: true });
  timerFunc = () => {
    counter = (counter + 1) % 2;
    textNode.data = String(counter);
  };
} else if (typeof setImmediate !== 'undefined') {
  timerFunc = () => setImmediate(flushCallbacks);
} else {
  timerFunc = () => setTimeout(flushCallbacks, 0);
}

export function nextTick(cb, ctx) {
  callbacks.push(() => {
    if (cb) {
      try {
        cb.call(ctx);
      } catch (e) {
        handleError(e, ctx, 'nextTick');
      }
    }
  });

  if (!pending) {
    pending = true;
    timerFunc();
  }

  // 不传 cb 则返回 Promise
  if (!cb && typeof Promise !== 'undefined') {
    return resolvePromise;
  }
}

// 使用方式
nextTick(() => {
  console.log('DOM 已更新');
});

// 或 async/await
await nextTick();
```

### Scheduler Queue

```js
// src/core/observer/scheduler.js
const queue = [];
let waiting = false;
let flushing = false;
let index = 0;

function queueWatcher(watcher) {
  const id = watcher.id;
  if (queue.indexOf(watcher) === -1) {  // 去重
    queue.push(watcher);

    if (!flushing) {
      waiting = true;
      nextTick(flushSchedulerQueue);
    }
  }
}

function flushSchedulerQueue() {
  flushing = true;
  queue.sort((a, b) => a.id - b.id);  // 按 id 排序（父组件优先）

  for (index = 0; index < queue.length; index++) {
    const watcher = queue[index];
    watcher.before?.();  // 生命周期 beforeUpdate
    watcher.run();
  }

  // 重置状态
  queue.length = 0;
  flushing = false;
  waiting = false;

  // 执行用户注册的 nextTick 回调
  resetSchedulerState();
}
```

## 数组/对象变更检测限制及解决方案

### 数组限制

```js
// ❌ 以下操作不会触发响应式更新
vm.items[indexOfItem] = newValue;    // 直接设索引
vm.items.length = newLength;          // 直接改长度
```

**解决方案：**

```js
// 方式一：Vue.set
Vue.set(vm.items, indexOfItem, newValue);

// 方式二：splice
vm.items.splice(indexOfItem, 1, newValue);

// 方式三：替换整个数组
vm.items = [...vm.items, newItem];
```

### 对象限制

```js
// ❌ Vue 2 无法检测到动态添加的属性
vm.newProperty = 'hello';
```

**解决方案：**

```js
// Vue.set
Vue.set(vm.someObject, 'newProperty', 'hello');

// 或
this.$set(this.someObject, 'newProperty', 'hello');
```

### $set / $delete 原理

```js
export function set(target, key, val) {
  if (Array.isArray(target) && isValidArrayIndex(key)) {
    // 数组场景：用 splice 触发响应式
    target.splice(key, 1, val);
    return val;
  }

  const ob = target.__ob__;
  if (key in target && !(key in Object.prototype)) {
    // 已存在的属性：直接赋值（已响应式）
    target[key] = val;
    return val;
  }

  // 新增属性
  ob.observeArray([val]);
  defineReactive(target, key, val);
  ob.dep.notify();  // 手动触发通知
  return val;
}
```

## 小结

| 特性 | 实现细节 | 影响 |
|------|---------|------|
| **劫持方式** | Object.defineProperty | 无法检测新增/删除属性、数组索引变更 |
| **依赖收集** | Dep.target 全局变量 + Stack | 利用 JS 单线程特性，简洁但需小心维护 |
| **深度响应式** | 递归 defineReactive | 大对象初始化开销高 |
| **数组处理** | 覆写 7 个变异方法 | 必须使用指定方法才能触发更新 |
| **批量更新** | nextTick + 排序去重队列 | 合并多次数据变化为一次渲染 |
| **$set / $delete** | 手动 defineReactive + notify | 补充 defineProperty 的能力缺口 |

Vue 2 的响应式系统在约束条件下做到了工程上的最佳平衡。下一章将看到 Vue 3 如何用 Proxy 彻底解决这些限制。
