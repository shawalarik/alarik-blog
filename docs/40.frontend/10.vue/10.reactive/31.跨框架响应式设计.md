这一模块从框架抽象层看响应式能力，提炼可复用的通用模型与工程实践。

## 为什么需要跨框架响应式

### 现实场景

```
1. 共享业务逻辑库（同时用于 Vue / React / Node.js）
2. 微前端架构中不同技术栈的组件通信
3. 设计系统 / 组件库需要支持多框架
4. 团队正在迁移框架，需要过渡方案
5. SSR 场景中服务端和客户端使用不同策略
```

## 框架无关的响应式库

### @vue/reactivity 独立使用

Vue 3 的响应式系统被提取为独立包，可以在任何 JavaScript 环境中使用：

```bash
npm install @vue/reactivity
```

```ts
// shared-store.ts —— 框架无关的状态管理
import {
  reactive,
  ref,
  computed,
  effect,
  stop,
} from '@vue/reactivity';

// ===== 定义状态 =====
interface AppState {
  user: User | null;
  theme: 'light' | 'dark';
  locale: string;
}

const state = reactive<AppState>({
  user: null,
  theme: 'dark',
  locale: 'zh-CN',
});

const isAuth = computed(() => !!state.user);

// 暴露 API
export function useSharedState() {
  return {
    state,
    isAuth,

    // Actions
    login(user: User) { state.user = user; },
    logout() { state.user = null; },
    setTheme(theme: AppState['theme']) { state.theme = theme; },

    // 订阅/取消
    subscribe(fn: () => void) {
      const runner = effect(fn);
      return () => stop(runner);
    },
  };
}
```

```tsx
// React 集成
import { useSyncExternalStore } from 'react';
import { useSharedState } from './shared-store';

function useAppState() {
  const store = useSharedState();
  
  return useSyncExternalStore(
    // subscribe —— 返回 unsubscribe 函数
    store.subscribe,  
    // getSnapshot —— 返回当前快照
    () => JSON.stringify(store.state),  // 或返回特定切片
    // 可选：服务端的快照
    () => JSON.stringify(initialServerState)
  );
}

function UserProfile() {
  const appState = useAppState();
  const data = JSON.parse(appState);  // 解析快照

  if (!data.user) return <LoginButton />;

  return <div>Welcome, {data.user.name}</div>;
}
```

```vue
<!-- Vue 集成 -->
<script setup>
import { toRefs } from 'vue';
import { useSharedState } from './shared-store';

// 直接在 Vue 中使用！@vue/reactivity 的 reactive 与 Vue 内置的是同一个实现
</script>

<template>
  <div :data-theme="state.theme">
    <p v-if="isAuth">Hello, {{ state.user?.name }}</p>
  </div>
</template>
```

### Nanostores —— 轻量级 Signal Store

```bash
npm install nanostores
```

```ts
import { atom, computed, onNotify } from 'nanostores';

// ===== Atom（基本 Signal Store）=====
export const $theme = atom<'light' | 'dark'>('dark');
export const $locale = atom('zh-CN');

// ===== Computed（派生 Store）=====
export const $isDark = computed($theme, (theme) => theme === 'dark');

// ===== Map Store（键值对集合）=====
export const $todos = map<Todo>({});

// ===== 使用 =====

// 在任何地方读取
console.log($theme.get());  // 'dark'

// 在任何地方修改
$theme.set('light');
$locale.set('en-US');

// Map Store 操作
$todos.setKey('todo-1', { id: 'todo-1', text: 'Learn', done: false });
$todos.setKey('todo-2', { id: 'todo-2', text: 'Build', done: true });
$todos.deleteKey('todo-1');

// ===== 框架集成 =====

// React:
// npm install @nanostores/react
import { useStore } from '@nanostores/react';
function ThemeToggle() {
  const theme = useStore($theme);
  return <button onClick={() => $theme.set(theme === 'dark' ? 'light' : 'dark')}>
    {theme}
  </button>;
}

// Vue:
// npm install @nanostores/vue
import { useStore } from '@nanostores/vue';
const theme = useStore($theme);  // Ref<string>

// Preact/Solid/VueUse/Svelte 都有对应集成包
```

### Signals 库对比

| 库 | 大小 (gzip) | 特点 |
|----|------------|------|
| **@vue/reactivity** | ~6KB | 功能最完整，Vue 官方维护 |
| **nanostores** | ~0.5KB | 极小、TypeScript 原生、多框架适配器齐全 |
| **@preact/signals** | ~2.5KB | Preact 生态，React 适配器成熟 |
| **@legendapp/state** | ~5KB | 高性能 + 细粒度 + 丰富的开发工具 |

## 响应式与不可变数据的通用模式

### 模式一：Signal Store + 不可变更新

```ts
// 结合两者的优点：Signal 提供细粒度通知，不可变性提供可预测性
import { atom } from 'nanostories';
import { produce } from 'immer';

type Todo = { id: string; text: string; done: boolean };
const $todos = atom<Todo[]>([]);

// 更新操作始终通过不可变方式
function addTodo(text: string) {
  $todos.set(produce(draft => {
    draft.push({ id: crypto.randomUUID(), text, done: false });
  }));
}

function toggleTodo(id: string) {
  $todos.set(produce(draft => {
    const todo = draft.find(t => t.id === id);
    if (todo) todo.done = !todo.done;
  }));
}

// 消费者获得不可变的快照
const todos = $todos.get();  // 普通数组，可以安全传递给任何库
```

### 模式二：Event Sourcing 风格

```ts
// 不直接存储状态，而是存储事件序列
// 状态由事件推导得出

type AppEvent =
  | { type: 'SET_THEME'; payload: 'light' | 'dark' }
  | { type: 'ADD_TODO'; payload: { text: string } }
  | { type: 'TOGGLE_TODO'; payload: { id: string } };

const $events = atom<AppEvent[]>([]);
const $state = computed($events, (events) => {
  return events.reduce((state, event) => reducer(state, event), initialState);
});

function dispatch(event: AppEvent) {
  $events.set([...$events.get(), event]);
}

// 好处：
// - 完整的操作历史（时间旅行）
// - 易于调试和回放
// - 天然支持撤销/重做
```

### 模式三：Adapter Pattern（适配器模式）

```ts
// 为不同框架提供统一的 Store 接口

interface IStore<T> {
  getState(): T;                    // 同步获取快照
  setState(partial: Partial<T>): void;   // 更新
  subscribe(listener: () => void): () => void;  // 订阅
  getSlice<K extends keyof T>(key: K): T[K];     // 获取切片
}

// Vue Adapter
class VueAdapter<T> implements IStore<T> {
  constructor(private state: Reactive<T>) {}

  getState() {
    return toRaw(this.state);  // 返回原始对象
  }

  setState(patch: Partial<T>) {
    Object.assign(this.state, patch);
  }

  subscribe(listener: () => void) {
    return watchEffect(() => {
      this.state;  // 触发依赖收集
      listener();
    }, { flush: 'sync' });  // 注意清理逻辑
  }
}

// React Adapter
class ReactAdapter<T> implements IStore<T> {
  private listeners = new Set<() => void>();

  constructor(private state: T) {}

  getState() { return this.state; }

  setState(patch: Partial<T>) {
    this.state = { ...this.state, ...patch };
    this.listeners.forEach(l => l());
  }

  subscribe(listener: () => void) {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }
}
```

## 依赖追踪与触发更新的统一抽象

### 最小统一接口设计

```ts
// ===== 核心类型定义 =====
interface ISignal<T> {
  readonly (): T;           // 读取（建立依赖）
  subscribe(listener: (v: T) => void): Unsubscribe;
}

interface IWritableSignal<T> extends ISignal<T> {
  (value: T): void;          // 写入（触发通知）
}

interface IComputedSignal<T> extends ISignal<T> {
  // 只读派生信号
}

interface IEffect {
  (): void;                  // 执行副作用
  destroy(): void;            // 销毁并取消订阅
}

// ===== 统一工厂函数 =====
function createSignal<T>(initial: T): IWritableSignal<T>;
function createComputed<T>(getter: () => T): IComputedSignal<T>;
function createEffect(fn: () => void): IEffect;

// ===== 不同实现的适配层 =====

// 基于 @vue/reactivity
function vueReactivitySystem() {
  const { ref, computed, effect, stop } = require('@vue/reactivity');

  return {
    createSignal<T>(initial: T) {
      const r = ref(initial) as any;
      r.subscribe = (fn) => { const e = effect(() => fn(r.value)); return () => stop(e); };
      return r;
    },
    createComputed: computed,
    createEffect: (fn) => { const e = effect(fn); e.destroy = () => stop(e); return e; },
  };
}

// 基于 nanostores
function nanoStoreSystem() {
  const { atom, computed, onNotify } = require('nanostores');

  return {
    createSignal<T>(initial: T): any {
      const s = atom(initial);
      s.subscribe = (fn) => onNotify(s, fn);
      return s;
    },
    createComputed(source, fn) {
      return computed(source, fn);
    },
    createEffect(fn) {
      let unsub: (() => void) | null = null;
      unsub = fn();  // 取决于具体实现
      return { destroy: () => unsub?.() };
    },
  };
}
```

### 实际案例：跨框架组件库

```tsx
// Design System Button Component
// 同时支持 React 和 Vue

// 1. 核心逻辑（框架无关）
// button-state.ts
import { atom, computed } from 'nanostores';

export interface ButtonProps {
  variant?: 'primary' | 'secondary' | 'ghost';
  size?: 'sm' | 'md' | 'lg';
  disabled?: boolean;
  loading?: boolean;
}

export function createButtonState(props: ButtonProps) {
  const $variant = atom(props.variant ?? 'primary');
  const $size = atom(props.size ?? 'md');
  const $disabled = atom(props.disabled ?? false);
  const $loading = atom(props.loading ?? false);

  const $classes = computed(
    [$variant, $size, $disabled, $loading],
    ([variant, size, disabled, loading]) => ({
      root: [
        'btn',
        `btn--${variant}`,
        `btn--${size}`,
        disabled && 'btn--disabled',
        loading && 'btn--loading',
      ].filter(Boolean).join(' '),
    })
  );

  return { $variant, $size, $disabled, $loading, $classes };
}

// 2. React 包装器
// Button.tsx
import { useStore } from '@nanostores/react';
import { createButtonState } from './button-state';

export function Button(props: ButtonProps & { children?: ReactNode }) {
  const state = createButtonState(props);
  const classes = use(state.$classes);

  return (
    <button className={classes.root} disabled={props.disabled}>
      {props.loading && <Spinner />}
      {props.children}
    </button>
  );
}

// 3. Vue 包装器
// Button.vue
<template>
  <button :class="classes.root" :disabled="$disabled">
    <Spinner v-if="$loading" />
    <slot />
  </button>
</template>

<script setup lang="ts">
import { useStore } from '@nanostores/vue';
import { createButtonState } from './button-state';

const props = defineProps<ButtonProps>();
const { $classes, $disabled, $loading } = createButtonState(props);
const classes = useStore($classes);
</script>
```

## 小结

| 方案 | 适用场景 | 优势 | 局限 |
|------|---------|------|------|
| **@vue/reactivity** | 需要 Vue 级别的完整功能 | 功能最丰富、生态好 | 包较大、API 是 Vue 风格 |
| **Nanostores** | 跨框架轻量状态共享 | 极小(~0.5KB)、适配器全、TypeScript 友好 | 功能相对基础 |
| **自定义统一接口** | 设计系统/组件库 | 完全控制语义 | 维护成本高 |
| **Event Sourcing** | 需要完整历史记录/协作编辑 | 天然可追溯、可重放 | 内存占用大、实现复杂 |

跨框架响应式设计的核心原则是**将状态管理与 UI 框架解耦**。选择合适的抽象层次——太深会引入不必要的复杂性，太浅则无法覆盖所有框架的差异。
