Vue Router 4 为 Composition API 提供了两个核心 Composable：`useRouter` 和 `useRoute`。本文聚焦它们的使用方式与最佳实践。

## 核心 Composable

### useRouter — 路由器实例

`useRouter()` 返回路由器实例，用于**编程式导航**和**全局控制**：

```vue
<script setup lang="ts">
import { useRouter } from 'vue-router'

const router = useRouter()

// ========== 编程式导航 ==========

// 基本导航
function goHome() {
  router.push('/')                           // 字符串路径
  router.push({ name: 'home' })             // 命名路由
  router.push({ path: '/user/42' })         // 路径 + 参数
}

// 带查询参数
function search(keyword: string) {
  router.push({
    path: '/search',
    query: { q: keyword, page: '1' }         // ?q=xxx&page=1
  })
}

// 带 hash
function goToSection(id: string) {
  router.push({ path: '/docs', hash: `#${id}` })  // /docs#section-id
}

// 替代当前位置（不产生历史记录）
function replaceCurrent() {
  router.replace('/new-location')
}

// 跨历史记录导航
function goBack() { router.back() }
function goForward() { router.forward() }
function go(n: number) { router.go(n) }  // 前进/后退 n 步

// ========== 高级导航 ==========

// 导航失败处理
async function navigateWithFallback() {
  try {
    // push/promise 模式（Vue Router 4 支持 Promise 返回值）
    await router.push({ name: 'profile', params: { id: 123 } })
    console.log('导航成功')
  } catch (err) {
    // 导航被守卫拦截或重定向时 reject
    if (isNavigationFailure(err, NavigationFailureType.aborted)) {
      console.log('导航被取消')
    } else if (isNavigationFailure(err, NavigationFailureType.duplicated)) {
      console.log('已在目标位置')
    } else {
      console.error('未知错误:', err)
    }
  }
}

// 完整导航选项
async function fullControlNav() {
  await router.push({
    path: '/target',
    // 控制滚动行为
    scrollBehavior: (to, from, savedPos) => savedPos || { top: 0 },
  })

  // 查询当前路由信息
  console.log('current route:', router.currentRoute.value.fullPath)
  
  // 添加路由后立即导航（动态路由场景）
  router.addRoute({ path: '/dynamic', component: DynamicComp })
  await router.push('/dynamic')

  // 导航完成后的回调（替代 afterEach）
  router.afterEach((to) => {
    document.title = to.meta.title || 'My App'
  })
}
</script>
```

### useRoute — 当前路由信息

`useRoute()` 返回当前路由的位置信息，用于**读取**当前状态：

```vue
<script setup lang="ts">
import { useRoute, watch } from 'vue-router'

const route = useRoute()

// ========== 读取路由属性 ==========

// 路径相关
console.log(route.path)           // "/users/123?tab=active#section"
console.log(route.fullPath)       // "/users/123?tab=active#section" (包含 query/hash)
console.log(route.name)           // "user-profile" (命名路由名)
console.log(route.redirectedFrom) // 重定向来源（如果有）

// 参数
console.log(route.params)         // { id: '123' } (动态路径参数)
console.log(route.query)          // { tab: 'active' } (查询参数)
console.log(route.hash)           // "#section"

// 元信息
console.log(route.meta)           // { requiresAuth: true, roles: ['admin'] }
console.log(route.matched)        // 匹配的路由记录数组

// ========== 响应式监听路由变化 ==========

// 监听整个路由对象变化
watch(() => route.fullPath, (newPath, oldPath) => {
  console.log(`路由从 ${oldPath} 变为 ${newPath}`)
}, { immediate: true })

// 只监听某个参数变化（更精确）
watch(() => route.params.id, (newId, oldId) => {
  console.log(`用户 ID 从 ${oldId} 变为 ${newId}`)
  fetchUserData(newId)
})

// 只监听 query 变化
watch(() => JSON.stringify(route.query), () => {
  handleSearchQuery(route.query)
})
</script>

<template>
  <div class="user-profile">
    <p>User ID: {{ route.params.id }}</p>
    <p>Current tab: {{ route.query.tab }}</p>
    <nav v-if="route.meta.breadcrumbs">
      <span v-for="(crumb, i) in route.meta.breadcrumbs" :key="i">
        {{ crumb }}
      </span>
    </nav>
  </div>
</template>
```

## 组合式函数封装最佳实践

### 封装路由相关的 Composable

```typescript
// composables/useAuthGuard.ts
import { useRouter, useRoute } from 'vue-router'
import { computed } from 'vue'

/**
 * 认证守卫 composable
 * 自动处理登录检测和权限校验
 */
export function useAuthGuard(requiredRoles?: string[]) {
  const router = useRouter()
  const route = useRoute()
  const authStore = useAuthStore()

  const isLoggedIn = computed(() => authStore.isLoggedIn)
  const hasPermission = computed(() => {
    if (!requiredRoles) return true
    return authStore.hasAnyRole(requiredRoles)
  })

  function checkAndRedirect(): boolean {
    if (!isLoggedIn.value) {
      router.push({
        name: 'login',
        query: { redirect: route.fullPath }
      })
      return false
    }
    if (!hasPermission.value) {
      router.push({ name: 'forbidden' })
      return false
    }
    return true
  }

  return { isLoggedIn, hasPermission, checkAndRedirect }
}
```

```typescript
// composables/usePagination.ts
import { useRouter, useRoute } from 'vue-router'
import { ref, watch, computed } from 'vue'

/**
 * 路由驱动的分页 composable
 * 将分页状态同步到 URL query 中
 */
export function usePagination(defaultPage = 1, defaultSize = 10) {
  const router = useRouter()
  const route = useRoute()

  const currentPage = ref(Number(route.query.page) || defaultPage)
  const pageSize = ref(Number(route.query.size) || defaultSize)

  // 同步到 URL
  async function syncUrl() {
    await router.replace({
      query: { ...route.query, page: String(currentPage.value), size: String(pageSize.value) }
    })
  }

  watch(currentPage, syncUrl)
  watch(pageSize, () => { currentPage.value = 1; syncUrl() })

  // 外部 URL 变化时同步回来（浏览器前进/后退）
  watch(() => route.query.page, (val) => {
    if (val) currentPage.value = Number(val)
  })

  function nextPage() { currentPage.value++ }
  function prevPage() { if (currentPage.value > 1) currentPage.value-- }
  function reset() { currentPage.value = 1; pageSize.value = defaultSize; syncUrl() }

  const offset = computed(() => (currentPage.value - 1) * pageSize.value)

  return { currentPage, pageSize, offset, nextPage, prevPage, reset }
}
```

## 在 `<script setup>` 中的无缝使用

### 典型页面组件示例

```vue
<script setup lang="ts">
// 所有 Composable 都可以在顶层直接调用
// （必须在 setup() 同步阶段，不能在条件语句或回调中）
import { useRoute, useRouter, onBeforeRouteLeave } from 'vue-router'
import { useAuthGuard } from '@/composables/useAuthGuard'
import { usePagination } from '@/composables/usePagination'

// 路由工具
const route = useRoute()
const router = useRouter()
const { checkAndRedirect } = useAuthGuard(['admin'])
const { currentPage, pageSize, offset, nextPage, prevPage } = usePagination()

// 数据获取
const { data: users, loading, error, refresh } = await useFetch(
  () => `/api/users?page=${currentPage.value}&size=${pageSize.value}`,
  { key: `users-${route.params.teamId}` }
)

// 页面守卫
onBeforeRouteLeave(() => {
  if (hasUnsavedEdits.value) {
    return confirm('有未保存的修改，确定离开吗？')
  }
})

// 初始化检查
checkAndRedirect()

// 操作方法
function editUser(id: number) {
  router.push({ name: 'user-edit', params: { id } })
}

function createUser() {
  router.push({ name: 'user-create', query: { teamId: route.params.teamId } })
}

function backToList() {
  router.back()  // 或 router.push({ name: 'team-users', params: { teamId: route.params.teamId } })
}
</script>

<template>
  <div class="user-list-page">
    <!-- 面包屑 -->
    <nav class="breadcrumbs">
      <router-link to="/">首页</router-link>
      <span>/</span>
      <router-link :to="{ name: 'teams' }">团队</router-link>
      <span>/</span>
      <span>{{ route.params.teamId }} 成员</span>
    </nav>

    <!-- 工具栏 -->
    <div class="toolbar">
      <button @click="createUser">+ 新建成员</button>
      <select v-model.number="pageSize">
        <option :value="10">10 条/页</option>
        <option :value="20">20 条/页</option>
        <option :value="50">50 条/页</option>
      </select>
    </div>

    <!-- 表格 -->
    <table v-if="!loading">
      <tr v-for="user in users" :key="user.id">
        <td>{{ user.name }}</td>
        <td><button @click="editUser(user.id)">编辑</button></td>
      </tr>
    </table>

    <!-- 分页 -->
    <div class="pagination">
      <button :disabled="currentPage <= 1" @click="prevPage">上一页</button>
      <span>第 {{ currentPage }} 页</span>
      <button @click="nextPage">下一页</button>
    </div>

    <!-- 导航链接的各种写法 -->
    <footer>
      <!-- 1. 声明式 -->
      <router-link :to="{ name: 'settings' }">设置</router-link>

      <!-- 2. 编程式 -->
      <a href="#" @click.prevent="backToList">返回列表</a>

      <!-- 3. 带参数 -->
      <router-link :to="{ path: `/reports/${route.params.teamId}` }">
        报告
      </router-link>
    </footer>
  </div>
</template>
```

## 常见陷阱

### 陷阱 1：在回调中调用 useRoute / useRouter

```vue
<script setup>
// ✅ 正确：在 setup 顶层调用
const router = useRouter()

async function fetchData() {
  const res = await api.get('/data')
  // 这里可以使用 router（闭包引用）
  if (res.code === 401) router.push('/login')
}

// ❌ 错误：在回调/条件/循环中调用
setTimeout(() => {
  const r = useRouter()  // ⚠️ 运行时错误！只能在 setup 同步期间调用
}, 1000)

if (someCondition) {
  const r = useRoute()  // ⚠️ 同样错误！
}
</script>
```

**原因**：`useRouter` / `useRoute` 依赖 Vue 的 `inject` 机制获取路由上下文，而 `inject` 只能在 `setup()` 同步执行期间工作。

**解决方案**：始终在顶层调用并赋值给变量。

### 陷阱 2：响应式丢失

```vue
<script setup>
import { useRoute, watchEffect } from 'vue-router'

const route = useRoute()

// ✅ 正确：在 computed/watch 中访问 route 属性（自动追踪依赖）
const userId = computed(() => route.params.id)

watchEffect(() => {
  console.log(route.query.search)  // 自动追踪 search 变化
})

// ⚠️ 注意：解构会丢失响应性
const { params, query } = route  // params 和 query 不是响应式的！
// 应该用 toRefs 或直接在模板中使用 route.xxx
</script>
```
