Pinia 是 Vue 生态系统中官方推荐的状态管理方案。本文剖析 Pinia 的设计哲学、API 设计与内部实现原理。

## 设计哲学

Pinia 的设计遵循三个核心原则：

```
1. 简单性 (Simplicity)
   ├── 没有 mutations（actions 统一处理同步+异步）
   ├── 没有 modules 嵌套（扁平 Store 结构）
   └── 没有 namespaced 的复杂概念

2. 类型安全 (TypeScript First)
   ├── 完整的 TypeScript 推断
   ├── defineStore 泛型自动推导
   └── IDE 自动补全体验极佳

3. 开发者体验 (DX)
   ├── DevTools 完美集成
   ├── storeToRefs 解决解构丢失响应性问题
   └── SSR / SPA 统一 API
```

## Store 定义方式对比

### Option Store（类似 Vuex）

```typescript
// stores/counter-option.ts
import { defineStore } from 'pinia'

export const useCounterOptionStore = defineStore('counterOption', {
  // State — 类似 Vue Options API 的 data()
  state: () => ({
    count: 0,
    incrementAmount: 1,
  }),

  // Getters — 类似 computed，支持跨 getter 引用
  getters: {
    doubleCount: (state) => state.count * 2,
    
    // 使用其他 getter
    tripleCount(): number {
      return this.doubleCount * 3  // this 指向 store 实例
    },
  },

  // Actions — 同时承担 mutation + action 的角色
  actions: {
    increment() {
      this.count += this.incrementAmount  // 可直接通过 this 访问 state/getters
    },

    incrementBy(amount: number) {
      this.count += amount
    },

    async fetchInitialValue() {
      // 异步操作也写在 actions 中
      const res = await api.getInitialCount()
      this.count = res.data
    },

    $reset() {
      // Pinia 内置的重置方法
      this.$reset()
    },
  },
})
```

### Setup Store（推荐）

```typescript
// stores/counter-setup.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useCounterSetupStore = defineStore('counterSetup', () => {
  // State — 直接声明 ref/reactive
  const count = ref(0)
  const incrementAmount = ref(1)

  // Getters — 直接使用 computed
  const doubleCount = computed(() => count.value * 2)
  const tripleCount = computed(() => doubleCount.value * 3)

  // Actions — 普通函数
  function increment() {
    count.value += incrementAmount.value
  }

  function incrementBy(amount: number) {
    count.value += amount
  }

  async function fetchInitialValue() {
    const res = await api.getInitialCount()
    count.value = res.data
  }

  // 返回所有需要在模板和外部使用的属性和方法
  return {
    count,
    incrementAmount,
    doubleCount,
    tripleCount,
    increment,
    incrementBy,
    fetchInitialValue,
  }
})
```

### 两种方式的适用场景

| 维度 | Option Store | Setup Store |
|------|-------------|------------|
| 学习曲线 | 低（Vuex 用户熟悉） | 中（需要理解组合式思维） |
| TypeScript | 较好 | **极佳**（完整类型推断） |
| 逻辑组织 | 按 type 分组 | **按功能分组**（关注点内聚） |
| 复用性 | 较弱 | **强**（可直接抽取为独立函数） |
| 推荐场景 | 简单 Store | 复杂 Store / 需要 Composable 复用 |

## 核心架构

### Pinia 内部结构

```
┌─────────────────────────────────────────────────────┐
│                      Pinia Instance                   │
│                                                       │
│  ┌─────────────┐  ┌─────────────┐  ┌──────────────┐  │
│  │  State      │  │  Getters    │  │   Actions     │  │
│  │  (reactive) │  │ (computed)  │  │ (functions)   │  │
│  └──────┬──────┘  └──────┬──────┘  └──────┬───────┘  │
│         │                │                │            │
│         └────────┬───────┴────────────────┘            │
│                  ▼                                     │
│         ┌──────────────────┐                           │
│         │   Store Instance  │  ← reactive(proxy({...}))│
│         │   (统一暴露)      │                           │
│         └────────┬─────────┘                           │
│                  │                                     │
│  ┌───────────────┼───────────────┬──────────────────┐ │
│  │               │               │                  │ │
│  ▼               ▼               ▼                  ▼ │
│ Template    Components    Other Stores    Plugins     │
│ (直接使用)   (storeToRefs)  (交叉引用)     (DevTools) │
└────────────────────────────────────────────────────---┘
```

### defineStore 的实现原理（简化）

```typescript
// Pinia 核心的简化实现思路

let pinia: Pinia | null = null

function defineStore<T>(
  idOrOptions: string | DefineStoreOptions<T>,
  setup?: DefineSetupStore<T>
): StoreDefinition<T> {
  // 统一处理两种形式
  const options: DefineStoreOptions<T> = isFunction(idOrOptions)
    ? { id: idOrOptions as string, setup: setup! }
    : (idOrOptions as DefineStoreOptions<T>)

  // 返回工厂函数（useXxxStore 形式）
  function useStore(): Store<T> {
    // 获取当前 Pinia 实例
    const currentPinia = pinia ?? activePinia
    
    // 单例模式：同一 id 的 store 只创建一次
    if (currentPinia._s.has(options.id)) {
      return currentPinia._s.get(options.id) as Store<T>
    }

    // 创建 Store 实例
    const store = createOptionsStore(options.id, options) as Store<T>
    // 或者 createSetupStore(options.id, options)

    // 注册到 Pinia
    currentPinia._s.set(options.id, store)

    // DevTools 注册
    if (__DEV__) {
      currentPinia._a.forEach(plugin => plugin(store))
    }

    return store
  }

  return useStore as StoreDefinition<T>
}
```

### Option Store 创建流程

```typescript
function createOptionsStore<
  Id extends string,
  S extends StateTree,
  G extends _GettersTree<S>,
  A extends _ActionsTree
>(id: Id, options: DefineStoreOptions<Id, S, G, A>): Store<Id, S, G, A> {
  // 1. 提取 state 函数
  const { state, getters, actions } = options

  // 2. 创建初始状态
  const initialState = state ? state() : {}

  // 3. 用 reactive 包装（这就是为什么可以直接 this.xxx 访问）
  const reactiveState = reactive(initialState)

  // 4. 处理 getters → 转为 computed
  const computedGetters: Record<string, ComputedRef> = {}
  for (const [key, getterFn] of Object.entries(getters ?? {})) {
    computedGetters[key] = computed(
      // getter 的 this 绑定到 reactiveState
      getterFn.bind(reactiveState)
    )
  }

  // 5. 处理 actions → 绑定到 store 上下文
  const boundActions: Record<string, Function> = {}
  for (const [key, actionFn] of Object.entries(actions ?? {})) {
    boundActions[key] = actionFn.bind(
      new Proxy(reactiveState, {
        get(target, prop) => {
          // this.state → 直接返回 state
          if (prop === '$state') return target
          // this.someGetter → 返回 computed
          if (prop in computedGetters) return computedGetters[prop]
          // this.someAction → 返回 action
          if (prop in boundActions) return boundActions[prop]
          return target[prop]
        }
      })
    )
  }

  // 6. 合并为统一的 store 对象
  const store = reactive({
    $id: id,
    $state: reactiveState,
    ...reactiveState,        // 展开 state（可直接 store.count 访问）
    ...computedGetters,       // 展开 getters（可直接 store.doubleCount 访问）
    ...boundActions,          // 展开 actions（可直接 store.increment() 调用）
    $reset() {              // 重置功能
      Object.assign(reactiveState, state())
    },
    $patch(partialState: Partial<S>) {
      Object.assign(reactiveState, partialState)
    },
    $dispose() {
      // 清理
    },
  }) as Store<Id, S, G, A>

  return store
}
```

## 跨 Store 通信

### 直接引用其他 Store

```typescript
// stores/user.ts
export const useUserStore = defineStore('user', () => {
  const name = ref('')
  const token = ref('')
  
  function login(credentials) { /* ... */ }
  function logout() { name.value = ''; token.value = '' }
  
  return { name, token, login, logout }
})

// stores/notification.ts
export const useNotificationStore = defineStore('notification', () => {
  const messages = ref([])
  
  function addNotification(text: string, type: 'success' | 'error') {
    // 可以直接在另一个 store 中调用
    const userStore = useUserStore()
    messages.value.push({
      text,
      type,
      user: userStore.name,  // 跨 store 数据访问
      time: Date.now()
    })
  }
  
  // 登出时清理通知
  function clearAll() { messages.value = [] }
  
  return { messages, addNotification, clearAll }
})
```

### provide / inject 模式（避免循环依赖）

```typescript
// stores/theme.ts
export const useThemeStore = defineStore('theme', () => {
  const mode = ref<'light' | 'dark'>('light')
  
  function toggle() {
    mode.value = mode.value === 'light' ? 'dark' : 'light'
  }
  
  return { mode, toggle }
})

// App.vue
<script setup>
import { useThemeStore } from './stores/theme'
import { provide } from 'vue'

provide('themeStore', useThemeStore())
</script>

// 任意深层组件
// stores/some-feature.ts
export const useFeatureStore = defineStore('feature', () => {
  // 使用 inject 获取，而非再次 useThemeStore()
  // 避免 store 间的硬耦合
  const theme = inject<ReturnType<typeof useThemeStore>>('themeStore')!
  
  const featureColor = computed(() => theme.mode.value === 'dark' ? '#fff' : '#333')
  
  return { featureColor }
})
```

## 插件系统

### 自定义 Pinia 插件

```typescript
// plugins/pinia-persist.ts
import type { PiniaPluginContext } from 'pinia'

/**
 * 持久化插件 — 自动将 state 保存到 localStorage
 */
export function piniaPersistPlugin(context: PiniaPluginContext) {
  const { store, options } = context
  
  // 只对标记了 persist 的 store 生效
  if (!(options as any).persist) return

  // 从 localStorage 恢复
  const savedState = localStorage.getItem(`pinia-${store.$id}`)
  if (savedState) {
    try {
      store.$patch(JSON.parse(savedState))
    } catch (e) {
      console.warn(`Failed to restore store "${store.$id}"`, e)
    }
  }

  // 监听变化，保存到 localStorage
  store.$subscribe(
    (mutation, state) => {
      localStorage.setItem(
        `pinia-${store.$id}`,
        JSON.stringify(state)
      )
    },
    { detached: true }  // detached = 即使 store 卸载也保持订阅
  )
}
```

```typescript
// main.ts
import { createPinia } from 'pinia'
import { piniaPersistPlugin } from './plugins/pinia-persist'

const pinia = createPinia()
pinia.use(piniaPersistPlugin)  // 全局注册插件

app.use(pinia)
```

## 响应式原理

Pinia 的响应式完全基于 `@vue/reactivity` 包：

```typescript
// Pinia 内部如何建立响应式连接
import { reactive, computed, toRefs, effectScope } from '@vue/reactivity'

function createSetupStore(id, setup) {
  const scope = effectScope(true)  // 创建独立作用域
  
  // setup() 在 scope 中运行，所有的 reactive/ref/computed 都被追踪
  const setupResult = scope.run(setup)! 
  
  // setupResult 本身就是一个普通对象
  // 但其中的值都是 ref / reactive / computed
  
  // 最终用 reactive 包装整个 store
  const store = reactive({
    $id: id,
    $state: {},  // 原始状态副本
    ...setupResult
  })
  
  // $state 用于 $reset 功能和序列化
  // 它是 setup 返回值的非响应式快照
  for (const [key, value] of Object.entries(setupResult)) {
    if (isRef(value)) {
      store.$state[key] = value.value
    } else if (isReactive(value)) {
      store.$state[key] = { ...value }
    }
  }
  
  return store
}
```

**关键点**：Pinia Store 就是一个 `reactive()` 对象，所以 Vue 的响应式系统天然适用——无需额外的订阅/发布机制。
