虽然 Vue 3 时代 Pinia 已成为官方推荐的状态管理方案，但大量存量 Vue 2 项目仍在使用 Vuex。本文系统梳理 Vuex 的核心概念，为维护遗留项目提供完整参考。

## 核心概念全景图

```
┌─────────────────────────────────────────────────────────┐
│                        Vuex Store                         │
│                                                           │
│  ┌─────────┐    ┌──────────┐    ┌──────────┐             │
│  │  State  │ ←── │ Mutations│ ←── │ Actions  │             │
│  │ (数据)  │    │ (同步变更)│    │ (异步逻辑)│             │
│  └────┬────┘    └─────┬────┘    └─────┬────┘             │
│       │               │               │                   │
│       ▼               ▼               ▼                   │
│  ┌─────────────────────────────────────┐                 │
│  │           Getters (计算属性)          │                 │
│  └─────────────────────────────────────┘                 │
│                                                           │
│  ┌─────────────────────────────────────┐                 │
│  │           Modules (模块拆分)         │                 │
│  └─────────────────────────────────────┘                 │
└─────────────────────────────────────────────────────────┘
```

## State — 单一数据源

### 设计原则

Vuex 采用 **单一状态树（Single Source of Truth）** 模式：

```
应用的全部状态集中在一个对象中：

State Tree:
{
  user: {
    profile: { name: 'Alice', avatar: '...' },
    preferences: { theme: 'dark', language: 'zh-CN' },
    token: 'eyJhbGci...'
  },
  cart: {
    items: [{ id: 1, qty: 2 }],
    total: 199.00
  },
  ui: {
    sidebarOpen: false,
    modalVisible: false
  }
}

原则:
1. 任何组件都可以访问 state
2. 只有 Mutation 能修改 state
3. 修改是可预测、可追踪的
```

### 定义 State

```javascript
// store/index.js
import Vue from 'vue'
import Vuex from 'vuex'

export default new Vuex.Store({
  // State 是一个函数（返回初始状态的对象）
  state: {
    // ========== 用户相关 ==========
    currentUser: null,
    token: null,
    isAuthenticated: false,

    // ========== 购物车 ==========
    cartItems: [],
    checkoutStatus: null,  // null | 'loading' | 'success' | 'error'

    // ========== UI 状态 ==========
    globalLoading: false,
    sidebarCollapsed: false,
    notifications: []
  }
})
```

### 在组件中访问 State

```vue
<!-- 方式 1：通过 $store.state 直接访问 -->
<template>
  <div v-if="$store.state.isAuthenticated">
    Welcome, {{ $store.state.currentUser.name }}!
  </div>
</template>

<script>
export default {
  computed: {
    // 方式 2：mapState 辅助函数（推荐）
    ...mapState({
      // 对象形式：自定义名称
      userName: state => state.currentUser?.name,
      isLoggedIn: state => state.isAuthenticated,
      
      // 数组形式：同名映射
      cartItems: state => state.cartItems,
      globalLoading: 'globalLoading'  // 字符串快捷方式
    })
  }
}
</script>
```

## Getters — 计算属性

Getters 相当于 Store 级别的 **computed**：

```javascript
getters: {
  // 基础 getter
  isLoggedIn: state => !!state.currentUser,
  
  // 依赖其他 getter
  displayName: (state, getters) => {
    if (!state.currentUser) return 'Guest'
    return `${state.currentUser.firstName} ${state.currentUser.lastName}`
  },
  
  // 返回函数（允许传参）
  getUserById: (state) => (id) => {
    return state.users.find(u => u.id === id)
  },
  
  // 购物车计算
  cartItemCount: state => state.cartItems.reduce((sum, item) => sum + item.quantity, 0),
  cartTotalPrice: (state, getters) => 
    state.cartItems.reduce((sum, item) => sum + item.price * item.quantity, 0),
  
  // 过滤型 getter
  completedTodos: state => state.todos.filter(t => t.completed),
  activeTodoCount: (state, getters) => 
    state.todos.length - getters.completedTodos.length
}
```

```vue
<!-- 在组件中使用 getters -->
<script>
import { mapGetters } from 'vuex'

export default {
  computed: {
    ...mapGetters([
      'isLoggedIn',          // 直接映射
      'displayName',
      'cartTotalPrice'
    ]),
    
    // 别名映射
    ...mapGetters({
      completed: 'completedTodos'
    })
  }
}
</script>

<template>
  <p>Total: {{ cartTotalPrice }} ({{ cartItemCount }} items)</p>
  <p v-if="isLoggedIn">Welcome, {{ displayName }}</p>
</template>
```

## Mutations — 同步变更

### 为什么必须有 Mutations？

```
Vuex 的核心理念: 所有的状态变更都必须经过 Mutation

原因:
1. 可追溯性 — DevTools 记录每一次 Mutation
2. 可预测性 — Mutation 必须同步，便于调试
3. 时间旅行调试 — 可以回放/撤销任意一次 Mutation
4. 严格的变更管道 — State → Mutation → View
```

### 定义 Mutations

```javascript
mutations: {
  // 基本形式: (state, payload) => {}
  SET_CURRENT_USER(state, user) {
    state.currentUser = user
    state.isAuthenticated = !!user
  },
  
  CLEAR_USER(state) {
    state.currentUser = null
    state.token = null
    state.isAuthenticated = false
  },
  
  SET_TOKEN(state, token) {
    state.token = token
  },
  
  // 购物车操作
  ADD_TO_CART(state, product) {
    const existing = state.cartItems.find(item => item.id === product.id)
    if (existing) {
      existing.quantity++
    } else {
      state.cartItems.push({ ...product, quantity: 1 })
    }
  },
  
  REMOVE_FROM_CART(state, productId) {
    const index = state.cartItems.findIndex(item => item.id === productId)
    if (index > -1) state.cartItems.splice(index, 1)
  },
  
  CLEAR_CART(state) {
    state.cartItems = []
    state.checkoutStatus = null
  },
  
  // UI 状态
  TOGGLE_SIDEBAR(state) {
    state.sidebarCollapsed = !state.sidebarCollapsed
  },
  
  SET_GLOBAL_LOADING(state, isLoading) {
    state.globalLoading = isLoading
  },
  
  PUSH_NOTIFICATION(state, notification) {
    state.notifications.push({
      ...notification,
      id: Date.now(),
      timestamp: new Date().toISOString()
    })
  },
  
  REMOVE_NOTIFICATION(state, notificationId) {
    state.notifications = state.notifications.filter(n => n.id !== notificationId)
  }
}
```

### 触发 Mutations

```javascript
// 方式 1: this.$store.commit
this.$store.commit('SET_CURRENT_USER', userData)
this.$store.commit('CLEAR_CART')

// 方式 2: mapMutations（组件 methods 中使用）
import { mapMutations } from 'vuex'
export default {
  methods: {
    ...mapMutations([
      'TOGGLE_SIDEBAR',
      'CLEAR_CART'
    ]),
    
    // 映射为不同名称
    ...mapMutations({
      setUser: 'SET_CURRENT_USER',
      setLoading: 'SET_GLOBAL_LOADING'
    }),
    
    // 自定义方法中调用
    handleLogin(userData) {
      this.setUser(userData)
      this.setLoading(false)
    }
  }
}
```

### Mutation 必须是同步的

```javascript
// ❌ 错误：Mutation 中执行异步操作
mutations: {
  async FETCH_DATA(state) {
    const res = await api.getData()  // DevTools 无法追踪！
    state.data = res.data
  }
}

// ✅ 正确：异步操作放在 Action 中
actions: {
  async fetchData({ commit }) {
    const res = await api.getData()
    commit('SET_DATA', res.data)  // Mutation 只做同步赋值
  }
}
```

## Actions — 异步操作

### Action vs Mutation 的分工

```
Mutation (同步):
  - 修改 State
  - 必须是纯同步函数
  - 第一个参数是 state

Action (异步):
  - 提交 Mutation
  - 可以包含任意异步操作
  - 第一个参数是 context ({ commit, dispatch, state, getters, rootState, rootGetters })
```

### 定义 Actions

```javascript
actions: {
  // ========== 用户认证 ==========
  async login({ commit, dispatch }, credentials) {
    commit('SET_GLOBAL_LOADING', true)
    try {
      const res = await authApi.login(credentials)
      commit('SET_TOKEN', res.token)
      commit('SET_CURRENT_USER', res.user)
      
      // 可以 dispatch 其他 action
      await dispatch('fetchUserProfile')
      
      // 成功通知
      commit('PUSH_NOTIFICATION', {
        type: 'success',
        message: '登录成功'
      })
      
      return res.user
    } catch (error) {
      commit('PUSH_NOTIFICATION', {
        type: 'error',
        message: error.message || '登录失败'
      })
      throw error
    } finally {
      commit('SET_GLOBAL_LOADING', false)
    }
  },
  
  logout({ commit }) {
    commit('CLEAR_USER')
    commit('CLEAR_CART')
    // 清除本地存储
    localStorage.removeItem('token')
  },
  
  async fetchUserProfile({ commit, state }) {
    if (!state.token) return
    
    try {
      const res = await userApi.getProfile(state.token)
      commit('SET_CURRENT_USER', { ...state.currentUser, ...res.data })
    } catch (error) {
      commit('PUSH_NOTIFICATION', { type: 'warning', message: '获取用户信息失败' })
    }
  },
  
  // ========== 购物车 ==========
  addToCart({ commit, state }, product) {
    commit('ADD_TO_CART', product)
    commit('PUSH_NOTIFICATION', {
      type: 'info',
      message: `"${product.name}" 已加入购物车`
    })
  },
  
  async checkout({ commit, state, getters }) {
    if (state.cartItems.length === 0) {
      commit('PUSH_NOTIFICATION', { type: 'warning', message: '购物车为空' })
      return
    }
    
    commit('SET_CHECKOUT_STATUS', 'loading')
    
    try {
      const order = await orderApi.create({
        items: state.cartItems,
        total: getters.cartTotalPrice
      })
      
      commit('CLEAR_CART')
      commit('SET_CHECKOUT_STATUS', 'success')
      commit('PUSH_NOTIFICATION', { type: 'success', message: '订单创建成功!' })
      
      return order
    } catch (error) {
      commit('SET_CHECKOUT_STATUS', 'error')
      commit('PUSH_NOTIFICATION', { type: 'error', message: error.message })
      throw error
    }
  }
}
```

### 触发 Actions

```javascript
// 方式 1: this.$store.dispatch
await this.$store.dispatch('login', { username: 'alice', password: '***' })
this.$store.dispatch('addToCart', { id: 1, name: 'Widget', price: 9.99 })

// 方式 2: mapActions
import { mapActions } from 'vuex'
export default {
  methods: {
    ...mapActions(['login', 'logout', 'addToCart', 'checkout']),
    
    // 包装为业务方法
    async handleLogin() {
      try {
        await this.login(this.form)
        this.$router.push('/dashboard')
      } catch (e) {
        // 错误已在 action 中处理
      }
    }
  }
}
```

## Modules — 模块拆分

### 基本结构

```
store/
├── index.js              # 根 store（组装所有模块）
├── modules/
│   ├── user.js           # 用户模块
│   ├── cart.js           # 购物车模块
│   ├── ui.js             # UI 状态模块
│   └── notifications.js   # 通知模块
```

```javascript
// store/modules/user.js
export default {
  namespaced: true,  // 关键：启用命名空间隔离
  
  state: () => ({
    currentUser: null,
    token: null,
    isAuthenticated: false
  }),
  
  getters: { /* ... */ },
  mutations: { /* ... */ },
  actions: { /* ... */ }
}

// store/modules/cart.js
export default {
  namespaced: true,
  
  state: () => ({
    items: [],
    status: null
  }),
  
  getters: {
    itemCount: state => state.items.reduce((sum, item) => sum + item.qty, 0),
    totalPrice: (state) => state.items.reduce((sum, item) => sum + item.price * item.qty, 0)
  },
  
  mutations: {
    ADD_ITEM(state, item) { state.items.push(item) },
    REMOVE_ITEM(state, id) { /* ... */ },
    CLEAR(state) { state.items = []; state.status = null }
  },
  
  actions: {
    async checkout({ commit }) { /* ... */ }
  }
}

// store/index.js
import Vue from 'vue'
import Vuex from 'vuex'
import user from './modules/user'
import cart from './modules/cart'
import ui from './modules/ui'
import notifications from './modules/notifications'

export default new Vuex.Store({
  modules: { user, cart, ui, notifications },
  
  strict: process.env.NODE_ENV !== 'production'  // 开启严格模式
})
```

### Namespaced 模块的访问方式

```javascript
// 访问 namespaced 模块

// State:
this.$store.state.user.currentUser
this.$store.state.cart.items

// Getters:
this.$store.getters['user/isLoggedIn']
this.$store.getters['cart/itemCount']

// Mutations:
this.$store.commit('user/SET_CURRENT_USER', data)
this.$store.commit('cart/ADD_ITEM', item)

// Actions:
this.$store.dispatch('user/login', credentials)
this.$store.dispatch('cart/checkout')

// 辅助函数:
...mapState('user', { userName: state => state.currentUser?.name })
...mapGetters('cart', ['itemCount', 'totalPrice'])
...mapMutations('cart', ['ADD_ITEM', 'REMOVE_ITEM'])
...mapActions('user', ['login', 'logout'])
```

## 严格模式

```javascript
export default new Vuex.Store({
  // 开启严格模式
  strict: process.env.NODE_ENV === 'development',
  
  // 效果：
  // 任何在 Mutation 之外修改 state 的操作都会抛出警告
  // 例如：
  //   - 直接修改 this.$store.state.xxx
  //   - 在 Action 中直接修改 state（应该通过 commit）
  //   - 异步回调中间接修改 state
  
  // 性能注意：
  // 严格模式通过深监听 state 来检测非法修改
  // 对于大型 state 对象会有性能开销
  // 所以只在 development 环境开启
})
```
