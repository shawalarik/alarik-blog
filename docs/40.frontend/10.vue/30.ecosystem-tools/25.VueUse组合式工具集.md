VueUse 是目前最流行的 Vue Composition API 工具库，提供了 200+ 个即拿即用的组合式函数。本文深入其核心工具的设计原理和使用技巧。

## 为什么需要 VueUse？

```
开发中反复遇到的需求:

✅ 监听窗口尺寸变化
✅ 获取鼠标位置
✅ LocalStorage 的响应式读写
✅ 防抖/节流
✅ 剪贴板操作
✅ 全屏控制
✅ 网络状态检测
✅ 元素可见性观察
✅ ...

每次都自己写？太累了！

VueUse 把这些常见需求封装成了经过充分测试的 composable。
```

## 核心工具实战

### 浏览器 API 封装

```vue
<script setup lang="ts">
import {
  useWindowSize,     // 响应式窗口尺寸
  useMouse,          // 响应式鼠标坐标
  useScroll,         // 滚动位置和方向
  useElementSize,   // 元素尺寸监听
  useElementVisibility // 元素可见性 (IntersectionObserver)
} from '@vueuse/core'

// ========== 窗口尺寸 ==========
const { width, height } = useWindowSize({ initialWidth: 1920, initialHeight: 1080 })

// 响应式断点判断
const isMobile = computed(() => width.value < 768)
const isTablet = computed(() => width.value >= 768 && width.value < 1024)

// ========== 鼠标追踪 ==========
const { x, y, sourceType } = useMouse({ touch: true })
// sourceType: 'mouse' | 'touch' | null

// ========== 滚动 ==========
const {
  x: scrollX,
  y: scrollY,
  isScrolling,        // 正在滚动中?
  directions          // { top, bottom, left, right } 滚动方向
} = useWindowScroll()

// 到达底部自动加载
watch(directions.bottom, (isBottom) => {
  if (isBottom) loadMoreData()
})

// ========== 元素尺寸 ==========
const cardRef = ref<HTMLDivElement>()
const { width: cardW, height: cardH } = useElementSize(cardRef)

// ========== 可见性 ==========
const targetRef = ref<HTMLElement>()
const { stop } = useElementVisibility(
  targetRef,
  { threshold: 0.5 }  // 50% 可见时触发
)
// 返回 { isVisible, pause, resume, stop }
</script>

<template>
  <div :class="{ mobile: isMobile, tablet: isTablet }">
    Mouse: ({{ Math.round(x) }}, {{ Math.round(y) }})
  </div>
  
  <div ref="cardRef" :style="{ width: cardW + 'px' }">
    Card size: {{ cardW }} x {{ cardH }}
  </div>
  
  <div v-show="isVisible" ref="targetRef">
    This element becomes visible!
  </div>
</template>
```

### 存储与状态持久化

```vue
<script setup lang="ts">
import {
  useStorage,         // 响应式 LocalStorage/SessionStorage
  useSessionStorage,
  useCookie,
} from '@vueuse/core'

// ========== useStorage ==========
// 自动同步到 localStorage，类型安全的响应式引用
const count = useStorage('vueuse-count', 0, localStorage, {
  // 序列化选项
  serializer: {
    read: JSON.parse,
    write: JSON.stringify
  },
  // 写入前的过滤
  writeDefaults: true,
  // 合并策略
  mergeDefaults: true
})

// 支持嵌套对象的路径写入
const settings = useStorage('settings', {
  theme: 'light',
  language: 'zh-CN',
  sidebarOpen: true
})

// 修改任何属性都会自动持久化
settings.value.theme = 'dark'  // → 自动保存到 localStorage['settings']

// ========== useSessionStorage ==========
const token = useSessionStorage('auth-token', '')

// ========== useCookie ==========
const consent = useCookie('cookie-consent', 'yes', {
  // Cookie 选项
  maxAge: 60 * 60 * 24 * 365,  // 1 年
  path: '/',
  sameSite: 'Strict'
})

// 同步读取 cookie
if (!consent.value) showConsentModal()
</script>
```

### 异步工具

```vue
<script setup lang="ts">
import {
  useAsyncQueue,       // 并发控制队列
  useFetch,            // 响应式数据请求
  until,               // 条件等待
  useTimeoutFn,        // 延迟执行
  promiseTimeout,      // Promise 超时
} from '@vueuse/core'

// ========== useFetch ==========
// 比 useAsyncData 更灵活的数据请求方案
const {
  data: users,
  error,
  isPending: loading,
  execute: refresh,
  canAbort,
  abort
} = useFetch('/api/users', {
  // 初始数据
  initialData: [],
  // 请求前拦截
  beforeFetch(ctx) => {
    ctx.options.headers = { Authorization: `Bearer ${token.value}` }
  },
  // 响应后处理
  afterFetch(ctx) => {
    if (ctx.response.status === 401) {
      redirectToLogin()
    }
    return ctx
  },
  // fetch 选项
  fetchOptions: { timeout: 10000 }
})

// 手动触发
async function handleRefresh() {
  await refresh()
}

// ========== until ==========
// 等待某个条件成立
const el = ref<HTMLElement | null>(null)

// 等待元素出现后再执行
until(el).then(() => {
  console.log('Element appeared:', el.value)
})

// 组合超时
await promiseTimeout(
  until(ref(target)),
  5000  // 最多等待 5 秒
)
if (target.value) {
  // 元素在 5 秒内出现了
}
```

### 动画与过渡

```vue
<script setup lang="ts">
import {
  useTransition,      // 过渡状态管理
  useAnimate,         // Web Animations API
  useIntervalFn,      // 定时器
  useRafFn,           // requestAnimationFrame
} from '@vueuse/core'
import { ref } from 'vue'

const show = ref(false)

// ========== useTransition ==========
// 管理进入/离开/取消状态的过渡
const {
  isLeaving,           // 正在离开
  isEnterTransitioning, // 正在进入
} = useTransition(show, {
  duration: 300,
  enterFrom: { opacity: 0, transform: 'translateY(20px)' },
  leaveTo: { opacity: 0, transform: 'translateY(-20px)' },
  onEnter: () => console.log('Entered!'),
  onLeave: () => console.log('Left!')
})

// toggle
function toggle() { show.value = !show.value }

// ========== useAnimate ==========
// 使用 Web Animations API 创建动画
const el = ref<HTMLElement>()

const { isAnimating, animate, set } = useAnimate(el, {
  initial: { scale: 1, rotate: '0deg' },
  entered: { scale: 1.05, rotate: '5deg' },
  left: { scale: 0.95, rotate: '-5deg' },
})

// hover 效果
function onMouseEnter() { animate('entered') }
function onMouseLeave() { animate('left') }

// ========== useIntervalFn ==========
// 清理友好的定时器（组件卸载时自动停止）
const { pause, resume } = useIntervalFn(() => {
  console.log('Tick!', new Date().toLocaleTimeString())
}, 1000)

// ========== useRafFn ==========
// 基于 rAF 的循环（适合动画帧级精度）
const { pause: pauseAnim } = useRafFn((deltaTime) => {
  // deltaTime: 距上一帧的毫秒数
  position.x += velocity.x * deltaTime
  position.y += velocity.y * deltaTime
})
</script>
```

## 设计原理剖析

### 响应式自动清理机制

```typescript
// useMouse 内部实现（简化）
export function useMouse(options = {}) {
  const x = ref(0)
  const y = ref(0)
  
  function update(event: MouseEvent) {
    x.value = event.clientX
    y.value = event.clientY
  }
  
  // 注册事件
  if (typeof window !== 'undefined') {
    window.addEventListener('mousemove', update)
    
    // ★ 关键：在组件卸载时自动清理
    tryOnScopeDispose(() => {
      window.removeEventListener('mousemove', update)
    })
  }
  
  return { x, y }
}
```

`tryOnScopeDispose` 是 Vue 3.3+ 的实验性 API，确保 effect scope 被销毁时清理副作用。这就是为什么 **VueUse 的所有 composable 都不需要手动 cleanup**——它们内部自动处理了。

### 如何编写自己的 Composable

```typescript
// composables/useDebounceFn.ts
import { type Ref } from 'vue'

/**
 * 防抖函数 Composable
 * 
 * @example
 * ```ts
 * const { pause, resume } = useDebounceFn(
 *   () => saveContent(),
 *   300  // 防抖延迟 ms
 * )
 * ```
 */
export function DebounceFilter<T extends (...args: any[]) => any>(
  fn: T,
  delay: number = 200
): [(...args: Parameters<T>) => void, () => void, () => void] {
  let timer: ReturnType<typeof setTimeout> | undefined
  
  function resetTimer() {
    if (timer) clearTimeout(timer)
  }
  
  function debounceFn(this: unknown, ...args: Parameters<T>) {
    resetTimer()
    timer = setTimeout(() => fn.apply(this, args), delay)
  }
  
  // 暂停防抖（立即执行下一次调用）
  function flush() {
    resetTimer()
    if (timer) {
      fn()
      timer = undefined
    }
  }
  
  // 取消待执行的调用
  function cancel() {
    resetTimer()
  }
  
  return [debounceFn, cancel, flush]
}

export { DebounceFilter as useDebounceFn }
```

```typescript
// composables/useNetworkStatus.ts
/**
 * 网络状态检测 Composable
 */
export function useNetworkStatus() {
  const online = navigator.onLine
  const offlineAt = ref<Date | null>(null)
  
  function updateOnlineStatus() {
    online.value = navigator.onLine
    if (!online.value) {
      offlineAt.value = new Date()
    }
  }
  
  if (typeof window !== 'undefined') {
    window.addEventListener('online', updateOnlineStatus)
    window.addEventListener('offline', updateOnlineStatus)
    
    tryOnScopeDispose(() => {
      window.removeEventListener('online', updateOnlineStatus)
      window.removeEventListener('offline', updateOnlineStatus)
    })
  }
  
  const isBackOnline = computed(() => {
    if (online.value && offlineAt.value) {
      return Date.now() - offlineAt.value.getTime() < 5000
    }
    return false
  })
  
  return { online: readonly(online), isBackOnline, offlineAt }
}
```

### Composable 设计规范

```
最佳实践:

1. 始终接受参数而非硬编码值
   ✓ useStorage(key, defaultValue)
   ✗ useLocalStorage() // 固定 key

2. 返回 ref 而非 raw value
   ✓ return { value: ref(initial) }
   ✗ return initialValue

3. 自动清理副作用
   ✓ tryOnScopeDispose(() => { cleanup() })
   ✗ 注册事件但不清理

4. 支持 SSR 安全
   ✓ if (typeof window !== 'undefined') { ... }
   ✗ 直接访问 window/document

5. 提供 pause/resume 控制（适用于异步/定时类 composable）
   ✓ const { pause, resume } = useIntervalFn(fn, 1000)

6. 使用 TypeScript 泛型
   ✓ export function useFoo<T>(initialValue: T): Ref<T>
   ✗ export function useFoo()  // 类型推断受限
```
