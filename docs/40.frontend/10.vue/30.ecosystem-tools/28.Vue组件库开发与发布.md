## 组件开发规范

### Props 类型定义

```typescript
// src/components/Button/Button.props.ts
import type { ExtractPropTypes } from 'vue-standardized-properties-and-emits'

export interface ButtonProps extends /* @vue-ignore */ ExtractPropTypes<typeof _Button> {
  /** 按钮类型 */
  type?: 'primary' | 'secondary' | 'ghost' | 'link'
  
  /** 按钮尺寸 */
  size?: 'small' | 'medium' | 'large'
  
  /** 禁用状态 */
  disabled?: boolean
  
  /** 加载状态 */
  loading?: boolean
  
  /** 块级元素 */
  tag?: 'button' | 'a' | 'router-link'
  
  /** 原生 HTML 属性透传 */
  htmlType?: string
  
  /** 自定义主题色 */
  color?: string
}
```

### 组件模板（SFC）

```vue
<!-- src/components/Button/Button.vue -->
<script setup lang="ts">
import { computed } from 'vue'
import { buttonProps } from './Button.props'

const props = withDefaults(defineProps<buttonProps>(), {
  type: 'primary',
  size: 'medium',
  disabled: false,
  loading: false,
  tag: 'button',
  htmlType: 'button'
})

// 类名组合
const classes = computed(() => [
  'btn',
  `btn--${props.type}`,
  `btn--${props.size}`,
  {
    'btn--disabled': props.disabled,
    'btn--loading': props.loading,
  }
])

// 点击事件（支持 native type="submit"）
function handleClick(e: MouseEvent) {
  if (props.disabled || props.loading) {
    e.preventDefault()
    return
  }
}
</script>

<template>
  <component
    :is="tag"
    :class="classes"
    :disabled="disabled || loading"
    :type="tag === 'button' ? htmlType : undefined"
    :aria-disabled="disabled"
    :aria-busy="loading"
    @click="handleClick"
  >
    <span v-if="loading" class="btn__loader" aria-hidden="true">
      <svg class="spinner" viewBox="0 0 24 24">...</svg>
    </span>
    <slot />
  </component>
</template>
```

## 构建配置

### Vite Library Mode 配置

```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import dts from 'vite-plugin-dts'
import { resolve } from 'path'

export default defineConfig({
  plugins: [
    vue(),
    dts(),  // 生成 .d.ts 声明文件
  ],
  
  build: {
    lib: {
      // ESM 入口
      entry: resolve(__dirname, 'src/index.ts'),
      name: 'MyComponentLib',
      formats: ['es', 'cjs']  // 同时输出 ES Module 和 CommonJS
    },
    rollupOptions: {
      external: ['vue', 'vue-router', 'pinia'],
      output: {
        preserveModules: true,  // 保持模块结构
        exports: 'named'  // 使用 named exports
      }
    },
    // CSS 提取
    cssCodeSplit: false,
    // 不压缩 CSS（使用者自行压缩）
    minify: 'esbuild',
    sourcemap: true,
    
    // 产物命名
    chunkFileNames: (chunkInfo) => {
      if (chunkInfo.name === 'style') return 'styles.css'
      return `[name].[hash].js`
    },
    entryFileNames: '[name].mjs'
  },
  
  resolve: {
    alias: {
      '@': resolve(__dirname, 'src')
    }
  },
  
  // 开发预览用的 playground 配置
  root: 'playground',
  server: {
    port: 3333,
    open: true
  }
})
```

## 样式隔离方案对比

### 方案选择

| 方案 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| **Scoped CSS** | 简单易用、Vue 原生支持 | 无法穿透 Shadow DOM | **大多数组件库首选** |
| **CSS Modules** | 类名哈希冲突概率极低 | 需要额外配置 | 需要极强样式隔离的场景 |
| **Shadow DOM** | 完全隔离、不受外部影响 | 表单/弹窗等交互复杂 | **Web Components 发布** |
| **CSS-in-JS** | 运行时样式、动态性强 | 包体增大、调试困难 | 需要高度动态化的组件 |

### 推荐：Scoped + BEM + CSS Variables

```scss
// styles/variables.scss
$lib-prefix: 'ml';  // MyLibrary 前缀

// 颜色系统（CSS Variables 支持 runtime 主题切换）
:root {
  --ml-color-primary: #42b983;
  --ml-color-secondary: #35495e;
  --ml-radius-md: 6px;
  --ml-spacing-sm: 8px;
  --ml-font-size-base: 14px;
}

[data-theme='dark'] {
  --ml-color-primary: #66cc99;
  --ml-color-secondary: #556b7e;
}

// styles/mixins.scss
@mixin respond-to($breakpoint) {
  @media screen and (min-width: $breakpoint) {
    @content;
  }
}
```

## 按需引入

### unplugin-vue-components 配置

```typescript
// vite.config.ts
import AutoImport from 'unplugin-auto-import/r-vite'
import Components from 'unplugin-vue-components/vite'
import { resolver as ElementPlusResolver } from 'unplugin-vue-components/resolvers'

export default defineConfig({
  plugins: [
    // 自动注册组件（无需手动 import）
    Components({
      resolvers: [ElementPlusResolver()],
      extensions: ['vue'],
      // 排除 playground 中的本地组件
      exclude: [resolve(__dirname, 'playground')]
    }),
    
    // 自动注册 API
    AutoImport({
      resolvers: [ElementPlusResolver()]
    })
  ]
})
```

## 发布到 npm

### 发布流程

```bash
# 1. 登录 npm（首次）
npm login

# 2. 检查 package.json
# 确保 name 唯一且未被占用
# 确保 version 符合 semver

# 3. 构建
pnpm build

# 4. 进入预发布模式（可选）
# npm publish --dry-run

# 5. 正式发布
npm publish --access public

# 6. 打标签
git tag v1.0.0
git push origin main --tags
```

### 版本管理（Semver 规范）

```
版本号格式: MAJOR.MINOR.PATCH

MAJOR (主版本): 不兼容的 API 变更
  1.0.0 → 2.0.0  (Button 的 props 结构变了)

MINOR (次版本): 向后兼容的新功能
  1.0.0 → 1.1.0  (新增 Icon 组件)

PATCH (修订版本): 向后兼容的问题修复
  1.0.0 → 1.0.1  (修复 Button disabled 状态的颜色 bug)

预发布版本:
  1.0.0-alpha.1  (alpha: 内部测试)
  1.0.0-beta.2   (beta: 公开测试)
  1.0.0-rc.3      (release candidate: 候选发布)
```

## CI/CD 自动化

```yaml
# .github/workflows/publish.yml
name: Publish to npm

on:
  push:
    tags: ['v*']  # 只有打 tag 时才触发

jobs:
  publish:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - uses: pnpm/action-setup@v3
        with:
          version: latest
      
      - run: pnpm install
        
      - run: pnpm build
        
      - uses: JS-DevTools/npm-publish@v3
        with:
          token: ${{ secrets.NPM_TOKEN }}
          access: public
```
