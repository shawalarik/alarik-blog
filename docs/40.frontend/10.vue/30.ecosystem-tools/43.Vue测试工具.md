建立完善的 Vue 应用测试体系，覆盖从单元测试到端到端（E2E）的全链路。本文基于 Vitest + Vue Test Utils 构建完整的测试工程。

## 测试技术栈选型

### 推荐组合

```
Vue 3 测试最佳实践:

┌─────────────────────────────────────┐
│              运行器 (Runner)               │
│                                         │
│   ┌───────────────────────────────┐     │
│   │        测试框架 (Framework)         │     │
│   │   Vitest (Vite 原生)                │     │
│   │   + @vue/test-utils (Vue 专用)       │     │
│   │   + @testing-library/vue (用户行为)    │     │
│   └───────────────────────────────┘     │
│                                         │
│   ├── Coverage (覆盖率)                   │
│   │   @vitest/coverage-istanbul          │
│   └── E2E                               │
│   │   Playwright / Cypress              │
│   └───────────────────────────────────┘
└─────────────────────────────────────┘

替代方案对比:
├─ Jest + vue-test-utils      ← Vue 2 时代常用，Vite 中需额外配置
├─ Mocha + chai             ← 灵活但配置繁琐
└─ Cypress                  ← E2E 强大，但学习成本高
```

## 项目配置

### 安装依赖

```bash
# 核心依赖
npm install -D vitest @vue/test-utils @testing-library/vue happy-dom @vitest/coverage-istanbul

# 可选：组件库测试工具
npm install -D @testing-library/jest-dom

# 类型定义（如果使用 TypeScript）
npm install -D vitest/jsdom jsdom @types/happy-dom
```

### Vitest 配置

```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import { resolve } from 'path'

export default defineConfig({
  test: {
    // 全局测试文件匹配模式
    include: [
      'src/**/*.test.{ts,tsx,js}',
      'tests/**/*.{test,spec}.{ts,tsx,js}'
    ],
    
    // 全局测试环境变量
    environment: {
      BASE_URL: 'http://localhost:3001',
      API_BASE_URL: 'http://localhost:8080',
    },
    
    // 覆盖率配置
    coverage: {
      provider: 'istanbul',           // istanbul 或 v8
      reports: ['text', 'html', 'lcov'], // 输出格式
      reporter: [ 'text', 'json-summary' ],
      
      // 包含的目录
      include: [
        'src/**/*.{vue,ts,tsx,js}',
        '!**/node_modules/**'
      ],
      exclude: [
        '**/*.d.ts',
        '**/*.test.ts',
        '**/__snapshots__/**'
      ],
      
      thresholds: {
        branches: 80,
        functions: 80,
        lines: 75,
        statements: 75
      },
      
      all: true,
      clean: true,
    },
    
    // 全局 setup 文件（每个测试文件前执行）
    setupFiles: ['./tests/setup.ts'],
    
    // CSS Modules 支持
    css: true,
  }
})
```

### Setup 文件（全局前置）

```typescript
// tests/setup.ts
import { beforeAll, afterAll, afterEach } from 'vitest'
import { config } from 'vitest/config'

// 全局 mock：API 调用
beforeAll(async () => {
  // Mock fetch (如果使用原生 fetch)
  vi.stubGlobal('fetch', (url, options) => {
    if (url.includes('/api/')) {
      return new Response(
        JSON.stringify({ 
          id: Math.floor(Math.random() * 100),
          name: 'Test User' 
        }),
        { status: 200, headers: { 'Content-Type': 'application/json' } }
      )
    }
  })

  // Mock localStorage
  const localStorage = {
    store: {}
  }
  vi.stubGlobal('localStorage', {
    get: (key) => localStorage.store[key] || null,
    set: (key, value) => { localStorage.store[key] = value; return undefined; },
    remove: (key) => { delete localStorage.store[key]; return undefined; },
    clear: () => { localStorage.store = {}; return undefined; },
    length: () => Object.keys(localStorage.store).length,
    key: (i) => Object.keys(localStorage.store)[i],
    getItem: (key) => localStorage.store[key] ?? null,
    setItem: (key, value) => { localStorage.store[key] = value; return undefined; },
    removeItem: (key) => { delete localStorage.store[key]; return undefined; },
    clear: () => { localStorage.store = {}; return undefined; }
  })
  vi.stubGlobalStorage(localStorage)

  // 清理 DOM
  document.body.innerHTML = ''
})

afterEach(() => {
  // 每个测试后清理 DOM 和副作用
  vi.restoreAllMocks()
  vi.clearAllTimers()
})
```

## 组件单元测试

### 基础组件测试

```typescript
// tests/components/Button/Button.test.tsx
import { describe, it, expect, beforeEach, afterEach } from 'vitest'
import { mount, flushPromises } from '@vue/test-utils'
import Button from '@/components/Button/Button.vue'
import { nextTick } from 'vue'

describe('Button Component', () => {
  
  // 渲染测试
  describe('rendering', () => {
    it('should render with default slot content', () => {
      const wrapper = mount(Button, {
        slots: { default: 'Click Me' }
      })
      
      expect(wrapper.text()).toBe('Click Me')
      expect(wrapper.element()).toHaveClass('btn')
    })
    
    it('should render custom type variant', () => {
      const wrapper = mount(Button, {
        props: { type: 'danger' },
        slots: { default: 'Delete' }
      })
      
      expect(wrapper.classes()).toContain('btn--danger')
    })
    
    it('should be disabled when disabled prop is true', () => {
      const wrapper = mount(Button, {
        props: { disabled: true },
        slots: { default: 'Disabled' }
      })
      
      expect(wrapper.attributes('disabled')).toBeDefined()
      expect(wrapper.element()).toBeDisabled()
    })
  })
  
  // 交互测试
  describe('interactions', () => {
    let wrapper: VueWrapper
    
    beforeEach(() => {
      wrapper = mount(Button, {
        slots: { default: 'Click Me' }
      })
    })
    
    afterEach(() => {
      wrapper.unmount()
    })
    
    it('emits click event when clicked', async () => {
      await wrapper.trigger('click')
      
      expect(wrapper.emitted()).toBeTruthy()
    })
    
    it('calls the onClick handler', async () => {
      const onClick = vi.fn()
      const wrapper = mount(Button, {
        props: { onClick },
        slots: { default: 'Submit' }
      })
      
      await wrapper.trigger('click')
      
      expect(onClick).toHaveBeenCalledOnce()
    })
    
    it('does not emit when disabled', async () => {
      const onClick = vi.fn()
      const wrapper = mount(Button, {
        props: { disabled: true, onClick },
        slots: { default: 'Cannot Click' }
      })
      
      await wrapper.trigger('click')  // 点击被忽略
      
      expect(onClick).not.toHaveBeenCalled()
    })
  })
})
```

### 表单组件测试

```typescript
// tests/components/Input/Input.test.tsx
import { describe, it, expect, vi } from 'vitest'
import { mount, flushPromises } from '@vue/test-utils'
import Input from '@/components/Input/Input.vue'
import { ref } from 'vue'

describe('Input Component', () => {
  
  describe('two-way binding (v-model)', () => {
    let wrapper: any
    
    beforeEach(() => {
      wrapper = mount(Input, {
        props: { modelValue: '' }
      })
    })
    
    afterEach(() => { wrapper.unmount() })
    
    it('binds value to input element', async () => {
      const input = wrapper.find('input')
      expect(input.element.value).toBe('')
      
      await input.setValue('hello')
      // 触发 input 事件以触发 v-model 更新
      await input.trigger('input')
      
      expect(input.element.value).toBe('hello')
      expect(wrapper.emitted('update:modelValue')).toHaveBeenCalledWith('hello')
    })
    
    it('validates based on rules', async () => {
      const wrapper = mount(Input, {
        props: { 
          modelValue: '',
          rules: { required: true }
        }
      })
      
      const input = wrapper.find('input')
      
      // 提交空值 → 应该显示验证错误
      await input.setValue('')
      await input.trigger('blur')
      
      expect(wrapper.text()).toContain('This field is required')
      expect(wrapper.find('.error-message').exists()).toBe(true)
    })
  })
})
```

## Composables 测试

```typescript
// composables/useAuth.test.ts
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import { ref } from 'vue'
import { useAuth } from '@/composables/useAuth'

describe('useAuth composable', () => {
  let auth: ReturnType<typeof useAuth>
  let pinia: { install: Function; restore: () => void }

  beforeEach(() => {
    // 创建真实的 Pinia 实例用于测试
    const { createPinia, setActiveStore } = require('pinia')
    pinia = createPinia({
      createSpy: false,
      fakeApp: true,
      plugins: [] as []
    })
    setActiveStore(pinia)
    
    auth = useAuth()
  })

  afterEach(() => {
    if (auth) auth.$reset?.()
  })

  it('starts with unauthenticated state', () => {
    expect(auth.isAuthenticated.value).toBe(false)
    expect(auth.user.value).toBeNull()
  })

  it('logs in successfully and sets user state', async () => {
    await auth.login('admin', 'password123')
    
    expect(auth.isAuthenticated.value).toBe(true)
    expect(auth.user.value).toEqual({ name: 'Admin', role: 'admin' })
  })

  it('clears authentication on logout', () => {
    auth.login('admin', 'password123')
    auth.logout()
    
    expect(auth.isAuthenticated.value).toBe(false)
    expect(auth.user.value).toBeNull()
  })
})
```

## Router Mock 与导航测试

```typescript
// tests/router/UserProfile.test.tsx
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import { createRouter, createMemoryHistory, createWebHistory } from 'vue-router'
import UserPage from '@/pages/users/[id].vue'

const routes = [
  { path: '/', component: HomeView },
  { path: '/users/:id', name: 'user-profile', component: UserPage },
]

describe('User Profile Page', () => {
  let router: ReturnType<typeof createRouter>
  let history: ReturnType<typeof createMemoryHistory>

  beforeEach(() => {
    router = createRouter({
      routes,
      history: createMemoryHistory(),  // 内存历史（不操作真实 URL）
      linkActiveClass: 'active',
      scrollBehavior: () => ({ left: 0, top: 0 })
    })
    
    router.push('/')  // 初始导航到首页
  })

  afterEach(() => {
    router.push('/')  // 重置回首页
  })

  it('renders correctly at /users/:id route', async () => {
    router.push('/users/42')
    
    const route = router.currentRoute.value
    expect(route.name).toBe('user-profile')
    expect(route.params.id).toBe('42')
  })
  
  it('handles route params changes', async () => {
    router.push('/users/42')
    expect(router.currentRoute.value.params.id).toBe('42')
    
    router.push('/users/99')
    expect(router.currentRoute.value.params.id).toBe('99')
  })
})
```

### E2E 测试示例

```typescript
// tests/e2e/user-flow.spec.ts
import { test, expect } from '@playwright/test'

test.describe('User login flow', () => {
  test('complete login flow', async ({ page }) => {
    // 访问到登录页
    await page.goto('/login')
    await expect(page).toHaveTitle('Login - My App')
    
    // 填入凭据
    await page.fillInput('[data-testid="username"]', 'admin')
    await page.fillInput('[data-testid="password"] => 'secret123', { secret: true })
    
    // 点击登录按钮
    await page.click('[data-testid="login-btn"]')
    
    // 应跳转到首页
    await expect(page).toHaveURL(/\//$/)
    await expect(page).toHaveTitle('Dashboard - My App')
    
    // 验证用户信息显示正确
    await expect(page.locator('[data-testid="user-name"]').toHaveText('Admin')
  })
})
```

## CI 集成示例

```yaml
# .github/workflows/ci.yml
name: Tests

on:
  push:
  branches: [main]
  pull_request:

jobs:
  unit:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
      - run: npm ci
      - run: npx vitest run --coverage
      - name: Upload coverage
        if: github.event_name != 'pull_request'
        uses: codecov/codecov-action@v4
        with:
          token: ${{ secrets.CODECOV_TOKEN }}
        flags: # 使用 V8 引擎报告
        files: ./coverage/
        flags: overwrite
        fail_ci_error: true
        verbose: true
      env:
        CODECOV_TOKEN: ${{ secrets.CODECOV_TOKEN }}

  e2e:
  needs: unit
    runs-on: ubuntu-latest
    steps:
      - uses: actions/coverage@v3
      - uses: nickvdp/playwright@v2
      - run: npx vitest run
      - name: Install deps
        working-directory: e2e
        run: npx playwright install --with-deps
      - name: Run E2E tests
        run: npx playwright test
```

### 编写可测试代码的原则

```
✅ DO:
  1. 每个测试只测一个行为
  2. 测试名称清晰描述预期行为
  3. Arrange-Act-Assert 模式 (Given-When-Then)
  4. 使用 data-* 属性选择器而非 CSS 类名
  5. 测试正常路径和边界情况
  ❌ DON'T:
  1. 测试内部实现细节
  2. 在测试中引入无关依赖
  3. 过度 mock 导致测试失去意义
  4. 让测试相互依赖
```
