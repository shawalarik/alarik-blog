# 自定义 Hooks 模式

## 1. 自定义 Hook 是什么

自定义 Hook 是一个**以 `use` 开头的普通 JavaScript 函数**，内部可以调用其他 Hooks。它让你将**组件逻辑提取到可复用的函数中**。

```jsx
// 最简自定义 Hook
function useDouble(number) {
  return number * 2;
}

// 使用
function Display({ value }) {
  const doubled = useDouble(value);
  return <span>{doubled}</span>;
}
```

### 1.1 命名规范

```jsx
// ✅ 以 use 开头
function useLocalStorage(key, initialValue) { /* ... */ }
function useFetch(url, options) { /* ... */ }
function useDebounce(value, delay) { /* ... */ }

// ❌ 不以 use 开头（违反 Hooks 规则，ESLint 无法检查）
function getLocalStorage(key, initialValue) { /* ... */ }
function fetchAPI(url) { /* ... */ }
```

> **为什么必须以 `use` 开头？**
> 1. 让开发者一眼看出这是一个 Hook（内部使用了其他 Hooks）
> 2. 让 `eslint-plugin-react-hooks` 能够检查其内部的 Hooks 规则是否被遵守
> 3. 符合社区约定，便于团队协作

## 2. 经典实战案例

### 2.1 useLocalStorage —— 持久化状态

```jsx
function useLocalStorage(key, initialValue) {
  // 惰性初始化：只在首次渲染时读取 localStorage
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.warn(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  // 同步到 localStorage
  const setValue = useCallback((value) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.warn(`Error setting localStorage key "${key}":`, error);
    }
  }, [key, storedValue]);

  // 监听其他标签页的变化
  const handleStorageChange = useCallback((e) => {
    if (e.key === key && e.newValue) {
      setStoredValue(JSON.parse(e.newValue));
    }
  }, [key]);

  useEffect(() => {
    window.addEventListener('storage', handleStorageChange);
    return () => window.removeEventListener('storage', handleStorageChange);
  }, [handleStorageChange]);

  return [storedValue, setValue];
}

// 使用
function SettingsPanel() {
  const [theme, setTheme] = useLocalStorage('theme', 'light');
  const [fontSize, setFontSize] = useLocalStorage('fontSize', 16);

  return (
    <div style={{ fontSize }}>
      <select value={theme} onChange={(e) => setTheme(e.target.value)}>
        <option value="light">Light</option>
        <option value="dark">Dark</option>
      </select>
      <input
        type="range"
        min="12"
        max="24"
        value={fontSize}
        onChange={(e) => setFontSize(Number(e.target.value))}
      />
    </div>
  );
}
```

### 2.2 useFetch —— 数据请求

```jsx
function useFetch(url, options = {}) {
  const { initialData, enabled = true, ...fetchOptions } = options;
  const [data, setData] = useState(initialData ?? null);
  const [loading, setLoading] = useState(enabled);
  const [error, setError] = useState(null);

  const execute = useCallback(async (overrideUrl?: string) => {
    if (!enabled) return;
    setLoading(true);
    setError(null);

    try {
      const response = await fetch(overrideUrl || url, fetchOptions);
      
      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }
      
      const json = await response.json();
      setData(json);
      return json;
    } catch (err) {
      setError(err instanceof Error ? err : new Error(String(err)));
      throw err;
    } finally {
      setLoading(false);
    }
  }, [url, enabled, fetchOptions]);

  useEffect(() => {
    if (enabled && url) {
      execute();
    }
  }, [execute, url, enabled]);

  const refetch = useCallback(() => execute(), [execute]);

  return { data, loading, error, refetch, execute };
}

// 使用
function UserProfile({ userId }) {
  const { data: user, loading, error, refetch } = useFetch(
    userId ? `/api/users/${userId}` : null,
    { enabled: !!userId }
  );

  if (loading) return <UserSkeleton />;
  if (error) return <ErrorMessage error={error} onRetry={refetch} />;
  if (!user) return <EmptyState />;

  return (
    <Card>
      <Avatar src={user.avatarUrl} />
      <h1>{user.name}</h1>
      <p>{user.bio}</p>
      <button onClick={refetch}>Refresh</button>
    </Card>
  );
}
```

### 2.3 useDebounce —— 防抖

```jsx
function useDebounce(value, delay) {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

// 使用
function SearchInput() {
  const [searchTerm, setSearchTerm] = useState('');
  const debouncedSearchTerm = useDebounce(searchTerm, 500);

  useEffect(() => {
    if (debouncedSearchTerm) {
      searchAPI(debouncedSearchTerm).then(setResults);
    }
  }, [debouncedSearchTerm]);

  return (
    <input
      value={searchTerm}
      onChange={(e) => setSearchTerm(e.target.value)}
      placeholder="Type to search..."
    />
  );
}
```

### 2.4 useToggle —— 开关状态

```jsx
function useToggle(initialValue = false) {
  const [value, setValue] = useState(initialValue);

  const toggle = useCallback(() => setValue(v => !v), []);
  const setTrue = useCallback(() => setValue(true), []);
  const setFalse = useCallback(() => setValue(false), []);

  return { value, toggle, setTrue, setFalse, setValue };
}

// 使用
function Modal({ title, children }) {
  const { value: isOpen, toggle, setFalse: close } = useToggle(false);

  return (
    <>
      <button onClick={toggle}>{title}</button>
      {isOpen && (
        <Dialog onClose={close}>
          {children}
        </Dialog>
      )}
    </>
  );
}
```

### 2.5 useMediaQuery —— 响应式查询

```jsx
function useMediaQuery(query) {
  const [matches, setMatches] = useState(() => window.matchMedia(query).matches);

  useEffect(() => {
    const mediaQuery = window.matchMedia(query);
    const handler = (event) => setMatches(event.matches);

    mediaQuery.addEventListener('change', handler);
    return () => mediaQuery.removeEventListener('change', handler);
  }, [query]);

  return matches;
}

// 使用
function ResponsiveLayout() {
  const isMobile = useMediaQuery('(max-width: 768px)');
  const isTablet = useMediaQuery('(min-width: 769px) and (max-width: 1024px)');
  const prefersDarkMode = useMediaQuery('(prefers-color-scheme: dark)');

  return isMobile ? <MobileNav /> : <DesktopNav />;
}
```

### 2.6 usePrevious —— 获取上一帧的值

```jsx
function usePrevious(value) {
  const ref = useRef();

  useEffect(() => {
    ref.current = value;  // 在 effect 中更新（渲染之后）
  }, [value]);

  // 返回的是上次渲染的值
  return ref.current;
}

// 使用
function CounterInfo({ count }) {
  const prevCount = usePrevious(count);

  return (
    <span>
      Current: {count}
      {prevCount !== undefined && prevCount !== count && (
        <em> (was {prevCount})</em>
      )}
    </span>
  );
}
```

### 2.7 useIntersectionObserver —— 可视区域检测

```jsx
function useIntersectionObserver(options = {}) {
  const elementRef = useRef(null);
  const [isIntersecting, setIsIntersecting] = useState(false);
  const [entry, setEntry] = useState(null);

  useEffect(() => {
    const element = elementRef.current;
    if (!element) return;

    const observer = new IntersectionObserver(([entry]) => {
      setIsIntersecting(entry.isIntersecting);
      setEntry(entry);
    }, options);

    observer.observe(element);
    return () => observer.disconnect();
  }, [options.threshold, options.root, options.rootMargin]);

  return { elementRef, isIntersecting, entry };
}

// 使用：懒加载图片 / 无限滚动
function LazyImage({ src, alt, placeholder }) {
  const { elementRef, isIntersecting } = useIntersectionObserver({ threshold: 0.1 });

  return (
    <div ref={elementRef} style={{ minHeight: 200 }}>
      {isIntersecting ? (
        <img src={src} alt={alt} loading="lazy" />
      ) : (
        placeholder ?? <Skeleton />
      )}
    </div>
  );
}
```

## 3. 自定义 Hook 设计原则

### 3.1 单一职责

```jsx
// ❌ 一个 Hook 做太多事情
function useUserAuthentication(loginUrl, registerUrl) {
  // 登录逻辑
  // 注册逻辑
  // Token 管理
  // Session 检测
  // 权限校验
  // 密码强度检测
  // ... 500 行
}

// ✅ 拆分为专注的小 Hook
function useLogin() { /* 登录 */ }
function useRegister() { /* 注册 */ }
function useAuthToken() { /* Token 管理 */ }
function usePermissions() { /* 权限 */ }
```

### 3.2 参数设计

```jsx
// ✅ 良好的参数设计
function useFetch<T = unknown>(
  url: string | null,
  options?: {
    initialData?: T;
    enabled?: boolean;
    retry?: number;
    staleTime?: number;
    onSuccess?: (data: T) => void;
    onError?: (error: Error) => void;
  }
): FetchResult<T>;

// 特点：
// 1. 泛型支持类型推断
// 2. options 对象方便扩展（不需要记住参数顺序）
// 3. 合理的默认值
// 4. 回调钩子
// 5. 返回值结构统一
```

### 3.3 返回值设计

```jsx
// ✅ 推荐的返回格式
return {
  // 状态值
  data,
  loading,
  error,
  
  // 操作方法
  refetch,
  mutate,
  reset,
  
  // 元信息
  isSuccess,
  isError,
  isFetched,
  updatedAt,
};

// 或对于简单场景，返回元组（方便重命名）
return [value, setValue, isDirty];  // 类似 useState 风格
```

### 3.4 错误处理

```jsx
function useCustomHook(param) {
  const [error, setError] = useState(null);

  // 1. 参数校验
  if (param === undefined) {
    console.warn('[useCustomHook] param is required');
    // 返回安全的默认值而非抛出异常
    return { data: null, loading: false, error: new Error('param required') };
  }

  // 2. 运行时错误边界
  try {
    // 危险操作
  } catch (err) {
    setError(err);
  }

  // 3. 清理函数中的错误
  useEffect(() => {
    const controller = new AbortController();
    
    doSomething({ signal: controller.signal })
      .catch(err => {
        if (err.name !== 'AbortError') {
          setError(err);
        }
      });

    return () => controller.abort();
  }, []);

  return { data, error };
}
```

## 4. Hook 组合模式

### 4.1 链式组合

```jsx
// 高层 Hook 组合低层 Hook
function useAuthenticatedApi(baseUrl) {
  const { token } = useAuthToken();           // 获取 token
  const headers = useMemo(() => ({            // 构建请求头
    Authorization: `Bearer ${token}`,
    'Content-Type': 'application/json',
  }), [token]);

  const { data, loading, error } = useFetch(  // 发起请求
    token ? baseUrl : null,
    { headers, enabled: !!token }
  );

  // 当 token 过期时刷新
  const { refresh } = useTokenRefresh(token);

  return { data, loading, error, refresh };
}
```

### 4.2 并行组合

```jsx
function useDashboardStats(dateRange) {
  // 多个独立的 Hook 并行运行
  const revenue = useFetch(`/api/stats/revenue?range=${dateRange}`);
  const users = useFetch(`/api/stats/users?range=${dateRange}`);
  const orders = useFetch(`/api/stats/orders?range=${dateRange}`);

  const isLoading = revenue.loading || users.loading || orders.loading;
  const hasError = revenue.error || users.error || orders.error;

  return {
    revenue: revenue.data,
    users: users.data,
    orders: orders.data,
    isLoading,
    hasError,
    refetch: () => { revenue.refetch(); users.refetch(); orders.refetch(); },
  };
}
```

## 6. 总结

```
┌──────────────────────────────────────────────────┐
│         自定义 Hook 要点速查                       │
├──────────────────────────────────────────────────┤
│ 定义: 以 use 开头的函数，内部调用其他 Hooks        │
│ 命名: 必须 use 前缀（ESLint 检查 + 语义清晰）     │
│ 原则: 单一职责、合理参数设计、完善错误处理         │
│ 返回: 对象（命名清晰）或 元组（灵活重命名）        │
│ 组合: Hook 调用 Hook，链式/并行组合               │
│ 测试: 纯函数特性使单元测试非常容易                 │
│ 参考: ahooks, @tanstack/react-query, zustand      │
└──────────────────────────────────────────────────┘
```
