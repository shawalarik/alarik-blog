# useReducer + Context 模式

## 1. 为什么组合使用？

将 `useReducer` 和 `Context` 组合，可以实现一个**类似 Redux 的轻量级全局状态管理方案**——无需安装任何第三方库：

```
Redux 架构:
  Store (单一数据源)
    ↓ dispatch(action)
  Reducer (纯函数) → 新 State
    ↓
  Store 更新 → 通知 subscribers (UI components)

useReducer + Context 模拟:
  Context.Provider (充当 Store)
    ↓ dispatch(action)
  useReducer (充当 Reducer) → 新 State
    ↓
  Context value 更新 → 触发 consumers 重渲染
```

## 2. 完整实现模板

### 2.1 定义 State 和 Action

```tsx
// types.ts
interface Todo {
  id: number;
  text: string;
  completed: boolean;
  createdAt: number;
}

interface TodosState {
  todos: Todo[];
  filter: 'all' | 'active' | 'completed';
  loading: boolean;
  error: string | null;
}

type TodosAction =
  | { type: 'FETCH_START' }
  | { type: 'FETCH_SUCCESS'; payload: Todo[] }
  | { type: 'FETCH_ERROR'; payload: string }
  | { type: 'ADD'; payload: string }
  | { type: 'TOGGLE'; payload: number }
  | { type: 'DELETE'; payload: number }
  | { type: 'SET_FILTER'; payload: TodosState['filter'] }
  | { type: 'CLEAR_COMPLETED' };
```

### 2.2 实现 Reducer

```tsx
// reducer.ts
const initialState: TodosState = {
  todos: [],
  filter: 'all',
  loading: false,
  error: null,
};

function todosReducer(state: TodosState, action: TodosAction): TodosState {
  switch (action.type) {
    case 'FETCH_START':
      return { ...state, loading: true, error: null };

    case 'FETCH_SUCCESS':
      return { ...state, loading: false, todos: action.payload };

    case 'FETCH_ERROR':
      return { ...state, loading: false, error: action.payload };

    case 'ADD': {
      const newTodo: Todo = {
        id: Date.now(),
        text: action.payload.trim(),
        completed: false,
        createdAt: Date.now(),
      };
      if (!newTodo.text) return state; // 忽略空输入
      return { ...state, todos: [...state.todos, newTodo] };
    }

    case 'TOGGLE':
      return {
        ...state,
        todos: state.todos.map(todo =>
          todo.id === action.payload
            ? { ...todo, completed: !todo.completed }
            : todo
        ),
      };

    case 'DELETE':
      return {
        ...state,
        todos: state.todos.filter(todo => todo.id !== action.payload),
      };

    case 'SET_FILTER':
      return { ...state, filter: action.payload };

    case 'CLEAR_COMPLETED':
      return {
        ...state,
        todos: state.todos.filter(todo => !todo.completed),
      };

    default:
      // TypeScript exhaustiveness check
      const _: never = action;
      return state;
  }
}

export { todosReducer, initialState };
export type { TodosState, TodosAction };
```

### 2.3 创建 Context 和 Provider

```tsx
// context.tsx
import { createContext, useContext, useReducer, Dispatch, ReactNode } from 'react';
import { todosReducer, initialState, TodosState, TodosAction } from './reducer';

// Context 值的结构
interface TodosContextValue {
  state: TodosState;
  dispatch: Dispatch<TodosAction>;
}

const TodosContext = createContext<TodosContextValue | null>(null);

function TodosProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(todosReducer, initialState);

  // 用 useMemo 保持 value 引用稳定？这里不需要因为 dispatch 引用稳定，
  // state 由 useReducer 管理
  const value: TodosContextValue = { state, dispatch };

  return (
    <TodosContext.Provider value={value}>
      {children}
    </TodosContext.Provider>
  );
}

function useTodosContext(): TodosContextValue {
  const ctx = useContext(TodosContext);
  if (!ctx) throw new Error('useTodosContext must be used within TodosProvider');
  return ctx;
}

export { TodosProvider, useTodosContext };
```

### 2.4 封装便捷 Hooks（可选但推荐）

```tsx
// hooks.ts
import { useTodosContext } from './context';
import { useCallback } from 'react';

// 直接访问 state
export function useTodosState() {
  return useTodosContext().state;
}

// 访问 dispatch 并封装常用操作
export function useTodosDispatch() {
  const { dispatch } = useTodosContext();
  
  return {
    addTodo: useCallback((text: string) => 
      dispatch({ type: 'ADD', payload: text })
    , [dispatch]),
    
    toggleTodo: useCallback((id: number) => 
      dispatch({ type: 'TOGGLE', payload: id })
    , [dispatch]),
    
    deleteTodo: useCallback((id: number) => 
      dispatch({ type: 'DELETE', payload: id })
    , [dispatch]),
    
    setFilter: useCallback((filter: TodosState['filter']) => 
      dispatch({ type: 'SET_FILTER', payload: filter })
    , [dispatch]),
    
    clearCompleted: useCallback(() => 
      dispatch({ type: 'CLEAR_COMPLETED' })
    , [dispatch]),
  };
}

// 组合 hook：同时提供 state 和 actions
export function useTodos() {
  const state = useTodosState();
  const actions = useTodosDispatch();
  
  // 派生数据
  const filteredTodos = state.todos.filter(todo => {
    if (state.filter === 'active') return !todo.completed;
    if (state.filter === 'completed') return todo.completed;
    return true;
  });

  const remainingCount = state.todos.filter(t => !t.completed).length;

  return {
    ...state,
    filteredTodos,
    remainingCount,
    ...actions,
  };
}
```

### 2.5 在组件中使用

```tsx
// App.tsx
import { TodosProvider } from './store/context';

function App() {
  return (
    <TodosProvider>
      <TodoApp />
    </TodosProvider>
  );
}

// TodoList.tsx
function TodoList() {
  const { filteredTodos, remainingCount, deleteTodo, toggleTodo } = useTodos();

  return (
    <div>
      <p>{remainingCount} items left</p>
      <ul>
        {filteredTodos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.completed}
              onChange={() => toggleTodo(todo.id)}
            />
            <span style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}>
              {todo.text}
            </span>
            <button onClick={() => deleteTodo(todo.id)}>×</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

## 3. 与 Redux 的对比

| 特征 | Redux | useReducer + Context |
|------|-------|-------------------|
| Store | 全局单一 Store | 每个 Context 一个 "mini-store" |
| 中间件 | 丰富（logger, thunk, saga） | 无（需自行实现异步） |
| DevTools | Redux DevTools（强大） | 需额外集成 react-devtools-hook |
| 时间旅行调试 | 原生支持 | 需自行实现 history |
| Action 类型检查 | 需工具辅助 | TypeScript 天然友好 |
| 代码量 | 较多（boilerplate） | 相对精简 |
| 性能 | connect selector 精准控制 | Context 全量通知（需拆分优化） |
| 异步 Action | thunk/saga/observable | useEffect + dispatch |
| 社区生态 | 庞大 | 无（纯 React 方案） |

## 4. 异步 Action 处理

useReducer 本身只支持同步 reducer。处理异步操作需要在 effect 中完成：

```tsx
// 在组件或自定义 Hook 中处理异步
function useTodosAsync() {
  const { state, dispatch } = useTodosContext();
  const { fetchStart, fetchSuccess, fetchError } = useTodosDispatch();

  const fetchTodos = useCallback(async () => {
    try {
      fetchStart();  // dispatch({ type: 'FETCH_START' })
      const res = await fetch('/api/todos');
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const todos: Todo[] = await res.json();
      fetchSuccess(todos);  // dispatch({ type: 'FETCH_SUCCESS', payload })
    } catch (err) {
      fetchError(err.message);  // dispatch({ type: 'FETCH_ERROR', payload })
    }
  }, [fetchStart, fetchSuccess, fetchError]);

  return { ...state, fetchTodos };
}
```

> 如果异步逻辑复杂（竞态、取消、轮询等），建议直接使用 **TanStack Query** 或 **RTK Query** 处理服务端状态。

## 5. 何时升级到第三方库

以下情况建议放弃 useReducer+Context，改用专业方案：

| 信号 | 推荐替代 |
|------|---------|
| 需要中间件（logger/persist/devtools） | **Zustand** 或 **Redux Toolkit** |
| 需要精确的性能优化（selector 粒度） | **Jotai** 或 **Zustand** |
| 大量服务端状态需要缓存管理 | **TanStack Query** |
| 复杂的状态机逻辑 | **XState** |
| 团队需要统一的架构规范 | **Redux Toolkit** |

## 6. 总结

```
┌──────────────────────────────────────────────────┐
│      useReducer + Context 模式要点                │
├──────────────────────────────────────────────────┤
│ 结构: Context(Provider) + useReducer(Reducer)    │
│ 流程: dispatch(action) → reducer计算新state → 更新Context
│ 封装: Provider + useXxx 自定义 Hook              │
│ 异步: useEffect 中 dispatch start/success/error  │
│ 优势: 零依赖、TS友好、适合中小项目               │
│ 劣势: 无中间件、Context 性能陷阱、无 DevTools     │
│ 进阶: 需要更多能力时迁移到 Zustand / RTK          │
└──────────────────────────────────────────────────┘
```
