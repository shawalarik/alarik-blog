# XState —— 有限状态机与状态图

## 1. 为什么需要状态机？

普通的 `useState` / `useReducer` 在处理**复杂状态转换逻辑**时力不从心：

```jsx
// ❌ 用 useState 管理复杂状态：容易出 bug
function Form() {
  const [status, setStatus] = useState('idle');  // idle | submitting | success | error
  const [error, setError] = useState(null);
  const [data, setData] = useState(null);
  
  async function handleSubmit() {
    if (status === 'submitting') return;  // 手动防重提交
    
    // 问题1：状态之间的合法转换关系不明确
    // 问题2：任何地方都可以随意 setStatus
    // 问题3：难以可视化状态流转
    setStatus('submitting');
    
    try {
      const res = await fetch('/api/submit');
      setData(await res.json());
      setStatus('success');
    } catch (err) {
      setError(err.message);
      setStatus('error');
    }
  }

  // 问题4：reset 时需要手动清理多个状态
  function reset() { setStatus('idle'); setError(null); setData(null); }
}
```

```
useState 的隐式问题:

  status 可以从任意值跳到任意值:
  idle → submitting ✓
  submitting → success ✓  
  submitting → error ✓
  idle → error ✗ (不应该直接发生!)
  success → submitting ✗ (应该先 reset)
  
  这些约束只能靠开发者"记住"，代码无法强制执行。
```

## 2. 核心概念

### 2.1 有限状态机（FSM）

XState 将状态和转换定义为**显式的、可验证的模型**：

```
┌─────────────────────────────────────────────┐
│              表单提交流转状态机               │
│                                             │
│   ┌───────┐   submit   ┌──────────┐        │
│   │ idle  │ ─────────→ │submitting │        │
│   └───┬───┘            └────┬─────┘        │
│       │                     │              │
│       │ reset         succ │ fail          │
│       │                     │              │
│       ▼                     ▼ ▼            │
│   ┌───────┐           ┌────────┐ ┌──────┐   │
│   │ idle* │           │success │ │error │   │
│   └───────┘           └────────┘ └──┬───┘   │
│                                  │         │
│                               retry│  reset  │
│                                  ▼         │
│                           ┌──────────┐     │
│                           │submitting│◄────┘
│                           └──────────┘
└─────────────────────────────────────────────┘
```

### 2.2 XState v5 新语法（TypeScript 优先）

```ts
import { setup, assign } from 'xstate';

const machine = setup({
  types: {} as {
    context: {
      retries: number;
      errorMsg: string | null;
      data: unknown;
    },
    events: { type: 'SUBMIT'; type: 'RETRY'; type: 'RESET' };
  },
  states: {
    idle: {
      on: { SUBMIT: 'submitting' },
    },
    submitting: {
      entry: [{ type: 'ASSIGN', retries: 0 }],
      invoke: {
        src: 'fetchData',
        onDone: { target: 'success', actions: ['assignData'] },
        onError: [
          { guard: ({ context }) => context.retries < 3, target: 'error', actions: ['incrementRetries'] },
          { target: 'error', actions: ['setFatalError'] },
        ],
      },
    },
    success: {
      on: { SUBMIT: 'submitting' }, // 可以重新提交
    },
    error: {
      on: { RETRY: 'submitting', RESET: 'idle' },
    },
  },
}).createMachine({
  context: { retries: 0, errorMsg: null, data: null },
});

// 状态机可视化：
// https://stately.ai/viz?input=...
```

## 3. 与 React 集成

### 3.1 @xstate/react

```tsx
import { useMachine } from '@xstate/react';

function SubmitForm() {
  const [state, send] = useMachine(formMachine);

  return (
    <form onSubmit={(e) => { e.preventDefault(); send({ type: 'SUBMIT' }); }}>
      {/* 根据 state 渲染不同 UI */}
      {state.matches('idle') && <SubmitButton />}
      
      {state.matches('submitting') && <LoadingSpinner />}
      
      {state.matches('success') && (
        <SuccessMessage data={state.context.data} />
      )}
      
      {state.matches('error') && (
        <ErrorMessage 
          message={state.context.errorMsg ?? 'Unknown error'} 
          onRetry={() => send({ type: 'RETRY' })}
          onReset={() => send({ type: 'RESET' })}
        />
      )}
    </form>
  );
}
```

### 3.2 Actor 模式（v5）

```tsx
import { createActorContext } from '@xstate/react';

// 定义 actor
export const formActor = createActorContext(formMachine);

function App() {
  return (
    <formActor.Provider>
      <FormComponent />
      <StatusDisplay />
    </formActor.Provider>
  );
}

// 子组件中消费
function StatusDisplay() {
  const state = formActor.useSelector((s) => s.value);
  return <div>Status: {JSON.stringify(state)}</div>;
}
```

## 4. 实战案例

### 4.1 审批流程状态机

```
                    ┌───────────┐
              提交   │  Draft   │
             ───────►│           │
                    └─────┬─────┘
                          │ submit
                          ▼
                    ┌───────────┐
              审核   │ Pending  │◄──────────┐
             ◄────── │ Review   │───────────┤ reject
              approve │           │            │
                    └─────┬─────┘
                          │ approve/reject
              ┌───────────┼───────────┐
              ▼           │           ▼
       ┌──────────┐       │    ┌──────────┐
       │ Approved │       │    │ Rejected │
       └─────┬────┘       │    └─────┬────┘
             │ revoke      │          │ appeal
             └────────────┼──────────┘
                          │
                    ┌─────▼─────┐
                    │ Pending  │ (重新审核)
                    └──────────┘
```

```ts
import { setup, assign } from 'xstate';

type ApprovalEvent =
  | { type: 'SUBMIT' }
  | { type: 'APPROVE' }
  | { type: 'REJECT' }
  | { type: 'APPEAL' }
  | { type: 'REVOKE' };

const approvalMachine = setup({
  types: {} as {
    context: {
      submitterId: string;
      reviewerId: string;
      reason: string;
      history: Array<{ action: string; at: Date; by: string }>;
    };
    events: ApprovalEvent;
  },
  states: {
    draft: {
      on: { SUBMIT: { target: 'pendingReview', actions: ['recordHistory'] } },
    },
    pendingReview: {
      on: {
        APPROVE: { target: 'approved', actions: ['recordHistory'] },
        REJECT: { target: 'rejected', actions: ['recordReason'] },
      },
    },
    approved: {
      on: { REVOKE: { target: 'draft', actions: ['recordHistory'] } },
    },
    rejected: {
      on: {
        APPEAL: { target: 'pendingReview', actions: ['recordHistory'] },
      },
    },
  },
}).createMachine({
  initialContext: {
    submitterId: '',
    reviewerId: '',
    reason: '',
    history: [],
  },
});
```

### 4.2 拖拽排序的状态管理

```tsx
import { setup, fromTransition } from 'xstate';
import { useMachine } from '@xstate/react';

const dragMachine = setup({
  types: {} as {
    context: {
      draggedItemId: string | null;
      dropZoneId: string | null;
      itemsOrder: string[];
    };
    events: { type: 'DRAG_START'; type: 'DRAG_OVER'; type: 'DROP'; type: 'CANCEL' };
  },
  states: {
    idle: {
      on: { DRAG_START: 'dragging' },
    },
    dragging: {
      on: {
        DRAG_OVER: { actions: ['updateDropZone'] },
        DROP: { target: 'idle', actions: ['reorderItems'] },
        CANCEL: { target: 'idle', actions: ['clearDrag'] },
      },
    },
  },
}).createMachine(/* ... */);

function SortableList() {
  const [snapshot, send] = useMachine(dragMachine);
  const isDragging = snapshot.matches('dragging');

  return (
    <ul className={isDragging ? 'dragging' : ''}>
      {snapshot.context.itemsOrder.map(id => (
        <SortableItem
          key={id}
          id={id}
          isDragged={snapshot.context.draggedItemId === id}
          isDropTarget={snapshot.context.dropZoneId === id}
          onDragStart={() => send({ type: 'DRAG_START', draggedItemId: id })}
          onDragOver={isDragging ? () => send({ type: 'DRAG_OVER', dropZoneId: id }) : undefined}
          onDrop={isDragging ? () => send({ type: 'DROP' }) : undefined}
        />
      ))}
    </ul>
  );
}
```

## 7. 总结

```
┌──────────────────────────────────────────────────┐
│         XState 要点速查                            │
├──────────────────────────────────────────────────┤
│ 核心: 显式定义有限状态机和合法转移路径            │
│ 优势: 状态可预测、可视化、可测试、防 bug          │
│ v5 API: setup() + TypeScript-first               │
│ React: @xstate/react (useMachine / Actor)        │
│ 工具: Stately Studio / Viz / Inspector DevTools  │
│ 适用: 复杂业务流程、审批流、游戏状态机           │
│ 不适用: 简单的布尔切换、数据获取缓存             │
│ 学习曲线: 中等（需理解 FSM/状态图概念）          │
└──────────────────────────────────────────────────┘
```
