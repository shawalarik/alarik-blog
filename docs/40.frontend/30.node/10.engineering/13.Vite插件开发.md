# Vite 插件开发实战

## 概述

Vite 插件基于 **Rollup 插件接口** 扩展，增加了 Vite 特有的钩子来处理开发服务器的需求。理解 Vite 插件开发意味着你同时掌握了 Rollup 和 Vite 双套接口。

```
┌─────────────────────────────────────────────────────────┐
│              Vite 插件钩子时序                            │
│                                                         │
│  服务器启动                                              │
│    config (Vite) → configResolved (Vite)                 │
│    configureServer (Vite)                                │
│                                                         │
│  请求中间件（仅 Dev）                                     │
│    configureServer → server.use()                        │
│                                                         │
│  每次请求（仅 Dev）                                       │
│    transformIndexHtml (Vite)                             │
│    resolveId → load → transform (Rollup)                 │
│                                                         │
│  构建阶段（Dev + Build）                                 │
│    buildStart (Rollup)                                   │
│    resolveId → load → transform (每模块)                 │
│    renderChunk (Rollup)                                  │
│    generateBundle (Rollup)                               │
│    writeBundle (Rollup)                                  │
│    closeBundle (Rollup)                                  │
│                                                         │
└─────────────────────────────────────────────────────────┘
```

## 一、插件基础结构

### 1.1 最小可用插件

```typescript
// my-vite-plugin.ts
import type { Plugin, ResolvedConfig } from "vite"

export default function myVitePlugin(options?: Options): Plugin {
  let config: ResolvedConfig

  return {
    // 插件名称
    name: "my-vite-plugin",

    // Vite 专属钩子：配置解析前
    config(userConfig, { command }) {
      // 可以修改用户配置
      // 返回的对象会深度合并
      return {
        resolve: {
          alias: {
            "@utils": options?.aliasPath ?? "/src/utils",
          },
        },
      }
    },

    // Vite 专属钩子：配置解析完成后
    configResolved(resolvedConfig) {
      config = resolvedConfig
    },

    // Vite 专属钩子：配置开发服务器
    configureServer(server) {
      // 添加自定义中间件
      server.middlewares.use((req, res, next) => {
        if (req.url === "/api/health") {
          res.writeHead(200, { "Content-Type": "application/json" })
          res.end(JSON.stringify({ status: "ok", timestamp: Date.now() }))
          return
        }
        next()
      })
    },

    // Rollup 钩子：模块转换
    transform(code, id) {
      if (!id.endsWith(".my-ext")) return null
      
      // 转换 .my-ext 文件为 JS
      return {
        code: `export default ${JSON.stringify(code.trim())}`,
        map: null,
      }
    },
  }
}

interface Options {
  aliasPath?: string
}
```

### 1.2 插件执行顺序控制

```typescript
export function myPlugin(): Plugin {
  return {
    name: "my-plugin",
    
    // 强制插件执行顺序（值越小越先执行）
    // pre: 在 Vite 核心插件之前
    // post: 在 Vite 核心插件之后（默认）
    enforce: "pre", // "post" | undefined (default)
    
    // 仅在特定模式下激活
    apply: "build", // "serve" | undefined (both)
  }
}
```

```typescript
export function apiProxyPlugin(apiBaseUrl: string): Plugin {
  let server: any

  return {
    name: "vite-plugin-api-proxy",
    configureServer(devServer) {
      server = devServer

      // API 代理
      devServer.middlewares.use("/api", createProxyMiddleware({
        target: apiBaseUrl,
        changeOrigin: true,
        pathRewrite: { "^/api": "" },
      }))

      // 自定义 WS 端点
      devServer.ws.on("custom:message", ({ data }, client) => {
        devServer.ws.send("custom:response", {
          message: `Echo: ${data}`,
        }, client)
      })

      // 打印访问 URL
      const printUrls = () => {
        const urls = devServer.resolvedUrls
        console.log("\n  ➜  Local:   ", urls.local[0])
        console.log("  ➜  Network: ", urls.network?.[0] ?? "not available")
      }

      // 延迟打印确保端口已分配
      devServer.httpServer?.once("listening", printUrls)
    },
  }
}
```

```typescript
export function htmlInjectPlugin(injectTags: HtmlTagDescriptor[]): Plugin {
  return {
    name: "vite-plugin-html-inject",
    transformIndexHtml(html, ctx) {
      return {
        html,
        tags: [
          // 注入到 <head>
          { tag: "link", attrs: { rel: "icon", href: "/favicon.ico" }, injectTo: "head" },
          // 注入到 <body> 末尾
          { tag: "script", attrs: { src: "/analytics.js" }, injectTo: "body" },
          // 条件注入
          ...(ctx.server ? [
            { tag: "script", attrs: { src: "/dev-tools.js" }, injectTo: "head" },
          ] : []),
          ...injectTags,
        ],
      }
    },
  }
}
```

### 2.4 handleHotUpdate — 自定义 HMR

```typescript
export function hmrCustomPlugin(): Plugin {
  return {
    name: "vite-plugin-hmr-custom",
    
    handleHotUpdate({ file, server, modules }) {
      // 只对特定文件类型做特殊处理
      if (file.endsWith(".config.ts")) {
        server.ws.send({
          type: "custom",
          event: "config-changed",
          data: { path: file },
        })
        
        // 让客户端自行处理
        return []
      }

      // 对 CSS 文件强制全屏刷新（默认行为是局部替换）
      if (file.endsWith(".global.css")) {
        server.ws.send({ type: "full-reload" })
        return []
      }

      // 默认 HMR 行为
      return modules
    },
  }
}
```

## 三、虚拟模块（Virtual Modules）

### 3.1 基本用法

```typescript
export function virtualEnvPlugin(): Plugin {
  // 虚拟模块必须以 virtual: 前缀开头
  const virtualModuleId = "virtual:env-config"

  return {
    name: "vite-plugin-virtual-env",
    resolveId(id) {
      if (id === virtualModuleId) {
        // 告诉 Vite 这是一个虚拟模块（没有真实文件）
        return "\0" + virtualModuleId  // \0 前缀防止其他插件处理
      }
      return null
    },
    
    load(id) {
      if (id === "\0" + virtualModuleId) {
        // 返回模块内容
        return `
          export const API_URL = "${process.env.API_URL ?? "http://localhost:3000"}"
          export const APP_VERSION = "${process.env.npm_package_version}"
          export const ENV = "${process.env.MODE}"
        `
      }
      return null
    },
  }
}

// 使用方式
// import { API_URL, APP_VERSION } from "virtual:env-config"
// TypeScript 声明
// declare module "virtual:env-config" {
//   export const API_URL: string
//   export const APP_VERSION: string
// }
```

### 3.2 动态虚拟模块

```typescript
export function svgIconsPlugin(globPattern: string): Plugin {
  let generatedModules: Record<string, string> = {}

  return {
    name: "vite-plugin-svg-icons",
    
    async buildStart() {
      // 构建开始时扫描 SVG 文件
      const files = await glob(globPattern)
      
      for (const filePath of files) {
        const content = await fs.readFile(filePath, "utf-8")
        const iconName = path.basename(filePath, ".svg")
        const moduleId = `virtual:svg-icon/${iconName}`
        
        generatedModules[moduleId] = `
          const svg = ${JSON.stringify(content)}
          export default svg
          export const name = "${iconName}"
        `
      }
    },

    resolveId(id) {
      if (id.startsWith("virtual:svg-icon/")) {
        return "\0" + id
      }
      return null
    },

    load(id) {
      if (id.startsWith("\0virtual:svg-icon/")) {
        return generatedModules[id.slice(1)] ?? "export default ''"
      }
      return null,
    },
  }
}
```

## 四、实战案例

### 4.1 案例：自动生成路由表插件

```typescript
// plugins/auto-route-plugin.ts
import * as fs from "fs/promises"
import * as path from "path"

interface RouteInfo {
  path: string
  component: string
  meta?: Record<string, unknown>
}

export function autoRoutePlugin(pagesDir: string): Plugin {
  const virtualModuleId = "virtual:auto-routes"
  
  async function generateRoutes(): Promise<RouteInfo[]> {
    const routes: RouteInfo[] = []
    
    async function walk(dir: string, basePath = "") {
      const entries = await fs.readdir(dir, { withFileTypes: true })
      
      for (const entry of entries) {
        const fullPath = path.join(dir, entry.name)
        const routePath = path.join(basePath, entry.name.replace(/\.(tsx?|jsx?)$/, ""))
        
        if (entry.isDirectory()) {
          // 递归子目录（忽略 components/ 等）
          if (!["components", "layouts", "shared"].includes(entry.name)) {
            await walk(fullPath, routePath)
          }
        } else if (/\.(tsx?|jsx?)$/.test(entry.name) && !entry.name.startsWith("_")) {
          routes.push({
            path: "/" + (routePath.toLowerCase() === "/index" ? "" : routePath.toLowerCase()),
            component: fullPath.replace(/\\/g, "/"),
          })
        }
      }
    }
    
    await walk(pagesDir)
    return routes
  }

  return {
    name: "vite-plugin-auto-routes",
    enforce: "pre",
    
    async handleHotUpdate({ file }) {
      if (file.startsWith(path.resolve(pagesDir))) {
        // pages 目录变更时通知客户端
        return [] // 让默认 HMR 处理单个文件
      }
    },

    resolveId(id) {
      return id === virtualModuleId ? "\0" + virtualModuleId : null
    },

    async load(id) {
      if (id !== "\0" + virtualModuleId) return null
      
      const routes = await generateRoutes()
      const imports = routes.map((r, i) => 
        `import Route_${i} from "${r.component}?lazy"`
      )
      const definitions = routes.map((r, i) =>
        `  { path: "${r.path}", component: () => Promise.resolve(Route_${i}), meta: ${JSON.stringify(r.meta ?? {})} }`
      )
      
      return `
${imports.join("\n")}

export const routes = [
${definitions.join(",\n")}
]

export default routes
      `
    },
  }
}
```

### 4.2 案例：Mock 数据插件

```typescript
// plugins/mock-dev-plugin.ts
import type { Connect, ViteDevServer } from "vite"

interface MockRule {
  method: "GET" | "POST" | "PUT" | "DELETE" | "PATCH"
  url: string | RegExp
  handler: (req: IncomingMessage, res: ServerResponse, params?: Record<string, string>) => any
}

export function mockDevPlugin(rules: MockRule[]): Plugin {
  let server: ViteDevServer

  return {
    name: "vite-plugin-mock-dev",
    apply: "serve", // 仅开发环境
    
    configureServer(devServer) {
      server = devServer
      
      devServer.middlewares.use(async (req, res, next) => {
        for (const rule of rules) {
          const match =
            typeof rule.url === "string"
              ? req.url?.startsWith(rule.url)
              : rule.url.test(req.url ?? "")

          if (match && req.method === rule.method) {
            // 提取 URL 参数
            const params: Record<string, string> = {}
            if (rule.url instanceof RegExp) {
              const urlMatch = (rule.url).exec(req.url ?? "")
              if (urlMatch) {
                // ... 提取命名组
              }
            }

            // 模拟延迟
            await new Promise(resolve => setTimeout(resolve, Math.random() * 300))

            const result = await rule.handler(req as any, res as any, params)
            
            res.setHeader("Content-Type", "application/json")
            res.setHeader("X-Mock-Response", "true")
            res.end(JSON.stringify(result))
            return
          }
        }
        
        next()
      })
    },
  }
}
```

## 五、插件开发最佳实践

| 原则 | 说明 |
|------|------|
| **命名规范** | `vite-plugin-{name}` 作为包名 |
| **类型安全** | 导出完整的 TypeScript 类型定义 |
| **Rollup 兼容** | 同时支持 Rollup 模式（生产构建），通过 `apply` 区分 |
| **虚拟模块** | 使用 `\0` 前缀标记，避免与其他插件冲突 |
| **错误提示** | 使用 `this.error()` 提供清晰的错误信息 |
| **性能意识** | `transform` 避免重度计算，善用 `buildStart` 预处理 |
| **配置合理** | 提供合理默认值，减少用户配置负担 |
