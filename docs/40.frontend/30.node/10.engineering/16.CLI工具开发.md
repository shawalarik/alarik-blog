# Node.js CLI 工具开发实战

## 概述

CLI（Command Line Interface）工具是前端工程化的重要组成部分——从 Vite、Webpack 到 ESLint、Prettier，几乎所有前端工具都是 CLI 应用。掌握 CLI 开发意味着你可以构建自己的自动化工具链。

```
┌──────────────────────────────────────────────────────┐
│              CLI 应用架构                              │
│                                                      │
│  用户输入                                             │
│    │                                                  │
│    ▼                                                  │
│  ┌──────────────┐                                   │
│  │ 参数解析器     │ ← Commander / Yargs / CAC         │
│  │ (Parser)     │                                    │
│  └──────┬───────┘                                   │
│         ▼                                           │
│  ┌──────────────┐   ┌──────────────┐                │
│  │ 业务逻辑      │ → │ 输出格式化   │                │
│  │ (Commands)   │   │ (Output)     │                │
│  └──────────────┘   └──────┬───────┘                │
│                            ▼                         │
│  ┌───────────┬───────────┬───────────┐              │
│  │ Console   │ Table/Box │ Interactive│             │
│  │ (log)     │ (cliui)   │ (Inquirer)│             │
│  └───────────┴───────────┴───────────┘              │
│                                                      │
└──────────────────────────────────────────────────────┘
```

## 一、基础框架选择

### 1.1 主流框架对比

| 框架 | Stars | 特点 | 适用场景 |
|------|-------|------|----------|
| **Commander** | ~28k | 轻量、TJ Holowaychuk 风格 | 中小型 CLI |
| **Yargs** | ~13k | 功能丰富、中间件模式 | 复杂 CLI / 企业级 |
| **CAC** | ~5k | 极简、类 Commander | 个人项目 |
| **oclif** | ~8k | 多命令框架、插件系统 | 大型 CLI（如 Heroku CLI） |

### 1.2 快速上手 Commander

```bash
npm install commander inquirer chalk ora commander
# 或
npx -y my-cli --init
```

```typescript
// src/index.ts
#!/usr/bin/env node
import { Command } from "commander"
import { version } from "../package.json" assert { type: "json" }

const program = new Command()
  .name("my-cli")
  .description("My awesome CLI tool")
  .version(version, "-v, --version", "显示版本号")
  .helpOption("-h, --help", "显示帮助信息")

// 注册子命令
program
  .command("create <project-name>")
  .description("创建新项目")
  .option("-t, --template <type>", "项目模板", "default")
  .option("--no-install", "跳过依赖安装")
  .action(async (name: string, options) => {
    console.log(`创建项目: ${name}`)
    console.log(`模板: ${options.template}`)
    // ... 创建逻辑
  })

program
  .command("deploy")
  .description("部署到远程服务器")
  .option("-e, --env <environment>", "目标环境", "production")
  .option("--dry-run", "仅预览不执行")
  .action((options) => {
    if (options.dryRun) {
      console.log("[Dry Run] 将部署到:", options.env)
      return
    }
    // ... 部署逻辑
  })

program.parse(process.argv)
```

## 二、参数解析

### 2.1 Commander 参数类型

```typescript
program
  .command("build")
  .description("构建项目")
  // 必选参数
  .argument("<input>", "输入文件")
  .argument("[output]", "输出文件", "dist")  // 可选，默认值
  
  // Options（可选标志）
  .option("-w, --watch", "监听文件变化", false)
  .option("-m, --mode <mode>", "构建模式", "production")
  .option("--sourcemap [type]", "生成 Source Map", "hidden")
  
  // 变长参数
  .option("--ext <extensions...>", "扩展名列表", ["js", "ts"])
  
  // 自定义选项处理
  .addOption(new Option("--debug").implies({ verbose: true }))
  .addOption(new Option("--log-level <level>").choices(["info", "warn", "error"]))
  
  .action((input, output, options) => {
    console.log({
      input,
      output,
      watch: options.watch,
      mode: options.mode,
      extensions: options.ext,
    })
  })
```

### 2.2 Yargs 风格（中间件模式）

```typescript
// yargs 更适合复杂场景
import yargs from "yargs"
import { hideBin } from "yargs/helpers"

yargs(hideBin(process.argv))
  .command(
    "serve [port]",
    "启动开发服务器",
    (yargs) => {
      return yargs
        .positional("port", { type: "number", default: 3000, describe: "端口号" })
        .option("open", { type: "boolean", alias: "o", describe: "自动打开浏览器" })
        .middleware([
          async (argv) => {
            // 全局前置中间件：检查端口占用
            if (await isPortInUse(argv.port)) {
              process.exit(1)
            }
            return argv
          },
        ])
    },
    async (argv) => {
      await startServer(argv.port, argv.open)
    }
  )
  .strict()       // 不允许未知参数
  .demandCommand(1, "至少需要一个命令")  // 至少一个命令
  .help()
  .parse()
```

## 三、交互式命令行

### 3.1 Inquirer.js — 问答式交互

```typescript
import inquirer from "inquirer"
import chalk from "chalk"

async function createProjectInteractive() {
  const answers = await inquirer.prompt([
    {
      type: "input",
      name: "name",
      message: "项目名称",
      default: "my-app",
      validate: (input: string) => {
        if (!/^[a-z][a-z0-9-]*$/.test(input)) {
          return "只允许小写字母、数字和连字符"
        }
        return true
      },
    },

    {
      type: "list",
      name: "template",
      message: "选择模板",
      choices: [
        { name: "React + TypeScript", value: "react-ts" },
        { name: "Vue 3 + TypeScript", value: "vue-ts" },
        { name: "Node.js API", value: "node-api" },
        { name: new chalk.dim("更多模板..."), value: "more" },
      ],
      filter: (val) => val === "more" ? null : val,
    },

    {
      type: "checkbox",
      name: "features",
      message: "选择功能",
      choices: [
        { name: "ESLint + Prettier", value: "lint" },
        { name: "Testing (Vitest)", value: "test" },
        { name: "Docker Support", value: "docker" },
        { name: "GitHub Actions CI", value: "ci" },
      ],
    },

    {
      type: "confirm",
      name: "gitInit",
      message: "初始化 Git 仓库？",
      default: true,
    },

    {
      type: "password",
      name: "token",
      message: "输入 API Token（可选）",
      mask: "*",
      when: () => false, // 条件性出现
    },
  ])

  console.log("\n" + chalk.green("✅ 配置完成！"))
  console.log(chalk.gray(JSON.stringify(answers, null, 2)))
}
```

### 3.2 Clack — 现代化 UI 库

```bash
npm install @clack/prompts
```

```typescript
import { text, select, confirm, multiselect, spinner, note } from "@clack/prompts"
import { bgMagenta, bold } from "chalk"

async function modernCli() {
  // 标题横幅
  console.log(`
  ${bgMagenta(bold(" My Awesome CLI "))} v1.0.0
`)

  const name = await text({
    message: "项目名称",
    placeholder: "my-project",
    validate(value) {
      if (value.length === 0) return "不能为空"
    },
  })

  const template = await select({
    message: "选择模板",
    options: [
      { value: "react", label: "React", hint: "最流行的选择" },
      { value: "vue", label: "Vue 3", hint: "渐进式框架" },
      { value: "svelte", label: "Svelte", hint: "编译时框架" },
    ],
  })

  const features = await multiselect({
    message: "额外功能",
    required: true,
    options: [
      { value: "ts", label: "TypeScript", checked: true },
      { value: "tailwind", label: "Tailwind CSS" },
      { value: "router", label: "Router" },
      { value: "pinia", label: "Pinia (状态管理)" },
    ],
  })

  const shouldDeploy = await confirm({ message: "立即部署？" })

  if (shouldDeploy) {
    const s = spinner()
    s.start("正在部署...")
    await sleep(2000)
    s.stop("部署成功！")
  }

  note(`项目 ${name} 已创建在 ./${name}`, "下一步")
}
```

## 四、输出美化

### 4.1 Chalk — 终端颜色

```typescript
import chalk from "chalk"

// 基础颜色
console.log(chalk.red("错误信息"))
console.log(chalk.green("成功"))
console.log(chalk.yellow("警告"))

console.log(`
```

### 4.2 Ora — 加载动画

```typescript
import ora from "ora"

const spinner = ora("加载中...").start()

await someAsyncTask()

spinner.succeed("加载完成!")

// 其他状态
// spinner.fail("操作失败")
// spinner.warn("有警告信息")
// spinner.info("提示信息")
// spinner.stop() // 静默停止

// 文本更新
let dots = ""
const interval = setInterval(() => {
  dots = ".".repeat(((dots.length + 1) % 4))
  spinner.text = `下载中${dots}`
}, 200)

// 完成后清除定时器
clearInterval(interval)
spinner.succeed("下载完毕!")
```

### 4.3 表格输出

```typescript
// 使用 cliui 或自建表格
function printTable(data: Record<string, string>[]) {
  if (data.length === 0) {
    console.log(chalk.gray("(无数据)"))
    return
  }

  const headers = Object.keys(data[0])
  const colWidths = headers.map(h =>
    Math.max(h.length, ...data.map(row => String(row[h]).length)) + 2
  )

  // 表头
  const headerRow = headers.map((h, i) => 
    chalk.bold(h.padEnd(colWidths[i]))
  ).join("│")

  // 分隔线
  const separator = colWidths.map(w => "─".repeat(w)).join("┼")

  // 数据行
  const dataRows = data.map(row =>
    headers.map((h, i) => String(row[h]).padEnd(colWidths[i])).join("│")
  )

  console.log(headerRow)
  console.log(separator)
  console.log(dataRows.join("\n" + separator + "\n"))
}

printTable([
  { name: "Vite", language: "TS", stars: "73k" },
  { name: "Webpack", language: "JS", stars: "64k" },
  { name: "esbuild", language: "Go", stars: "44k" },
])
```
