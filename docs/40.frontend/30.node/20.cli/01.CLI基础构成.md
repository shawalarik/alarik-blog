## 概述

CLI（Command Line Interface，命令行接口）是用户通过文本命令与程序进行交互的方式。Node.js 提供了丰富的内置模块来构建 CLI 工具，理解其基础构成是开发命令行工具的第一步。

## 命令行参数解析（process.argv）

`process.argv` 是一个数组，包含启动 Node.js 进程时传入的命令行参数。

```js
// argv.js
console.log(process.argv);
```

执行 `node argv.js one two three`，输出：

```
[
  'C:\\Program Files\\node\\node.exe',  // process.argv[0]: 可执行文件路径
  'D:\\project\\argv.js',               // process.argv[1]: 脚本文件路径
  'one',                                // process.argv[2]: 第一个参数
  'two',                                // process.argv[3]: 第二个参数
  'three'                               // process.argv[4]: 第三个参数
]
```

### 解析参数

实际使用中通常需要跳过前两个元素：

```js
const args = process.argv.slice(2);

args.forEach((arg, index) => {
  console.log(`参数 ${index}: ${arg}`);
});
```

### 简单的键值对解析

```js
function parseArgs(args) {
  const result = {};
  for (let i = 0; i < args.length; i++) {
    if (args[i].startsWith('--')) {
      const key = args[i].slice(2);
      // 处理 --key=value 格式
      if (key.includes('=')) {
        const [k, v] = key.split('=');
        result[k] = v;
      } else if (args[i + 1] && !args[i + 1].startsWith('-')) {
        // 处理 --key value 格式
        result[key] = args[++i];
      } else {
        // 布尔标志
        result[key] = true;
      }
    } else if (args[i].startsWith('-') && args[i].length > 2) {
      // 处理短选项组合如 -abc
      const chars = args[i].slice(1).split('');
      chars.forEach(char => {
        result[char] = true;
      });
    }
  }
  return result;
}

const parsed = parseArgs(process.argv.slice(2));
console.log(parsed);
// node cli.js --port=3000 -v --name test
// 输出: { port: '3000', v: true, name: 'test' }
```

## 输出流（stdout / stderr）

Node.js 提供三个标准流：

| 流 | 描述 | 文件描述符 |
|---|---|---|
| `process.stdin` | 标准输入 | 0 |
| `process.stdout` | 标准输出 | 1 |
| `process.stderr` | 标准错误输出 | 2 |

### stdout 与 stderr 的区别

```js
// 正常输出使用 stdout
console.log('这是正常信息');       // 内部写入 stdout
process.stdout.write('直接输出\n');

// 错误信息使用 stderr
console.error('这是错误信息');     // 内部写入 stderr
process.stderr.write('错误输出\n');
```

**为什么区分？**

- `stdout`：程序正常输出，可通过 `>` 重定向到文件
- `stderr`：诊断和错误信息，即使重定向 stdout 也会显示

```bash
# 只将正常输出写入文件，错误仍显示在终端
node app.js > output.log 2>&1

# 将正常和错误输出分别重定向
node app.js > output.log 2> error.log
```

### 实际应用示例

```js
function log(message) {
  process.stdout.write(`${message}\n`);
}

function error(message) {
  process.stderr.write(`\x1b[31m[ERROR]\x1b[0m ${message}\n`);
}

function warn(message) {
  process.stderr.write(`\x1b[33m[WARN]\x1b[0m ${message}\n`);
}

log('服务已启动...');
warn('配置文件未找到，使用默认配置');
error('连接数据库失败');
```

## 退出码

进程退出时会返回一个退出码，表示程序执行状态：

| 退出码 | 含义 |
|---|---|
| 0 | 成功 |
| 1 | 通用错误 |
| 2 | 命令用法不正确 |
| 127 | 命令未找到 |
| 128+N | 被信号 N 终止 |
| 130 | 被 Ctrl+C 终止（SIGINT=2 + 128） |

### 使用退出码

```js
// 成功退出
process.exit(0);  // 或 process.exit()

// 错误退出
process.exit(1);

// 自定义退出码
function validateConfig(config) {
  if (!config.port || config.port < 1024 || config.port > 65535) {
    process.stderr.write('Error: 无效的端口号\n');
    process.exit(2);
  }
}
```

### 进程退出事件

```js
process.on('exit', (code) => {
  console.log(`进程退出，退出码: ${code}`);
});

process.on('uncaughtException', (err) => {
  console.error('未捕获异常:', err.message);
  process.exit(1);
});

// SIGINT 信号处理（Ctrl+C）
process.on('SIGINT', () => {
  console.log('\n收到终止信号，正在清理资源...');
  // 执行清理操作
  cleanup();
  process.exit(130);
});
```

## 交互式输入（readline）

`readline` 模块提供了逐行读取流的接口，适合创建交互式命令行。

### 基本用法

```js
const readline = require('readline');

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

rl.question('请输入你的名字: ', (answer) => {
  console.log(`你好, ${answer}!`);
  rl.close();
});
```

### 多步骤交互

```js
const readline = require('readline');

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

async function prompt(question) {
  return new Promise((resolve) => {
    rl.question(question, resolve);
  });
}

async function main() {
  console.log('=== 项目初始化向导 ===\n');

  const name = await prompt('项目名称: ');
  const version = await prompt('版本号 (默认 1.0.0): ') || '1.0.0';
  const author = await prompt('作者: ');
  const description = await prompt('项目描述: ');

  console.log('\n=== 配置确认 ===');
  console.log(`名称: ${name}`);
  console.log(`版本: ${version}`);
  console.log(`作者: ${author}`);
  console.log(`描述: ${description}`);

  rl.close();
}

main();
```

### 密码输入（隐藏字符）

```js
const readline = require('readline');

rl.question('请输入密码: ', {
  hideEchoBack: true,   // 隐藏输入（需要特定终端支持）
  mask: '*',            // 用 * 替代显示
}, (password) => {
  console.log(`密码已接收，长度: ${password.length}`);
  rl.close();
});
```

### 读取多行输入

```js
const readline = require('readline');

const rl = readline.createInterface({
  input: process.stdin,
  terminal: false,  // 非终端模式，适合管道输入
});

const lines = [];

rl.on('line', (line) => {
  lines.push(line);
});

rl.on('close', () => {
  console.log(`共读取 ${lines.length} 行`);
  console.log(lines.join('\n'));
});
```

## 完整示例：简易 CLI 工具

```js
#!/usr/bin/env node

const readline = require('readline');

// 解析命令行参数
function parseArgs(argv) {
  const args = argv.slice(2);
  const options = { help: false, version: false };

  for (let i = 0; i < args.length; i++) {
    switch (args[i]) {
      case '-h':
      case '--help':
        options.help = true;
        break;
      case '-v':
      case '--version':
        options.version = true;
        break;
      default:
        options.input = args[i];
    }
  }

  return options;
}

// 显示帮助信息
function showHelp() {
  const help = `
Usage: my-cli [options] [input]

Options:
  -h, --help     显示帮助信息
  -v, --version  显示版本号

Examples:
  $ my-cli hello
  $ my-cli --version
`;
  process.stdout.write(help);
}

// 交互模式
function interactiveMode() {
  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
  });

  rl.question('> ', async (input) => {
    if (input === 'exit') {
      rl.close();
      process.exit(0);
    }
    process.stdout.write(`你输入了: ${input}\n\n`);
    interactiveMode();  // 循环等待输入
  });

  rl.on('close', () => {
    process.stdout.write('再见!\n');
    process.exit(0);
  });
}

// 主函数
function main() {
  const options = parseArgs(process.argv);

  if (options.help) {
    showHelp();
    process.exit(0);
  }

  if (options.version) {
    process.stdout.write('my-cli v1.0.0\n');
    process.exit(0);
  }

  if (options.input) {
    process.stdout.write(`处理输入: ${options.input}\n`);
    process.exit(0);
  }

  // 无参数进入交互模式
  interactiveMode();
}

main();
```

## 小结

| 模块/对象 | 用途 | 核心API |
|---|---|---|
| `process.argv` | 获取命令行参数 | 数组访问、slice |
| `process.stdout` / `stderr` | 输出流管理 | write() |
| `process.exit()` | 设置退出码 | exit(code) |
| `readline` | 交互式输入 | createInterface()、question() |

掌握这些基础知识后，可以进一步学习专业的命令行参数解析库（如 Commander.js、Yargs）来构建更复杂的 CLI 工具。
