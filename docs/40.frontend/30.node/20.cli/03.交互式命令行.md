## 概述

优秀的 CLI 工具不仅要能正确解析参数，还需要良好的交互体验——包括友好的问题引导、清晰的进度展示和即时的视觉反馈。本章介绍 Inquirer.js、Ora 等交互式组件的使用方法。

## Inquirer.js 问题类型

Inquirer.js 是最流行的交互式命令行界面套件，通过问答式流程收集用户输入。

### 安装

```bash
npm install inquirer @inquirer/core @inquirer/prompts
```

> 注意：Inquirer 已拆分为独立包，`inquirer` 为兼容层，新项目可直接使用各子包。

### input 类型 — 文本输入

```js
import input from '@inquirer/input';

const answer = await input({
  message: '请输入项目名称:',
  default: 'my-project',
  required: true,           // 必填
  validate: (value) => {
    if (!/^[a-z][a-z0-9-]*$/.test(value)) {
      return '只能包含小写字母、数字和连字符，且以字母开头';
    }
    return true;
  },
  transformer: (value) => {
    // 显示时的变换效果
    return `📁 ${value}`;
  }
});

console.log('项目名称:', answer);
```

### confirm 类型 — 确认提示

```js
import confirm from '@inquirer/confirm';

const shouldContinue = await confirm({
  message: '是否继续?',
  default: false,
});

if (shouldContinue) {
  console.log('用户选择了继续');
} else {
  console.log('用户取消了操作');
  process.exit(0);
}
```

### list 类型 — 单选列表

```js
import select from '@inquirer/select';

const framework = await select({
  message: '选择前端框架:',
  choices: [
    { name: 'Vue 3', value: 'vue', description: '渐进式 JavaScript 框架' },
    { name: 'React', value: 'react', description: '用于构建用户界面的库' },
    { name: 'Svelte', value: 'svelte', description: '编译时框架' },
    { name: 'Angular', value: 'angular', description: '企业级平台' },
    { name: 'Nuxt', value: 'nuxt', disabled: true },  // 禁用选项
  ],
  default: 'vue',
  loop: false,  // 到达底部后不再循环
});

console.log('选择的框架:', framework);
```

### checkbox 类型 — 多选框

```js
import checkbox from '@inquirer/checkbox';

const features = await checkbox({
  message: '选择需要的功能 (空格选中):',
  choices: [
    { name: 'TypeScript 支持', value: 'typescript', checked: true },
    { name: 'ESLint', value: 'eslint', checked: true },
    { name: 'Prettier', value: 'prettier' },
    { name: 'Jest 单元测试', value: 'jest' },
    { name: 'CI/CD 配置', value: 'cicd' },
    { name: 'Docker', value: 'docker' },
  ],
  required: true,
  validate: (answers) => {
    if (answers.length === 0) {
      return '至少选择一项功能';
    }
    return true;
  },
});

console.log('选择的功能:', features);
```

### password 类型 — 密码输入

```js
import password from '@inquirer/password';

const token = await password({
  message: '请输入 API Token:',
  mask: '*',         // 掩码字符
  validate: (value) => {
    if (value.length < 16) {
      return 'Token 长度至少为 16 个字符';
    }
    return true;
  },
});
```

### 组合使用：完整脚手架流程

```js
import input from '@inquirer/input';
import select from '@inquirer/select';
import checkbox from '@inquirer/checkbox';
import confirm from '@inquirer/confirm';
import password from '@inquirer/password';

async function scaffold() {
  console.log('\n🚀 项目初始化向导\n');

  const projectName = await input({
    message: '项目名称:',
    default: 'my-app',
    validate: (val) => /^[a-z][\w-]*$/.test(val) ? true : '命名不规范',
  });

  const framework = await select({
    message: '选择框架:',
    choices: [
      { name: 'Vue 3', value: 'vue' },
      { name: 'React', value: 'react' },
      { name: 'Solid', value: 'solid' },
    ],
  });

  const features = await checkbox({
    message: '选择功能:',
    choices: [
      { name: 'TypeScript', value: 'ts', checked: true },
      { name: 'Router', value: 'router' },
      { name: 'Pinia/Zustand', value: 'state' },
      { name: 'Testing', value: 'test' },
    ],
  });

  const usePrivateRegistry = await confirm({
    message: '是否使用私有 npm registry?',
    default: false,
  });

  let registryToken = null;
  if (usePrivateRegistry) {
    registryToken = await password({
      message: '输入 Registry Token:',
    });
  }

  const confirmed = await confirm({
    message: `\n确认创建项目 "${projectName}"?`,
    default: true,
  });

  if (!confirmed) {
    console.log('取消操作');
    return;
  }

  // 输出汇总
  console.log('\n========== 配置摘要 ==========');
  console.log(`项目名称: ${projectName}`);
  console.log(`框架: ${framework}`);
  console.log(`功能: ${features.join(', ')}`);
  console.log(`私有仓库: ${usePrivateRegistry ? '是' : '否'}`);

  // 执行创建逻辑...
}

scaffold().catch(console.error);
```

## 进度条（Ora）

Ora 提供优雅的终端加载动画，常用于异步操作的进度反馈。

### 安装

```bash
npm install ora
```

### 基础用法

```js
import ora from 'ora';

const spinner = ora('正在下载资源...').start();

try {
  // 模拟耗时操作
  await new Promise(resolve => setTimeout(resolve, 2000));
  spinner.succeed('资源下载完成!');
} catch (error) {
  spinner.fail('下载失败: ' + error.message);
}
```

### 状态控制

```js
const spinner = ora();

spinner.start();              // 开始并显示默认文本
spinner.text = '处理中...';   // 更新文本
spinner.color = 'yellow';     // 更改颜色
spinner.indent = 4;           // 缩进

spinner.succeed('完成!');     // 绿色 ✓ 完成
spinner.fail('失败!');        // 红色 ✗ 失败
spinner.warn('警告!');        // 黄色 ⚠ 警告
spinner.info('提示!');        // 蓝色 ℹ 信息
spinner.stop();               // 停止并清除
spinner.stopAndPersist({      // 停止并保留最后一帧
  symbol: '❤',
  text: '已完成'
});
```

### 自定义样式

```js
import ora from 'ora';

const spinner = ora({
  text: '加载中...',
  prefixText: '[my-app]',     // 前缀文字
  spinner: 'moon',            // 预设动画样式
  color: 'cyan',             // 文字颜色
  hideCursor: true,          // 隐藏光标
  interval: 120,             // 刷新间隔(ms)
  isSilent: false,           // 静默模式
}).start();
```

Ora 内置多种预设动画：`dots`（默认）、`line`、`arrow`、`bouncingBar`、`material`、`moon`、`runner` 等。

### Promise 包装

```js
import ora from 'ora';

// 直接包装 Promise
await ora('安装依赖中...').run(async () => {
  await execAsync('npm install');
});

// 使用 start/promise 模式
const result = await ora({
  text: '获取远程数据...',
  discardStdin: false,  // 不丢弃 stdin
}).start().promise(
  fetchData(),
  { success: '获取成功!', fail: '获取失败!' }
);
```

## 加载动画进阶

### MultiProgress 多任务进度

```bash
npm install cli-progress
```

```js
import cliProgress from 'cli-progress';
import { MultiBar } from 'cli-progress';

const multiBar = new MultiBar({
  format: '{bar} {percentage}% | {filename}',
  barCompleteChar: '\u2588',
  barIncompleteChar: '\u2591',
  hideCursor: true,
});

// 创建多个进度条
const bar1 = multiBar.create(100, 0, { filename: 'file1.zip' });
const bar2 = multiBar.create(50, 0, { filename: 'file2.zip' });
const bar3 = multiBar.create(30, 0, { filename: 'file3.zip' });

// 分别更新
bar1.update(40);
bar2.update(20);
bar3.update(15);

// 全部完成后关闭
multiBar.stop();
```

### Listr 任务列表

Listr 可以按顺序或并行地展示多个任务的执行状态。

```bash
npm install listr
```

```js
import Listr from 'listr';

const tasks = new Listr([
  {
    title: '安装依赖',
    task: () =>
      new Listr([
        {
          title: '安装 npm 包',
          task: () => execAsync('npm install'),
        },
        {
          title: '验证依赖',
          task: () => execAsync('npm ls'),
        },
      ], { concurrent: false }),
  },
  {
    title: '运行构建',
    task: () => execAsync('npm run build'),
  },
  {

    skip: () => !hasTestScript && '没有测试脚本',
    task: () => execAsync('npm test'),
  },
  {
    title: '发布',
    task: () => publishPackage(),
  },
], { exitOnError: false });

tasks.run().catch(console.error);
```

## 用户输入校验

### 校验策略

```js
class Validator {
  static required(msg = '此字段必填') {
    return (val) => (val && String(val).trim()) ? true : msg;
  }

  static number(min, max) {
    return (val) => {
      const num = Number(val);
      if (isNaN(num)) return '请输入有效数字';
      if (min !== undefined && num < min) return `不能小于 ${min}`;
      if (max !== undefined && num > max) return `不能大于 ${max}`;
      return true;
    };
  }

  static url(msg = 'URL 格式不正确') {
    try {
      new URL(val);
      return true;
    } catch {
      return msg;
    }
  }

  static email(msg = '邮箱格式不正确') {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(val) || msg;
  }

  static pattern(regex, msg) {
    return regex.test(val) || msg;
  }

  static minLength(n) {
    return val => String(val).length >= n || `长度至少 ${n} 位`;
  }

  static maxLength(n) {
    return val => String(val).length <= n || `长度不能超过 ${n} 位`;
  }
}

// 使用
import input from '@inquirer/input';

const age = await input({
  message: '年龄:',
  validate: Validator.number(0, 150),
});
```

### 校验装饰器（链式组合）

```js
function composeValidators(...validators) {
  return (value) => {
    for (const validator of validators) {
      const result = validator(value);
      if (result !== true) return result;
    }
    return true;
  };
}

// 使用
const portValidator = composeValidators(
  Validator.required(),
  Validator.number(1024, 65535),
);

const port = await input({
  message: '端口号:',
  validate: portValidator,
});
```

## 小结

| 库/工具 | 功能 | 适用场景 |
|---------|------|---------|
| **Inquirer** | 交互式问答（7+ 种问题类型） | 脚手架、配置向导 |
| **Ora** | 加载旋转动画 | 异步操作反馈 |
| **cli-progress** | 多进度条 | 并行下载/上传 |
| **Listr** | 任务列表状态 | 多步骤构建流程 |
