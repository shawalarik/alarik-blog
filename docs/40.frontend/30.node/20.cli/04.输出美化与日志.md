## 概述

CLI 工具的输出质量直接影响用户体验。纯文本输出虽然功能完备，但缺乏视觉层次感。本章介绍颜色样式、边框装饰、表格布局以及结构化日志管理等美化技术。

## Chalk 颜色样式

Chalk 是终端字符串样式化的事实标准库，被数万个 npm 包依赖。

### 安装

```bash
npm install chalk
```

> Chalk v5+ 为 ESM only，CommonJS 项目需使用 v4。

### 基础颜色

```js
import chalk from 'chalk';

console.log(chalk.red('红色文字'));
console.log(chalk.green('绿色文字'));
console.log(chalk.blue('蓝色文字'));
console.log(chalk.yellow('黄色文字'));
console.log(chalk.cyan('青色文字'));
console.log(chalk.magenta('洋红'));
console.log(chalk.white('白色'));
console.log(chalk.gray('灰色'));
console.log(chalk.black('黑色'));

// 背景色
console.log(chalk.bgRed('红底白字'));
console.log(chalk.bgGreen.bgBlue('背景叠加'));
```

### 样式修饰

```js
console.log(chalk.bold('粗体'));          // 加粗
console.log(chalk.italic('斜体'));        // 斜体
console.log(chalk.underline('下划线'));   // 下划线
console.log(chalk.strikethrough('删除线')); // 删除线
console.log(chalk.dim('暗淡'));           // 降低亮度
console.log(chalk.inverse('反转'));       // 反转前景/背景色
console.log(chalk.hidden('隐藏'));        // 不可见（用于密码）
console.log(chalk.overline('上划线'));    // 上划线
```

### 链式调用

```js
// 组合多种样式
console.log(chalk.red.bold.underline('红色粗体下划线'));

// 嵌套使用（不会污染外部样式）
console.log(chalk.red(`正常红色 ${chalk.bold('加粗')} 继续红色`));

// RGB 颜色
console.log(chalk.rgb(255, 136, 0).bold('自定义橙色'));

// HEX 颜色
console.log(chalk.hex('#FF8800').bgHex('#1a1a2e').bold('主题色'));

// ANSI 256 色
console.log(chalk.ansi256(196).bgAnsi256(236)('256 色模式'));
```

### 模板字面量扩展

```js
const name = 'World';
const error = chalk.red.bold;
const warning = chalk.yellow;

console.log(`
  ${chalk.blue('Hello')} ${chalk.green(name)}!
  ${error('错误:')} 连接失败
  ${warning('警告:')} 配置文件过期
`);

// 使用 tagged template literal
import chalkTemplate from 'chalk/source/tags.js';

const version = '2.0.0';
console.log(chalkTemplate`
  {red 错误:} 操作失败
  {green 成功:} 版本 {bold ${version}} 已部署
  {yellow 警告:} 即将废弃的 API
  {blue 信息:} 查看 {underline https://example.com/docs}
`);
```

### 条件着色

```js
import chalk from 'chalk';

// 根据级别着色
function logLevel(level, ...args) {
  const colors = {
    info: chalk.blue,
    warn: chalk.yellow,
    error: chalk.red,
    success: chalk.green,
  };
  const color = colors[level] || chalk.white;
  console.log(color(`[${level.toUpperCase()}]`), ...args);
}

// 支持禁用所有颜色（CI 环境、管道输出）
if (!process.stdout.isTTY) {
  chalk.level = 0;  // 禁用颜色
}
```

Chalk 的 level 系统：

| Level | 说明 |
|-------|------|
| 0 | 禁用所有颜色 |
| 1 | 基础 16 色 |
| 2 | ANSI 256 色 |
| 3 | TrueColor (1600万色) |

## Boxen 边框

Boxen 用于给文本添加漂亮的边框盒子，适合用于标题、公告、重要提示等场景。

### 安装

```bash
npm install boxen
```

### 基础用法

```js
import boxen from 'boxen';

console.log(boxen('Hello, World!', {
  padding: 1,           // 内边距
  margin: 1,            // 外边距
  borderStyle: 'round', // 边框样式
  borderColor: 'green', // 边框颜色
  borderBackground: false,
}));
```

输出效果：

```
╭──────────────────────╮
│                      │
│   Hello, World!      │
│                      │
╰──────────────────────╯
```

### 边框样式

Boxen 提供多种预设边框样式：

```js
const styles = [
  'single',      // ┌─┐ │ └─┘
  'double',      // ╔═╗ ║ ╚═╝
  'round',       // ╭─╮ │ ╰─╯
  'bold',        // ┏━┓ ┃ ┗━┛
  'singleDouble',// ╭═╮ │ ╰═╯
  'doubleSingle',// ╔─║ ║ ╚─╝
  'classic',     // +-+ | + -+
  'arrow',       // >-- | >--
  'dots',        // ··· · ···
];

styles.forEach(style => {
  console.log(boxen(`Style: ${style}`, { borderStyle: style }));
});
```

### 实际应用示例

```js
import boxen from 'boxen';
import chalk from 'chalk';

// 欢迎信息
function showWelcome(version) {
  console.log(boxen(
    chalk.cyan.bold(' My Awesome CLI ') +
    chalk.gray(` v${version}\n\n`) +
    chalk.white('快速、现代的命令行工具') +
    chalk.dim('\nhttps://github.com/user/my-cli'),
    {
      padding: 1,
      margin: { top: 1, bottom: 1, left: 2, right: 2 },
      borderStyle: 'round',
      borderColor: 'cyan',
      titleAlignment: 'center',
    }
  ));
}

// 错误提示
function showError(message) {
  console.error(boxen(
    chalk.red.bold(' ERROR ') + chalk.red(` ${message}`),
    {
      padding: { top: 0, bottom: 0, left: 1, right: 1 },
      borderStyle: 'single',
      borderColor: 'red',
    }
  ));
}

// 成功提示
function showSuccess(message) {
  console.log(boxen(
    chalk.green.bold(' SUCCESS ') + chalk.green(` ${message}`),
    {
      padding: { top: 0, bottom: 0, left: 1, right: 1 },
      borderStyle: 'round',
      borderColor: 'green',
    }
  ));
}
```

## CLI 表格（cli-table3）

cli-table3 用于在终端中渲染美观的表格数据。

### 安装

```bash
npm install cli-table3
```

### 基础用法

```js
import Table from 'cli-table3';

const table = new Table({
  head: ['名称', '版本', '许可证'],
  colWidths: [20, 12, 15],
});

table.push(
  ['vue', '^3.4.0', 'MIT'],
  ['react', '^18.2.0', 'MIT'],
  ['typescript', '^5.3.0', 'Apache-2.0'],
);

console.log(table.toString());
```

输出效果：

```
┌──────────────────┬────────────┬───────────────┐
│ 名称             │ 版本       │ 许可证        │
├──────────────────┼────────────┼───────────────┤
│ vue              │ ^3.4.0     │ MIT           │
│ react            │ ^18.2.0    │ MIT           │
│ typescript       │ ^5.3.0     │ Apache-2.0    │
└──────────────────┴────────────┴───────────────┘
```

### 高级表格

```js
import Table from 'cli-table3';
import chalk from 'chalk';

const table = new Table({
  characters: {
    top: '═', 'top-mid': '╦', 'top-left': '╔', 'top-right': '╗',
    bottom: '═', 'bottom-mid': '╩', 'bottom-left': '╚', 'bottom-right': '╝',
    left: '║', 'left-mid': '╠',
    mid: '─', 'mid-mid': '╬',
    right: '║', 'right-mid': '╣',
    middle: '│',
  },
  style: { 'padding-left': 1, 'padding-right': 1 },
  head: [
    chalk.cyan('Package'),
    chalk.cyan('Version'),
    chalk.cyan('Status'),
  ],
});

table.push(
  [{ colSpan: 1, content: chalk.green('✓ core'), hAlign: 'center' }],
  [{ colSpan: 1, content: chalk.green('✓ utils'), hAlign: 'center' }],
  [{ colSpan: 1, content: chalk.yellow('⚠ deprecated'), hAlign: 'center' }],
);

console.log(table.toString());
```

## 日志分级（Consola / Signale）

### Consola

Consola 是基于 chalk 的增强日志库，提供日志级别过滤、 reporters 系统等能力。

#### 安装

```bash
npm install consola
```

#### 基础用法

```js
import consola from 'consola';

consola.info('普通信息');
consola.success('操作成功');
consola.warn('警告信息');
consola.error('错误信息');
consola.fatal('致命错误');
consola.start('开始处理');
consola.debug('调试信息');
consola.trace('堆栈追踪');
consola.ready('就绪');
consola.box('带框的信息');

// 占位符
consola.info('用户 %s 已登录，ID: %d', 'Alice', 1234);
consola.info('配置详情: %O', { port: 3000, host: 'localhost' });

// 日志级别
consola.level = consolaLogLevel.Fatal;  // 只显示致命及以上
```

#### 自定义 Reporter

```js
import consola from 'consola';

// 同时输出到终端和文件
consola.wrapConsole();
consola.reporters.push(
  new consola.FancyReporter(),      // 彩色终端输出
  new consola.JSONReporter(),       // JSON 格式（适合 CI）
  new consola.BasicReporter(),      // 简洁输出
);
```

### Signale

Signale 提供更丰富的预设日志类型和可定制的时间戳、徽章等。

#### 安装

```bash
npm install signale
```

#### 基础用法

```js
import signale from 'signale';

signale.success('Operation successful');
signale.debug('Debugging info');
signale.pending('Write release notes for %s', '1.2.0');
signale.watch('File %s changed', 'config.json');
signale.complete('Task complete');

// 定时器
const timer = signale.start('Building project');
setTimeout(() => timer.complete('Build finished!'), 2000);

// 作用域实例
const logger = signale.scope('database', 'mysql');
logger.info('Connection established');
logger.error('Query failed');
```

#### 自定义 Logger

```js
import Signale from 'signale';

const custom = new Signale({
  types: {
    deploy: {
      badge: '🚀',
      color: 'magenta',
      label: 'deploy',
      logLevel: 'info',
    },
    rollback: {
      badge: '⏪',
      color: 'yellow',
      label: 'rollback',
      logLevel: 'warn',
    },
  },
  secrets: ['apiKey', 'token'],  // 自动隐藏敏感字段
  timestamp: () => new Date().toISOString(),
});

custom.deploy('Deploying to production');
custom.rollback('Rolling back to v1.0.0');
custom.info({ apiKey: 'sk-xxx', user: 'admin' });  // apiKey 会被隐藏
```

### 对比

| 特性 | Consola | Signale |
|------|---------|---------|
| 预设类型 | 8 种基础类型 | 20+ 预设类型 |
| 自定义类型 | 通过 reporter | 通过 types 配置 |
| 时间戳 | 可选 | 可选 |
| 敏感信息隐藏 | 手动处理 | 内置 secrets 支持 |
| 计时器 | 无 | 有（start/complete） |
| 作用域 | 无 | scope() 方法 |
| JSON 输出 | 内置 JSONReporter | 需自行配置 |

## 综合示例：统一输出管理

```js
import chalk from 'chalk';
import boxen from 'boxen';
import consola from 'consola';

class Logger {
  #name;

  constructor(name) {
    this.#name = name;
  }

  #format(level, message, meta) {
    const timestamp = new Date().toLocaleTimeString();
    const colors = {
      info: chalk.blue,
      success: chalk.green,
      warn: chalk.yellow,
      error: chalk.red,
    };
    const icon = { info: 'ℹ', success: '✔', warn: '⚠', error: '✗' };
    const color = colors[level] || chalk.white;
    return `${color(`[${timestamp}]`)} ${icon[level]} ${message}`;
  }

  info(message) {
    console.log(this.#format('info', message));
  }

  success(message) {
    console.log(this.#format('success', message));
  }

  warn(message) {
    console.warn(this.#format('warn', message));
  }

  error(message) {
    console.error(this.#format('error', message));
  }

  banner(text, options = {}) {
    const defaults = { style: 'round', color: 'cyan', padding: 1 };
    const opts = { ...defaults, ...options };
    console.log(boxen(chalk[opts.color].bold(text), {
      padding: opts.padding,
      borderStyle: opts.style,
      borderColor: opts.color,
    }));
  }

  table(headers, rows) {
    import('cli-table3').then(({ default: Table }) => {
      const t = new Table({ head: headers.map(h => chalk.cyan(h)) });
      rows.forEach(row => t.push(row));
      console.log(t.toString());
    });
  }
}

export const logger = new Logger('my-cli');

// 使用
logger.banner('My CLI Tool v1.0');
logger.info('正在初始化...');
logger.success('构建成功！');
logger.warn('使用了过期的配置');
logger.error('网络请求失败');
```

## 小结

| 工具 | 职责 | 核心价值 |
|------|------|---------|
| **Chalk** | 字符串样式化 | 颜色、粗体、斜体等终端样式 |
| **Boxen** | 边框盒子 | 标题、公告、重要提示的视觉强调 |
| **cli-table3** | 表格渲染 | 结构化数据的美观呈现 |
| **Consola** | 结构化日志 | 日志级别过滤、多 reporter 输出 |
| **Signale** | 高级日志 | 丰富预设、计时器、作用域、敏感信息隐藏 |

合理组合这些工具可以让 CLI 输出既美观又专业。下一章将探讨如何通过子进程执行系统命令。
