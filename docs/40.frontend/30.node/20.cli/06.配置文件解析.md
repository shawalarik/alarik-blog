## 概述

CLI 工具通常需要支持多种方式的配置输入：代码中的默认值、项目配置文件、用户级配置、环境变量和命令行参数。理解如何正确读取和合并这些配置源，是构建专业 CLI 工具的关键技能。

## 读取 package.json

`package.json` 是 Node.js 项目中最常见的配置载体，许多工具从中读取元数据和自定义配置。

### 读取自身 package.json

```js
import { readFileSync } from 'fs';
import { fileURLToPath } from 'url';
import { dirname, join } from 'path';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const pkgPath = join(__dirname, '..', 'package.json');
const pkg = JSON.parse(readFileSync(pkgPath, 'utf-8'));

console.log(`工具名称: ${pkg.name}`);
console.log(`版本号: ${pkg.version}`);
console.log(`描述: ${pkg.description}`);
```

### 读取项目的 package.json

```js
import { readFileSync, existsSync } from 'fs';
import { resolve } from 'path';

function findPkgJson(startDir) {
  let dir = resolve(startDir);

  while (dir !== dirname(dir)) {
    const pkgPath = join(dir, 'package.json');
    if (existsSync(pkgPath)) {
      return JSON.parse(readFileSync(pkgPath, 'utf-8'));
    }
    dir = dirname(dir);
  }

  return null;
}

const pkg = findPkgJson(process.cwd());

// 读取工具自定义字段（convention: "工具名"Config）
const myCliConfig = pkg?.myCliConfig;
const customField = pkg?.['my-cli'];  // 也可能用连字符
```

## JSON / YAML / TOML 解析

### JSON 配置

```js
import { readFileSync, existsSync } from 'fs';
import { resolve } from 'path';

function loadJsonConfig(filename) {
  const filePath = resolve(process.cwd(), filename);
  if (!existsSync(filePath)) return null;

  const content = readFileSync(filePath, 'utf-8');
  return JSON.parse(content);
}

const config = loadJsonConfig('.myclirc.json');
```

支持注释的 JSON（JSONC）:

```bash
npm install jsonc-parser
```

```js
import { parse as parseJsonc } from 'jsonc-parser';

const content = readFileSync('.myclirc.json', 'utf-8');
// 支持注释和尾部逗号
const errors = [];
const config = parseJsonc(content, errors, { allowTrailingComma: true });

if (errors.length > 0) {
  console.warn('配置文件有语法错误');
}
```

### YAML 配置

YAML 因其可读性好，被广泛用作 CLI 配置格式。

```bash
npm install yaml js-yaml
```

```yaml
# .myclirc.yml
server:
  port: 3000
  host: localhost

logging:
  level: info
  file: ./logs/app.log

features:
  - auth
  - cache
  - metrics
```

```js
// 使用 yaml（推荐，ESM 原生支持）
import { readFile } from 'fs/promises';
import { parse } from 'yaml';

const content = await readFile('.myclirc.yml', 'utf-8');
const config = parse(content);
console.log(config.server.port);  // 3000
```

```js
// 使用 js-yaml（传统方案）
import yaml from 'js-yaml';
import { readFileSync } from 'fs';

const config = yaml.load(readFileSync('.myclirc.yml', 'utf-8'));
```

### TOML 配置

### 配置文件发现策略

```js
import { existsSync } from 'fs';
import { resolve } from 'path';

// 按优先级查找配置文件
const CONFIG_FILES = [
  '.myclirc.json',      // JSON 格式
  '.myclirc.yaml',      // YAML 格式
  '.myclirc.yml',       // YAML 格式
  '.myclirc.toml',      // TOML 格式
  '.myclirc',           // 无扩展名（通常为 JSON/YAML）
  'mycli.config.js',    // JS 动态配置
  'mycli.config.ts',    // TS 动态配置
  'mycli.config.cjs',   // CommonJS 动态配置
  'mycli.config.mjs',   // ESM 动态配置
];

function discoverConfig(dir = process.cwd()) {
  for (const file of CONFIG_FILES) {
    const filePath = resolve(dir, file);
    if (existsSync(filePath)) return { path: filePath, file };
  }
  return null;
}

const found = discoverConfig();
if (found) {
  console.log(`发现配置文件: ${found.file}`);
}
```

## 配置优先级

### 标准优先级链

CLI 工具应遵循业界公认的配置优先级（低→高）：

```
默认值（代码内） < 配置文件 < 环境变量 < 命令行参数
```

### 完整实现

```js
import { readFileSync, existsSync } from 'fs';
import { resolve, join } from 'path';
import { parse as parseYaml } from 'yaml';

// ====== 1. 默认值 ======
const DEFAULT_CONFIG = {
  port: 3000,
  host: 'localhost',
  open: false,
  verbose: false,
  output: 'dist',
  features: [],
};

// ====== 2. 配置文件 ======
function loadFileConfig() {
  const candidates = ['.myclirc.json', '.myclirc.yml', '.myclirc.yaml'];

  for (const file of candidates) {
    const path = resolve(process.cwd(), file);
    if (!existsSync(path)) continue;

    const raw = readFileSync(path, 'utf-8');
    if (file.endsWith('.json')) return JSON.parse(raw);
    if (file.endsWith('.yml') || file.endsWith('.yaml')) return parseYaml(raw);
  }

  return {};
}

// ====== 3. 环境变量 ======
function loadEnvConfig() {
  const envPrefix = 'MY_CLI_';  // 环境变量前缀
  const envConfig = {};

  for (const [key, value] of Object.entries(process.env)) {
    if (key.startsWith(envPrefix)) {
      // MY_CLI_PORT -> port
      const configKey = key
        .slice(envPrefix.length)
        .toLowerCase();

      // 自动类型转换
      envConfig[configKey] = coerceValue(value);
    }
  }

  return envConfig;
}

function coerceValue(value) {
  if (value === 'true') return true;
  if (value === 'false') return false;
  if (/^\d+$/.test(value)) return parseInt(value, 10);
  if (/^\d+\.\d+$/.test(value)) return parseFloat(value);
  return value;
}

// ====== 4. 命令行参数 ======
function loadCliArgs() {
  // 此处使用 Commander/Yargs 解析后的结果
  // 简化的手动演示
  const args = process.argv.slice(2);
  const cliConfig = {};

  for (let i = 0; i < args.length; i++) {
    if (args[i].startsWith('--')) {
      const key = args[i].slice(2).replace(/-([a-z])/g, (_, c) => c.toUpperCase());
      if (args[i + 1] && !args[i + 1].startsWith('-')) {
        cliConfig[key] = args[++i];
      } else {
        cliConfig[key] = true;
      }
    }
  }

  return cliConfig;
}

// ====== 合并配置 ======
function mergeConfig(sources) {
  return sources.reduce((merged, source) => deepMerge(merged, source), {});
}

function deepMerge(target, source) {
  const result = { ...target };

  for (const [key, value] of Object.entries(source)) {
    if (
      typeof value === 'object' &&
      value !== null &&
      !Array.isArray(value) &&
      typeof target[key] === 'object' &&
      target[key] !== null &&
      !Array.isArray(target[key])
    ) {
      result[key] = deepMerge(target[key], value);
    } else {
      result[key] = value;
    }
  }

  return result;
}

// ====== 主流程 ======
function resolveConfig() {
  const fileConfig = loadFileConfig();
  const envConfig = loadEnvConfig();
  const cliConfig = loadCliArgs();

  const finalConfig = mergeConfig([
    DEFAULT_CONFIG,
    fileConfig,
    envConfig,
    cliConfig,
  ]);

  return finalConfig;
}

const config = resolveConfig();
console.log(config);
```

### 使用 cosmiconfig（推荐）

`cosmiconfig` 是专门解决配置发现与加载问题的库，被 ESLint、Prettier、PostCSS 等主流工具使用。

```bash
npm install cosmiconfig
```

```js
import { cosmiconfig } from 'cosmiconfig';

const explorer = cosmiconfig('mycli', {
  searchPlaces: [
    'package.json',
    '.myclirc.json',
    '.myclirc.yaml',
    '.myclirc.yml',
    '.myclirc.toml',
    '.myclirc',
    'mycli.config.js',
    'mycli.config.ts',
    'mycli.config.cjs',
    'mycli.config.mjs',
  ],
  loaders: {
    '.toml': async (filepath) => {
      const { parse } = await import('smol-toml');
      return parse(await fs.readFile(filepath, 'utf-8'));
    },
  },
});

const result = await explorer.search(process.cwd());

if (result) {
  console.log('配置来源:', result.filepath);
  console.log('配置内容:', result.config);
  // result.isEmpty — 配置是否为空
  // result.config — 解析后的配置对象
} else {
  console.log('未找到配置文件，使用默认值');
}
```

### 使用 rc 库

`rc` 是另一个轻量级配置解决方案，遵循 `.rc` 文件约定。

```bash
npm install rc
```

```js
import rc from 'rc';

const config = rc('mycli', DEFAULT_CONFIG, {
  // 可指定额外配置路径
  config: [
    resolve(process.cwd(), 'mycli.config.json'),
  ]
});

// 自动查找顺序：
// 1. 命令行参数 (--mycli_port 8080)
// 2. 环境变量 (MY_CLI_port=8080)
// 3. ./myclirc (当前目录)
// 4. ~/.myclirc (用户主目录)
// 5. ~/.mycli/config (用户配置目录)
// 6. /etc/myclirc (全局配置)
// 7. package.json 的 "mycli" 字段
// 8. 默认值
```

## Schema 校验

使用 Zod 或 Joi 对配置进行类型校验：

```bash
npm install zod
```

```js
import { z } from 'zod';

const ConfigSchema = z.object({
  server: z.object({
    port: z.number().int().min(1).max(65535).default(3000),
    host: z.string().ip().default('localhost'),
  }),
  logging: z.object({
    level: z.enum(['debug', 'info', 'warn', 'error']).default('info'),
    file: z.string().optional(),
  }).optional(),
  features: z.array(z.string()).default([]),
});

type Config = z.infer<typeof ConfigSchema>;

// 使用
const rawConfig = loadFileConfig();
const validated = ConfigSchema.parse(rawConfig);  // 校验失败抛出异常
const safeResult = ConfigSchema.safeParse(rawConfig);  // 安全校验

if (!safeResult.success) {
  console.error('配置校验失败:');
  safeResult.error.errors.forEach(err => {
    console.error(`  ${err.path.join('.')}: ${err.message}`);
  });
  process.exit(1);
}
```

## 小结

| 层级 | 来源 | 优先级 | 特点 |
|------|------|--------|------|
| 1 | 代码默认值 | 最低 | 保证可用性 |
| 2 | 配置文件 | 低 | 持久化，团队共享 |
| 3 | 环境变量 | 中 | 适合 CI/CD、密钥 |
| 4 | 命令行参数 | 最高 | 临时覆盖，用户意图 |

| 格式 | 库 | 特点 |
|------|-----|------|
| JSON | 原生 / JSONC Parser | 广泛支持，可带注释 |
| YAML | yaml / js-yaml | 可读性强，支持复杂类型 |
| TOML | smol-toml | Rust 生态友好 |
| 发现与加载 | cosmiconfig / rc | 自动查找、多格式支持 |
| 校验 | Zod / Joi | 类型安全，错误提示友好 |
