## 概述

CLI 工具经常需要调用其他程序或 shell 命令——比如执行 `git clone`、运行 `npm install`、调用编译器等。Node.js 通过 `child_process` 模块提供了完整的子进程管理能力。本章从原生 API 出发，逐步介绍封装方案和跨平台最佳实践。

## child_process 执行 Shell 命令

Node.js 的 `child_process` 模块提供四种主要方式创建子进程：

| 方法 | 特点 | 适用场景 |
|------|------|---------|
| `exec` | 缓存全部输出到回调，支持 shell 语法 | 短命令、少量输出 |
| `execSync` | `exec` 的同步版本 | 脚本初始化阶段 |
| `execFile` | 直接执行文件，不经过 shell | 高性能、安全要求高 |
| `spawn` | 基于 Stream，实时输出流 | 长时间运行的命令 |
| `fork` | 专门用于执行 Node.js 模块 | Node.js 进程间通信 |

### exec — 执行 Shell 命令

```js
import { exec } from 'child_process';

exec('ls -la', (error, stdout, stderr) => {
  if (error) {
    console.error(`执行出错: ${error.message}`);
    return;
  }
  if (stderr) {
    console.warn(`警告: ${stderr}`);
  }
  console.log(`输出:\n${stdout}`);
});
```

```js
// Promise 封装
function execPromise(command, options = {}) {
  return new Promise((resolve, reject) => {
    exec(command, options, (error, stdout, stderr) => {
      if (error) reject({ error, stderr });
      else resolve(stdout.trim());
    });
  });
}

// 使用
try {
  const gitBranch = await execPromise('git branch --show-current');
  console.log('当前分支:', gitBranch);
} catch ({ error, stderr }) {
  console.error('获取分支失败:', stderr);
}
```

### execFile — 直接执行程序

```js
import { execFile } from 'child_process';

execFile('node', ['--version'], (error, stdout, stderr) => {
  if (error) throw error;
  console.log('Node 版本:', stdout.trim());
});
```

**exec vs execFile 的区别：**

```js
// exec 会经过 shell 解释，支持通配符、管道、变量替换
exec('cat *.md | wc -l', callback);

// execFile 直接执行，不支持 shell 语法，更安全高效
execFile('cat', ['file.md'], callback);  // ✅
execFile('cat *.md', callback);          // ❌ 不支持通配符
```

### spawn — 流式处理

spawn 最适合长时间运行的命令或大数据量的实时处理。

```js
import { spawn } from 'child_process';

const child = spawn('npm', ['install'], {
  cwd: './my-project',
  stdio: 'pipe',  // 'pipe' | 'inherit' | 'ignore' | ['pipe','inherit','pipe']
  env: { ...process.env, NODE_ENV: 'production' },
  shell: true,     // 是否通过 shell 执行
  detached: false, // 是否脱离父进程
});

// 实时捕获 stdout
child.stdout.on('data', (chunk) => {
  const line = chunk.toString().trim();
  if (line) console.log(line);
});

// 实时捕获 stderr
child.stderr.on('data', (chunk) => {
  const line = chunk.toString().trim();
  if (line) console.warn(line);
});

// 进程退出
child.on('close', (code) => {
  if (code === 0) {
    console.log('命令执行成功');
  } else {
    console.error(`命令以退出码 ${code} 结束`);
  }
});

// 进程错误（如命令不存在）
child.on('error', (err) => {
  console.error('启动失败:', err.code === 'ENOENT' ? '命令不存在' : err);
});
```

### stdio 选项详解

```js
// inherit — 让子进程共享父进程的标准流（直接显示在终端）
spawn('npm', ['run', 'build'], { stdio: 'inherit' });

// ignore — 忽略输出
spawn('npm', ['run', 'lint'], { stdio: ['ignore', 'ignore', 'pipe'] });

// 数组形式精确控制每个流
spawn('git', ['push'], {
  stdio: [
    'inherit',  // stdin 来自终端
    'pipe',     // stdout 可编程处理
    'pipe',     // stderr 可编程处理
  ]
});

// 额外的 IPC 通道（配合 fork 使用）
spawn('node', ['worker.js'], {
  stdio: ['pipe', 'pipe', 'pipe', 'ipc']
});
```

### fork — Node.js 进程间通信

```js
// parent.js
import { fork } from 'child_process';

const worker = fork('./worker.js');

worker.on('message', (result) => {
  console.log('子进程结果:', result);
});

worker.send({ type: 'compute', data: [1, 2, 3, 4, 5] });


// worker.js
process.on('message', (msg) => {
  if (msg.type === 'compute') {
    const sum = msg.data.reduce((a, b) => a + b, 0);
    process.send({ sum, count: msg.data.length });
  }
});
```

### 进程树管理

```js
import { spawn } from 'child_process';

// detached + unref 使子进程脱离父进程独立运行
const child = spawn('node', ['server.js'], {
  detached: true,
  stdio: 'ignore',
});

child.unref();  // 允许父进程退出而不等待子进程
console.log('守护进程已启动，PID:', child.pid);
```

## execa 封装

execa 是社区广泛使用的 child_process 封装库，提供了更友好的 Promise API 和跨平台支持。

### 安装

```bash
npm install execa
```

### 基础用法

```js
import execa from 'execa';

// 基本 Promise 用法
const { stdout } = await execa('git', ['branch', '--show-current']);
console.log('当前分支:', stdout);

// 同步版本
const resultSync = execaSync('node', ['--version']);

// $ 方便方法
const { all } = await execa$({ all: true })`npm run build`;
```

### 高级选项

```js
import execa from 'execa';

const subprocess = execa('npm', ['run', 'watch'], {
  cwd: './packages/app',
  env: { NODE_ENV: 'development' },
  timeout: 60_000,           // 超时时间 (ms)
  buffer: false,             // 不缓冲，使用 stream
  reject: true,              // 非0退出码时 reject
  all: true,                 // 合并 stdout 和 stderr 到 all 流
  cleanup: true,             // 父进程退出时杀死子进程
  preferLocal: true,         // 优先使用本地 node_modules 中的命令
  localDir: './',            // 本地命令搜索目录
  stdin: 'pipe',             // 控制 stdin 行为
  killSignal: 'SIGTERM',     // 终止信号
  gracefulCancel: false,     // 是否优雅取消
  cancelSignal: AbortSignal, // 取消信号（替代 timeout）
});

// 实时输出
subprocess.stdout.on('data', (data) => {
  process.stdout.write(data);
});

// 取消执行
const controller = new AbortController();
setTimeout(() => controller.abort(), 5000);

try {
  await execa({ signal: controller.signal })`long-running-command`;
} catch (error) {
  if (error.isCanceled) {
    console.log('命令已被取消');
  }
}
```

### 错误处理

```js
import execa from 'execa';

try {
  await execa('npm', ['run', 'nonexistent-script']);
} catch (error) {
  console.log('短消息:', error.shortMessage);
  console.log('原始消息:', error.originalMessage);
  console.log('命令:', error.command);
  console.log('退出码:', error.exitCode);
  console.log('失败原因:', error.failed);
  console.log('是否超时:', error.timedOut);
  console.log('是否被取消:', error.isCanceled);
  console.log('stdout:', error.stdout);
  console.log('stderr:', error.stderr);
}
```

### 实际应用：Git 操作封装

```js
import execa from 'execa';

class GitHelper {
  #cwd;

  constructor(cwd = process.cwd()) {
    this.#cwd = cwd;
  }

  async run(args, options = {}) {
    const result = await execa('git', args, {
      cwd: this.#cwd,
      ...options,
    });
    return result.stdout.trim();
  }

  async getCurrentBranch() {
    return this.run(['branch', '--show-current']);
  }

  async getStatus() {
    return this.run(['status', '--porcelain']);
  }

  async commit(message) {
    await this.run(['commit', '-m', message]);
  }

  async push(remote = 'origin', branch) {
    branch = branch || await this.getCurrentBranch();
    return this.run(['push', remote, branch]);
  }

  async hasChanges() {
    const status = await this.getStatus();
    return status.trim().length > 0;
  }
}

const git = new GitHelper('./my-project');
console.log(await git.getCurrentBranch());
```

## 跨平台兼容（cross-spawn）

不同操作系统对 shell 命令的处理存在差异，cross-spawn 解决了这个问题。

### 为什么需要跨平台处理？

```js
// Windows 下的问题：
// - 无法直接执行不带 .cmd/.exe 后缀的命令
// - 引号和转义行为不一致
// - PATH 变量分隔符不同（: vs ;）

// Linux/Mac：
spawn('rm', ['-rf', 'dist']);  // ✅ 正常工作

// Windows：
spawn('rm', ['-rf', 'dist']);  // ❌ rm 不是 Windows 命令
```

### 安装与使用

```bash
npm install cross-spawn
```

```js
import crossSpawn from 'cross-spawn';

// 自动处理跨平台差异
const result = crossSpawn.sync('npm', ['run', 'build'], {
  stdio: 'inherit',
});

if (result.status !== 0) {
  process.exit(result.status ?? 1);
}
```

### 常见的跨平台注意事项

```js
// 1. 使用 path.posix.join 处理路径（避免反斜杠）
import path from 'path';
const filePath = path.posix.join('src', 'components', 'App.tsx');

// 2. 使用 os.homedir() 获取主目录
import os from 'os';
const homeDir = os.homedir();

// 3. 使用环境变量代替硬编码路径
const configPath = process.env.XDG_CONFIG_HOME ||
  path.join(os.homedir(), '.config');

// 4. shebang 处理
// Windows 不识别 shebang，需要 .cmd shim
```

## 实时输出管道

### 将子进程输出转发到父进程

```js
import { spawn } from 'child_process';

function pipeOutput(childProcess) {
  if (childProcess.stdout) {
    childProcess.stdout.pipe(process.stdout);
  }
  if (childProcess.stderr) {
    childProcess.stderr.pipe(process.stderr);
  }
  if (process.stdin.isTTY && childProcess.stdin) {
    process.stdin.pipe(childProcess.stdin);
  }
}

const child = spawn('webpack', ['serve'], { shell: true });
pipeOutput(child);
```

### 双向通信管道

```js
import { spawn } from 'child_process';

const python = spawn('python3', ['script.py'], {
  stdio: ['pipe', 'pipe', 'pipe'],
});

// 向 Python 进程发送数据
python.stdin.write(JSON.stringify({ data: [1, 2, 3] }));
python.stdin.end();

// 收集响应
let output = '';
python.stdout.on('data', (chunk) => {
  output += chunk.toString();
});

python.on('close', (code) => {
  console.log('Python 返回:', output);
});
```

### 流式处理大量输出

```js
import { spawn } from 'child_process';
import { createWriteStream } from 'fs';

const logStream = createWriteStream('build.log');

const build = spawn('npm', ['run', 'build'], {
  stdio: ['ignore', 'pipe', 'pipe'],
});

// 同时输出到终端和日志文件
build.stdout.on('data', (chunk) => {
  const str = chunk.toString();
  process.stdout.write(str);
  logStream.write(str);
});

build.stderr.on('data', (chunk) => {
  const str = chunk.toString();
  process.stderr.write(str);
  logStream.write(`[ERROR] ${str}`);
});
```

## 小结

| 方案 | 适用场景 | 特点 |
|------|---------|------|
| `child_process.exec` | 短命令、少量输出 | Shell 语法支持，缓存全部输出 |
| `child_process.spawn` | 长时间运行、大数据量 | Stream 模式，实时处理 |
| `child_process.execFile` | 安全性优先 | 不经 shell，性能更好 |
| `child_process.fork` | Node.js 子进程 | 内置 IPC 通信通道 |
| **execa** | 日常开发推荐 | Promise API，完善错误处理 |
| **cross-spawn** | 跨平台需求 | 解决 Windows/Linux 差异 |

核心原则：简单场景用 execa 封装；需要细粒度控制时回到原生 API；始终考虑跨平台兼容性。
