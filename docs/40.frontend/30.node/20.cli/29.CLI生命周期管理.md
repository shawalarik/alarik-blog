## 概述

一个健壮的 CLI 工具需要完善的生命周期管理——从启动初始化到正常结束或异常退出，每个阶段都需要恰当的处理。本章涵盖命令行钩子、错误捕获、资源清理和优雅退出的完整实践。

## 命令行钩子（before / after）

钩子机制允许在命令执行的各个阶段注入自定义逻辑，是实现可扩展性的核心手段。

### 生命周期阶段

典型的 CLI 生命周期：

```
init → beforeCommand → command → afterCommand → cleanup → exit
  ↓        ↓            ↓          ↓            ↓        ↓
 初始化   前置钩子     执行命令    后置钩子     资源清理  退出
```

### 基础钩子系统实现

```js
class HookSystem {
  #hooks = {
    before: [],
    after: [],
    error: [],
    finally: [],
  };

  on(event, handler) {
    if (this.#hooks[event]) {
      this.#hooks[event].push(handler);
    }
    return this;  // 链式调用
  }

  async emit(event, context) {
    const handlers = this.#hooks[event] || [];

    for (const handler of handlers) {
      try {
        await handler(context);
      } catch (error) {
        if (event !== 'error') {
          await this.emit('error', { error, originalEvent: event, ...context });
        }
      }
    }
  }
}

// 使用
const hooks = new HookSystem()

  .on('before', async (ctx) => {
    console.log(`准备执行命令: ${ctx.command}`);
    ctx.startTime = Date.now();
  })

  .on('after', async (ctx) => {
    const duration = Date.now() - ctx.startTime;
    console.log(`命令执行完毕，耗时: ${duration}ms`);
  })

  .on('error', async (ctx) => {
    console.error(`执行出错: ${ctx.error.message}`);
  })

  .on('finally', async (ctx) => {
    console.log('生命周期结束');
  });
```

### Commander.js 钩子集成

```js
import { Command } from 'commander';

const program = new Command();

// 全局前置钩子
program.hook('preAction', (thisCommand, actionCommand) => {
  console.log(`[Hook] preAction: ${actionCommand.name()}`);
});

// 全局后置钩子
program.hook('postAction', (thisCommand, actionCommand) => {
  console.log(`[Hook] postAction: ${actionCommand.name()}`);
});

program
  .command('build')
  .option('--watch', '监听模式')
  .hook('preAction', (thisCmd, cmd) => {
    // 命令级前置钩子
    console.log('构建前检查依赖...');
  })
  .action(async (options) => {
    console.log('执行构建...');
  });

program.parse();
```

### Yargs 中间件作为钩子

```js
import yargs from 'yargs';
import { hideBin } from 'yargs/helpers';

yargs(hideBin(process.argv))

  // 全局中间件（相当于前置钩子）
  .middleware(async (argv) => {
    argv._startTime = Date.now();
    console.log(`[${new Date().toISOString()}] 开始执行`);
  }, true)

  // 后置处理可在 command action 末尾添加
  .command('deploy', '部署应用', {}, async (argv) => {
    try {
      await doDeploy(argv);
    } finally {
      const duration = Date.now() - argv._startTime;
      console.log(`耗时: ${duration}ms`);
    }
  })

  .parse();
```

## 错误捕获（try-catch + 退出码）

### 分层错误处理

```js
// 定义自定义错误类
class CliError extends Error {
  constructor(message, code = 1) {
    super(message);
    this.name = 'CliError';
    this.code = code;       // 退出码
    this.timestamp = new Date();
  }
}

class ValidationError extends CliError {
  constructor(message) {
    super(message, 2);  // 参数错误使用退出码 2
    this.name = 'ValidationError';
  }
}

class ConfigError extends CliError {
  constructor(message) {
    super(message, 78);  // 配置错误使用退出码 78
    this.name = 'ConfigError';
  }
}

class NetworkError extends CliError {
  constructor(message) {
    super(message, 3);
    this.name = 'NetworkError';
  }
}
```

### 统一错误处理器

```js
import process from 'process';

function setupGlobalHandlers() {
  // 未捕获异常
  process.on('uncaughtException', (error) => {
    console.error('\n✗ 未捕获异常:');
    console.error(`  ${error.message}`);
    if (process.env.DEBUG) {
      console.error(error.stack);
    }
    process.exit(1);
  });

  // 未处理的 Promise rejection
  process.on('unhandledRejection', (reason, promise) => {
    console.error('\n✗ 未处理的 Promise 拒绝:');
    console.error(`  ${reason}`);
    process.exit(1);
  });
}

// 主函数包装器
async function main() {
  try {
    setupGlobalHandlers();

    const config = loadConfig();
    await validateConfig(config);

    const result = await executeCommand(config);
    process.exit(0);

  } catch (error) {
    await handleError(error);
  }
}

async function handleError(error) {
  if (error instanceof CliError) {
    console.error(`\n${getErrorIcon(error)} ${error.message}`);
    process.exit(error.code);
  }

  // 未知错误
  console.error('\n✗ 发生意外错误:');
  if (process.env.DEBUG) {
    console.error(error.stack);
  } else {
    console.error(`  ${error.message}`);
    console.error('  使用 --debug 查看详细堆栈');
  }
  process.exit(1);
}

function getErrorMsg(error) {
  const icons = {
    ValidationError: '⚠ ',
    ConfigError: '⚙ ',
    NetworkError: '🌐 ',
    CliError: '✗ ',
  };
  return icons[error.name] || '✗ ';
}

main();
```

### 命令级错误隔离

```js
import { program } from 'commander';

program
  .command('build')
  .action(async () => {
    try {
      await buildProject();
    } catch (error) {
      // 命令内部错误不影响其他命令
      console.error(`构建失败: ${error.message}`);
      // 不要在这里 process.exit()！
      // Commander 会处理非零退出
      throw error;  // 重新抛出让 Commander 处理
    }
  });

program.configureOutput({
  writeErr: (str) => process.stderr.write(chalk.red(str)),
  outputError: (err, write) =>
    write(chalk.red(`错误: ${err.message}\n`)),
});
```

## 清理临时资源

### 资源追踪器

```js
class ResourceManager {
  #resources = [];
  #cleaned = false;

  register(description, cleanupFn) {
    this.#resources.push({ description, cleanupFn });
  }

  registerTempFile(filePath) {
    this.register(`临时文件: ${filePath}`, async () => {
      const fs = await import('fs/promises');
      try {
        await fs.unlink(filePath);
      } catch (e) {
        // 忽略文件不存在的错误
      }
    });
  }

  registerSubprocess(childProcess) {
    this.register(`子进程 PID:${childProcess.pid}`, async () => {
      if (!childProcess.killed) {
        childProcess.kill('SIGTERM');
      }
    });
  }

  registerServer(server) {
    this.register('HTTP 服务器', async () => {
      return new Promise((resolve) => {
        server.close(resolve);
      });
    });
  }

  async cleanup() {
    if (this.#cleaned) return;
    this.#cleaned = true;

    // 按注册的逆序清理（后注册的资源先清理）
    for (let i = this.#resources.length - 1; i >= 0; i--) {
      const { description, cleanupFn } = this.#resources[i];
      try {
        await cleanupFn();
        console.log(`✓ 清理完成: ${description}`);
      } catch (error) {
        console.warn(`⚠ 清理失败: ${description} - ${error.message}`);
      }
    }

    this.#resources = [];
  }
}

// 使用
const resources = new ResourceManager();

// 注册各类资源
resources.registerTempFile('/tmp/my-cli-temp-config.json');
resources.registerSubprocess(childProcess);
resources.registerServer(httpServer);
```

### Finally 处理与信号处理

```js
import process from 'process';

function setupExitHandling(resources) {
  let isExiting = false;

  async function gracefulShutdown(signal) {
    if (isExiting) return;  // 防止重复触发
    isExiting = true;

    console.log(`\n收到 ${signal} 信号，正在清理...`);

    try {
      await resources.cleanup();
      console.log('资源清理完毕，退出');
    } catch (error) {
      console.error('清理过程中出错:', error.message);
    }

    process.exit(signal === 'SIGINT' ? 130 : 128 + getSignalNumber(signal));
  }

  // 退出信号
  process.on('SIGINT', () => gracefulShutdown('SIGINT'));   // Ctrl+C
  process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));  // kill 命令

  // 进程正常退出
  process.on('exit', (code) => {
    if (!isExiting) {
      // 非信号触发的退出，同步清理
      // 注意: exit 事件中不能执行异步操作
    }
  });
}
```

## 完整的生命周期管理器

```js
import process from 'process';
import chalk from 'chalk';

class CliLifecycle {
  #hooks = { before: [], after: [], error: [], finally: [] };
  #resources = new ResourceManager();
  #startTime;
  #exiting = false;

  constructor(options = {}) {
    this.options = {
      name: options.name || 'cli',
      version: options.version || '1.0.0',
      debug: options.debug || false,
    };

    this.#setupSignalHandlers();
    this.#setupErrorHandlers();
  }

  // ===== 钩子注册 =====
  before(fn) { this.#hooks.before.push(fn); return this; }
  after(fn) { this.#hooks.after.push(fn); return this; }
  onError(fn) { this.#hooks.error.push(fn); return this; }
  finally(fn) { this.#hooks.finally.push(fn); return this; }

  // ===== 资源管理 =====
  resource(desc, fn) { this.#resources.register(desc, fn); return this; }

  // ===== 执行命令 =====
  async run(commandFn) {
    this.#startTime = Date.now();

    try {
      // 前置阶段
      await this.#emitHooks('before', { phase: 'before' });

      // 执行命令
      const result = await commandFn({
        resource: (desc, fn) => this.resource(desc, fn),
        startTime: this.#startTime,
      });

      // 后置阶段
      await this.#emitHooks('after', {
        phase: 'after',
        result,
        duration: Date.now() - this.#startTime,
      });

      return result;

    } catch (error) {
      // 错误阶段
      await this.#emitHooks('error', {
        phase: 'error',
        error,
        duration: Date.now() - this.#startTime,
      });
      throw error;

    } finally {
      // 最终阶段
      await this.#emitHooks('finally', { phase: 'finally' });
      await this.#resources.cleanup();
    }
  }

  // ===== 内部方法 =====
  async #emitHooks(event, context) {
    for (const fn of this.#hooks[event]) {
      await fn(context);
    }
  }

  #setupSignalHandlers() {
    const signals = ['SIGINT', 'SIGTERM'];
    signals.forEach(sig => {
      process.on(sig, async () => {
        if (this.#exiting) return;
        this.#exiting = true;
        console.log(`\n${chalk.yellow(`收到 ${sig} 信号...`)}`);
        await this.#resources.cleanup();
        process.exit(128 + process.signals[sig]);
      });
    });
  }

  #setupErrorHandlers() {
    process.on('uncaughtException', async (error) => {
      console.error(chalk.red('未捕获异常:'), error.message);
      if (this.options.debug) console.error(error.stack);
      await this.#resources.cleanup();
      process.exit(1);
    });

    process.on('unhandledRejection', async (reason) => {
      console.error(chalk.red('未处理的拒绝:'), reason);
      await this.#resources.cleanup();
      process.exit(1);
    });
  }
}

// ===== 使用示例 =====
const cli = new CliLifecycle({ name: 'my-cli', debug: true })

  .before(async (ctx) => {
    console.log(chalk.cyan('▸ 初始化...'));
    console.log(chalk.cyan(`  版本: ${cli.options.version}`));
  })

  .after(async (ctx) => {
    console.log(chalk.green(`✓ 完成 (${ctx.duration}ms)`));
  })

  .onError(async (ctx) => {
    console.error(chalk.red(`✗ 失败: ${ctx.error.message}`));
  })

  .finally(async () => {
    console.log(chalk.gray('  生命周期结束'));
  })

  .run(async ({ resource }) => {
    // 注册临时资源
    resource('示例资源', async () => {
      console.log('  清理示例资源');
    });

    // 主要业务逻辑
    console.log(chalk.bold('  执行核心任务...'));

    await new Promise(r => setTimeout(r, 500));  // 模拟工作

    return { status: 'ok' };
  })
  .catch(() => process.exit(1));
```

## 小结

| 关注点 | 技术方案 | 关键要点 |
|--------|---------|---------|
| **钩子系统** | before/after/finally 事件模型 | 支持链式注册，逆序清理 |
| **错误捕获** | 自定义错误类 + 分层处理 | 区分退出码含义，保留原始堆栈 |
| **资源清理** | ResourceManager 追踪器 | 注册/清理配对，防止泄漏 |
| **信号处理** | SIGINT/SIGTERM 监听 | 优雅退出，防止重复触发 |
| **防御编程** | uncaughtException/unhandledRejection | 最后的安全网 |

良好的生命周期管理使 CLI 工具在任何退出路径下都能保持一致性和可靠性。
