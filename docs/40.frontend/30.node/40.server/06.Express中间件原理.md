## 概述

Express 是 Node.js 最流行的 Web 框架，其核心设计理念就是**中间件（Middleware）**模式。理解 Express 中间件的执行模型、`next()` 机制以及错误处理流程，对于构建健壮的 Web 服务至关重要。

## 中间件执行模型

### 什么是中间件？

中间件是一个函数，它可以：
1. **访问请求对象 (`req`)**
2. **访问响应对象 (`res`)**
3. **调用下一个中间件 (`next`)**
4. **结束请求-响应循环**

```js
// 中间件的基本签名
function middleware(req, res, next) {
  // 1. 执行前置逻辑
  console.log(`${req.method} ${req.url}`);
  
  // 2. 决定是否传递给下一个中间件
  next();
  
  // 或者直接结束响应
  // res.send('Hello');
}
```

### Express 请求生命周期

```
客户端请求
    │
    ▼
┌─────────────────────────────────────┐
│  中间件栈（按 app.use 注册顺序）      │
│                                     │
│  ┌───────────┐                      │
│  │ middleware1│ ← req 进入          │
│  │   ↓ next()│                     │
│  ├───────────┤                      │
│  │ middleware2│                      │
│  │   ↓ next()│                     │
│  ├───────────┤                      │
│  │ router     │ ← 路由处理          │
│  │   ↓ next()│                     │
│  ├───────────┤                      │
│  │ error handler│                    │
│  └───────────┘                      │
└─────────────────────────────────────┘
    │
    ▼
  响应返回客户端
```

### app.use — 注册中间件

```js
const express = require('express');
const app = express();

// 全局中间件（所有请求都经过）
app.use((req, res, next) => {
  console.log(`[${new Date().toISOString()}] ${req.method} ${req.url}`);
  next();  // 必须调用！否则请求挂起
});

// 路径前缀中间件
app.use('/api', (req, res, next) => {
  // 只对 /api/* 路径生效
  req.apiRequest = true;
  next();
});

// 多个中间件串联
app.use(middleware1, middleware2, middleware3);

// 路径 + 方法级中间件
app.get('/', handler);
app.post('/users', handler);

// 数组形式注册多个中间件
app.get('/admin', [authMiddleware, adminCheck, adminHandler]);

// 启动服务器
app.listen(3000);
```

### 内置中间件

```js
// 解析 JSON 请求体（Express 4.16+）
app.use(express.json({ limit: '10mb' }));

// 解析 URL 编码的请求体
app.use(express.urlencoded({ extended: true }));

// 静态文件服务
app.use(express.static('public'));
app.use(express.static('dist', { maxAge: '1y' }));

// Cookie 解析
const cookieParser = require('cookie-parser');
app.use(cookieParser());
```

## req 对象详解

```js
app.use((req, res, next) => {
  // --- 基本信息 ---
  req.app;             // Express 应用实例引用
  req.baseUrl;         // 路由挂载的基础路径
  req.body;            // 已解析的请求体（需要 json/urlencoded 中间件）
  req.cookies;         // cookie 对象（需要 cookie-parser）
  req.hostname;        // 主机名（不含端口）
  req.ip;              // 客户端 IP（需 trust proxy）
  req.ips;             // 代理链 IP 列表
  req.method;          // HTTP 方法
  req.originalUrl;     // 原始 URL（含查询字符串）
  req.params;          // 路由参数（如 /user/:id 中的 id）
  req.path;            // URL 路径部分（不含查询字符串）
  req.protocol;        // 'http' 或 'https'
  req.query;           // 查询参数对象（自动解析）
  req.route;           // 当前匹配的路由对象
  req.secure;          // 是否 HTTPS
  
  // --- 参数获取 ---
  req.get('Content-Type');   // 获取指定请求头
  req.get('accept');         
  req.accepts('html');       // 检查 Accept 头是否包含某类型
  
  // --- 属性设置 ---
  req.myCustomData = {};     // 中间件间传递数据的惯用方式
  
  next();
});
```

### req.params vs req.query

```js
// 定义路由
app.get('/users/:userId/posts/:postId', (req, res) => {
  // URL: /users/42/posts/100?sort=date&order=desc
  
  req.params;  // { userId: "42", postId: "100" }  ← 来自路径段
  req.query;   // { sort: "date", order: "desc" }   ← 来自 ? 后面的查询字符串
  req.path;    // "/users/42/posts/100"
});
```

## res 对象详解

```js
app.use((req, res, next) => {
  
  // --- 状态码 ---
  res.status(201);        // 设置状态码（可链式调用）
  res.sendStatus(404);    // 设置状态并发送状态文本
  
  // --- 响应头 ---
  res.set('X-Custom-Header', 'value');   // 设置单个或多个头
  res.type('.html');                      // 根据 extension 设置 Content-Type
  res.type('application/json');
  
  // --- 响应发送 ---
  res.send('<h1>Hello</h1>');             // 自动根据类型设置 Content-Type
  res.json({ name: 'Alice', age: 30 });   // 发送 JSON
  res.jsonp({ name: 'Alice' });           // JSONP（需 query callback）
  res.render('index', { title: 'Home' }); // 渲染模板引擎
  res.sendFile(__dirname + '/file.pdf');  // 发送文件
  res.download('/report.pdf');           // 触发浏览器下载
  res.redirect('/login');                 // 重定向（302）
  res.redirect(301, '/new-url');          // 永久重定向
  res.end();                              // 结束响应（无内容）
  
  // --- Cookie ---
  res.cookie('token', 'xxx', {
    maxAge: 900000,      // 过期时间（毫秒）
    secure: true,        // 仅 HTTPS
    httpOnly: true,      // 防 XSS
    sameSite: 'lax',     // CSRF 防护
    signed: true,        // 签名 cookie
    domain: '.example.com',
    path: '/',
  });
  res.clearCookie('token');               // 清除 cookie
  
  // --- 本地变量（传给模板）---
  res.locals.user = currentUser;
  
  next();
});
```

## next() 机制深度解析

### next() 的三种调用方式

```js
// 方式一：无参数 → 传递给下一个中间件
app.use((req, res, next) => {
  req.timestamp = Date.now();
  next();  // 继续往下走
});

// 方式二：传入字符串 → 视为错误，跳到错误处理中间件
app.use((req, res, next) => {
  const token = req.headers.authorization;
  if (!token) {
    next('Unauthorized');  // 跳转到 err middleware
    return;
  }
  next();
});

// 方式三：传入 Error 对象 → 跳到错误处理中间件
app.use((req, res, next) => {
  db.query('SELECT * FROM users', (err, result) => {
    if (err) {
      next(err);  // 传递错误
      return;
    }
    req.users = result;
    next();
  });
});
```

### next('route') — 跳过当前路由的其他中间件

```js
app.get('/data',
  (req, res, next) => {
    if (req.query.format === 'xml') {
      // 跳过后续中间件，直接找下一个匹配的路由
      return next('route');
    }
    next();  // 继续当前路由链
  },
  (req, res) => {
    // JSON 格式返回
    res.json({ data: 'json response' });
  }
);

app.get('/data', (req, res) => {
  // XML 格式返回（next('route') 时到达这里）
  res.type('application/xml');
  res.send('<data>xml response</data>');
});
```

### next() 未调用的后果

```js
// ⚠️ 危险：忘记调用 next()
app.use((req, res, next) => {
  // 如果某种条件不满足就忘了 next()
  if (req.url !== '/special') {
    // 没有 next() → 请求永远挂起！浏览器一直 loading
  }
  // 只有 /special 能正常响应
  res.end('Special page');
});

// ✅ 安全做法：确保每个分支都有出口
app.use((req, res, next) => {
  if (req.url !== '/special') {
    return next();  // ← 不要忘了 return 或 else
  }
  res.end('Special page');
});
```

## 错误处理中间件

### 四参数格式 `(err, req, res, next)`

```js
// 错误处理中间件必须放在最后（其他中间件之后）
app.use((err, req, res, next) => {
  console.error(err.stack);

  // 开发环境：返回详细错误
  if (app.get('env') === 'development') {
    res.status(500).json({
      error: err.message,
      stack: err.stack,
    });
    return;
  }

  // 生产环境：隐藏详细信息
  res.status(500).json({
    error: 'Internal Server Error',
    requestId: req.id,  // 用于排查
  });
});
```

### 异步错误处理

```js
// ❌ 问题：async 函数中的错误不会被 Express 捕获
app.get('/data', async (req, res, next) => {
  const data = await riskyOperation();  // 如果抛出异常？
  res.json(data);                       // Express 不会捕获这个异常！
});

// ✅ 方案一：try/catch 包裹
app.get('/data', async (req, res, next) => {
  try {
    const data = await riskyOperation();
    res.json(data);
  } catch (err) {
    next(err);  // 手动传递给错误中间件
  }
});

// ✅ 方案二：封装 asyncHandler 工具
function asyncHandler(fn) {
  return (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}

// 使用方式
app.get('/data', asyncHandler(async (req, res) => {
  const data = await riskyOperation();
  res.json(data));
}));
```

### 自定义错误类

```js
// 自定义 HTTP 错误
class HttpError extends Error {
  constructor(statusCode, message) {
    super(message);
    this.statusCode = statusCode;
    this.name = 'HttpError';
  }
}

class NotFoundError extends HttpError {
  constructor(resource = 'Resource') {
    super(404, `${resource} not found`);
    this.name = 'NotFoundError';
  }
}

class UnauthorizedError extends HttpError {
  constructor(message = 'Unauthorized') {
    super(401, message);
    this.name = 'UnauthorizedError';
  }
}

// 使用
app.get('/users/:id', asyncHandler(async (req, res) => {
  const user = await User.findById(req.params.id);
  if (!user) throw new NotFoundError('User');
  res.json(user);
}));
```

### 分层错误处理

```js
// 404 处理（放在所有路由之后）
app.use((req, res, next) => {
  res.status(404).json({ error: `${req.method} ${req.url} not found` });
});

// 全局错误处理（放在 404 之后）
app.use((err, req, res, next) => {
  // 已知 HTTP 错误
  if (err instanceof HttpError) {
    return res.status(err.statusCode).json({ error: err.message });
  }

  // SyntaxError（JSON 解析失败等）
  if (err instanceof SyntaxError && err.status === 400 && 'body' in err) {
    return res.status(400).json({ error: 'Invalid JSON payload' });
  }

  // 未知错误
  console.error('Unexpected error:', err);
  res.status(500).json({ error: 'Internal Server Error' });
});
```

## 中间件顺序的重要性

```
正确的顺序                          错误的顺序
─────────                        ─────────
1. logger                        1. router
2. cors                          2. auth（无法读取 body）
3. json body parser              3. json parser（太晚）
4. urlencoded parser             4. logger
5. auth                          5. cors
6. router                        6. 错误处理（放太早）
7. 404 handler                   （路由找不到错误处理）
8. 错误处理
```

### 最佳实践顺序模板

```js
const express = require('express');
const helmet = require('helmet');
const cors = require('cors');
const morgan = require('morgan');

const app = express();

// ===== 第一层：安全与基础设施 =====
app.use(helmet());                  // 安全头
app.use(cors(options));             // CORS 配置
app.use(morgan('dev'));             // 日志
app.use(express.json());            // Body 解析
app.use(express.urlencoded());

// ===== 第二层：认证与会话 =====
app.use(session(config));
app.use(passport.initialize());
app.use(passport.session());

// ===== 第三层：业务逻辑 =====
app.use('/api/auth', authRoutes);
app.use('/api/users', userRoutes);
app.use('/api/posts', postRoutes);

// ===== 第四层：错误兜底 =====
app.use(notFoundHandler);
app.use(globalErrorHandler);

module.exports = app;
```
