## 概述

在前端开发中，API 代理是解决跨域问题的标准方案。无论是 webpack-dev-server、Vite 还是自建的开发服务器，都依赖代理中间件将浏览器的请求转发到后端服务。本文深入讲解代理的实现原理与最佳实践。

## 为什么需要开发代理？

```
开发环境的跨域问题：

浏览器 (localhost:5173)          后端 API (api.example.com)
┌──────────────────┐             ┌──────────────────┐
│ fetch('/api/user')│ ──✕ X ──→ │  GET /api/user     │
│ (不同源)          │  CORS阻止   │  返回用户数据      │
└──────────────────┘             └──────────────────┘

解决方案 —— 开发代理：

浏览器 (localhost:5173)          Dev Server (localhost:5173)      Backend (:3001)
┌──────────────────┐            ┌──────────────────┐            ┌──────────┐
│ fetch('/api/user')│ ─── OK ─→ │ /api/user 代理     │ ───OK ─→  │ /api/user │
│ 同源请求 ✓        │            │ 转发到 localhost:3001│           │ 返回数据  │
└──────────────────┘            └──────────────────┘            └──────────┘
```

## http-proxy-middleware 使用

### 基本配置

```js
const express = require('express');
const { createProxyMiddleware } = require('http-proxy-middleware');

const app = express();

// 最简配置
app.use('/api', createProxyMiddleware({
  target: 'http://localhost:3001',  // 目标服务器
  changeOrigin: true,                // 修改 Origin 头为目标地址
}));

// 完整配置
app.use('/api', createProxyMiddleware({
  target: 'http://api.example.com',
  changeOrigin: true,
  
  // 路径重写
  pathRewrite: {
    '^/api/old': '/api/new',        // /api/old/... → /api/new/...
    '^/api': '',                    // /api/... → /...（去掉前缀）
  },
  
  // 请求头修改
  on: {
    proxyReq: (proxyReq, req, res) => {
      // 注入自定义请求头
      proxyReq.setHeader('X-Custom-Header', 'dev-proxy');
      // 注入认证 Token
      if (process.env.API_TOKEN) {
        proxyReq.setHeader('Authorization', `Bearer ${process.env.API_TOKEN}`);
      }
    },
    
    proxyRes: (proxyRes, req, res) => {
      // 修改响应头（例如移除安全限制头）
      proxyRes.headers['x-custom-response'] = 'proxied';
    },
    
    error: (err, req, res) => {
      console.error('Proxy error:', err.message);
      res.status(502).json({ error: 'Backend unavailable' });
    },
  },
  
  // 日志
  logger: console,
}));
```

### Vite 中的代理配置

```js
// vite.config.ts
import { defineConfig } from 'vite';

export default defineConfig({
  server: {
    port: 5173,
    proxy: {
      // 字符串简写
      '/api': 'http://localhost:3001',
      
      // 选项对象
      '/api/v2': {
        target: 'http://api.example.com',
        changeOrigin: true,
        rewrite: (path) => path.replace(/^\/api\/v2/, ''),
      },
      
      // 正则表达式（注意要用 ^ 开头）
      '^/fallback': {
        target: 'https://jsonplaceholder.typicode.com',
        changeOrigin: true,
        rewrite: (path) => path.replace(/^\/fallback/, ''),
      },
      
      // 多目标代理（通过 configure）
      '/api': {
        target: 'http://localhost:3001',
        changeOrigin: true,
        configure: (proxy, options) => {
          proxy.on('proxyReq', (proxyReq, req) => {
            // 动态决定目标
            if (req.headers['x-target-env'] === 'staging') {
              proxyReq.setHeader('host', 'staging-api.example.com');
            }
          });
        },
        
        // WebSocket 代理
        ws: true,
      },
      
      // bypass 函数（条件绕过代理）
      '/api/public': {
        target: 'http://localhost:3001',
        bypass(req) {
          if (req.headers.accept?.includes('html')) {
            // 浏览器请求 HTML 页面时不代理（用于 SSR 场景）
            return '/index.html';
          }
        },
      },
    },
  },
});
```

### webpack-dev-server 中的配置

```js
// webpack.config.js
module.exports = {
  devServer: {
    port: 3000,
    proxy: [
      {
        context: ['/api', '/auth'],
        target: 'http://backend:8080',
        changeOrigin: true,
        pathRewrite: { '^/api': '' },
        secure: false,           // 不验证 SSL 证书
        headers: {
          'X-Dev-Proxy': 'webpack',
        },
        onProxyReq(proxyReq, req) {
          // Bearer token injection
          if (process.env.PROXY_TOKEN) {
            proxyReq.setHeader('Authorization', `Bearer ${process.env.PROXY_TOKEN}`);
          }
        },
        onError(err, req, res) {
          res.end('Proxy failed');
        },
      },
    ],
  },
};
```

## 代理核心原理解析

### 从零实现一个简易代理

```js
const http = require('node:http');
const https = require('node:http');
const { URL } = require('node:url');

function createProxy(target) {
  return function proxyMiddleware(req, res) {
    const targetUrl = new URL(target);
    
    // 合并目标 URL 和请求路径
    const proxyOptions = {
      hostname: targetUrl.hostname,
      port: targetUrl.port || (targetUrl.protocol === 'https:' ? 443 : 80),
      path: req.url,
      method: req.method,
      headers: {
        ...req.headers,
        host: targetUrl.host,          // 关键：改写 Host 头
        origin: targetUrl.origin,      // 关键：改写 Origin
      },
    };
    
    // 转发请求到目标
    const proxyReq = http.request(proxyOptions, (proxyRes) => {
      // 转发响应头
      res.writeHead(proxyRes.statusCode, proxyRes.headers);
      
      // pipe 响应体（流式传输，高效）
      proxyRes.pipe(res);
    });
    
    // 转发请求体（POST/PUT/PATCH）
    req.pipe(proxyReq);
    
    // 错误处理
    proxyReq.on('error', (err) => {
      console.error('Proxy request failed:', err.message);
      if (!res.headersSent) {
        res.writeHead(502, { 'Content-Type': 'application/json' });
      }
      res.end(JSON.stringify({ error: 'Bad Gateway', detail: err.message }));
    });
  };
}

// 使用
const express = require('express');
const app = express();
app.use('/api', createProxy('http://localhost:3001'));
```

### changeOrigin 的作用

```js
// 没有 changeOrigin
Host: localhost:5173          // 保持原始（可能导致后端拒绝）
Origin: http://localhost:5173

// 有 changeOrigin
Host: backend-api.example.com  // 改写为目标地址
Origin: https://backend-api.example.com

// 为什么重要？
// 很多后端框架（Spring Security, Django CSRF 等）会校验 Host/Origin 头
// changeOrigin 让后端认为请求来自它自己
```

## 路径重写规则

### 常见场景

```js
// 场景 1：去除前缀
// 前端: /api/users → 后端: /users
{
  pathRewrite: { '^/api': '' }
}

// 场景 2：替换前缀
// 前端: /api/v1/users → 后端: /v2/users
{
  pathRewrite: { '^/api/v1': '/v2' }
}

// 场景 3：多路径重写
{
  pathRewrite: function (path, req) {
    // 根据动态条件重写
    if (path.includes('/old-endpoint')) {
      return path.replace('/old-endpoint', '/new-endpoint');
    }
    return path;
  }
}

// 场景 4：添加版本前缀
// 前端: /users → 后端: /api/v3/users
{
  pathRewrite: { '^/(.*)$': '/api/v3/$1' }
}
```

### Vite 的 rewrite 函数

```js
// vite.config.ts — rewrite 可以是一个函数
proxy: {
  '/legacy-api': {
    target: 'http://legacy-service:4000',
    rewrite: (path) =>
      path
        .replace(/^\/legacy-api\/v1/, '/v2')
        .replace(/^\/legacy-api/, ''),
  },
},
```

## 请求头修改技巧

### 注入认证信息

```js
on: {
  proxyReq: (proxyReq, req) => {
    // 方式 1：固定 Token
    proxyReq.setHeader('Authorization', `Bearer ${process.env.DEV_API_TOKEN}`);
    
    // 方式 2：透传用户的 Cookie 到后端
    if (req.headers.cookie) {
      proxyReq.setHeader('Cookie', req.headers.cookie);
    }
    
    // 方式 3：添加调试标识
    proxyReq.setHeader('X-Forwarded-For', req.socket.remoteAddress);
    proxyReq.setHeader('X-Real-IP', req.socket.remoteAddress);
    proxyReq.setHeader('X-Proxied-By', 'dev-server');
    
    // 方式 4：模拟特定 User-Agent
    proxyReq.setHeader('User-Agent', 'MyApp/1.0 DevProxy');
  },
}
```

### 响应头处理

```js
on: {
  proxyRes: (proxyRes, req, res) => {
    // 解决 CORS（如果后端没配）
    proxyRes.headers['Access-Control-Allow-Origin'] = '*';
    proxyRes.headers['Access-Control-Allow-Methods'] = 'GET,POST,PUT,DELETE';
    proxyRes.headers['Access-Control-Allow-Headers'] = 'Content-Type,Authorization';
    
    // 移除可能造成问题的头
    delete proxyRes.headers['x-frame-options'];
    
    // 缓存控制
    if (req.url.includes('/static/')) {
      proxyRes.headers['cache-control'] = 'public, max-age=31536000';
    }
  },
}
```

## WebSocket 代理

### 配置方法

```js
// Express + http-proxy-middleware
app.use('/socket.io', createProxyMiddleware({
  target: 'ws://localhost:3001',
  ws: true,              // 启用 WebSocket 代理
  changeOrigin: true,
}));

// Vite 配置
proxy: {
  '/socket.io': {
    target: 'ws://localhost:3001',
    ws: true,
  },
},

// webpack-dev-server 配置
devServer: {
  proxy: {
    '/socket.io': {
      target: 'ws://localhost:3001',
      ws: true,
    },
  },
};
```

### WebSocket 代理原理

```javascript
// WebSocket 握手升级过程：
// 1. 客户端发送带 Upgrade: websocket 头的 HTTP 请求
// 2. 代理服务器拦截此请求
// 3. 代理服务器向目标发起相同的升级请求
// 4. 建立 双向 TCP 隧道（客户端 ↔ 代理 ↔ 目标）
// 5. 之后的数据帧透明转发

// 原生实现 WebSocket 代理的关键：
const http = require('http');
const httpProxy = require('http-proxy');

const proxy = httpProxy.createProxyServer({
  ws: true,
  target: 'ws://localhost:3001',
});

const server = http.createServer(app);
server.on('upgrade', (req, socket, head) => {
  // 拦截 WebSocket 升级请求
  proxy.ws(req, socket, head);
});

server.listen(5173);
```

## 代理性能优化

### 连接池（keepAlive）

```js
const http = require('http');
const agent = new http.Agent({
  keepAlive: true,        // 复用连接
  keepAliveMsecs: 30000,  // 保活时间
  maxSockets: Infinity,   // 无限并行
  maxFreeSockets: 256,    // 空闲连接数
});

// 传入 agent
createProxyMiddleware({
  target: 'http://localhost:3001',
  agent: agent,
});
```

### 响应压缩

```js
on: {
  proxyRes: (proxyRes, req, res) => {
    // 如果目标未压缩，代理可以压缩
    const encoding = req.headers['accept-encoding'];
    if (encoding?.includes('gzip') && !proxyRes.headers['content-encoding']) {
      // 使用 compression 中间件
    }
  },
}
```

### 请求超时与重试

```js
const proxy = createProxyMiddleware({
  target: 'http://localhost:3001',
  timeout: 30000,          // 30 秒超时
  proxyTimeout: 30000,     // 后端响应超时
  
  on: {
    error: (err, req, res) => {
      // 超时时返回友好提示
      if (err.code === 'ECONNRESET' || err.code === 'ETIMEDOUT') {
        res.status(504).json({ error: 'Gateway Timeout' });
      }
    },
    proxyReq: (proxyReq) => {
      proxyReq.setTimeout(30000, () => {
        proxyReq.destroy();
      });
    },
  },
});
```
