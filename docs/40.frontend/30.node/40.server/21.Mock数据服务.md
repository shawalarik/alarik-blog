## 概述

Mock 数据服务让前端团队可以在没有真实后端的情况下独立开发和测试。好的 Mock 方案不仅能模拟数据，还能模拟延迟、错误、分页等真实场景。本文介绍基于 Express/Koa 的 Mock 服务集成方案。

## Mock 方案对比

| 方案 | 类型 | 优点 | 缺点 | 适用场景 |
|------|------|------|------|---------|
| **JSON 文件** | 静态 | 最简单 | 无法动态生成 | 固定数据 |
| **Express Mock Server** | 自建 | 灵活可控 | 需维护代码 | 中大型项目 |
| **MSW (Mock Service Worker)** | Service Worker | 拦截网络请求 | 学习成本 | 现代 SPA 项目 |
| **json-server** | 即用型 | CRUD 自动生成 | 定制受限 | 快速原型 |
| **Mock.js** | 数据生成 | 随机数据丰富 | 需配合其他方案 | 数据构造 |

### 入口文件

```js
// mock/index.js
const express = require('express');
const cors = require('cors');
const userRoutes = require('./routes/users');
const productRoutes = require('./routes/products');
const orderRoutes = require('./routes/orders');

const app = express();

app.use(cors({ origin: '*' }));
app.use(express.json());

// 注册 Mock 路由
app.use('/api/users', userRoutes);
app.use('/api/products', productRoutes);
app.use('/api/orders', orderRoutes);

// 健康检查
app.get('/health', (req, res) => {
  res.json({ status: 'ok', service: 'mock-server', timestamp: Date.now() });
});

const PORT = process.env.MOCK_PORT || 3800;
app.listen(PORT, () => {
  console.log(`🚀 Mock Server running at http://localhost:${PORT}`);
  console.log(`   API Base: http://localhost:${PORT}/api`);
});
```

### 用户 Mock API 示例

```js
// mock/routes/users.js
const express = require('express');
const router = express.Router();
const { randomDelay } = require('../utils/delay');
const { generateUsers } = require('../utils/random');
const { paginate } = require('../utils/paginate');

let users = generateUsers(50);  // 生成 50 条初始数据
let nextId = users.length + 1;

// 获取用户列表（支持搜索、排序、分页）
router.get('/', randomDelay(100, 500), (req, res) => {
  let result = [...users];
  
  // 搜索
  const { keyword, role, status } = req.query;
  if (keyword) {
    const kw = keyword.toLowerCase();
    result = result.filter(u =>
      u.name.toLowerCase().includes(kw) ||
      u.email.toLowerCase().includes(kw)
    );
  }
  if (role) result = result.filter(u => u.role === role);
  if (status) result = result.filter(u => u.status === status);
  
  // 排序
  const { sort = 'createdAt', order = 'desc' } = req.query;
  result.sort((a, b) => {
    const valA = a[sort] ?? '';
    const valB = b[sort] ?? '';
    const cmp = typeof valA === 'string'
      ? valA.localeCompare(valB)
      : valA - valB;
    return order === 'desc' ? -cmp : cmp;
  });
  
  // 分页
  const paginatedData = paginate(result, req.query);
  
  res.json({
    code: 0,
    data: paginatedData.items,
    pagination: paginatedData.meta,
  });
});

// 获取单个用户
router.get('/:id', randomDelay(50, 200), (req, res) => {
  const user = users.find(u => u.id === Number(req.params.id));
  if (!user) {
    return res.status(404).json({ code: 404, message: 'User not found' });
  }
  res.json({ code: 0, data: user });
});

// 创建用户
router.post('/', randomDelay(200, 600), (req, res) => {
  const { name, email, role } = req.body;
  
  if (!name || !email) {
    return res.status(400).json({ code: 400, message: 'Name and email required' });
  }
  
  const newUser = {
    id: nextId++,
    name,
    email,
    role: role || 'member',
    status: 'active',
    avatar: `https://i.pravatar.cc/150?img=${nextId}`,
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
  };
  
  users.push(newUser);
  res.status(201).json({ code: 0, data: newUser });
});

// 更新用户
router.put('/:id', randomDelay(200, 500), (req, res) => {
  const idx = users.findIndex(u => u.id === Number(req.params.id));
  if (idx === -1) {
    return res.status(404).json({ code: 404, message: 'User not found' });
  }
  
  users[idx] = {
    ...users[idx],
    ...req.body,
    id: users[idx].id,          // 防止修改 ID
    updatedAt: new Date().toISOString(),
  };
  
  res.json({ code: 0, data: users[idx] });
});

// 删除用户
router.delete('/:id', randomDelay(100, 300), (req, res) => {
  const idx = users.findIndex(u => u.id === Number(req.params.id));
  if (idx === -1) {
    return res.status(404).json({ code: 404, message: 'User not found' });
  }
  
  const deleted = users.splice(idx, 1)[0];
  res.json({ code: 0, data: deleted, message: 'Deleted successfully' });
});

module.exports = router;
```

## Mock 工具函数

### 模拟延迟

```js
// mock/utils/delay.js
function randomDelay(min = 100, max = 800) {
  return (req, res, next) => {
    const ms = Math.floor(Math.random() * (max - min + 1)) + min;
    setTimeout(next, ms);
  };
}

// 固定延迟
function fixedDelay(ms) {
  return (req, res, next) => setTimeout(next, ms);
}

// 按概率模拟慢请求
function occasionalSlowDown(slowProb = 0.1, slowMs = 3000) {
  return (req, res, next) => {
    if (Math.random() < slowProb) {
      console.log(`⏳ Slow request simulated: ${req.method} ${req.url}`);
      return setTimeout(next, slowMs);
    }
    next();
  };
}

module.exports = { randomDelay, fixedDelay, occasionalSlowDown };
```

### 随机数据生成

```js
// mock/utils/random.js
const NAMES = [
  '张三', '李四', '王五', '赵六', '钱七',
  'Alice', 'Bob', 'Charlie', 'Diana', 'Edward',
];

const DOMAINS = ['gmail.com', 'outlook.com', 'qq.com', '163.com'];
const ROLES = ['admin', 'editor', 'member'];
const STATUSES = ['active', 'inactive', 'suspended'];

function randomItem(arr) {
  return arr[Math.floor(Math.random() * arr.length)];
}

function randomNumber(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}

function generateUsers(count = 20) {
  return Array.from({ length: count }, (_, i) => ({
    id: i + 1,
    name: randomItem(NAMES) + String.fromCharCode(65 + (i % 26)),
    email: `user${i + 1}@${randomItem(DOMAINS)}`,
    avatar: `https://i.pravatar.cc/150?img=${(i % 70) + 1}`,
    role: randomItem(ROLES),
    status: randomItem(STATUSES),
    phone: `1${randomNumber(30000000000, 99999999999)}`,
    department: ['技术部', '产品部', '市场部', '运营部'][i % 4],
    joinDate: new Date(2023, randomNumber(0, 11), randomNumber(1, 28)).toISOString(),
    lastLogin: new Date(Date.now() - randomNumber(0, 86400000 * 30)).toISOString(),
    createdAt: new Date(2023, randomNumber(0, 11), randomNumber(1, 28)).toISOString(),
    updatedAt: new Date().toISOString(),
  }));
}

function generateProducts(count = 30) {
  const categories = ['电子产品', '服装', '食品', '图书', '家居'];
  const brands = ['Apple', 'Samsung', 'Nike', 'IKEA', 'Uniqlo'];
  
  return Array.from({ length: count }, (_, i) => ({
    id: i + 1,
    name: `Product ${String.fromCharCode(65 + (i % 26))}-${i + 1}`,
    category: randomItem(categories),
    brand: randomItem(brands),
    price: parseFloat((Math.random() * 10000 + 10).toFixed(2)),
    stock: randomNumber(0, 500),
    rating: parseFloat((Math.random() * 2 + 3).toFixed(1)),
    sales: randomNumber(0, 10000),
    image: `https://picsum.photos/seed/prod${i + 1}/400/300`,
    description: `这是商品 ${i + 1} 的详细描述`,
    status: Math.random() > 0.1 ? 'online' : 'offline',
    createdAt: new Date(2024, randomNumber(0, 11), randomNumber(1, 28)).toISOString(),
  }));
}

module.exports = { generateUsers, generateProducts, randomItem, randomNumber };
```

### 分页工具

```js
// mock/utils/paginate.js
function paginate(items, query = {}) {
  const page = Number(query.page) || 1;
  const pageSize = Number(query.pageSize) || 10;
  
  const total = items.length;
  const totalPages = Math.ceil(total / pageSize);
  
  // 校正 page 范围
  const validPage = Math.max(1, Math.min(page, totalPages || 1));
  const start = (validPage - 1) * pageSize;
  const end = start + pageSize;
  
  return {
    items: items.slice(start, end),
    meta: {
      page: validPage,
      pageSize,
      total,
      totalPages,
      hasNext: validPage < totalPages,
      hasPrev: validPage > 1,
    },
  };
}

module.exports = { paginate };
```

## 模拟错误响应

### 错误率注入

```js
// 按概率模拟各种错误
function simulateErrors(errorConfig = {}) {
  const {
    rate_400 = 0.05,    // 5% 概率返回 400
    rate_401 = 0.02,    // 2% 概率返回 401
    rate_403 = 0.02,
    rate_404 = 0.05,
    rate_500 = 0.03,    // 3% 概率返回 500
    rate_timeout = 0.01, // 1% 概率超时
  } = errorConfig;
  
  return (req, res, next) => {
    const rand = Math.random();
    
    if (rand < rate_400) {
      return res.status(400).json({
        code: 400,
        message: 'Bad Request: Invalid parameters',
        errors: [{ field: 'email', message: 'Email format invalid' }],
      });
    }
    
    if (rand < rate_400 + rate_401) {
      return res.status(401).json({
        code: 401,
        message: 'Unauthorized: Token expired or invalid',
      });
    }
    
    if (rand < rate_400 + rate_401 + rate_403) {
      return res.status(403).json({
        code: 403,
        message: 'Forbidden: Insufficient permissions',
      });
    }
    
    if (rand < rate_400 + rate_401 + rate_403 + rate_404) {
      return res.status(404).json({
        code: 404,
        message: 'Not Found: Resource does not exist',
      });
    }
    
    if (rand < rate_400 + rate_401 + rate_403 + rate_404 + rate_500) {
      return res.status(500).json({
        code: 500,
        message: 'Internal Server Error',
        traceId: `trace-${Date.now()}`,
      });
    }
    
    if (rand < rate_400 + rate_401 + rate_403 + rate_404 + rate_500 + rate_timeout) {
      // 永远不响应（超时模拟）
      return;  // 不调用 next()，请求挂起
    }
    
    next();
  };
}

// 使用：仅在开发环境启用错误模拟
if (process.env.MOCK_ERRORS === 'true') {
  app.use(simulateErrors({
    rate_400: 0.1,    // 开发时可提高错误率
    rate_500: 0.05,
  }));
}
```

## 与 Vite/Webpack 集成

### Vite 插件方式

```js
// vite.mock-plugin.js
export default function mockPlugin(options = {}) {
  const { enabled = true, prefix = '/api' } = options;
  let mockServer;
  
  return {
    name: 'vite-plugin-mock',
    configureServer(server) {
      if (!enabled) return;
      
      // 使用中间件模式
      server.middlewares.use(async (req, res, next) => {
        if (!req.url?.startsWith(prefix)) return next();
        
        try {
          // 导入并使用 mock 路由
          const { handler } = await import('./mock/handler.js');
          await handler(req, res);
        } catch (e) {
          next(e);
        }
      });
    },
  };
}

// vite.config.ts
import mockPlugin from './vite.mock-plugin';

export default {
  plugins: [
    mockPlugin({
      enabled: process.env.VITE_MOCK === 'true',
    }),
  ],
};
```
