## 概述

HMR（Hot Module Replacement，热模块替换）是现代开发体验的核心能力之一——修改代码后无需刷新页面即可看到变更效果。HMR 的本质是**开发服务器与浏览器之间通过 WebSocket 进行实时通信**。本文深入剖析 HMR 的通信协议、消息格式和客户端重连机制。

## HMR 整体架构

```
┌──────────────────────────────────────────────────────────┐
│                      开发工作流                            │
│                                                          │
│  编辑器修改文件                                            │
│       │                                                  │
│       ▼                                                  │
│  文件监听 (chokidar / Watcher API)                        │
│       │                                                  │
│       ▼                                                  │
│  ┌─────────────┐                                         │
│  │ Dev Server   │  1. 分析变更的模块                      │
│  │              │  2. 确定受影响的模块（依赖图）            │
│  │              │  3. 构建热更新载荷                       │
│  └──────┬──────┘                                         │
│         │                                                │
│         │ WebSocket 消息                                  │
│         │ {"type": "update", "updates": [...]}           │
│         ▼                                                │
│  ┌─────────────┐                                         │
│  │ Browser Client│  4. 接收 HMR 载荷                      │
│  │ (HMR Runtime)│  5. 加载新模块代码                       │
│  │              │  6. 执行 accept 回调                     │
│  │              │  7. 若不可热更新 → full reload          │
│  └─────────────┘                                         │
└──────────────────────────────────────────────────────────┘
```

## WebSocket 通信建立

### 服务端：创建 WebSocket 服务

```js
const { createServer } = require('node:http');
const { WebSocketServer } = require('ws');

const server = createServer();
const wss = new WebSocketServer({ server, path: '/ws-hmr' });

wss.on('connection', (ws, req) => {
  console.log('[HMR] Client connected:', req.socket.remoteAddress);
  
  // 发送连接确认
  ws.send(JSON.stringify({
    type: 'connected',
    data: { timestamp: Date.now(), version: '1.0.0' },
  }));
  
  ws.on('message', (raw) => {
    try {
      const msg = JSON.parse(raw.toString());
      handleMessage(ws, msg);
    } catch (e) {
      console.warn('[HMR] Invalid message:', raw.toString());
    }
  });
  
  ws.on('close', (code, reason) => {
    console.log('[HMR] Client disconnected:', code, reason.toString());
  });
  
  ws.on('error', (err) => {
    console.error('[HMR] WebSocket error:', err.message);
  });
});

server.listen(5173);
console.log('Dev server with HMR at ws://localhost:5173/ws-hmr');
```

### 客户端：连接 HMR Socket

```js
// hmr-client.js（注入到浏览器）
let socket;

function connectHMR(base = '') {
  const protocol = location.protocol === 'https:' ? 'wss:' : 'ws:';
  const host = location.host;
  const url = `${protocol}//${host}${base}/ws-hmr`;
  
  socket = new WebSocket(url);
  
  socket.addEventListener('open', () => {
    console.log('[HMR] Connected to dev server');
  });
  
  socket.addEventListener('message', ({ data }) => {
    handleHMRMessage(JSON.parse(data));
  });
  
  socket.addEventListener('close', (event) => {
    console.log('[HMR] Connection closed, reconnecting...', event.code);
    scheduleReconnect();
  });
  
  socket.addEventListener('error', () => {
    console.error('[HMR] Connection error');
  });
}

// 初始连接
connectHMR();
```

## HMR 消息协议

### 标准消息类型

```typescript
// HMR 消息类型定义
interface HMRMessage {
  type: MessageType;
  data?: any;
  error?: string;
  timestamp?: number;
}

type MessageType =
  | 'connected'          // 连接成功
  | 'update'             // 模块更新
  | 'full-reload'        // 全页面重载
  | 'custom'             // 自定义事件
  | 'error'              // 错误通知
  | 'prune'              // 移除不再使用的模块
  | 'style-update'       // CSS 更新（特殊快速通道）
```

### update 消息（核心）

```js
// 服务端发送 update 消息
function sendUpdate(clients, updates) {
  const message = JSON.stringify({
    type: 'update',
    data: {
      // 更新的模块列表
      updates: updates.map(({ acceptedPath, timestamp, moduleId }) => ({
        type: 'js-update',          // 或 'css-update', 'html-update'
        path: acceptedPath,         // 客户端接受的路径
        acceptedPath: acceptedPath,
        timestamp: timestamp,
        moduleId: moduleId,          // 模块唯一标识
      })),
      // 变更原因
      cause: 'file-change',
    },
    timestamp: Date.now(),
  });
  
  clients.forEach(client => {
    if (client.readyState === WebSocket.OPEN) {
      client.send(message);
    }
  });
}
```

### 各消息类型的完整格式

```js
// ===== 1. connected =====
{ type: 'connected', data: { timestamp: 1700000000000 } }

// ===== 2. update（JS 模块）=====
{
  type: 'update',
  data: {
    updates: [{
      type: 'js-update',
      path: '/src/components/Button.vue',
      acceptedPath: '/src/components/Button.vue?vue&type=script&lang.ts&t=1700000000000',
      timestamp: 1700000000000,
      moduleId: '/src/components/Button.vue',
    }],
  }
}

// ===== 3. style-update（CSS 热更新）=====
{
  type: 'update',
  data: {
    updates: [{
      type: 'css-update',
      path: '/src/styles/main.css',
      acceptedPath: '/src/styles/main.css?t=1700000000001',
      timestamp: 1700000000001,
    }]
  }
}

// ===== 4. full-reload =====
{ type: 'full-reload', data: { path: '/about' } }

// ===== 5. custom =====
{ type: 'custom', data: { event: 'config-changed', payload: { theme: 'dark' } } }

// ===== 6. error =====
{ type: 'error', error: 'Failed to compile: Unexpected token', data: { file: '/src/App.tsx', line: 42 } }

// ===== 7. prune =====
{ type: 'prune', data: { staleModuleIds: ['/src/deprecated/old-util.ts'] } }
```

### 客户端处理 update 消息

```js
// hmr-client.js
async function handleHMRMessage(payload) {
  switch (payload.type) {
    case 'connected':
      console.log('[HMR] 🟢 Ready');
      break;
      
    case 'update': {
      const { updates } = payload.data;
      // 过滤出当前客户端关心的更新
      const acceptedUpdates = updates.filter(update => 
        hotModulesMap.has(update.moduleId)
      );
      
      if (acceptedUpdates.length === 0) {
        // 没有接受任何更新的模块 → 全部重载
        return fullReload();
      }
      
      // 并行加载新模块代码
      await Promise.all(
        acceptedUpdates.map(update => fetchUpdate(update))
      );
      
      // 执行 accept 回调（替换模块导出）
      for (const update of acceptedUpdates) {
        const mod = hotModulesMap.get(update.moduleId);
        if (mod?.callback) {
          await mod.callback(mod.deps);
        }
      }
      
      // 清理旧模块
      pruneStaleModules(updates);
      
      // 通知 UI 更新
      dispatchCustomEvent('vite:after-update', { updates });
      console.log(`[HMR] 🔥 Updated ${acceptedUpdates.length} module(s)`);
      break;
    }
    
    case 'full-reload':
      fullReload(payload.data?.path);
      break;
      
    case 'error':
      console.error('[HMR] ❌', payload.error);
      showErrorOverlay(payload);
      break;
      
    case 'custom':
      dispatchCustomEvent(`vite:${payload.data.event}`, payload.data.payload);
      break;
      
    case 'prune':
      pruneModules(payload.data.staleModuleIds);
      break;
  }
}

// 通过动态 import 加载新模块代码
async function fetchUpdate(update) {
  const { path, acceptedPath, timestamp } = update;
  const mod = await import(`${acceptedPath}?t=${timestamp}`);
  // mod 包含最新的模块导出
  return mod;
}
```

## HMR Accept 机制

### 模块声明接受 HMR

```js
// src/components/Button.jsx
// 声明：当此模块被更新时，执行什么操作
import.meta.hot?.accept((newExports) => {
  // newExports 是模块的新导出
  // 在这里可以：
  // 1. 更新组件引用
  // 2. 触发局部重新渲染
  // 3. 保留组件内部状态
  
  console.log('[HMR] Button component updated:', newExports);
  rerenderComponent(newExports.default);
});

// 声明依赖模块的更新
import.meta.hot?.accept(['./utils', './theme'], ([newUtils, newTheme]) => {
  // ./utils 或 ./theme 更新时触发
  applyNewUtils(newUtils);
  applyNewTheme(newTheme);
});

// 声明自我失效（强制父模块重新加载）
import.meta.hot?.dispose(() => {
  // 清理副作用：定时器、事件监听器、DOM 引用
  cleanupTimers();
  removeEventListeners();
});
```

### HMR 边界（HMR Boundary）

```
main.tsx
  └→ App.tsx (accepts children)
       ├→ Header.tsx (accepts self)
       │    └→ Logo.tsx
       ├→ Button.tsx (accepts self)
       └→ Layout.tsx (accepts self)
            └→ Sidebar.tsx
                 └→ NavItem.tsx

当 Logo.tsx 变更时：
  Logo.tsx 热更新 ✅
  Header.tsx 的 accept 回调被触发（收到新的 Logo）
  App.tsx 无感知（Logo 的变化被 Header 封装了）

当 main.tsx 变更时：
  无法热更新（入口模块通常不支持 HMR）
  → Full Reload（整个页面刷新）
```

## 客户端重连机制

### 断线重连策略

```js
// hmr-client.js
let reconnectTimer = null;
let retryCount = 0;
const MAX_RETRIES = 30;
const BASE_DELAY = 1000;
const MAX_DELAY = 30000;

function scheduleReconnect() {
  if (reconnectTimer) return;
  
  // 指数退避算法
  const delay = Math.min(BASE_DELAY * Math.pow(2, retryCount), MAX_DELAY);
  retryCount++;
  
  console.log(`[HMR] Reconnecting in ${delay}ms (attempt ${retryCount}/${MAX_RETRIES})`);
  
  reconnectTimer = setTimeout(() => {
    reconnectTimer = null;
    
    if (retryCount > MAX_RETRIES) {
      console.error('[HMR] Max retries reached. Giving up.');
      showReconnectFailedUI();
      return;
    }
    
    connectHMR();
  }, delay);
}

// 连接成功后重置计数器
socket.addEventListener('open', () => {
  retryCount = 0;
  if (reconnectTimer) {
    clearTimeout(reconnectTimer);
    reconnectTimer = null;
  }
});
```

### 重连时间线

```
尝试    延迟时间     累计等待
────    ────────     ────────
 1      1,000ms      1s
 2      2,000ms      3s
 3      4,000ms      7s
 4      8,000ms      15s
 5     16,000ms      31s
 6     30,000ms      61s  ← 达到上限
...
30     30,000ms     ~15min ← 最终放弃
```

### 页面可见性重连

```js
// 当标签页从后台切回前台时，立即检查连接
document.addEventListener('visibilitychange', () => {
  if (document.visibilityState === 'visible') {
    if (!socket || socket.readyState !== WebSocket.OPEN) {
      console.log('[HMR] Page visible, reconnecting...');
      connectHMR();
    }
    // 即使连接看似正常，也发送 ping 检测
    if (socket?.readyState === WebSocket.OPEN) {
      socket.send(JSON.stringify({ type: 'ping' }));
    }
  }
});
```

## Vite HMR 协议细节

### Vite 的独特优化

```js
// Vite 的 CSS HMR 是独立的快速通道
// CSS 更新不走通用 update 流程

// 服务端检测到 .css 文件变更时：
{
  type: 'custom',
  data: {
    event: 'vite:update-style',
    data: {
      path: '/src/index.css',
      timestamp: Date.now(),
      content: '/* 新的 CSS 内容 */',  // 直接推送内容
    }
  }
}

// 客户端接收后：
// 直接创建/替换 <link> 标签或更新 <style>
// 无需模块重新加载 → 极快（< 10ms）
```

### Vite HMR 与 SSR 的兼容

```js
// SSR 环境下的 HMR 处理
if (import.meta.hot) {
  // 客户端 HMR
  import.meta.hot.accept();
  
  if (import.meta.env.SSR) {
    // SSR 环境：通知 dev server 重新渲染该模块
    import.meta.hot.on('vite:before-full-reload', () => {
      // 清除模块缓存
      delete require.cache[moduleId];
    });
  }
}
```
