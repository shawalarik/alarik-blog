## 概述

开发服务器虽然只在本地运行，但安全防护同样不可忽视。目录遍历攻击、CORS 配置不当、环境变量泄露等问题在开发环境也时有发生。本文系统讲解开发服务器的安全最佳实践。

## 防止目录遍历攻击

### 攻击原理

```
恶意请求：
GET /../../../etc/passwd
GET /..%2f..%2f..%2fetc%2fpasswd   (URL 编码)
GET /....//....//....//etc/passwd    (双重编码)

目标：读取服务器上任意文件！
```

### 防御实现

```js
const path = require('node:path');

const ROOT_DIR = path.resolve(__dirname, 'public');

function safePath(reqPath) {
  // 解码 URL 编码
  const decoded = decodeURIComponent(reqPath);
  
  // 规范化路径（解析 .. 和 .）
  const resolved = path.join(ROOT_DIR, decoded);
  
  // 关键检查：规范化后的路径必须在根目录内
  if (!resolved.startsWith(ROOT_DIR)) {
    return null;  // 尝试逃逸！
  }
  
  return resolved;
}

// 在静态文件处理中使用
const http = require('node:http');
const fs = require('node:fs');

const server = http.createServer((req, res) => {
  const url = new URL(req.url, `http://${req.headers.host}`);
  const safeFilePath = safePath(url.pathname);
  
  if (!safeFilePath) {
    res.statusCode = 403;
    res.end('Forbidden: Path traversal detected');
    console.warn(`⚠️ Path traversal attempt from ${req.socket.remoteAddress}: ${req.url}`);
    return;
  }
  
  // 继续正常文件服务...
});
```

### 更严格的白名单机制

```js
const ALLOWED_EXTENSIONS = new Set([
  '.html', '.css', '.js', '.json', '.png', '.jpg',
  '.svg', '.ico', '.woff', '.woff2', '.ttf', '.map',
  '.txt', '.xml', '.wasm'
]);

function isSafeRequest(url) {
  const pathname = url.pathname;
  
  // 检查扩展名
  const ext = path.extname(pathname).toLowerCase();
  if (!ALLOWED_EXTENSIONS.has(ext)) {
    return false;
  }
  
  // 检查隐藏文件（以 . 开头）
  if (path.basename(pathname).startsWith('.')) {
    return false;
  }
  
  // 检查敏感文件名
  const sensitiveFiles = [
    '.env', '.git', 'package.json', '.DS_Store',
    '.htaccess', 'web.config', '.npmrc',
  ];
  if (sensitiveFiles.some(f => pathname.includes(f))) {
    return false;
  }
  
  return true;
}
```

## 请求过滤

### IP 白名单

```js
const ALLOWED_IPS = new Set([
  '::1',           // IPv6 localhost
  '127.0.0.1',     // IPv4 localhost
  '::ffff:127.0.0.1',  // IPv4-mapped IPv6
]);

function ipFilter(req, res) {
  // 信任代理时获取真实 IP
  const forwarded = req.headers['x-forwarded-for'];
  const clientIp = forwarded ? forwarded.split(',')[0].trim() : req.socket.remoteAddress;
  
  if (!ALLOWED_IPS.has(clientIp) && process.env.NODE_ENV === 'development') {
    // 开发环境可以放宽，但记录警告
    console.warn(`🔍 Non-local access from IP: ${clientIp}`);
  } else if (!ALLOWED_IPS.has(clientIp)) {
    res.statusCode = 403;
    res.end('Access denied');
    return false;
  }
  return true;
}
```

### 方法过滤

```js
const ALLOWED_METHODS = new Set(['GET', 'HEAD', 'OPTIONS']);

function methodFilter(req, res) {
  if (!ALLOWED_METHODS.has(req.method)) {
    res.statusCode = 405;
    res.setHeader('Allow', [...ALLOWED_METHODS].join(', '));
    res.end('Method Not Allowed');
    return false;
  }
  return true;
}
```

### 路径过滤黑名单

```js
const BLOCKED_PATTERNS = [
  /\.env/i,
  /\.git\//i,
  /\/node_modules\//i,
  /package-lock\.json/i,
  /\.npmrc/i,
  /webpack\.config/i,
];

function pathFilter(req, res) {
  const url = req.url;
  
  for (const pattern of BLOCKED_PATTERNS) {
    if (pattern.test(url)) {
      res.statusCode = 403;
      res.end('Forbidden');
      return false;
    }
  }
  return true;
}
```

## CORS 配置

### 开发环境的 CORS 最佳配置

```js
// 开发环境：宽松但可控的 CORS
const corsOptions = {
  origin: [
    'http://localhost:5173',     // Vite 默认端口
    'http://localhost:3000',     // 可能的其他端口
    'http://127.0.0.1:5173',
    /^http:\/\/localhost:\d+$/,  // 匹配 localhost 的任意端口
  ],
  methods: ['GET', 'POST', 'PUT', 'DELETE', 'PATCH', 'OPTIONS'],
  allowedHeaders: [
    'Content-Type',
    'Authorization',
    'Accept',
    'X-Requested-With',
  ],
  credentials: true,
  maxAge: 86400,              // 预检缓存 24 小时
  optionsSuccessStatus: 204,
};

// Express 中使用
const cors = require('cors');
app.use(cors(corsOptions));

// Koa 中使用
app.use(cors(corsOptions));
```

### 动态 Origin 校验

```js
function dynamicCorsOrigin(ctx) {
  const requestOrigin = ctx.get('Origin');
  
  if (!requestOrigin) return '';  // 非浏览器请求
  
  // 允许的开发域名
  const allowedOrigins = [
    'http://localhost:5173',
    'http://localhost:3000',
    `http://${ctx.hostname}:*`,  // 同主机任意端口
  ];
  
  for (const pattern of allowedOrigins) {
    if (pattern === '*' || requestOrigin === pattern) {
      return requestOrigin;
    }
    // 通配符匹配
    if (pattern.endsWith(':*') && requestOrigin.startsWith(pattern.slice(0, -2))) {
      return requestOrigin;
    }
  }
  
  // 不匹配 → 返回空字符串（拒绝）
  return '';
}
```

### CORS 预检请求（OPTIONS）

```js
// 手动处理 OPTIONS（不依赖中间件）
app.options('*', (req, res) => {
  const origin = dynamicCorsOrigin({ get: (h) => req.headers[h.toLowerCase()] });
  
  if (origin) {
    res.setHeader('Access-Control-Allow-Origin', origin);
    res.setHeader('Access-Control-Allow-Methods', 'GET,POST,PUT,DELETE,PATCH');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type,Authorization');
    res.setHeader('Access-Control-Max-Age', '86400');
    res.setHeader('Access-Control-Allow-Credentials', 'true');
  }
  res.sendStatus(204);
});
```

## HTTPS 本地证书（mkcert）

### 为什么开发也需要 HTTPS？

```
需要 HTTPS 的场景：
├── Service Worker（必须 HTTPS 或 localhost）
├── geolocation API（部分浏览器限制）
├── clipboard API（需要安全上下文）
├── HTTP/2 推送测试
├── cookie 的 Secure 属性
└── 第三方 OAuth 回调
```

### 使用 mkcert 生成本地 CA 证书

```bash
# 安装 mkcert
# macOS:
brew install mkcert
nss-db   # Firefox 支持

# Windows (chocolatey):
choco install mkcert

# Linux:
apt install mkcert libnss3-tools  # Ubuntu/Debian

# 初始化本地 CA（只需执行一次）
mkcert -install
# ✓ Created a new local CA at "..."
# ⚠ The local CA is now installed in system trust store! ⚠

# 生成为你的开发域名签名的证书
# 单域名
mkcert localhost
# → localhost.pem  +  localhost-key.pem

# 多域名（含局域网 IP）
mkcert localhost 127.0.0.1 ::1 192.168.1.100 myapp.local
# → localhost+4.pem  +  localhost+4-key.pem

# 通配符（所有子域）
mkcert "*.example.test"
# → _wildcard.example.test.pem  +  key-file
```

### Node.js 中使用自签名证书

```js
const https = require('node:https');
const fs = require('node:fs');

const server = https.createServer({
  key: fs.readFileSync('./localhost-key.pem'),
  cert: fs.readFileSync('./localhost.pem'),
}, app);  // Express/Koa 应用

server.listen(443, () => {
  console.log('HTTPS Dev Server: https://localhost');
});

// 如果同时需要 HTTP（自动重定向到 HTTPS）
const http = require('node:http');
http.createServer((req, res) => {
  res.writeHead(301, { Location: `https://localhost${req.url}` });
  res.end();
}).listen(80);
```

## 环境变量保护

### 不要将敏感信息暴露给前端

```js
// ❌ 错误做法
app.get('/api/config', (req, res) => {
  res.json({
    API_KEY: process.env.API_KEY,        // 泄露给前端！
    DATABASE_URL: process.env.DB_URL,     // 泄露！
    SECRET_KEY: process.env.SECRET,       // 泄露！
  });
});

// ✅ 正确做法
app.get('/api/config', (req, res) => {
  res.json({
    // 仅暴露前端需要的信息
    API_BASE_URL: '/api/v1',
    APP_NAME: 'My App',
    VERSION: '1.2.0',
    ENVIRONMENT: process.env.NODE_ENV,     // 只暴露环境名称
    
    // 敏感信息绝对不返回
  });
});
```

### 服务端环境变量隔离

```bash
# .env.development（可提交到 Git，不含密钥）
VITE_APP_TITLE=My App
VITE_API_BASE=/api

# .env.local（Git 忽略，含本地密钥）
API_KEY=sk_live_xxxxx
DB_PASSWORD=secret123

# .env.production（生产环境）
VITE_API_BASE=https://api.example.com
```

```js
// .gitignore
.env.local
.env.*.local
*.key
*-key.pem
secrets/
```

### 防止错误信息泄露堆栈

```js
// 生产/预发布环境隐藏详细错误
app.use((err, req, res, next) => {
  const isDev = process.env.NODE_ENV !== 'production';
  
  res.status(err.status || 500).json({
    error: err.message || 'Internal Server Error',
    ...(isDev ? { stack: err.stack } : {}),  // 仅开发环境暴露 stack
    ...(isDev ? { details: err.details } : {}),
    
    // 生产环境额外字段
    ...(!isDev ? {
      requestId: req.id || generateId(),
      timestamp: new Date().toISOString(),
    } : {}),
  });
});

// 全局未捕获异常处理
process.on('uncaughtException', (err) => {
  console.error('Uncaught Exception:', err.message);
  // 生产环境：不要打印完整 stack 到日志
  // 发送到日志聚合服务（Sentry/DataDog）
});

process.on('unhandledRejection', (reason, promise) => {
  console.error('Unhandled Rejection:', reason);
});
```
