## 概述

静态资源的处理是开发服务器和生产服务器的核心功能之一。高效的静态资源策略能显著提升页面加载速度。本文详解静态目录挂载、缓存策略、压缩算法以及 ETag 协商缓存的实现。

## 静态目录挂载

### Express 方式

```js
const express = require('express');
const path = require('node:path');

const app = express();

// 基础挂载
app.use(express.static('public'));

// 指定路径前缀挂载
app.use('/assets', express.static(path.join(__dirname, 'dist/assets')));
app.use('/images', express.static('uploads'));

// 多个目录按优先级查找
app.use(express.static('dist'));       // 优先查找 dist
app.use(express.static('public'));     // dist 找不到再找 public

// 详细选项
app.use('/static', express.static('public', {
  dotfiles: 'ignore',          // ignore | allow | deny
  etag: true,                  // 生成 ETag
  extensions: ['html'],        // 自动补全扩展名（/about → /about.html）
  index: ['index.html'],       // 目录默认文件
  lastModified: true,          // 设置 Last-Modified 头
  maxAge: '1d',               // Cache-Control max-age
  redirect: true,              // 路径以 / 结尾时重定向到 index
  setHeaders: (res, filePath) => {
    // 自定义响应头
    if (filePath.endsWith('.js')) {
      res.set('Content-Type', 'application/javascript; charset=utf-8');
    }
  },
}));
```

### Koa 方式

```js
const Koa = require('koa');
const koaStatic = require('koa-static');
const mount = require('koa-mount');
const app = new Koa();

// 根路径挂载
app.use(koaStatic('dist'));

// 子路径挂载（需要 koa-mount）
app.use(mount('/assets', koaStatic('public/assets')));

// 自定义选项
app.use(koaStatic('public', {
  maxage: 86400000,             // 24 小时（毫秒）
  hidden: false,               // 是否允许访问隐藏文件（. 开头）
  gzip: true,                  // 当客户端接受 gzip 且存在 .gz 文件时提供
  brotli: true,                // 同上但针对 .br 文件
  setHeaders: (res, path) => {
    if (path.endsWith('.wasm')) {
      res.set('Content-Type', 'application/wasm');
    }
  },
}));
```

### 原生 Node.js 实现

```js
const fs = require('node:fs');
const path = require('node:path');
const mime = require('mime-types');  // npm 包或自行维护 MIME 表

class StaticServer {
  constructor(rootDir, options = {}) {
    this.rootDir = path.resolve(rootDir);
    this.maxAge = options.maxAge || 0;
    this.dotfiles = options.dotfiles || 'ignore';
    this.etag = options.etag !== false;
    this.lastModified = options.lastModified !== false;
    this.fallthrough = options.fallthrough !== false;  // 404 时是否传递给下一个中间件
  }

  middleware() {
    return async (req, res, next) => {
      const url = new URL(req.url, `http://${req.headers.host}`);
      let filePath = path.join(this.rootDir, decodeURIComponent(url.pathname));
      
      // 规范化并安全检查
      filePath = path.normalize(filePath);
      if (!filePath.startsWith(this.rootDir)) {
        res.statusCode = 403;
        res.end('Forbidden');
        return;
      }
      
      // 隐藏文件检查
      const basename = path.basename(filePath);
      if (basename.startsWith('.') && this.dotfiles === 'ignore') {
        return next();  // 或 403
      }
      
      // 目录 → 尝试 index.html
      try {
        const stat = await fs.promises.stat(filePath);
        if (stat.isDirectory()) {
          filePath = path.join(filePath, 'index.html');
        }
        
        // 条件请求：If-None-Match / If-Modified-Since
        if (this.checkConditional(req, stat, res)) {
          return;
        }
        
        // 读取并发送
        const data = await fs.promises.readFile(filePath);
        const ext = path.extname(filePath);
        const contentType = mime.lookup(ext) || 'application/octet-stream';
        
        this.setCacheHeaders(res, ext);
        res.writeHead(200, {
          'Content-Type': `${contentType}; charset=utf-8`,
          'Content-Length': data.length,
          ...(this.etag ? { 'ETag': this.generateETag(stat) } : {}),
          ...(this.lastModified ? { 'Last-Modified': stat.mtime.toUTCString() } : {}),
        });
        res.end(data);
        
      } catch (err) {
        if (err.code === 'ENOENT') {
          if (this.fallthrough) return next();
          res.writeHead(404);
          res.end('Not Found');
        } else {
          res.writeHead(500);
          res.end('Internal Error');
        }
      }
    };
  }

  checkConditional(req, stat, res) {
    // ETag 检查
    if (this.etag && req.headers['if-none-match']) {
      const clientEtag = req.headers['if-none-match'];
      const serverEtag = `"${stat.size}-${Date.parse(stat.mtime)}"`;
      if (clientEtag === serverEtag || clientEtag === '*') {
        res.writeHead(304);
        res.end();
        return true;
      }
    }
    
    // Last-Modified 检查
    if (this.lastModified && req.headers['if-modified-since']) {
      const clientTime = Date.parse(req.headers['if-modified-since']);
      if (clientTime >= stat.mtime.getTime()) {
        res.writeHead(304);
        res.end();
        return true;
      }
    }
    
    return false;
  }

  setCacheHeaders(res, ext) {
    const cacheMap = {
      '.html': 'no-cache',
      '.css':  'public, max-age=31536000, immutable',
      '.js':   'public, max-age=31536000, immutable',
      '.png':  'public, max-age=31536000, immutable',
      '.jpg':  'public, max-age=31536000, immutable',
      '.svg':  'public, max-age=31536000, immutable',
      '.woff':'public, max-age=31536000, immutable',
      '.woff2':'public, max-age=31536000, immutable',
      '.ico':  'public, max-age=604800',
      '.wasm': 'public, max-age=31536000, immutable',
    };
    
    const cacheValue = cacheMap[ext] || `public, max-age=${this.maxAge}`;
    res.setHeader('Cache-Control', cacheValue);
  }

  generateETag(stat) {
    return `"${stat.size}-${Date.parse(stat.mtime)}"`;
    // 更精确的方式是用文件的 SHA-1 hash
  }
}

// 使用
const staticServer = new StaticServer('public', { maxAge: 86400000 });
app.use(staticServer.middleware());
```

## 缓存策略（Cache-Control）

### 指令详解

| 指令 | 含义 | 适用场景 |
|------|------|---------|
| `public` | 可被任何缓存存储（CDN、代理） | 静态资源 |
| `private` | 只有浏览器可缓存（CDN 不能） | HTML 页面 |
| `no-cache` | 每次需验证新鲜度（协商缓存） | 入口 HTML |
| `no-store` | 完全不缓存 | 敏感数据 |
| `max-age=N` | N 秒内直接用缓存 | 版本化的 JS/CSS |
| `immutable` | max-age 期间不再发送条件请求 | 带 hash 的资源 |
| `must-revalidate` | 过期后必须向源站验证 | API 数据 |
| `stale-while-revalidate=N` | 过期后 N 秒内可用旧版同时后台更新 | 字体等大文件 |

### 推荐配置方案

```
资源类型                    Cache-Control                          原因
─────────────────────────────────────────────────────────────────────
HTML 入口文件              no-cache                               随时可能更新
带 hash 的 JS/CSS          public, max-age=31536000, immutable     hash 变则内容变
图片（无hash）            public, max-age=86400                   一天更新一次可接受
字体文件                  public, max-age=31536000, immutable     极少变动
API JSON 响应             no-cache, must-revalidate              数据实时性要求高
第三方库 CDN             公共 CDN 自行设置                       交给 CDN 管理
```

### Express 中配置

```js
const express = require('express');
const app = express();

// 不同路径不同缓存策略
app.use((req, res, next) => {
  const url = req.url;
  
  if (url.match(/\.[a-f0-9]{8}\.(js|css)$/)) {
    // 带内容 hash 的资源：强缓存一年
    res.set('Cache-Control', 'public, max-age=31536000, immutable');
  } else if (url.match(/\.(js|css|png|jpe?g|svg|woff2?)$/)) {
    // 无 hash 的静态资源：短时间缓存
    res.set('Cache-Control', 'public, max-age=86400');
  } else if (url.match(/\.html?$/)) {
    // HTML：每次验证
    res.set('Cache-Control', 'no-cache');
  } else {
    // API：不缓存
    res.set('Cache-Control', 'no-store');
  }
  
  next();
});

app.use(express.static('dist'));
```

## gzip / brotli 压缩

### 为什么需要压缩？

```
原始大小 vs 压缩后：

文件类型         原始       gzip(-9)    brotli(11)
─────────────────────────────────────────────────
main.js (500KB)   500KB      120KB       95KB
styles.css (80KB)  80KB      18KB        14KB
index.html (10KB)  10KB       3KB         2.5KB
vendor.js (2MB)    2MB        480KB       380KB

平均压缩率：gzip ~70%, brotli ~80%
首屏加载时间减少：50%~70%
```

### Express 中使用 compression

```js
const compression = require('compression');

app.use(compression({
  level: 6,              // 压缩级别 1-9（越大越慢越省空间）
  threshold: 1024,       // 小于 1KB 的不压缩（压缩开销不值得）
  filter: (req, res) => {
    if (req.headers['x-no-compression']) return false;
    // 只压缩可压缩的类型
    return compression.filter(req, res);
  },
  // 已压缩的文件不再压缩（如 .gz, .br）
}));

// 或者手动检测 Accept-Encoding
app.use((req, res, next) => {
  const encoding = req.headers['accept-encoding'] || '';
  if (encoding.includes('br')) {
    res.set('Content-Encoding', 'br');
    // 使用 brotli 流
  } else if (encoding.includes('gzip')) {
    res.set('Content-Encoding', 'gzip');
    // 使用 zlib.createGzip() 流
  }
  next();
});
```

### Nginx 配置（生产推荐）

```nginx
# 启用 gzip 和 brotli 动态压缩
gzip on;
gzip_vary on;
gzip_proxied any;
gzip_comp_level 6;
gzip_types
    text/plain
    text/css
    text/xml
    text/javascript
    application/javascript
    application/json
    application/xml+rss
    image/svg+xml
    font/woff2;

brotli on;
brotli_comp_level 6;
brotli_types
    text/plain
    text/css
    text/javascript
    application/javascript
    application/json
    image/svg+xml
    font/woff2;

# 静态资源优先提供预压缩文件
gzip_static on;    # 先找 .gz 文件
brotli_static on;   # 再找 .br 文件
# 都没有才动态压缩
```

## ETag 与协商缓存

### 工作原理

```
首次请求（200 + ETag）：
Client                              Server
  │ ── GET /style.css ─────────────→ │
  │ ←── 200 OK                      │
  │      ETag: "abc123"              │
  │      Content-Length: 18000       │
  │      Body: [CSS 内容]            │
  │                                  │
再次请求（304 Not Modified）：
  │ ── GET /style.css ─────────────→ │
  │    If-None-Match: "abc123"      │
  │ ←── 304 Not Modified            │
  │      （无 body！）                │
  │      节省带宽：~18KB             │
```

### ETag 生成策略

```js
// 策略 1：基于文件属性（快速但不精确）
function weakETag(stat) {
  return `W/"${stat.size}-${Math.floor(stat.mtimeMs)}"`;
  // W/ 表示弱校验（适用于动态生成的内容）
}

// 策略 2：基于文件内容哈希（精确但较慢）
const crypto = require('node:crypto');
async function strongETag(filePath) {
  const content = await fs.promises.readFile(filePath);
  const hash = crypto.createHash('md5').update(content).digest('hex');
  return `"${hash}"`;
}

// 策略 3：折中方案（修改时间 + 大小）
function fastETag(stat) {
  return `"${stat.mtimeMs.toString(16)}-${stat.size.toString(16)}"`;
}
```

### Last-Modified 与 ETag 的配合

```js
app.use(async (req, res, next) => {
  const filePath = resolvePath(req.url);
  const stat = await fs.promises.stat(filePath);
  
  const etag = fastETag(stat);
  const lastModified = stat.mtime.toUTCString();
  
  // 设置响应头
  res.set('ETag', etag);
  res.set('Last-Modified', lastModified);
  
  // 检查 If-None-Match（优先级更高）
  const ifNoneMatch = req.headers['if-none-match'];
  if (ifNoneMatch && (ifNoneMatch === etag || ifNoneMatch === '*')) {
    res.sendStatus(304);
    return;
  }
  
  // 检查 If-Modified-Since
  const ifModifiedSince = req.headers['if-modified-since'];
  if (ifModifiedSince && new Date(ifModifiedSince) >= stat.mtime) {
    res.sendStatus(304);
    return;
  }
  
  next();
});
```

### Vary 头的重要性

```js
// 当同一 URL 对不同用户可能返回不同内容时，必须设置 Vary
app.use((req, res, next) => {
  // 根据 Accept-Encoding 返回不同压缩版本
  if (req.headers['accept-encoding']) {
    res.set('Vary', 'Accept-Encoding');
  }
  
  // 根据语言返回不同内容
  if (req.headers['accept-language']) {
    res.set('Vary', 'Accept-Language');
  }
  
  next();
});

// ⚠️ 不设 Vary 的问题：
// 用户 A 收到 gzip 版本，缓存了
// 用户 B（不支持 gzip）请求同一个 URL
// CDN 直接返回 A 的 gzip 缓存 → B 无法解压！
```

## 性能优化清单

| 优化项 | 方法 | 预期收益 |
|--------|------|---------|
| **版本化文件名** | `[name].[contenthash:8].js` | 强缓存命中 100% |
| **Brotli 压缩** | 预生成 `.br` 文件 | 比 gzip 省 15~25% 体积 |
| **HTTP/2 Push** | 主动推送关键 CSS/JS | 减少 1 个 RTT |
| **Preload/Prefetch** | `<link rel="preload">` | 提示浏览器优先级 |
| **字体优化** | `font-display: swap` | 减少文字不可见时间 |
| **图片格式** | WebP/AVIF 替代 JPEG | 体积减小 25~50% |
| **CDN 分发** | 全球边缘节点 | 延迟降低 50ms+ |
| **DNS 预解析** | `<link rel="dns-prefetch">` | DNS 查询提前完成 |
| **连接预建立** | `<link rel="preconnect">` | TCP/TLS 握手提前 |
