## 概述

在实际项目中，中间件不是孤立使用的——它们需要**组合、编排、复用**，并在出现异常时有统一的**错误边界**兜底。本文探讨中间件的组合模式、全局错误捕获策略、优雅降级方案以及实用的日志中间件。

## 中间件组合模式

### 串行组合（Pipeline）

```js
// 最常见的模式：一个接一个执行
function compose(middlewares) {
  return function composedMiddleware(ctx, next) {
    let index = -1;
    
    function dispatch(i) {
      if (i <= index) throw new Error('next() called multiple times');
      index = i;
      const fn = middlewares[i];
      if (!fn) return Promise.resolve(next?.());
      try {
        return Promise.resolve(fn(ctx, dispatch.bind(null, i + 1)));
      } catch (err) {
        return Promise.reject(err);
      }
    }
    
    return dispatch(0);
  };
}

// 使用
const pipeline = compose([logger, auth, rateLimit, router]);
pipeline(ctx, finalHandler);
```

### 并行组合（Fan-out）

```js
// 同时运行多个独立中间件，全部完成后继续
function parallel(...middlewares) {
  return async (ctx, next) => {
    await Promise.all(
      middlewares.map(mw => mw(ctx, () => Promise.resolve()))
    );
    await next();
  };
}

// 示例：并行收集元数据
app.use(parallel(
  async (ctx) => { ctx.metadata.geo = await lookupGeo(ctx.ip); },
  async (ctx) => { ctx.metadata.device = parseDeviceUA(ctx.ua); },
  async (ctx) => { ctx.metadata.abTest = await getABTestGroup(ctx.userId); },
));
```

### 条件组合（Branching）

```js
// 根据条件选择不同的中间件链
function condition(testFn, trueChain, falseChain) {
  return async (ctx, next) => {
    if (testFn(ctx)) {
      return compose(trueChain)(ctx, next);
    } else if (falseChain) {
      return compose(falseChain)(ctx, next);
    }
    return next();
  };
}

// 使用示例
app.use(condition(
  ctx => ctx.path.startsWith('/api/'),
  [jsonParser, apiLogger, validateSchema],
  [staticServe]
));
```

### 分组组合（Grouping）

```js
// 将相关中间件分组为逻辑单元
function group(prefix, ...middlewares) {
  const composed = compose(middlewares);
  return (ctx, next) => {
    if (!ctx.path.startsWith(prefix)) return next();
    ctx.state.routePrefix = prefix;
    return composed(ctx, next);
  };
}

// 使用
app.use(group('/admin/', adminAuth, roleCheck('admin'), adminRouter));
app.use(group('/api/v1/', apiKeyAuth, rateLimit(apiLimits), v1Router));
app.use(group('/api/v2/', oauth2Auth, v2Router));
```

### 装饰器组合（Wrapping）

```js
// 为现有中间件添加横切关注点
function withTimeout(ms, middleware) {
  return async (ctx, next) => {
    const timeoutId = setTimeout(() => {
      ctx.status = 504;
      ctx.body = { error: 'Gateway Timeout' };
    }, ms);
    
    try {
      await middleware(ctx, next);
    } finally {
      clearTimeout(timeoutId);
    }
  };
}

function withRetry(maxRetries, middleware) {
  return async (ctx, next) => {
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        await middleware(ctx, next);
        return;  // 成功则退出
      } catch (err) {
        if (attempt === maxRetries) throw err;
        console.warn(`Retry ${attempt}/${maxRetries} for ${ctx.method} ${ctx.url}`);
        await new Promise(r => setTimeout(r, 100 * attempt));  // 指数退避
      }
    }
  };
}

function withMetrics(name, middleware) {
  return async (ctx, next) => {
    const start = Date.now();
    try {
      await middleware(ctx, next);
    } finally {
      const duration = Date.now() - start;
      metrics.histogram(`${name}_duration_ms`, duration);
      metrics.counter(`${name}_total`, { method: ctx.method, status: ctx.status });
    }
  };
}

// 组合使用
app.use('/slow-api', withTimeout(5000, withRetry(3, withMetrics('slow_api', slowHandler))));
```

## 全局错误捕获

### Express 错误边界

```js
// error-boundary.js
class AppError extends Error {
  constructor(statusCode, message, code, details) {
    super(message);
    this.statusCode = statusCode;
    this.code = code;
    this.details = details;
    this.isOperational = true;  // 标记为预期内错误
  }
}

// 全局异步错误包装
function asyncHandler(fn) {
  return (req, res, next) =>
    Promise.resolve(fn(req, res, next)).catch(next);
}

// 全局错误处理中间件（必须放在最后）
function globalErrorHandler(err, req, res, _next) {
  // 记录错误
  const errorInfo = {
    message: err.message,
    stack: err.stack,
    url: req.originalUrl,
    method: req.method,
    body: req.body,
    query: req.query,
    params: req.params,
    headers: req.headers,
    ip: req.ip,
    timestamp: new Date().toISOString(),
    requestId: req.id,
  };
  
  // 区分已知/未知错误
  if (err instanceof AppError || err.isOperational) {
    // 已知业务错误
    logger.warn('Operational error', errorInfo);
    return res.status(err.statusCode || 400).json({
      code: err.code || 'ERROR',
      message: err.message,
      ...(err.details && { details: err.details }),
    });
  }
  
  // 未知错误（可能是 bug）
  logger.error('Unexpected error', errorInfo);
  
  // 开发环境返回详细信息
  if (process.env.NODE_ENV === 'development') {
    return res.status(500).json({
      code: 'INTERNAL_ERROR',
      message: err.message,
      stack: err.stack,
      requestId: req.id,
    });
  }
  
  // 生产环境隐藏细节
  res.status(500).json({
    code: 'INTERNAL_ERROR',
    message: 'An unexpected error occurred',
    requestId: req.id,
  });
}

module.exports = { AppError, asyncHandler, globalErrorHandler };

// app.js
const express = require('express');
const { asyncHandler, globalErrorHandler } = require('./error-boundary');

const app = express();

// 所有路由使用 asyncHandler 包装
app.get('/users/:id', asyncHandler(async (req, res) => {
  const user = await User.findById(req.params.id);
  if (!user) throw new AppError(404, 'User not found', 'NOT_FOUND');
  res.json(user);
}));

// 错误处理放在最后
app.use(globalErrorHandler);
```

### Koa 错误边界

```js
// Koa 的错误会自然冒泡
app.use(async (ctx, next) => {
  try {
    await next();
  } catch (err) {
    // 所有未被下游处理的错误都会到达这里
    ctx.status = err.status || err.statusCode || 500;
    
    if (err.expose || process.env.NODE_ENV === 'development') {
      ctx.body = { error: err.message };
    } else {
      ctx.body = { error: 'Internal Server Error' };
    }
    
    // 触发应用级别的 error 事件
    ctx.app.emit('error', err, ctx);
  }
});

// 全局监听
app.on('error', (err, ctx) => {
  // 集中日志记录、告警通知
  logToService(err, {
    url: ctx?.url,
    status: ctx?.status,
    ip: ctx?.ip,
  });
});
```

### 进程级未捕获异常

```js
// process-level-error-handlers.js
process.on('uncaughtException', (err) => {
  console.error('[FATAL] Uncaught Exception:', err);
  // 严重错误：记录后退出（Node.js 不保证状态一致）

process.on('unhandledRejection', (reason, promise) => {
  console.error('[WARN] Unhandled Rejection at:', promise, 'reason:', reason);
  // 不要退出进程（Promise rejection 不一定致命）
  // 但应该记录并监控
});

process.on('SIGTERM', () => {
  console.log('[INFO] SIGTERM received, shutting down gracefully');
  gracefulShutdown();
});

process.on('SIGINT', () => {
  console.log('[INFO] SIGINT received (Ctrl+C)');
  gracefulShutdown();
});

async function gracefulShutdown() {
  // 1. 停止接收新请求
  server.close(() => {
    console.log('[INFO] HTTP server closed');
  });
  
  // 2. 等待进行中的请求完成（最多 10 秒）
  setTimeout(() => {
    console.log('[WARN] Force exiting after timeout');
    process.exit(0);
  }, 10000);
  
  // 3. 清理资源（数据库连接、定时器等）
  await cleanupResources();
  
  process.exit(0);
}
```

## 优雅降级

### 功能降级策略

```js
// degradation.js
class DegradationManager {
  constructor(options = {}) {
    this.failureCounts = new Map();  // 各功能的失败计数
    this.circuitStates = new Map();  // 断路器状态: closed/open/half-open
    this.threshold = options.threshold || 5;       // 失败阈值
    this.resetTime = options.resetTime || 60000;   // 重置时间（毫秒）
  }
  
  // 执行带降级的功能
  async execute(featureName, primaryFn, fallbackFn) {
    const state = this.getCircuitState(featureName);
    
    if (state === 'open') {
      // 断路器开启 → 直接走降级
      return this.runWithFallback(featureName, fallbackFn);
    }
    
    try {
      const result = await primaryFn();
      this.recordSuccess(featureName);
      return result;
    } catch (err) {
      this.recordFailure(featureName);
      
      if (this.shouldOpenCircuit(featureName)) {
        this.openCircuit(featureName);
        console.log(`[Degradation] Circuit OPEN for "${featureName}"`);
      }
      
      return this.runWithFallback(featureName, fallbackFn);
    }
  }
  
  async runWithFallback(featureName, fallbackFn) {
    try {
      return await fallbackFn();
    } catch (fallbackErr) {
      console.error(`[Degradation] Fallback failed for "${featureName}":`, fallbackErr.message);
      throw fallbackErr;  // 最终降级也失败，抛出
    }
  }
  
  // ... getCircuitState, recordSuccess, recordFailure, shouldOpenCircuit ...
}

// 使用示例
const degradation = new DegradationManager({ threshold: 3 });

app.get('/recommendations', async (req, res) => {
  const recommendations = await degradation.execute(
    'ml-recommendations',
    // 主逻辑：调用 ML 推荐服务
    async () => fetchFromMLService(req.userId),
    // 降级逻辑：返回热门列表
    async () => fetchPopularItems()
  );
  
  res.json(recommendations);
});
```

## 日志中间件

### 请求日志（Morgan 风格）

```js
// logger.js
function requestLogger(options = {}) {
  const {
    immediate = false,       // true: 请求进入即记录；false: 响应完成后记录
    skip = () => false,      // 条件跳过特定请求
    format = 'combined',     // 输出格式
  } = options;
  
  return (req, res, next) => {
    if (skip(req, res)) return next();
    
    const start = Date.now();
    const id = generateUniqueId();
    req.id = id;  // 附加请求 ID
    
    if (immediate) {
      logEntry(format, { req, id, start });
    }
    
    // 监听响应结束事件
    const originalEnd = res.end;
    res.end = function (...args) {
      const duration = Date.now() - start;
      logEntry(format, { req, res, id, duration });
      originalEnd.apply(this, args);
    };
    
    next();
  };
}

function logEntry(format, { req, res, id, start, duration }) {
  const entry = {
    id,
    time: new Date().toISOString(),
    method: req.method,
    url: req.originalUrl || req.url,
    status: res?.statusCode ?? '-',
    duration: duration ?? 0,
    size: res?.getHeader('content-length') ?? '-',
    referrer: req.headers.referer || req.headers.referrer || '-',
    userAgent: req.headers['user-agent'] || '-',
    ip: req.ip || req.connection.remoteAddress,
  };
  
  // 彩色控制台输出
  const statusColor = entry.status < 400 ? '\x1b[32m' :  // 绿色
                        entry.status < 500 ? '\x1b[33m' :  // 黄色
                        '\x1b[31m';                        // 红色
  
  console.log(
    `\x1b[90m${entry.id}\x1b[0m ` +
    `\x1b[36m${entry.method}\x1b[0m ` +
    `\x1b[34m${entry.url}\x1b[0m ` +
    `${statusColor}${entry.status}\x1b[0m ` +
    `\x1b[90m${entry.duration}ms\x1b[0m`
  );
  
  // 同时输出到结构化日志（供 ELK/Datadog 消费）
  writeStructuredLog(entry);
}

// 使用
app.use(requestLogger({ skip: (req) => req.path === '/health' }));
```

```js
function errorLogger() {
  return (err, req, res, next) => {
    const errorLog = {
      timestamp: new Date().toISOString(),
      requestId: req.id,
      error: {
        name: err.name,
        message: err.message,
        stack: err.stack,
        code: err.code,
        statusCode: err.statusCode || err.status,

      request: {
        method: req.method,
        url: req.originalUrl,
        query: req.query,
        params: req.params,
        body: sanitizeBody(req.body),
        headers: sanitizeHeaders(req.headers),

      },
    };
    
    // 按严重程度分级输出
    if (errorLog.error.statusCode >= 500) {
      console.error(JSON.stringify(errorLog));  // ERROR
      alertTeam(errorLog);                     // 发送告警
    } else if (errorLog.error.statusCode >= 400) {
      console.warn(JSON.stringify(errorLog));   // WARN
    } else {
      console.info(JSON.stringify(errorLog));  // INFO
    }
    
    next(err);  // 继续传递给下一个错误处理器
  };
}
```

### 性能日志

```js
function performanceLogger(threshold = 1000) {
  return (req, res, next) => {
    const start = process.hrtime.bigint();
    
    res.on('finish', () => {
      const durationNs = Number(process.hrtime.bigint() - start);
      const durationMs = durationNs / 1_000_000;
      
      // 记录超过阈值的慢请求
      if (durationMs > threshold) {
        slowRequestAlert({
          url: req.originalUrl,
          method: req.method,
          duration: `${durationMs.toFixed(2)}ms`,
          threshold: `${threshold}ms`,
        });
      }
      
      // 上报性能指标
      metrics.record('request_duration_ms', durationMs, {
        route: req.route?.path || req.path,
        method: req.method,
        status: res.statusCode,
      });
    });
    
    next();
  };
}

// 使用
app.use(performanceLogger(2000));  // 超过 2s 视为慢请求
```

## 中间件架构总览

```
推荐的中间件顺序（Express）：

1️⃣ helmet()                    — 安全头
2️⃣ cors()                      — CORS 配置
3️⃣ compression()               — 响应压缩
4️⃣ requestLogger()              — 请求日志
5️⃣ express.json()               — Body 解析
6️⃣ express.urlencoded()         — URL-encoded 解析
7️⃣ cookie-parser()              — Cookie 解析
8️⃣ session()                    — 会话管理
9️⃣ passport.initialize()        — 认证初始化
🔟 authMiddleware()             — 身份验证（有状态）
1️⃣1️⃣ rateLimiter()              — 限流
1️⃣2️⃣ routes                     — 业务路由
1️⃣3️⃣ notFoundHandler()          — 404 兜底
1️⃣4️⃣ globalErrorHandler()        — 错误最终兜底
```
