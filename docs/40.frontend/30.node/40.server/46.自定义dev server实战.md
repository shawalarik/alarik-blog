## 概述

理解 Vite/Webpack DevServer 的最好方式就是自己动手搭建一个。本文从零开始构建一个类似 Vite 的简易开发服务器，支持 Vue/React 单文件组件编译、HMR 热更新、API 代理等功能，串联前面所学的所有中间件技术。

```

│   ├── index.ts                 # 入口 & CLI
```

## 第一步：基础 HTTP 服务器

```ts
// src/server.ts
import http from 'node:http';
import { createLogger } from './middleware/logger.js';
import type { DevServerConfig } from './config.js';

export function createDevServer(config: DevServerConfig) {
  const server = http.createServer(async (req, res) => {
    // 请求 ID（用于日志追踪）
    const rid = crypto.randomUUID();
    (req as any).__rid = rid;
    
    try {
      // 中间件管道
      await handleRequest(config, req, res);
    } catch (err: any) {
      handleError(err, req, res);
    }
  });

  // 启动
  return new Promise<http.Server>((resolve) => {
    server.listen(config.port, config.host, () => {
      console.log(`
  ══════════════════════════════════════
    🚀 My Dev Server
    ➜ Local:   http://${config.host}:${config.port}/
    ➜ Network: http://${getLocalIP()}:${config.port}/
  ══════════════════════════════════════`);
      resolve(server);
    });
  });
}

async function handleRequest(config: DevServerConfig, req: any, res: any) {
  const { middleware } = config;
  
  // 按顺序执行中间件
  for (const mw of middleware) {
    const shouldContinue = await mw(req, res, config);
    if (shouldContinue === false) break;  // 中间件决定终止
  }
  
  // 未被任何中间件处理 → 404
  if (!res.writableEnded) {
    res.writeHead(404, { 'Content-Type': 'text/plain' });
    res.end('Not Found');
  }
}
```

## 第二步：Vue SFC 编译器

```ts
// src/transformer/vue-transformer.ts
import * as compiler from '@vue/compiler-sfc';

interface TransformResult {
  code: string;
  map?: string;
  deps?: string[];
}

/**
 * 将 Vue Single File Component (.vue) 编译为 ES Module
 */
export async function transformVue(sourceCode: string, filename: string): Promise<TransformResult> {
  const { descriptor } = compiler.parse(sourceCode, { filename });
  const deps: string[] = [];
  let scriptCode = '';
  let templateCode = '';

  // --- 处理 <script> ---
  if (descriptor.script || descriptor.scriptSetup) {
    const { code, bindings } = compiler.compileScript(descriptor, {
      id: genId(filename),
      inline: false,  // 不内联渲染函数（方便调试）
    });

    scriptCode = `
${code}

const __sfc__Options = ${(descriptor.scriptSetup?.attrs.setup ? 'undefined' : '{}')}
if (typeof __sfc__Options !== 'undefined') {
  Object.assign(__sfc__.options, __sfc__Options)
}

// 收集依赖
${
  descriptor.styles?.length 
    ? `import '${filename}?vue&type=style&index=0&lang.css';`
    : ''
}`;
  }

  // --- 处理 <template> ---
  if (descriptor.template) {
    const { code: renderCode } = compiler.compileTemplate({
      source: descriptor.template.content,
      filename,
      id: genId(filename),
      scoped: descriptor.styles.some(s => s.scoped),
      ssr: false,
      compilerOptions: {
        isCustomElement: (tag) => tag.startsWith('x-'),
      },
    });

    templateCode = `
/* @__PURE__ */ import { createElementBlock as _createElementBlock, openBlock as _openBlock, createVNode as _createVNode } from 'vue'

function _sfc_render(_ctx, _cache, $props, $setup, $data, $options) {
  return (${renderCode})
}`;

    deps.push('vue');  // 声明对 vue 的依赖
  }

  // --- 组装最终代码 ---
  const fullCode = `
import { defineComponent as _defineComponent } from 'vue'

${scriptCode}

${templateCode}

const __sfc__ = _defineComponent({
  ${templateCode ? 'render: _sfc_render,' : ''}
  __file: '${filename}',
})

export default __sfc__
`;

  return { code: fullCode, deps };
}

function genId(filepath: string): string {
  // 为每个文件生成唯一的 scope ID（用于 scoped CSS）
  const hash = crypto
    .createHash('md5')
    .update(filepath.replace(/\\/g, '/'))
    .digest('hex')
    .slice(0, 8);
  return `data-v-${hash}`;
}
```

## 第三步：CSS Transformer

```ts
// src/transformer/css-transformer.ts
import postcss from 'postcss';
import autoprefixer from 'autoprefixer';

interface CSSTransformResult {
  code: string;
  map?: any;
}

/**
 * 编译 CSS（支持 PostCSS 插件链）
 */
export async function transformCSS(
  sourceCode: string,
  filename: string,
  options?: { scoped?: boolean; scopedId?: string }
): Promise<CSSTransformResult> {
  const plugins = [autoprefixer()];
  
  // 如果是 Vue scoped CSS
  if (options?.scoped && options.scopedId) {
    plugins.unshift(scopedPlugin(options.scopedId));
  }

  const result = await postcss(plugins).process(sourceCode, {
    from: filename,
    map: { inline: false },
  });

  return {
    code: result.css,
    map: result.map?.toString(),
  };
}

/**
 * Vue scoped CSS 插件
 */
function scopedPlugin(scopedId: string): postcss.Plugin {
  return {
    postcssPlugin: 'my-dev-scoped',
    Rule(rule) {
      rule.selectors = rule.selectors.map(sel => {
        return `${sel}[${scopedId}]`;
      });
    },
  };
}
```

## 第四步：HMR WebSocket 实现

```ts
// src/hrm.ts
import { WebSocketServer, WebSocket } from 'ws';

type ClientMessage =
  | { type: 'ping' }
  | { type: 'custom-event'; event: string; data: any };

type ServerMessage =
  | { type: 'connected'; data: { timestamp: number } }
  | { type: 'update'; data: UpdatePayload }
  | { type: 'full-reload'; data: { path?: string } }
  | { type: 'error'; error: string; data?: any }
  | { type: 'custom'; data: { event: string; payload: any } };

interface UpdatePayload {
  updates: Array<{
    type: 'js-update' | 'css-update' | 'html-update';
    path: string;
    acceptedPath: string;
    timestamp: number;
    moduleId: string;
  }>;
}

export class HRMManager {
  private wss!: WebSocketServer;
  private clients: Set<WebSocket> = new Set();

  attach(server: http.Server, path: string = '/ws-hmr') {
    this.wss = new WebSocketServer({ server, path });

    this.wss.on('connection', (ws) => {
      this.clients.add(ws);

      // 连接确认
      this.send(ws, {
        type: 'connected',
        data: { timestamp: Date.now() },
      });

      ws.on('message', (raw) => {
        try {
          const msg: ClientMessage = JSON.parse(raw.toString());
          this.handleClientMessage(ws, msg);
        } catch {
          // 忽略非法消息
        }
      });

      ws.on('close', () => this.clients.delete(ws));
      ws.on('error', () => this.clients.delete(ws));
    });
  }

  private handleClientMessage(_ws: WebSocket, msg: ClientMessage) {
    switch (msg.type) {
      case 'ping':
        // 心跳 pong（WebSocket 层面已处理）
        break;
      }
  }

  /** 广播更新消息 */
  sendUpdate(updates: UpdatePayload['updates']) {
    this.broadcast({
      type: 'update',
      data: { updates, cause: 'file-change' },
    });
  }

  /** 通知全页面重载 */
  triggerFullReload(path?: string) {
    this.broadcast({ type: 'full-reload', data: { path } });
  }

  /** 发送错误遮罩 */
  sendErrorOverlay(error: string, data?: any) {
    this.broadcast({ type: 'error', error, data });
  }

  /** 发送自定义事件 */
  emitCustomEvent(event: string, payload: any) {
    this.broadcast({ type: 'custom', data: { event, payload } });
  }

  private broadcast(message: ServerMessage) {
    const data = JSON.stringify(message);
    for (const client of this.clients) {
      if (client.readyState === WebSocket.OPEN) {
        client.send(data);
      }
    }
  }

  private send(ws: WebSocket, message: ServerMessage) {
    if (ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify(message));
    }
  }
}
```

## 第五步：模块转换中间件

```ts
// src/middleware/module-transformer.ts
import fs from 'node:fs/promises';
import path from 'node:path';
import { transformVue } from '../transformer/vue-transformer.js';
import { transformCSS } from '../transformer/css-transformer.js';
import type { HRMManager } from '../hrm.js';

/**
 * 核心中间件：拦截 .vue/.jsx/.tsx/.css 等，
 * 实时编译后返回给浏览器
 */
export function moduleTransformer(hrm: HRMManager) {
  return async (req: any, res: any) => {
    const url = req.parsedUrl.pathname;
    
    // 只处理需要编译的文件类型
    if (!needsTransform(url)) return true;  // pass through

    const filePath = resolveModulePath(url);
    
    try {
      let code: string;
      let contentType: string;
      const timestamp = Date.now();

      switch (path.extname(url)) {
        case '.vue':
          const source = await fs.readFile(filePath, 'utf-8');
          const vueResult = await transformVue(source, filePath);
          code = vueResult.code;
          contentType = 'application/javascript; charset=utf-8';
          
          // 注入 HMR accept 代码
          code += `\nimport.meta.hot?.accept(() => console.log("[HMR] ${url} updated"));\n`;
          hrm.sendUpdate([{
            type: 'js-update',
            path: url,
            acceptedPath: url,
            timestamp,
            moduleId: url,
          }]);
          break;

        case '.css':
        case '.scss':
        case '.less':
          const cssSource = await fs.readFile(filePath, 'utf-8');
          const cssResult = await transformCSS(cssSource, filePath);
          code = `
const style = document.createElement('style');
style.setAttribute('type', 'text/css');
style.textContent = ${JSON.stringify(cssResult.code)};
document.head.appendChild(style);
import.meta.hot?.accept(() => {
  style.textContent = ${JSON.stringify(cssResult.code)};
})`;
          contentType = 'application/javascript; charset=utf-8';
          hrm.sendUpdate([{
            type: 'css-update',
            path: url,
            acceptedPath: url,
            timestamp,
          }]);
          break;

        case '.jsx':
        case '.tsx':
          // 使用 esbuild 或 sucrase 转换 JSX
          code = await transformJSX(await fs.readFile(filePath, 'utf-8'), url);
          contentType = 'application/javascript; charset=utf-8';
          break;

        default:
          return true;  // 不认识的类型放行
      }

      res.writeHead(200, {
        'Content-Type': contentType,
        'Cache-Control': 'no-cache, no-store, must-revalidate',
        'Access-Control-Allow-Origin': '*',
      });
      res.end(code);

    } catch (err: any) {
      // 编译错误 → 发送到浏览器的错误遮罩
      hrm.sendErrorOverlay(err.message, {
        file: filePath,
        line: err.loc?.line,
        column: err.loc?.column,
        stack: err.stack,
      });

      res.writeHead(500, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: err.message }));
    }
  };
}

function needsTransform(url: string): boolean {
  const ext = path.extname(url);
  return ['.vue', '.jsx', '.tsx', '.css', '.scss', '.less'].includes(ext);
}
```

## 第六步：组装完整的 Dev Server

```ts
// src/index.ts
import chokidar from 'chokidar';
import { createDevServer } from './server.js';
import { HRMManager } from './hrm.js';
import { moduleTransformer } from './middleware/module-transformer.js';
import { staticMiddleware } from './middleware/static.js';
import { proxyMiddleware } from './middleware/proxy.js';
import { spafallback } from './middleware/spa-fallback.js';
import { requestLogger } from './middleware/logger.js';
import type { DevServerConfig } from './config.js';

async function main() {
  const hrm = new HRMManager();
  const config: DevServerConfig = {
    port: 5173,
    host: '0.0.0.0',
    root: process.cwd(),
    middleware: [
      requestLogger(),                           // 1. 日志
      proxyMiddleware([{                         // 2. API 代理
        context: '/api',
        target: 'http://localhost:3001',
        changeOrigin: true,
      }]),
      staticMiddleware({                         // 3. 静态文件
        dir: 'public',
        maxAge: 0,  // 开发环境禁用缓存
      }),
      moduleTransformer(hrm),                    // 4. 模块实时编译
      spafallback({ html: 'index.html' }),       // 5. SPA 回退
    ],
  };

  // 创建并启动服务器
  const server = await createDevServer(config);

  // 附加 HMR WebSocket
  hrm.attach(server);

  // 文件监听
  const watcher = chokidar.watch(['src/**/*.{vue,jsx,tsx,ts,css}', 'public/**/*'], {
    cwd: config.root,
    ignored: /node_modules/,
    persistent: true,
    ignoreInitial: true,
  });

  watcher.on('change', (filepath) => {
    const relativePath = '/' + path.relative(config.root, filepath).replace(/\\/g, '/');
    console.log(`  🔄 Changed: ${relativePath}`);
    
    // 判断变更类型
    if (/\.(vue|jsx|tsx|ts)$/.test(relativePath)) {
      // JS/Vue 变更 → HMR update
      // moduleTransformer 中间件下次请求时会重新编译
      hrm.sendUpdate([{
        type: relativePath.endsWith('.vue') ? 'js-update' : 'js-update',
        path: relativePath,
        acceptedPath: relativePath,
        timestamp: Date.now(),
        moduleId: relativePath,
      }]);
    } else if (/\.(css|scss|less)$/.test(relativePath) && !relativePath.includes('?')) {
      // CSS 变更 → 快速通道
      hrm.sendUpdate([{
        type: 'css-update',
        path: relativePath,
        acceptedPath: relativePath,
        timestamp: Date.now(),
      }]);
    } else if (relativePath.endsWith('.html')) {
      // HTML 变更 → Full Reload
      hrm.triggerFullReload();
    }
  });

  // 优雅关闭
  process.on('SIGINT', () => {
    watcher.close();
    server.close(() => process.exit(0));
  });
}

main().catch(console.error);
```

## 第七步：HMR 客户端注入

```ts
// 通过 HTML 注入脚本方式注入 HMR 客户端
function injectHRMClient(html: string, port: number): string {
  const protocol = 'ws';
  const host = 'localhost:' + port;
  const hmrScript = `
<script type="module">
// My Dev Server HMR Runtime
(function(){
  const socket = new WebSocket('${protocol}://${host}/ws-hmr');
  const hotModulesMap = new Map();
  
  socket.addEventListener('message', async ({data}) => {
    const payload = JSON.parse(data);
    switch(payload.type){
      case 'update':
        for(const update of payload.data.updates){
          if(update.type === 'css-update'){
            // CSS 快速替换
            const links = document.querySelectorAll('link[rel="stylesheet"]');
            links.forEach(link=>{
              const newLink = link.cloneNode();
              newLink.href = link.href.split('?')[0]+'?t='+Date.now();
              link.replaceWith(newLink);
            });
            continue;
          }
          // JS 模块热更新
          const mod = await import(update.acceptedPath+'?t='+update.timestamp);
          const entry = hotModulesMap.get(update.moduleId);
          if(entry?.callback) entry.callback(mod);
        }
        console.log('[HMR] Updated '+payload.data.updates.length+' module(s)');
        break;
      case 'full-reload':
        location.reload();
        break;
      case 'error':
        console.error('[HMR]', payload.error);
        break;
    }
  });
  
  // 注册 HMR accept
  window.__hmr_accept = function(id, callback){
    hotModulesMap.set(id, { callback });
  };
})();
</script>`;

  return html.replace('</head>', hmrScript + '</head>');
}
```
