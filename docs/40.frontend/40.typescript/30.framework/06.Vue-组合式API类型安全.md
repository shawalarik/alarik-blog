# Vue 组合式 API 类型安全

> **一句话总结**：组合式函数（Composable）是 Vue 3 代码复用的核心形态。为它们编写精确的类型定义，不仅能消除 `any` 污染，更能让调用者获得完整的 IDE 补全和编译期校验——这是区分"能用 Vue 3"和"用好 Vue 3 + TS"的关键分水岭。

## 一、什么是类型安全的组合式函数？

```
┌─────────────────────────────────────────────────────────────┐
│                                                             │
│  普通的 JavaScript 函数                                     │
│  ─ 输入：参数                                                 │
│  ─ 输出：结果                                                 │
│  ─ 关注：功能正确                                             │
│                                                             │
│  类型安全的组合式函数                                         │
│  ─ 输入：类型化的参数 + Ref/Reactive 约束                      │
│  ─ 输出：类型化的响应式数据 + 方法                              │
│  ─ 关注：功能正确 + 类型流转正确 + 响应性保持                  │
│                                                             │
│  本质差异：Composable 不仅返回值，还返回"活的"类型连接          │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

**类比理解**：

想象你经营一家**工具租赁店**（Composable Factory）：

- **无类型的版本**：顾客来借工具，你给他一个黑盒子。他不知道里面是什么，只能碰运气使用。有时好用，有时砸脚。
- **有类型的版本**：每个工具箱上都贴着清晰的标签："内含：扳手（8-14mm）、螺丝刀（十字）、说明书"。顾客不仅知道有什么，还能在租借前就确认工具适合他的任务。

**组合式函数就是那个贴好标签的工具箱。**

## 二、编写类型化 Composable 的黄金法则

### 法则一：泛型是灵魂

```typescript
/**
 * ❌ 反面：硬编码类型 —— 失去复用价值
 */
function useUserData() {
  const user = ref<UserProfile | null>(null)
  const loading = ref(false)
  // ...
  return { user, loading }
}

// 这个 composable 只能给 UserProfile 用，换 Product 就得重写


/**
 * ✅ 正面：泛型参数化 —— 一份代码适配所有类型
 */
function useAsyncData<T>(
  url: MaybeRef<string>,
  options?: AsyncDataOptions<T>
): UseAsyncDataReturn<T> {
  const data = shallowRef<T | null>(null)
  const error = ref<Error | null>(null)
  const loading = ref(false)

  async function execute() {
    loading.value = true
    error.value = null
    try {
      const res = await fetch(unref(url))
      if (!res.ok) throw new Error(`HTTP ${res.status}`)
      data.value = await res.json()
      options?.onSuccess?.(data.value)
    } catch (e) {
      error.value = e instanceof Error ? e : new Error(String(e))
      options?.onError?.(error.value!)
    } finally {
      loading.value = false
    }
  }

  // 支持 immediate 选项
  if (options?.immediate !== false) {
    execute()
  }

  return { data: readonly(data), error: readonly(error), loading: readonly(loading), execute }
}

// 同一份代码，服务所有实体：
const { data: user } = useAsyncData<UserProfile>('/api/user')     // data: Ref<UserProfile | null>
const { data: posts } = useAsyncData<Post[]>('/api/posts')       // data: Ref<Post[] | null>
const { data: config } = useAsyncData<AppConfig>('/api/config')   // data: Ref<AppConfig | null>
```

### 法则二：返回值结构标准化

```typescript
/**
 * 推荐的标准返回结构
 */
interface StandardComposableReturn<T> {
  // 响应式状态（必须 readonly 保护）
  state: Readonly<ShallowRef<T>>
  // 派生计算状态
  derived: { /* computed values */ }
  // 操作方法
  actions: { /* functions */ }
  // 工具方法
  utilities: { /* reset, clear, etc. */ }
}

/**
 * 实际例子：分页查询 composable
 */
function usePaginatedQuery<T extends { id: string | number }>(
  fetchFn: (params: PaginationParams) => Promise<PaginatedResult<T>>,
  initialPageSize: number = 20
) {
  // ===== 状态 =====
  const items = ref<T[]>([])
  const currentPage = ref(1)
  const pageSize = ref(initialPageSize)
  const totalItems = ref(0)
  const totalPages = ref(0)
  const isLoading = ref(false)
  const error = ref<Error | null>(null)

  // ===== 派生状态 =====
  const hasNextPage = computed(() => currentPage.value < totalPages.value)
  const hasPrevPage = computed(() => currentPage.value > 1)
  const isFirstPage = computed(() => currentPage.value === 1)
  const isLastPage = computed(() => currentPage.value >= totalPages.value)
  const rangeStart = computed(() => (currentPage.value - 1) * pageSize.value + 1)
  const rangeEnd = computed(() => Math.min(currentPage.value * pageSize.value, totalItems.value))

  // ===== 操作 =====
  async function fetchPage(page: number = currentPage.value) {
    isLoading.value = true
    error.value = null
    try {
      const result = await fetchFn({ page, pageSize: pageSize.value })
      items.value = result.items
      totalItems.value = result.total
      totalPages.value = Math.ceil(result.total / pageSize.value)
      currentPage.value = page
    } catch (e) {
      error.value = e instanceof Error ? e : new Error(String(e))
    } finally {
      isLoading.value = false
    }
  }

  function goToPage(page: number) {
    if (page >= 1 && page <= totalPages.value) {
      fetchPage(page)
    }
  }

  function nextPage() { goToPage(currentPage.value + 1) }
  function prevPage() { goToPage(currentPage.value - 1) }

  // ===== 工具 =====
  function reset() {
    items.value = []
    currentPage.value = 1
    totalItems.value = 0
    totalPages.value = 0
  }

  // 初始加载
  fetchPage()

  // ===== 标准化返回 =====
  return {
    // State（readonly 保护外部不能直接修改）
    items: readonly(items),
    currentPage: readonly(currentPage),
    pageSize: readonly(pageSize),
    totalItems: readonly(totalItems),
    totalPages: readonly(totalPages),
    isLoading: readonly(isLoading),
    error: readonly(error),

    // Derived（只读的计算属性）
    hasNextPage,
    hasPrevPage,
    isFirstPage,
    isLastPage,
    rangeStart,
    rangeEnd,

    // Actions（修改状态的方法）
    fetchPage,
    goToPage,
    nextPage,
    prevPage,

    // Utilities
    reset,
  }
}
```

### 法则三：MaybeRef 让参数更灵活

```typescript
/**
 * MaybeRef 工具类型 —— 让参数既可以是静态值也可以是 ref
 * 这是 VueUse 库的核心工具之一
 */

type MaybeRef<T> = T | Ref<T>

// 手动实现 unref
function _unref<T>(r: MaybeRef<T>): T {
  return isRef(r) ? r.value : r
}

// 应用到 composable 参数
function useWindowTitle(title: MaybeRef<string>) {
  const actualTitle = ref(_unref(title))

  watch(title, (newVal) => {
    actualTitle.value = _unref(newVal)
  })

  watch(actualTitle, (val) => {
    document.title = val
  })

  // 初始化
  document.title = actualTitle.value

  return actualTitle
}

// ===== 使用 =====

// 传静态字符串
const t1 = useWindowTitle('My App')

// 传 ref
const dynamicTitle = ref('Dynamic App')
const t2 = useWindowTitle(dynamicTitle)

// 传 computed
const appTitle = computed(() => `${appName.value} - ${userStatus.value}`)
const t3 = useWindowTitle(appTitle)
```

## 三、实战案例集

### 案例 1：useLocalStorage —— 持久化存储

```typescript
/**
 * 类型安全的 localStorage 封装
 * 
 * 特性：
 * - 泛型 T 适配任意存储数据
 * - 自动序列化/反序列化
 * - 损坏数据自动恢复
 * - SSR 安全（no window 时返回初始值）
 */
function useLocalStorage<T>(
  key: string,
  initialValue: T,
  options: {
    serializer?: {
      read: (raw: string) => T
      write: (value: T) => string
    }
    listenToStorageChanges?: boolean
  } = {}
) {
  const { serializer = defaultSerializer, listenToStorageChanges = false } = options

  // 从存储读取
  let readValue: T
  try {
    const raw = localStorage.getItem(key)
    readValue = raw ? serializer.read(raw) : initialValue
  } catch {
    readValue = initialValue
    // 清除损坏数据
    if (typeof window !== 'undefined') {
      localStorage.removeItem(key)
    }
  }

  // 创建响应式数据
  const data = ref<T>(readValue) as Ref<T>

  // 写入存储
  function write(value: T) {
    data.value = value
  }

  // 监听变化 → 持久化
  watch(
    data,
    newValue => {
      try {
        localStorage.setItem(key, serializer.write(newValue))
        
        // 跨 Tab 同步
        if (listenToStorageChanges) {
          window.dispatchEvent(new CustomEvent('local-storage-update', {
            detail: { key, newValue: serializer.write(newValue) },
          }))
        }
      } catch (e) {
        console.error(`[useLocalStorage] Failed to save "${key}":`, e)
      }
    },
    { deep: true }
  )

  // 监听来自其他 Tab 的更新
  if (listenToStorageChanges && typeof window !== 'undefined') {
    window.addEventListener('local-storage-update', ((e: Event) => {
      const customEvent = e as CustomEvent
      if (customEvent.detail?.key === key) {
        const newValue = serializer.read(customEvent.detail.newValue)
        data.value = newValue
      }
    }) as EventListener)
  }

  // 移除
  function remove() {
    data.value = initialValue
    localStorage.removeItem(key)
  }

  return {
    data: readonly(data),  // Readonly<Ref<T>>
    write,
    remove,
  }
}

// 默认序列化器
const defaultSerializer = {
  read: <T>(v: string): T => JSON.parse(v),
  write: <T>(v: T): string => JSON.stringify(v),
}

// ===== 使用 =====

// 简单类型
const { data: theme } = useLocalStorage<'light' | 'dark' | 'auto'>('app-theme', 'auto')

// 复杂对象
interface UserPreferences {
  fontSize: number
  language: string
  sidebarCollapsed: boolean
  recentFiles: string[]
}

const { data: prefs, remove: resetPrefs } = useLocalStorage<UserPreferences>('user-prefs', {
  fontSize: 16,
  language: 'zh-CN',
  sidebarCollapsed: false,
  recentFiles: [],
})

prefs.value.fontSize = 18           // 自动保存到 localStorage
prefs.value.recententFiles.push('doc.md')  // 深度监听，自动持久化
```

### 案例 2：useWebSocket —— 类型化的 WebSocket 连接

```typescript
/**
 * WebSocket composable —— 带自动重连、心跳检测、消息类型安全
 */
type WSStatus = 'connecting' | 'open' | 'closing' | 'closed' | 'error'
type WSMessageHandler<T = unknown> = (data: T, event: MessageEvent) => void

interface UseWSOptions<T = unknown> {
  url: string
  protocols?: string | string[]
  heartbeatInterval?: number
  reconnect: boolean | { retries: number; delay: number; onMaxRetry?: () => void }
  onMessage?: WSMessageHandler<T>
  onError?: (event: Event) => void
  onClose?: (event: CloseEvent) => void
  onOpen?: (event: Event) => void
}

interface UseWSReturn<T = unknown> {
  status: Readonly<Ref<WSStatus>>
  lastMessage: Readonly<Ref<T | null>>
  isConnected: Readonly<ComputedRef<boolean>>
  send: (data: T) => void
  connect: () => void
  disconnect: (code?: number, reason?: string) => void
}

function useWebSocket<T = unknown>(options: UseWSOptions<T>): UseWSReturn<T> {
  const {
    url,
    protocols,
    heartbeatInterval = 30000,
    reconnect = false,
    onMessage,
    onError,
    onClose,
    onOpen,
  } = options

  const status = ref<WSStatus>('closed')
  const lastMessage = ref<T | null>(null)
  let ws: WebSocket | null = null
  let heartbeatTimer: ReturnType<typeof setInterval> | null = null
  let retryCount = 0
  let maxRetries = typeof reconnect === 'object' ? reconnect.retries : 0
  let retryDelay = typeof reconnect === 'object' ? reconnect.delay : 1000

  function connect() {
    if (ws?.readyState === WebSocket.OPEN) return

    status.value = 'connecting'

    try {
      ws = protocols
        ? new WebSocket(url, protocols)
        : new WebSocket(url)

      ws.onopen = (event) => {
        status.value = 'open'
        retryCount = 0
        startHeartbeat()
        onOpen?.(event)
      }

      ws.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data) as T
          lastMessage.value = data
          onMessage?.(data, event)
        } catch {
          // 非JSON消息（如文本/二进制）
          lastMessage = event.data as unknown as T
          onMessage?.(lastMessage.value, event)
        }
      }

      ws.onerror = (event) => {
        status.value = 'error'
        onError?.(event)
      }

      ws.onclose = (event) => {
        status.value = 'closed'
        stopHeartbeat()
        onClose?.(event)
        
        // 自动重连
        if (reconnect && !event.wasClean) {
          attemptReconnect()
        }
      }
    } catch (e) {
      status.value = 'error'
      console.error('[WebSocket] Connection failed:', e)
    }
  }

  function attemptReconnect() {
    if (reconnect === false) return
    
    if (retryCount < maxRetries) {
      retryCount++
      setTimeout(connect, retryDelay * retryCount)
    } else {
      if (typeof reconnect === 'object' && reconnect.onMaxRetry) {
        reconnect.onMaxRetry()
      }
    }
  }

  function startHeartbeat() {
    stopHeartbeat()
    heartbeatTimer = setInterval(() => {
      if (ws?.readyState === WebSocket.OPEN) {
        ws.send(JSON.stringify({ type: '__heartbeat__' }))
      }
    }, heartbeatInterval)
  }

  function stopHeartbeat() {
    if (heartbeatTimer) {
      clearInterval(heartbeatTimer)
      heartbeatTimer = null
    }
  }

  function send(data: T) {
    if (ws?.readyState === WebSocket.OPEN) {
      ws.send(typeof data === 'string' ? data : JSON.stringify(data))
    } else {
      console.warn('[WebSocket] Cannot send: not connected')
    }
  }

  function disconnect(code = 1000, reason = 'Normal closure') {
    stopHeartbeat()
    ws?.close(code, reason)
    status.value = 'closing'
  }

  // 自动连接
  connect()

  return {
    status: readonly(status),
    lastMessage: readonly(lastMessage),
    isConnected: computed(() => status.value === 'open'),
    send,
    connect,
    disconnect,
  }
}

// ===== 使用 =====

interface ChatMessage {
  type: 'text' | 'image' | 'system'
  content: string
  senderId: string
  timestamp: number
}

const chat = useWebSocket<ChatMessage>({
  url: 'wss://chat.example.com/ws',
  protocols: 'chat-v1',
  heartbeatInterval: 25000,
  reconnect: { retries: 5, delay: 2000 },

  onMessage(msg, event) {
    switch (msg.type) {
      case 'text':
        console.log(`${msg.senderId}: ${msg.content}`)
        break
      case 'system':
        console.log(`[System] ${msg.content}`)
        break
    }
  },
})

// chat.lastMessage.value → ChatMessage | null
// chat.isConnected.value → boolean
// chat.send({ type: 'text', content: 'Hello!', senderId: 'me', timestamp: Date.now() })
```

### 案例 3：useValidation —— 表单验证

```typescript
/**
 * 类型驱动的表单验证 composable
 * 
 * 利用 TypeScript 的类型信息自动推导验证规则
 */
interface ValidationRule<T> {
  field: keyof T
  label: string
  validate: (value: T[keyof T], allValues: T) => string | null
  trigger?: 'blur' | 'input' | 'change'
}

interface UseValidationReturn<T> {
  values: Ref<T>
  errors: Ref<Partial<Record<keyof T, string[]>>>
  touched: Ref<Set<keyof T>>
  isValid: ComputedRef<boolean>
  register: (field: keyof T, rules: ValidationRule<T>[]) => void
  validateField: (field: keyof T) => string[]
  validateAll: () => boolean
  reset: () => void
}

function useValidation<T extends Record<string, unknown>>(
  initialValues: T,
  rules: ValidationRule<T>[] = []
): UseValidationReturn<T> {
  const values = ref<T>({ ...initialValues }) as Ref<T>
  const errors = ref<Partial<Record<keyof T, string[]>>({}) as any
  const touched = ref(new Set<keyof T>()) as Ref<Set<keyof T>>

  const ruleMap = new Map<keyof T, ValidationRule<T>[]>()

  // 注册规则
  rules.forEach(rule => {
    const existing = ruleMap.get(rule.field) ?? []
    existing.push(rule)
    ruleMap.set(rule.field, existing)
  })

  function register(field: keyof T, newRules: ValidationRule<T>[]) {
    ruleMap.set(field, newRules)
  }

  function validateField(field: keyof T): string[] {
    touched.value.add(field)
    const fieldRules = ruleMap.get(field) ?? []
    const fieldErrors: string[] = []

    for (const rule of fieldRules) {
      const error = rule.validate(values.value[field], values.value)
      if (error) fieldErrors.push(error)
    }

    if (fieldErrors.length > 0) {
      errors.value[field] = fieldErrors
    } else {
      delete (errors.value as any)[field]
    }

    return fieldErrors
  }

  function validateAll(): boolean {
    const allFields = Array.from(new Set(rules.map(r => r.field)))
    const results = allFields.map(f => validateField(f))
    return results.every(r => r.length === 0)
  }

  isValid = computed(() => {
    return Object.keys(errors.value).length === 0 &&
      Array.from(ruleMap.keys()).every(field => touched.value.has(field))
  })

  function reset() {
    values.value = { ...initialValues }
    errors.value = {}
    touched.value.clear()
  }

  return { values, errors, touched, isValid, register, validateField, validateAll, reset }
}

// ===== 使用 =====

interface LoginForm {
  username: string
  password: string
  rememberMe: boolean
}

const form = useValidation<LoginForm>(
  { username: '', password: '', rememberMe: false },
  [
    {
      field: 'username',
      label: '用户名',
      validate: (val) => (!val ? '请输入用户名' : val.length < 3 ? '至少3个字符' : null),
      trigger: 'blur',
    },
    {
      field: 'password',
      label: '密码',
      validate: (val) => {
        if (!val) return '请输入密码'
        if (val.length < 8) return '密码至少8位'
        if (!/[A-Z]/.test(val)) return '需包含大写字母'
        if (!/[0-9]/.test(val)) return '需包含数字'
        return null
      },
    },
    {
      field: 'rememberMe',
      label: '记住我',
      validate: () => null, // 始终有效
    },
  ]
)

// form.values.username → string
// form.errors.username → string[] | undefined
// form.validateAll() → boolean
```

## 总结

编写类型安全的组合式函数，本质上是**在 JavaScript 的灵活性之上叠加 TypeScript 的精确性**。三个核心要素——泛型参数化（复用的基础）、标准返回结构（一致性的保障）、MaybeRef（灵活性的来源）——共同构成了高质量 Composable 的骨架。当你写的每个 useXxx 函数都能让调用者在零配置的情况下获得完整的类型保护和 IDE 体验时，你就真正掌握了 Vue 3 + TypeScript 的协同之道。
