# React 事件处理器与 DOM 类型

> **一句话总结**：React 的事件系统封装了原生 DOM 事件，`SyntheticEvent`、`ChangeEvent`、`MouseEvent` 等类型构成了事件处理的完整类型体系。理解它们之间的继承关系和属性差异，是编写无 bug 事件处理代码的关键。

## 一、事件类型层次结构

```
                    Event (原生事件)
                       │
          ┌────────────┼────────────┐
          │            │            │
   UIEvent    FocusEvent  MouseEvent
     │            │            │
     └──┬──┘       └──┬──┘        └──┬──┘
        │              │              │
   KeyboardEvent  ClipboardEvent   PointerEvent
                                │
                          WheelEvent
```

**类比理解**：

把 DOM 事件类型想象成一个**家族树**：
- **Event** 是老祖宗，所有事件的共同祖先
- **UIEvent** 是用户界面事件分支的老祖宗
- **MouseEvent**、**KeyboardEvent** 等是具体的"孙辈"，各有自己的专属特征

React 的 `SyntheticEvent<T>` 就是这个家族树的"TypeScript 镜像"——它保留了完整的类型层级关系。

## 二、SyntheticEvent —— React 的事件包装

### 2.1 什么是 SyntheticEvent？

```typescript
/**
 * SyntheticEvent<T> 的核心作用：
 * 
 * 1. 跨浏览器兼容：统一不同浏览器的差异
 * 2. 类型安全：提供完整的 TypeScript 类型
 * 3. 性能优化：使用事件池（Event Pooling）
 *
 * 泛型参数 T = 底层原生事件类型
 */

// React 中的事件处理函数签名
function handleClick(event: React.MouseEvent<HTMLButtonElement>) {
  // event 是 SyntheticEvent<MouseEvent>
  // 它拥有 MouseEvent 的所有属性 + React 额外添加的属性
}

function handleChange(event: React.ChangeEvent<HTMLInputElement>) {
  // event 是 SyntheticEvent<Event>
  // 它拥有原生 ChangeEvent 的所有属性
}
```

### 2.2 SyntheticEvent 的通用属性

```typescript
interface BaseSyntheticEvent<E = Event> {
  // ===== 来自原生 Event 的属性 =====
  bubbles: boolean                  // 是否冒泡
  cancelable: boolean               // 可否取消默认行为
  currentTarget: EventTarget        // 当前绑定目标
  defaultPrevented: boolean           // 是否调用了 preventDefault()
  eventPhase: number                // 事件阶段（1=捕获 2=目标 3=冒泡）
  isTrusted: boolean                // 是否可信（用户触发 vs 脚本生成）
  target: EventTarget                // 事件原始目标
  timeStamp: number                 // 事件发生时间戳
  type: string                      // 事件类型 ('click', 'change'...)

  // ===== React 特有属性 =====
  nativeEvent: E                     // 底层原生事件引用
}
```

### 2.3 为什么不直接用原生事件类型？

```typescript
// ❌ 使用原生 MouseEvent 的问题
function handleNativeClick(event: MouseEvent) {
  console.log(event.target)         // EventTarget | null — 太宽泛
  console.log(event.currentTarget)  // EventTarget | null — 同上
  // 缺少 React 特有的方法如 isPropagationStopped()
  // 在 React 17+ 中可能无法正确工作
}


// ✅ 使用 React.MouseEvent
function handleReactClick(event: React.MouseEvent<HTMLButtonElement>) {
  console.log(event.currentTarget.value) // HTMLButtonElement — 精确类型！
  console.log(event.nativeEvent)      // 可以回退到原生事件
  
  // React 提供的便捷方法
  event.isPropagationStopped()          // 是否已阻止冒泡
  event.isDefaultPrevented()             // 是否已阻止默认行为
  event.persist()                        // 将事件从池中取出（异步场景需要）
}
```

## 三、各事件类型详解

### 3.1 MouseEvent —— 最常用的事件类型

```typescript
/**
 * MouseEvent 的关键属性（React.MouseEvent 继承了全部）
 */
interface MouseEventType {
  // 坐标位置
  altKey: boolean                   // Alt 键是否按下
  button: number                   // 鼠标按键 (0=左 1=中 2=右 3=后退 4=前进)
  buttons: number                  // 同时按下的键位掩码
  clientX: number                  // 相对于视口的 X 坐标
  clientY: number                  // 相对于视口的 Y 坐标
  ctrlKey: boolean                  // Ctrl 键是否按下
  metaKey: boolean                  // Meta/Command 键是否按下
  offsetX: number                  // 相对于目标元素的 X 偏移
  offsetY: number                  // 相对于目标元素的 Y 偏移
  pageX: number                   // 相对于文档的 X 坐标
  pageY: number                   // 相对于文档的 Y 坐标
  screenX: number                  // 相对于屏幕的 X 坐标
  screenY: number                  // 相对于屏幕的 Y 坐标
  shiftKey: boolean                 // Shift 键是否按下
  movementX: number                // 上一次鼠标移动的 X 差值
  movementY: number                // 上一次鼠标移动的 Y 差值
  
  // 目标信息
  relatedTarget: EventTarget | null  // 相关目标（如 mouseleave 时离开进入的元素)
  
  // 按钮详情
  which: number                    // 按键码（已被 button 替代，但保留兼容）
}

// ===== 实用示例 =====
function DraggableComponent() {
  const [position, setPosition] = useState({ x: 0, y:  })
  const [isDragging, setIsDragging] = useState(false)
  const dragOffset = useRef({ x: 0, y: 0 })

  function handleMouseDown(e: React.MouseEvent<HTMLDivElement>) => {
    setIsDragging(true)
    // 精确获取相对于元素的偏移
    const rect = e.currentTarget.getBoundingClientRect()
    dragOffset.current = {
      x: e.clientX - rect.left,
      y: e.clientY - rect.top,
    }
    
    e.preventDefault() // 阻止文本选择
  }

  function handleMouseMove(e: React.MouseEvent<HTMLDocumentElement>) {
    if (!isDragging) return
    
    setPosition({
      x: e.clientX - dragOffset.current.x,
      y: e.clientY - dragOffset.current.y,
    })
  }

  function handleMouseUp() {
    setIsDragging(false)
  }

  return (
    <div
      className="draggable"
      style={{
        position: 'relative',
        left: position.x,
        top: position.y,
        cursor: isDragging ? 'grabbing' : 'grab',
        userSelect: 'none',
        touchAction: 'none',
      }}
      onMouseDown={handleMouseDown}
      onMouseMove={handleMouseMove}
      onMouseUp={handleMouseUp}
    >
      Drag me!
    </div>
  )
}
```

```typescript
  const selectedValue = e.target.value  // string

}

function handleRangeChange(e: React.ChangeEvent<HTMLInputElement>) {
  const value = Number(e.target.value)  // 需要手动转换！
  console.log(`Range value: ${value}`)
}

// input[type="file"] → target.files 是 FileList
function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
  const files = e.target.files  // FileList（类数组，不是普通数组）
  const file = files[0]           // File | undefined
  if (file) {
    console.log(`File name: ${file.name}, size: ${file.size} type: ${file.type}`)
  }
}
```

```typescript
  // 键码相关

  console.log(`Code: ${e.code}`)          // 物理按键码（'KeyA', 'Enter', 'ArrowUp'...）
  console.log(`Which: ${e.which}`)        // 数字 keyCode（已废弃但保留）
  console.log(`Ctrl: ${e.ctrlKey} Alt: ${e.altKey} Shift: {e.shiftKey}`)

  if (e.key === 's' && (e.ctrlKey || e.metaKey)) {
    e.preventDefault()
    console.log('Save shortcut triggered!')
  }
  
  // 特殊键
  if (e.key === 'Tab') {
    // Tab 键逻辑
  }
  if (e.key === 'Escape') {
    // ESC 取消操作
  }
  if (e.key === 'Enter') {
    // Enter 提交
  }
}

// Focus 事件（常用于表单验证时机控制）
function handleBlur(e: React.FocusEvent<HTMLInputElement>) {
  // e.relatedTarget —— 失去焦点的元素
  // 当从 inputA 切换到 inputB 时：
  // inputA 的 blur 事件：relatedTarget = inputB
  // inputB 的 focus 事件：relatedTarget = inputA
  
  // 典用于"失焦时校验"
  validateField(e.currentTarget as HTMLInputElement)
}

function validateField(input: HTMLInputElement) {
  if (!input.value.trim()) {
    input.classList.add('error')
    input.setCustomValue('此字段不能为空')
  } else {
    input.classList.remove('error')
    input.setCustomValue('')
  }
}
```

```typescript
// 触摸事件（移动端开发必备）
function handleTouchStart(e: React.TouchEvent<HTMLDivElement>) {
  // touches: TouchList —— 当前触摸点列表
  // changedTouches: TouchList —— 变化的触摸点
  // targetTouches: TouchList —— 目标元素上的触摸点
  
  const touch = e.touches[0]

// Pointer Event —— 统一的指针 API
function handlePointerDown(e: React.PointerEvent<HTMLDivElement>) {
  // pointerId —— 唯一标识符（支持多指/多点交互）
  // pointerType —— 'mouse' | 'touch' | 'pen'
  // pressure —— 压力值（0~1，支持压感设备）
  // width/height —— 接触区域的尺寸
  // tiltX/tiltY —— 笔倾斜角度（手写笔）
  
  switch (e.pointerType) {
    case 'mouse':
      console.log('Mouse down')
      break
    case 'touch':
      console.log('Touch down')
      break
    case 'pen':
      console.log('Pen down, pressure:', e.pressure)
      break
  }
}
```

### 3.5 拖截事件默认行为的正确方式

```typescript
/**
 * ⚠️ 常见错误：在 React 中调用 preventDefault 的时机问题
 */

// ❌ 错误：在异步操作之后调用 preventDefault（太晚了）
async function badHandleLinkClick(e: React.MouseEvent<HTMLAnchorElement>) {
  await someAsyncOperation()   // 先执行了其他操作
  e.preventDefault()                 // 再尝试阻止默认行为 —— 但浏览器已经导航了！
}

// ✅ 正确：先阻止，再执行异步操作
async function goodHandleLinkClick(e: React.MouseEvent<HTMLAnchorElement>) {
  e.preventDefault()                  // 立即阻止默认导航
  await someAsyncOperation()       // 然后执行业务逻辑
  router.push(newDestination)
}

  if (!confirmed) return  // 用户取消
```

## 四、事件委托的类型安全模式

```typescript
/**
 * 事件委托（Event Delegation）是前端性能优化的核心技术
 * TypeScript 让委托模式的实现更加安全
 */

// ===== 类型安全的通用事件委托器 =====
function useDelegatedEventHandler<
  K extends keyof HTMLElementEventMap
>(
  parentRef: Ref<HTMLElement | null>,
  eventType: K,
  selector: string,
  handler: (event: HTMLElementEventMap[K], element: HTMLElement) => void,
  options?: AddEventListenerOptions
): void {
  useEffect(() => {
    const el = parentRef.current
    if (!el) return

    // 使用事件委托：监听父元素，检查事件目标
    const listener = (event: HTMLElementEventMap[K]) => {
      // 找到匹配 selector 的后代元素
      const target = (event.target as HTMLElement).closest(selector)
      if (target && el.contains(target)) {
        handler(event, target)
      }
    }

    el.addEventListener(eventType, listener, options)

    return () => el.removeEventListener(eventType, listener)
  }, [parentRef, handler, selector, eventType])
}

// ===== 使用：表格行点击 =====
function TableWithDelegation() {
  const tableRef = useRef<HTMLTableElement>(null)

  useDelegatedEventHandler(
    tableRef,
    'click',
    'tr[data-id]',
    (event, row) => {
      const id = row.getAttribute('data-id')
      console.log(`Clicked row with ID: ${id}`)
      
      // event 是 MouseEvent（click 对应的是 MouseEvent）
      if (event.ctrlKey) {
        openInNewTab(`/rows/${id}`)
      } else {
        navigateTo(`/rows/${id}`)
      }
    }
  )

  return (
    <table ref={tableRef}>
      <tbody>
        {rows.map(row => (
          <tr key={row.id} data-id={row.id}>
            <td>{row.name}</td>
            <td>{row.status}</td>
          </tr>
        ))}
      </tbody>
    </table>
  )
}
```

## 五、自定义事件类型

```typescript
/**
 * 有时你需要创建自定义组件事件或扩展原生事件类型
 */

// ===== 方式一：接口继承（推荐）=====
interface CustomEventData {
  detail: { itemId: string; action: string; timestamp: number }
  bubble?: boolean
  cancelable?: boolean
}

// 自定义 Hook 封装
function useCustomEventListener(
  elementRef: Ref<HTMLElement | null>,
  eventName: string,
  handler: (detail: CustomEventData['detail']) => void
): void {
  useEffect(() => {
    const el = elementRef.current
    if (!el) return

    const listener = (event: Event) => {
      const ce = event as CustomEvent<CustomEventData['detail']>
      handler(ce.detail)
    }

    el.addEventListener(eventName, listener)

    return () => el.removeEventListener(eventName, listener)
  }, [elementRef, handler, eventName])
}


// ===== 方式二：增强原生事件类型 =====
// 为原生事件添加自定义属性
declare global {
  interface MouseEvent {
    /** 数据属性 —— 由外部注入的业务数据 */
    dataPayload?: unknown
    /** 触发来源标记 */
    sourceId?: string
  }
}

// 现在 MouseEvent 自动携带这些额外字段
function handleClick(e: React.MouseEvent) {
  // ✅ e.dataPayload 存在
  // ✅ e.sourceId 存在
  console.log((e as any).dataPayload)
}
```

## 总结

React 事件类型体系的掌握程度直接决定了你写事件处理代码的效率和安全性。核心要点：

1. **`React.MouseEvent<E>`** 等 SyntheticEvent 包装器同时提供跨浏览器兼容和精确的元素类型——这是对原生 `MouseEvent` 的全面升级。

2. **`ChangeEvent<T>` 的泛型参数 T** 决定了 `target.value` 的类型——`HTMLInputElement` 给出 string、`HTMLInputElement`（checkbox）给出 boolean。不要忘记 range 类型的 value 需要手动 `Number()` 转换。

3. **`KeyboardEvent` 的 `e.key` vs `e.code` vs `e.which`**——`key` 用于可读字符、`code` 用于物理按键名（推荐）、`which` 已废弃。

4. **`preventDefault` 必须在异步操作之前调用**——这是 React 事件处理中最容易踩的坑之一。

5.事件委托配合 `closest()` 和泛型约束可以实现高效且类型安全的批量事件管理。
