# 文件上传进阶 —— input type="file"、多文件、拖拽上传、分片上传

> **本文的目标**：突破基础 [FormData 文件上传](/40.frontend/50.javascript/20.browser-api/20.网络请求与实时通信/140.FormData文件上传.md) 的范畴，掌握生产环境中的高级文件上传技术——多文件选择、拖拽区域实现、大文件分片上传与断点续传，以及文件类型的客户端校验。

## 一、input[type=file] 的完全掌控

`<input type="file">` 是浏览器提供的原生文件选择控件。虽然它的默认外观简陋，但其背后的 `FileList` / `File` API 却非常强大：

```html
<!-- 基础用法 -->
<input type="file" id="basic">

<!-- 限制文件类型（通过 accept 属性） -->
<input type="file" accept="image/*">
<input type="file" accept=".pdf,.doc,.docx">
<input type="file" accept="audio/*,video/*">

<!-- 多文件选择（multiple 属性） -->
<input type="file" id="multi" multiple accept="image/*">

<!-- 隐藏原生按钮，自定义触发 -->
<input type="file" id="hidden-file" style="display:none">
<button onclick="document.getElementById('hidden-file').click()">
    选择文件
</button>

<!-- 目录选择（webkitdirectory） -->
<input type="file" webkitdirectory>
```

### File 对象的属性与方法

当用户选择了文件后，`input.files` 返回的是一个 `FileList` 类数组对象，其中的每一项都是 `File` 对象（继承自 `Blob`）：

```javascript
const input = document.querySelector('#multi');

// 监听文件选择变化
input.addEventListener('change', (e) => {
    const files = e.target.files;   // FileList 对象
    console.log('选中了', files.length, '个文件');

    Array.from(files).forEach(file => {
        console.log('---');
        console.log('文件名:', file.name);           // "photo.jpg"
        console.log('大小:', formatSize(file.size));  // "2.5 MB"
        console.log('类型:', file.type);             // "image/jpeg"
        console.log('最后修改:', new Date(file.lastModified));  // Date 对象
        console.log('是否图片?', file.type.startsWith('image/'));

        // File 继承自 Blob，所以可以使用 Blob 的所有方法
        // file.text() → Promise<string>   (读取为文本)
        // file.arrayBuffer() → Promise<ArrayBuffer>  (读取为二进制缓冲区)
        // file.stream() → ReadableStream   (流式读取)
        // URL.createObjectURL(file) → string  (创建临时URL用于预览)
    });
});

function formatSize(bytes) {
    if (bytes < 1024) return bytes + ' B';
    if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB';
    return (bytes / 1048576).toFixed(1) + ' MB';
}
```

### 文件类型校验

不要只依赖 `accept` 属性——它只是过滤了文件选择对话框的显示范围，用户仍然可以选择"所有文件"来绕过限制。真正的校验需要在 JavaScript 中进行：

```javascript
function validateFile(file, rules = {}) {
    const errors = [];

    // 类型校验
    if (rules.accept && rules.accept.length > 0) {
        const allowedTypes = rules.accept;
        const fileType = file.type.toLowerCase();
        const ext = '.' + file.name.split('.').pop().toLowerCase();

        const typeMatch = allowedTypes.some(rule =>
            rule.startsWith('.') ? rule === ext : fileType.includes(rule.replace('*', ''))
        );
        if (!typeMatch) errors.push(`不支持的文件类型: ${file.type || '未知类型'}`);
    }

    // 大小校验
    if (rules.maxSize && file.size > rules.maxSize) {
        errors.push(`文件过大: ${formatSize(file.size)}（最大允许 ${formatSize(rules.maxSize)}）`);
    }

    if (rules.minSize && file.size < rules.minSize) {
        errors.push(`文件过小: ${formatSize(file.size)}（最小需要 ${formatSize(rules.minSize)}）`);
    }

    return { valid: errors.length === 0, errors };
}

// 使用
const result = validateFile(selectedFile, {
    accept: ['image/png', 'image/jpeg', '.gif'],
    maxSize: 5 * 1024 * 1024,   // 5MB
    minSize: 100               // 至少100字节
});

if (!result.valid) {
    showErrors(result.errors);
}
```

## 二、拖拽上传：打造现代化的拖放区域

```javascript
class DropZoneUploader {
    constructor(zoneId, options = {}) {
        this.zone = document.getElementById(zoneId);
        this.input = this.zone.querySelector('[type="file"]') ||
                     document.createElement('input');
        this.previewGrid = options.previewGrid;
        this.acceptedTypes = options.accept || ['image/*'];
        this.maxSize = options.maxSize || 10 * 1024 * 1024;
        this.maxFiles = options.maxFiles || 10;
        this.selectedFiles = [];

        this._bindEvents();
    }

    _bindEvents() {
        // 点击触发文件选择
        this.zone.addEventListener('click', () => this.input.click());

        // 拖拽进入区域
        this.zone.addEventListener('dragenter', (e) => {
            e.preventDefault();
            this.zone.classList.add('drag-over');
        });

        // 拖拽悬停（持续触发）
        this.zone.addEventListener('dragover', (e) => {
            e.preventDefault();  // 必须阻止默认行为才能触发 drop！
        });

        // 拖拽离开
        this.zone.addEventListener('dragleave', (e) => {
            e.preventDefault();
            // 只有真正离开了 dropzone 才移除样式
            if (!this.zone.contains(e.relatedTarget)) {
                this.zone.classList.remove('drag-over');
            }
        });

        // 放置文件
        this.zone.addEventListener('drop', (e) => {
            e.preventDefault();
            this.zone.classList.remove('drag-over');

            const files = Array.from(e.dataTransfer.files);
            this.handleFiles(files);
        });

        // input 选择文件
        this.input.addEventListener('change', (e) => {
            this.handleFiles(Array.from(e.target.files));
            e.target.value = '';   // 清空以便重复选择同一文件
        });
    }

    handleFiles(files) {
        if (this.selectedFiles.length + files.length > this.maxFiles) {
            alert(`最多只能选择 ${this.maxFiles} 个文件`);
            return;
        }

        files.forEach(file => {
            const validation = this._validateFile(file);
            if (!validation.valid) {
                showToast(validation.error, 'error');
                return;
            }

            this.selectedFiles.push(file);
            this._createPreview(file);
        });
    }

    _validateFile(file) {
        if (file.size > this.maxSize) {
            return { valid: false, error: `${file.name} 超过大小限制` };
        }

        if (this.acceptedTypes.length > 0) {
            const isAccepted = this.acceptedTypes.some(type => {
                if (type.endsWith('/*')) {
                    return file.type.startsWith(type.slice(0, -1));
                }
                return file.type === type || file.name.endsWith(type);
            });
            if (!isAccepted) {
                return { valid: false, error: `${file.name} 不支持的文件类型` };
            }
        }

        return { valid: true, error: null };
    }

    _createPreview(file) {
        if (!this.previewGrid) return;

        const item = document.createElement('div');
        item.className = 'preview-item';

        // 使用 URL.createObjectURL 创建预览图（比 FileReader 更高效）
        const img = document.createElement('img');
        img.src = URL.createObjectURL(file);
        img.alt = file.name;
        img.onload = () => URL.revokeObjectURL(img.src);   // 加载完成后释放内存

        const btn = document.createElement('button');
        btn.className = 'remove-btn';
        btn.innerHTML = '&times;';
        btn.onclick = () => {
            item.remove();
            const idx = this.selectedFiles.indexOf(file);
            if (idx > -1) this.selectedFiles.splice(idx, 1);
        };

        item.appendChild(img);
        item.appendChild(btn);
        this.previewGrid.appendChild(item);
    }

    getFiles() { return this.selectedFiles; }
}

// 初始化
new DropZoneUploader('dropZone', {
    previewGrid: document.getElementById('previewGrid'),
    accept: ['image/png', 'image/jpeg', 'image/gif'],
    maxSize: 10 * 1024 * 1024,
    maxFiles: 9
});
```

## 三、大文件分片上传与断点续传

当文件体积超过几十 MB 时，一次性的整体上传会面临诸多问题：网络波动导致整个上传失败需要重来、服务器可能限制了单个请求的大小、无法展示精细的上传进度……**分片上传（Chunked Upload）** 是这类问题的标准解决方案：

```javascript
/**
 * ChunkedUploader —— 分片上传器
 *
 * 核心思路：
 * 1. 将大文件按固定大小切分为多个 chunk
 * 2. 逐个上传每个 chunk，携带 chunk 索引和文件 hash
 * 3. 服务端接收所有 chunk 后合并
 * 4. 记录已完成的 chunk，支持断点续传
 */

class ChunkedUploader {
    constructor(file, options = {}) {
        this.file = file;
        this.chunkSize = options.chunkSize || 2 * 1024 * 1024;   // 默认2MB一片
        this.uploadUrl = options.uploadUrl || '/api/upload/chunk';
        this.mergeUrl = options.mergeUrl || '/api/upload/merge';
        this.checkUrl = options.checkUrl || '/api/upload/check';  // 断点续传检查
        this.maxRetries = options.maxRetries || 3;

        this.totalChunks = Math.ceil(this.file.size / this.chunkSize);
        this.uploadedChunks = new Set();   // 已上传的 chunk 索引集合
        this.fileHash = null;              // 文件的唯一标识（用于续传）

        this.onProgress = options.onProgress || (() => {});
    }

    // 计算文件hash（简化版：使用文件名+大小+修改时间）
    async calculateHash() {
        const content = `${this.file.name}-${this.file.size}-${this.file.lastModified}`;
        const encoder = new TextEncoder();
        const data = encoder.encode(content);
        const hashBuffer = await crypto.subtle.digest('SHA-256', data);
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        this.fileHash = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
        return this.fileHash;
    }

    // 检查哪些chunk已经上传过了（断点续传的核心）
    async checkUploadedChunks() {
        await this.calculateHash();

        const res = await fetch(`${this.checkUrl}?hash=${this.fileHash}&filename=${encodeURIComponent(this.file.name)}`);

        if (res.ok) {
            const data = await res.json();
            this.uploadedChunks = new Set(data.uploadedChunks || []);
            console.log(`断点续传: 已有 ${this.uploadedChunks.size}/${this.totalChunks} 个分片`);

            if (this.uploadedChunks.size === this.totalChunks) {
                console.log('文件实际上已经上传完毕！');
                return 'completed';
            }

            if (data.fileId) {
                this.serverFileId = data.fileId;
            }
        }

        return 'partial';
    }

    // 获取第 n 个分片的 Blob 数据
    getChunk(chunkIndex) {
        const start = chunkIndex * this.chunkSize;
        const end = Math.min(start + this.chunkSize, this.file.size);
        return this.file.slice(start, end);
    }

    // 上传单个分片
    async uploadChunk(chunkIndex, retryCount = 0) {
        if (this.uploadedChunks.has(chunkIndex)) {
            return;   // 已上传过，跳过
        }

        const chunk = this.getChunk(chunkIndex);
        const formData = new FormData();
        formData.append('chunk', chunk);
        formData.append('index', String(chunkIndex));
        formData.append('total', String(this.totalChunks));
        formData.append('hash', this.fileHash);
        formData.append('filename', this.file.name);
        if (this.serverFileId) {
            formData.append('fileId', this.serverFileId);
        }

        try {
            const res = await fetch(this.uploadUrl, { method: 'POST', body: formData });

            if (!res.ok) throw new Error(`Chunk ${chunkIndex}: HTTP ${res.status}`);

            const data = await res.json();

            if (data.fileId) this.serverFileId = data.fileId;

            this.uploadedChunks.add(chunkIndex);
            this._updateProgress();

        } catch (err) {
            console.warn(`分片 ${chunkIndex} 上传失败 (${retryCount + 1}/${this.maxRetries}):`, err.message);

            if (retryCount < this.maxRetries - 1) {
                await sleep(1000 * (retryCount + 1));  // 递增延迟重试
                return this.uploadChunk(chunkIndex, retryCount + 1);
            }

            throw err;   // 重试耗尽，抛出错误
        }
    }

    // 开始上传所有分片（可控制并发数）
    async startUpload(concurrency = 3) {
        const status = await this.checkUploadedChunks();
        if (status === 'completed') {
            this.onProgress({ percent: 100, uploaded: this.file.size, total: this.file.size });
            return { completed: true, message: '已完成' };
        }

        // 找出还需要上传的分片索引
        const pendingChunks = [];
        for (let i = 0; i < this.totalChunks; i++) {
            if (!this.uploadedChunks.has(i)) pendingChunks.push(i);
        }

        console.log(`开始上传 ${pendingChunks.length} 个剩余分片（并发: ${concurrency}）`);

        // 并发上传控制
        const executing = new Set();

        for (const chunkIdx of pendingChunks) {
            const promise = this.uploadChunk(chunkIdx).catch(err => {
                throw err;   // 让外层 catch 处理
            });

            executing.add(promise);
            promise.finally(() => executing.delete(promise));

            if (executing.size >= concurrency) {
                await Promise.race(executing);   // 有空位了再继续
            }
        }

        await Promise.all(executing);   // 等待所有进行中的任务完成

        // 所有分片上传完毕，通知服务端合并
        return this._mergeChunks();
    }

    // 通知服务端合并所有分片
    async _mergeChunks() {
        const res = await fetch(this.mergeUrl, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                filename: this.file.name,
                hash: this.fileHash,
                totalChunks: this.totalChunks,
                fileSize: this.file.size,
                fileId: this.serverFileId
            })
        });

        if (!res.ok) throw new Error(`合并失败: HTTP ${res.status}`);

        return res.json();
    }

    _updateProgress() {
        const uploadedSize = this.uploadedChunks.size * this.chunkSize;
        const percent = Math.round(
            (this.uploadedChunks.size / this.totalChunks) * 100
        );

        this.onProgress({
            percent: Math.min(percent, 99),   // 合并期间保持99%
            chunksDone: this.uploadedChunks.size,
            chunksTotal: this.totalChunks,
            uploaded: uploadedSize,
            total: this.file.size
        });
    }
}

function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

// ===== 使用示例 =====
document.querySelector('#start-upload').onclick = async () => {
    const file = fileInput.files[0];
    if (!file) return;

    const uploader = new ChunkedUploader(file, {
        chunkSize: 2 * 1024 * 1024,   // 2MB per chunk
        maxRetries: 3,
        onProgress(info) {
            updateUI(info);
            console.log(`进度: ${info.chunksDone}/${info.chunksTotal} 分片 (${info.percent}%)`);
        }
    });

    try {
        const result = await uploader.startUpload(3);   // 3并发
        console.log('上传完成!', result.url);
    } catch (err) {
        console.error('上传失败:', err);
        showError('上传失败，请稍后重试');
    }
};
```

> **哲学思考**：分片上传体现了一种**分而治之**的经典工程智慧。面对一个巨大的问题（上传一个500MB的文件），与其试图一口气解决它（然后因为任何一点小失误就前功尽弃），不如把它拆成若干个小问题逐个击破。每个分片的失败只影响自身而不影响其他分片，失败后的重试代价也极低。这不仅是文件上传的策略，也是软件架构中微服务设计、分布式计算、增量部署等众多领域的共同思维模型。人类在面对巨大挑战时也是这样做的——不会试图一天建成罗马城，而是先铺一块砖，再砌一面墙，最后才有宏伟的建筑。分片上传就是数字世界里的"搬砖法"——朴实无华但极其有效。
