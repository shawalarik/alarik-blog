# 封装 IndexedDB —— Promise 化、idb 库的使用

在 [《IndexedDB》](/40.frontend/50.javascript/20.browser-api/30.数据存储与二进制/160.IndexedDB.md) 中我们学习了原生 API，但它的回调模式会让代码迅速陷入嵌套深渊。本文将用 Promise 重写所有操作，并介绍社区成熟的 idb 库。

## 一、回调地狱：原生 API 之痛

IndexedDB 的所有操作都基于事件驱动——`request.onsuccess`、`request.onerror`……连续执行多个操作时代码会层层嵌套：

```javascript
// 原生写法 —— 查询 → 更新 → 插入日志，三层嵌套
request.onsuccess = () => {
    const db = request.result;
    const tx1 = db.transaction(['users'], 'readwrite');
    const req1 = tx1.objectStore('users').get(1);
    
    req1.onsuccess = () => {
        req1.result.lastLogin = Date.now();
        const tx2 = db.transaction(['users', 'logs'], 'readwrite');
        const req2 = tx2.objectStore('users').put(req1.result);   // 第二层
        
        req2.onsuccess = () => {
            const req3 = tx2.objectStore('logs')
                .add({ userId: 1, action: 'login' });             // 第三层
            req3.onsuccess = () => console.log('全部完成！');
            req3.onerror = () => console.error('日志写入失败');
        };
    };
};
```

每层都要处理 success 和 error，每多一步就多一层缩进。这种代码不仅难读更难维护。

### 生活类比

原生回调就像手工记账——每记一笔翻一页新纸（开新事务），写完要检查有没有算错（error 处理），连续几笔就把本子翻乱了。Promise 封装则像换成了电子表格——输入公式自动计算，数据清晰排列，修改一处不影响其他部分。

## 二、从零构建 Promise 封装

核心思路很简单：把每个基于 `request.onsuccess/request.onerror` 的操作包裹成返回 Promise 的函数：

```javascript
class IDBHelper {
    constructor(dbName, version, upgradeCallback) {
        this.dbName = dbName;
        this.version = version;
        this.upgradeCallback = upgradeCallback;
        this._dbPromise = null;   // 缓存数据库连接
    }

    // 单例模式获取数据库连接
    getDB() {
        if (!this._dbPromise) {
            this._dbPromise = new Promise((resolve, reject) => {
                const request = indexedDB.open(this.dbName, this.version);

                request.onupgradeneeded = (event) => {
                    if (this.upgradeCallback) {
                        this.upgradeCallback(
                            event.target.result,
                            event.oldVersion,
                            event.newVersion
                        );
                    }
                };

                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);

                // 连接意外断开时清除缓存以便重连
                request.result?.addEventListener?.('versionchange', () => {
                    this._dbPromise = null;
                    request.result.close();
                });
            });
        }
        return this._dbPromise;
    }

    // 核心：将 IDBRequest 包装为 Promise
    _promisify(request) {
        return new Promise((resolve, reject) => {
            request.onsuccess = () => resolve(request.result);
            request.onerror = () => reject(request.error);
        });
    }

    // ===== CRUD 操作 =====

    async get(storeName, key) {
        const db = await this.getDB();
        return this._promisify(
            db.transaction(storeName, 'readonly')
              .objectStore(storeName).get(key)
        );
    }

    async getAll(storeName, query = null) {
        const db = await this.getDB();
        const store = db.transaction(storeName, 'readonly')
                          .objectStore(storeName);
        return query ? this._promisify(store.getAll(query))
                     : this._promisify(store.getAll());
    }

    async put(storeName, data) {
        const db = await this.getDB();
        return this._promisify(
            db.transaction(storeName, 'readwrite')
              .objectStore(storeName).put(data)
        );
    }

    async add(storeName, data) {
        const db = await this.getDB();
        return this._promisify(
            db.transaction(storeName, 'readwrite')
              .objectStore(storeName).add(data)
        );
    }

    async delete(storeName, key) {
        const db = await this.getDB();
        return this._promisify(
            db.transaction(storeName, 'readwrite')
              .objectStore(storeName).delete(key)
        );
    }

    async count(storeName, query = null) {
        const db = await this.getDB();
        const store = db.transaction(storeName, 'readonly')
                          .objectStore(storeName);
        return query ? this._promisify(store.count(query))
                     : this._promisify(store.count());
    }

    async clear(storeName) {
        const db = await this.getDB();
        return this._promisify(
            db.transaction(storeName, 'readwrite')
              .objectStore(storeName).clear()
        );
    }
}
```

### 使用封装后的代码对比

```javascript
// 初始化
const idb = new IDBHelper('MyAppDB', 1, (db) => {
    db.createObjectStore('users', { keyPath: 'id', autoIncrement: true });
    db.createObjectStore('posts', { keyPath: 'id', autoIncrement: true })
       .createIndex('userId', 'userId');
});

// 现在！同样的逻辑变得如此清爽：
async function handleUserLogin(userId) {
    const user = await idb.get('users', userId);
    if (!user) throw new Error('用户不存在');

    user.lastLogin = Date.now();
    await idb.put('users', user);

    await idb.add('logs', { userId, action: 'login', time: Date.now() });
    console.log('全部完成！');   // 扁平的！没有嵌套！
}

// 调用
handleUserLogin(1).catch(console.error);
```

从三层嵌套变成了线性的 `async/await` 流程。可读性提升了不止一个量级。

## 三、进阶封装：索引查询与游标

基础 CRUD 只是起点，真正的威力在于索引查询和游标遍历：

```javascript
// 在 IDBHelper 类中继续添加方法：

// 通过索引查询单条
async getByIndex(storeName, indexName, value) {
    const db = await this.getDB();
    const store = db.transaction(storeName, 'readonly')
                      .objectStore(storeName);
    const index = store.index(indexName);
    return this._promisify(index.get(value));
}

// 通过索引范围查询多条
async getAllByIndexRange(storeName, indexName, range) {
    const db = await this.getDB();
    const store = db.transaction(storeName, 'readonly')
                      .objectStore(storeName);
    const index = store.index(indexName);
    return this._promisify(index.getAll(range));
}

// 通过索引范围 + 游标遍历（处理大量数据时比 getAll 更节省内存）
async *cursorByIndex(storeName, indexName, range, direction = 'next') {
    const db = await this.getDB();
    const store = db.transaction(storeName, 'readonly')
                      .objectStore(storeName);
    const index = store.index(indexName);
    
    const request = range 
        ? index.openCursor(range, direction)
        : index.openCursor(null, direction);

    return new Promise((resolve, reject) => {
        const results = [];
        request.onsuccess = (event) => {
            const cursor = event.target.result;
            if (cursor) {
                results.push({ key: cursor.key, value: cursor.value });
                cursor.continue();   // 移动到下一条记录
            } else {
                resolve(results);    // 遍历完毕
            }
        };
        request.onerror = () => reject(request.error);
    });
}
```

### 实际使用场景示例

```javascript
// 场景1：按邮箱查找用户（唯一索引）
const user = await idb.getByIndex('users', 'email', 'alarik@example.com');

// 场景2：价格区间筛选商品
const cheapProducts = await idb.getAllByIndexRange(
    'products',
    'price',
    IDBKeyRange.bound(10, 100)
);

// 场景3：按状态批量更新订单
const pendingOrders = await idb.getAllByIndexRange(
    'orders',
    'status',
    IDBKeyRange.only('pending')
);

for (const order of pendingOrders) {
    order.status = 'processing';
    order.updatedAt = Date.now();
    await idb.put('orders', order);
}

console.log(`已将 ${pendingOrders.length} 个订单标记为处理中`);
```

## 四、社区方案：idb 库

虽然自己封装不难，但生产项目中推荐使用 **idb**（Jake Archibald 维护的轻量库）。它提供了更完善的类型支持、更简洁的 API，以及事务自动提交等便利功能：

```bash
# 安装
npm install idb
# 或通过 CDN
# <script src="https://unpkg.com/idb@8/build/umd.js"></script>
```

```javascript
import { openDB } from 'idb';

// 打开数据库
const db = await openDB('MyAppDB', 1, {
    upgrade(db) {
        // 创建 object store
        const userStore = db.createObjectStore('users', { 
            keyPath: 'id', autoIncrement: true 
        });
        userStore.createIndex('email', 'email', { unique: true });

        db.createObjectStore('posts', { keyPath: 'id', autoIncrement: true })
           .createIndex('userId', 'userId');
    }
});

// CRUD 操作 —— 极简 API
await db.put('users', { name: 'Alarik', email: 'a@ex.com' });
const user = await db.get('users', 1);
const allUsers = await db.getAll('users');
await db.delete('users', 1);

// 事务 —— 自动提交
await db.transaction('rw', ['users', 'logs'], async (tx) => {
    await tx.store.users.put(updatedUser);
    await tx.store.logs.add(logEntry);
    // 事务结束时自动 commit
});

// 索引查询
const byEmail = await db.getFromIndex('users', 'email', 'a@ex.com');
const priceRange = await db.getAllFromIndex(
    'products', 
    'price', 
    IDBKeyRange.bound(10, 100)
);

// 游标遍历
let count = 0;
for await (const cursor of db.cursor('users')) {
    console.log(cursor.value.name);
    count++;
    if (count >= 10) break;   // 可以提前终止
}

// 关闭
db.close();
```

### idb vs 自行封装的对比

| 特性 | 自行封装 | idb |
|------|---------|-----|
| 代码量 | ~150 行 | 零（引入即用） |
| 类型支持 | 需自行编写 | 内置 TypeScript 类型 |
| 事务管理 | 手动创建 | `transaction()` 自动提交 |
| 游标 | 需手动迭代 | `for await...of` 原生支持 |
| 键范围 | 手动构造 | 同上 |
| 包体积 | 零依赖 | ~1KB gzipped |
| 社区维护 | 自己维护 | Jake Archibald (Chrome 团队) |

> **哲学思考**：封装是对复杂性的管理策略，而非消除手段。IndexedDB 的原生 API 并非设计得不好——它的事件驱动模型在浏览器环境中是合理的选择（与 DOM 事件系统一致）。但这种一致性并不总是意味着对开发者友好。Promise 封装做的事情是**在不改变底层行为的前提下，改变表达方式**——就像把文言文翻译成白话文，意思没变但理解门槛大幅降低。而选择使用成熟的第三方库如 idb，则是另一种智慧：**承认有些轮子不需要重新发明**。你的时间应该花在解决业务问题上，而不是重复造一个已经经过大量实战检验的工具。好的工程师知道什么时候该自己动手，什么时候该站在巨人的肩膀上。
