# 性能监控 —— PerformanceObserver、Lighthouse、Web Vitals

## 为什么需要性能监控

没有度量就没有改进。你无法优化一个你看不见的东西。性能监控就是给 Web 应用装上"仪表盘"——让那些隐藏在后台的加载时间、渲染延迟和交互卡顿变得**可见、可量化、可追踪**。

这就像开车时看仪表盘。速度表告诉你是否超速，油量表告诉你该不该加油，发动机温度表告诉你有没有过热。如果没有这些仪表，你只能靠感觉判断——而人的感觉是极其不靠谱的，尤其是当问题逐渐恶化的时候。性能监控的作用就是在用户抱怨之前就发现问题。

## Performance API 基础

浏览器提供了丰富的 Performance API 用于采集各种性能数据。它是所有前端性能监控的基础设施。

### performance.timing（传统方式）

```javascript
/**
 * Navigation Timing —— 页面加载各阶段耗时分析
 *
 * 时间线：
 * navigationStart → fetchStart → domainLookupStart →
 * connectStart → requestStart → responseStart → responseEnd →
 * domLoading → domInteractive → domComplete → loadEventStart
 */
function analyzeNavigationTiming() {
  const timing = performance.timing;

  const metrics = {
    // DNS 查询耗时
    dns: timing.domainLookupEnd - timing.domainLookupStart,
    // TCP 连接耗时（含 TLS 握手）
    tcp: timing.connectEnd - timing.connectStart,
    // 请求到响应首字节（TTFB）
    ttfb: timing.responseStart - timing.requestStart,
    // 内容下载时间
    download: timing.responseEnd - timing.responseStart,
    // DOM 解析 + 构建渲染树
    domProcessing: timing.domComplete - timing.responseEnd,
    // DOM 解析完成到 load 事件触发
    loadComplete: timing.loadEventEnd - timing.domComplete,

    // 关键汇总指标
    totalPageLoad: timing.loadEventEnd - timing.navigationStart,   // 总加载时间
    readyToInteract: timing.domInteractive - timing.navigationStart, // 可交互时间
    firstPaint: timing.responseStart - timing.navigationStart,      // 首次绘制（近似）
  };

  console.table(metrics);
  return metrics;
}

// 使用
const navMetrics = analyzeNavigationTiming();

if (navMetrics.ttfb > 800) {
  console.warn('⚠ TTFB 过长:', navMetrics.ttfb, 'ms');
}
```

### PerformanceEntry 和 getEntries

`performance.getEntries()` 返回页面生命周期中记录的所有性能条目：

```javascript
/**
 * 资源加载分析
 * 列出所有已加载的资源及其耗时分布
 */
function analyzeResourceTiming() {
  const resources = performance.getEntriesByType('resource');

  // 按类型分组统计
  const byType = {};
  let totalSize = 0;

  resources.forEach(r => {
    const type = getResourceType(r.name);
    if (!byType[type]) {
      byType[type] = { count: 0, totalDuration: 0, slowest: 0 };
    }
    byType[type].count++;
    byType[type].totalDuration += r.duration;
    if (r.duration > byType[type].slowest) {
      byType[type].slowest = Math.round(r.duration);
    }

    if (r.transferSize) totalSize += r.transferSize;
  });

  console.group('%c📊 资源加载报告', 'font-size:14px');
  for (const [type, stats] of Object.entries(byType)) {
    const avg = (stats.totalDuration / stats.count).toFixed(0);
    console.log(
      `${type}: ${stats.count} 个资源, 平均 ${avg}ms, 最慢 ${stats.slowest}ms`
    );
  }
  console.log(`总传输大小: ${(totalSize / 1024).toFixed(1)} KB`);
  console.groupEnd();

  return { byType, totalSize };
}

function getResourceType(url) {
  if (url.match(/\.(js|jsx|mjs)(\?.*)?$/i)) return 'JavaScript';
  if (url.match(/\.css(\?.*)?$/i)) return 'CSS';
  if (url.match(/\.(jpe?g|png|gif|webp|svg|avif)(\?.*)?$/i)) return 'Image';
  if (url.match(/\.(woff2?|ttf|otf)(\?.*)?$/i)) return 'Font';
  return 'Other';
}

analyzeResourceTiming();
```

## PerformanceObserver

`PerformanceObserver` 是现代浏览器推荐的性能数据采集方式。它采用异步回调模式，不会阻塞主线程，并且支持 `buffered: true` 获取历史条目。

### 观察 Long Tasks（长任务）

长任务（超过 50ms 的主线程任务）是导致 FID 高、交互卡顿的直接原因：

```javascript
/**
 * Long Task 监控器
 * 长任务是用户体验的最大杀手之一
 * 它们阻塞了浏览器的所有交互响应
 */
class LongTaskMonitor {
  constructor(options = {}) {
    this.threshold = options.threshold || 50;  // 默认 50ms 阈值
    this.reportUrl = options.reportUrl;
    this.entries = [];
    this._init();
  }

  _init() {
    if (!('PerformanceLongTaskTiming' in window)) {
      console.warn('[LongTaskMonitor] 当前浏览器不支持 Long Task API');
      return;
    }

    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        this._processEntry(entry);
      }
    });

    observer.observe({ entryTypes: ['longtask'] });
  }

  _processEntry(entry) {
    const attribution = entry.attribution?.[0];
    const data = {
      duration: Math.round(entry.duration),
      startTime: Math.round(entry.startTime),
      name: attribution?.containerName || entry.name,
      containerType: attribution?.containerType,
      containerId: attribution?.containerId,
      containerSrc: attribution?.containerSrc,
    };

    this.entries.push(data);

    // 控制台告警
    console.groupCollapsed(
      `%c⚠ 长任务: ${data.duration}ms`,
      data.duration > 200 ? 'color:#e74c3c;font-weight:bold' : 'color:#e67e22'
    );
    console.log('来源:', data.name);
    console.log('容器类型:', data.containerType);
    console.log('开始时间:', `${(data.startTime / 1000).toFixed(2)}s`);
    console.groupEnd();

    // 上报严重长任务
    if (data.duration > 200 && this.reportUrl) {
      this._report(data);
    }
  }

  _report(data) {
    navigator.sendBeacon(this.reportUrl, JSON.stringify({
      type: 'long-task',
      ...data,
      url: location.href,
      timestamp: Date.now(),
    }));
  }

  /** 获取统计数据摘要 */
  getSummary() {
    if (this.entries.length === 0) return null;

    const durations = this.entries.map(e => e.duration);
    durations.sort((a, b) => a - b);

    return {
      count: this.entries.length,
      totalBlockingTime: durations.reduce((a, b) => a + b, 0) -
                           (this.threshold * this.entries.length),
      avg: Math.round(durations.reduce((a, b) => a + b, 0) / durations.length),
      p50: durations[Math.floor(durations.length * 0.5)],
      p90: durations[Math.floor(durations.length * 0.9)],
      p99: durations[Math.floor(durations.length * 0.99)],
      max: durations[durations.length - 1],
    };
  }
}

// ====== 使用 ======
const longTaskMonitor = new LongTaskMonitor({
  threshold: 50,
  reportUrl: '/api/performance/long-tasks',
});

// 页面卸载时输出摘要
window.addEventListener('visibilitychange', () => {
  if (document.visibilityState === 'hidden') {
    const summary = longTaskMonitor.getSummary();
    console.log('Long Task 摘要:', summary);
  }
});
```

### 观察 Layout Shift（布局偏移）

```javascript
/**
 * CLS 精确监控
 * 区分用户主动滚动/输入引起的偏移和意外偏移
 */
class LayoutShiftMonitor {
  constructor(options = {}) {
    this.clsValue = 0;
    this.sessionEntries = [];
    this.sessionValue = 0;
    this.lastSessionTime = 0;
    this.onUpdate = options.onUpdate || null;
    this._init();
  }

  _init() {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        this._processShift(entry);
      }
    });
    observer.observe({ type: 'layout-shift', buffered: true });
  }

  _processShift(entry) {
    // 忽略用户操作引起的偏移
    if (entry.hadRecentInput) return;

    const now = entry.startTime;
    const timeSinceLastSession = now - this.lastSessionTime;

    // 新 session 规则：间隔超过 1 秒或距首个条目不超过 5 秒
    if (
      this.sessionEntries.length > 0 &&
      timeSinceLastSession > 1000 &&
      now - this.sessionEntries[0].startTime < 5000
    ) {
      this.clsValue += this.sessionValue;
      this._resetSession();
    } else if (!this.sessionValue || timeSinceLastSession >= 5000) {
      this._resetSession();
    }

    this.sessionValue += entry.value;
    this.sessionEntries.push(entry);

    this.onUpdate?.({
      clsValue: +(this.clsValue + this.sessionValue).toFixed(4),
      lastShift: {
        value: entry.value.toFixed(4),
        sources: entry.sources?.map(s => ({
          node: s.node?.tagName,
          previousRect: s.previousRect,
          currentRect: s.currentRect,
        })),
      },
    });
  }

  _resetSession() {
    this.sessionValue = 0;
    this.sessionEntries = [];
    this.lastSessionTime = performance.now();
  }

  getValue() {
    return +(this.clsValue + this.sessionValue).toFixed(4);
  }
}

// 使用：实时显示 CLS 值
const clsMonitor = new LayoutShiftMonitor({
  onUpdate(data) {
    const badge = document.getElementById('cls-badge');
    if (badge) {
      badge.textContent = `CLS: ${data.clsValue}`;
      badge.style.color = data.clsValue > 0.25 ? '#e74c3c' :
                          data.clsValue > 0.1 ? '#e67e22' : '#27ae60';
    }
  },
});
```

### 自定义性能标记

对于业务逻辑中的关键节点，可以添加自定义标记来追踪耗时：

```javascript
/**
 * 业务流程性能追踪工具
 */
class PerfTracker {
  constructor(label) {
    this.label = label;
    this.marks = new Map();
  }

  mark(name) {
    const markName = `${this.label}-${name}`;
    performance.mark(markName);
    this.marks.set(name, performance.now());
    return this;
  }

  measure(from, to) {
    const measureName = `${this.label}-${from}-to-${to}`;
    try {
      performance.measure(measureName, `${this.label}-${from}`, `${this.label}-${to}`);
      const entries = performance.getEntriesByName(measureName);
      const duration = entries[0]?.duration ?? 0;
      console.log(`[${this.label}] ${from} → ${to}: ${Math.round(duration)}ms`);
      return duration;
    } catch (err) {
      console.error(`测量失败: ${from} → ${to}`, err.message);
      return -1;
    }
  }

  /**
   * 自动计算相邻标记之间的耗时
   */
  measureAll() {
    const names = Array.from(this.marks.keys());
    const results = [];

    for (let i = 1; i < names.length; i++) {
      results.push({
        from: names[i - 1],
        to: names[i],
        duration: this.measure(names[i - 1], names[i]),
      });
    }

    console.table(results.map(r => ({
      阶段: `${r.from} → ${r.to}`,
      耗时: `${Math.round(r.duration)}ms`,
    })));

    // 清理 marks 避免内存泄漏
    names.forEach(n => performance.clearMarks(`${this.label}-${n}`));
    results.forEach(r =>
      performance.clearMeasures(`${this.label}-${r.from}-to-${r.to}`)
    );

    return results;
  }
}

// ====== 实际使用场景 ======

const tracker = new PerfTracker('checkout-flow');

// 标记各个关键节点
tracker.mark('start');              // 用户点击"结算"
// ... 显示结算弹窗 ...
tracker.mark('modal-shown');        // 弹窗动画结束
// ... 加载地址列表 ...
tracker.mark('address-loaded');     // 地址数据就绪
// ... 调用支付接口 ...
tracker.mark('payment-called');     // 发起支付请求
tracker.mark('payment-response');   // 收到支付结果

// 输出完整的时间线
tracker.measureAll();
/* 输出示例：
┌───────────────┬──────────┐
│     阶段       │   耗时   │
├───────────────┼──────────┤
│ start→modal   │ 120ms    │
│ modal→address │ 340ms    │
│ address→pay   │ 89ms     │
│ pay→response  │ 1856ms   │
└───────────────┴──────────┘
*/
```

## User Timing 与自定义指标

除了浏览器预定义的性能条目，开发者可以定义自己的业务指标：

```javascript
/**
 * 用户感知性能指标定义
 * 这些指标更贴近真实体验而非技术细节
 */
function trackUserPerceivedMetrics() {
  // 1. 可交互时间（TTI）—— 用户真正能操作的时机
  const ttiMark = () => {
    performance.mark('user-interactive');

    // 计算从导航开始到可交互的总时长
    performance.measure(
      'time-to-interactive',
      'navigationStart',
      'user-interactive'
    );

    const measure = performance.getEntriesByName('time-to-interactive')[0];
    console.log('TTI:', Math.round(measure.duration), 'ms');
  };

  // 2. 首个有意义的内容绘制（FMP）近似
  // 通过 MutationObserver 监测 DOM 变化量来判断
  function detectFMP() {
    let maxScore = 0;
    let fmpTime = 0;
    let lastScoreTime = 0;

    const observer = new MutationObserver((mutations) => {
      // 计算本次变更的影响分值
      let score = 0;
      mutations.forEach(m => {
        score += m.addedNodes.length * 10;           // 新增节点权重高
        score += m.removedNodes.length * 5;          // 删除节点次之
        score += (m.target as Element)?.offsetHeight ?? 0; // 可见内容高度
      });

      if (score > maxScore) {
        maxScore = score;
        fmpTime = performance.now();
      }

      // 如果连续 2 秒内没有更大的变化，认为 FMP 已发生
      clearTimeout(detectFMP._timer);
      detectFMP._timer = setTimeout(() => {
        observer.disconnect();
        performance.mark('fmp-detected');
        console.log('FMP 近似值:', Math.round(fmpTime), 'ms');
      }, 2000);
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true,
      attributes: true,
    });
  }

  // 3. 骨架屏消失时间（Skeleton to Content Time）
  function trackSkeletonToContent() {
    const skeletonEl = document.querySelector('.skeleton-loader');
    if (!skeletonEl) return;

    const observer = new IntersectionObserver(([entry]) => {
      if (entry && !entry.isIntersecting) {
        // 骨架屏被移除或隐藏
        performance.mark('skeleton-hidden');
        performance.measure(
          'skeleton-to-content',
          'navigationStart',
          'skeleton-hidden'
        );
        console.log(
          '骨架屏→内容:',
          Math.round(performance.getEntriesByName('skeleton-to-content')[0]?.duration ?? 0),
          'ms'
        );
        observer.disconnect();
      }
    }, { root: document.body });

    observer.observe(skeletonEl);
  }

  return { ttiMark, detectFMP, trackSkeletonToContent };
}
```

## 数据上报策略

收集到的性能数据需要上报给服务端进行分析。上报本身也有性能开销，需要精心设计。

### sendBeacon vs fetch/XMLHttpRequest

```javascript
/**
 * 性能数据上报器
 * 优先使用 sendBeacon，确保页面关闭时不丢数据
 */
class PerfReporter {
  constructor(options = {}) {
    this.endpoint = options.endpoint || '/api/performance';
    this.batchSize = options.batchSize || 20;
    this.flushInterval = options.flushInterval || 5000;
    this.queue = [];
    this._started = false;
  }

  /**
   * 上报单条数据
   */
  report(data) {
    this.queue.push({
      ...data,
      url: location.href,
      ua: navigator.userAgent,
      screen: `${screen.width}x${screen.height}`,
      timestamp: Date.now(),
    });

    if (this.queue.length >= this.batchSize) {
      this.flush();
    }
  }

  /**
   * 批量发送队列中的数据
   */
  flush() {
    if (this.queue.length === 0) return;

    const payload = JSON.stringify(this.queue.splice(0));

    if (navigator.sendBeacon) {
      // ✅ sendBeacon：异步发送，不阻塞页面关闭
      // 即使页面正在 unload 也能成功发送
      navigator.sendBeacon(this.endpoint, payload);
    } else {
      // Fallback: 使用 fetch keepalive
      fetch(this.endpoint, {
        method: 'POST',
        body: payload,
        keepalive: true,
        headers: { 'Content-Type': 'application/json' },
      }).catch(() => {
        // 最终 fallback: 存储到 sessionStorage 下次再发
        const pending = JSON.parse(sessionStorage.getItem('_perf_pending') || '[]');
        pending.push(...JSON.parse(payload));
        sessionStorage.setItem('_perf_pending', JSON.stringify(pending.slice(-50)));
      });
    }
  }

  /**
   * 启动定时刷新和页面离开自动上报
   */
  start() {
    if (this._started) return;
    this._started = true;

    // 定时批量上报
    setInterval(() => this.flush(), this.flushInterval);

    // 页面隐藏/卸载时立即上报剩余数据
    const flushOnHide = () => this.flush();
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden') flushOnHide();
    });
    window.addEventListener('pagehide', flushOnHide);
  }
}

// 全局实例
const perfReporter = new PerfReporter({
  endpoint: '/api/performance/events',
  batchSize: 15,
});

perfReporter.start();

// 在其他监控模块中使用
perfReporter.report({
  type: 'metric',
  name: 'LCP',
  value: lcpValue,
});
```

### 采样率控制

不是每个用户的每一条数据都需要上报。合理的采样率既能保证数据的代表性，又能降低成本：

```javascript
/**
 * 采样率管理器
 * 支持基于用户 ID 的稳定采样和动态调整
 */
class SamplingManager {
  constructor(baseRate = 1) {
    this.baseRate = baseRate;
    // 为当前会话生成稳定的采样决策
    this.isSampled = this._computeSamplingDecision();
  }

  _computeSamplingDecision() {
    // 基于 user agent 或随机数生成稳定决定
    const seed = this._hashCode(navigator.userAgent + location.pathname);
    return (seed % 100) < (this.baseRate * 100);
  }

  _hashCode(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      hash = ((hash << 5) - hash + str.charCodeAt(i)) | 0;
    }
    return Math.abs(hash);
  }

  shouldReport() {
    return this.isSampled;
  }

  /**
   * 对异常情况强制采样
   * 即使在低采样率下也捕获关键错误
  */
  shouldForceReport(severity = 'error') {
    if (severity === 'critical') return true;
    if (severity === 'error') return Math.random() < 0.5; // 50% 错误采样率
    return this.shouldReport();
  }
}

const sampler = new SamplingManager(0.1); // 10% 采样率

// 在采集中使用
if (sampler.shouldReport() || sampler.shouldForceReport('error')) {
  perfReporter.report(data);
}
```

**哲学思考**：性能监控的本质是建立一种**反馈回路**。没有反馈的系统就像蒙着眼睛开车——你可能觉得自己开得很好，但早已偏离了路线。监控数据不是为了追求数字上的完美，而是为了发现真实的瓶颈在哪里、优化的投入产出比如何。最好的监控系统不是最详细的那个，而是能够帮助你做出正确决策的那个。
