# HTTP 服务器 —— http.createServer、请求与响应对象

> **核心内容**：从零搭建 HTTP 服务器、`IncomingMessage`（请求对象）详解、`ServerResponse`（响应对象）完整 API、路由设计模式、中间件机制原理、静态文件服务、HTTPS 配置。

## 最简单的 HTTP 服务器

```js
const http = require('http');

const server = http.createServer((req, res) => {
  res.writeHead(200, { 'Content-Type': 'text/plain' });
  res.end('Hello World\n');
});

server.listen(3000, () => {
  console.log('服务器运行在 http://localhost:3000');
});
```

这短短几行代码背后，Node.js 做了大量工作：解析 HTTP 协议、管理 TCP 连接、处理请求/响应生命周期。理解每一层的工作方式，才能写出高效的服务器代码。

## 请求对象 IncomingMessage

每次收到请求时，回调函数的 `req` 参数就是一个 `IncomingMessage` 实例，它是 `ReadableStream` 的子类：

```js
http.createServer((req, res) => {
  console.log({
    // === URL 相关 ===
    url: req.url,              // 完整路径含查询字符串：/users?page=1&limit=10
    method: req.method,        // HTTP 方法：GET / POST / PUT / DELETE

    // === HTTP 头部 ===
    headers: req.headers,      // 所有头部（小写键名）
    rawHeaders: req.rawHeaders, // 原始头部（未处理的键值对数组）

    // === 连接信息 ===
    socket: req.socket,        // 底层 net.Socket 对象
    connection: req.connection,// 同 socket（兼容旧版）
    remoteAddress: req.socket.remoteAddress, // 客户端 IP
    remotePort: req.socket.remotePort,       // 客户端端口

    // === HTTP 版本 ===
    httpVersion: req.httpVersion,            // '1.1' 或 '2.0'
    httpVersionMajor: req.httpVersionMajor,
    httpVersionMinor: req.httpVersionMinor,
  });

  // 解析 URL 和查询字符串
  const url = new URL(req.url, `http://${req.headers.host}`);
  console.log({
    pathname: url.pathname,     // '/users'
    searchParams: Object.fromEntries(url.searchParams), // { page: '1', limit: '10' }
  });
}).listen(3000);
```

### 读取请求体

因为 `req` 是可读流，读取 POST/PUT 的请求体需要手动收集数据：

```js
function readBody(req) {
  return new Promise((resolve, reject) => {
    const chunks = [];

    req.on('data', (chunk) => {
      chunks.push(chunk);
    });

    req.on('end', () => {
      const buffer = Buffer.concat(chunks);
      resolve(buffer);
    });

    req.on('error', reject);

    // 设置超时和大小限制
    req.setTimeout(10000, () => {
      reject(new Error('请求体读取超时'));
      req.destroy(); // 终止连接
    });

    req.on('close', () => {
      if (!req.complete) {
        reject(new Error('客户端提前关闭连接'));
      }
    });
  });
}

// 使用示例
http.createServer(async (req, res) => {
  if (req.method === 'POST') {
    const contentType = req.headers['content-type'] || '';

    try {
      const buffer = await readBody(req);

      if (contentType.includes('application/json')) {
        const json = JSON.parse(buffer.toString('utf-8'));
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ received: json }));
      } else if (contentType.includes('application/x-www-form-urlencoded')) {
        const params = new URLSearchParams(buffer.toString('utf-8'));
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(Object.fromEntries(params)));
      } else {
        // 纯文本或其他格式
        res.writeHead(200, { 'Content-Type': 'text/plain' });
        res.end(`收到 ${buffer.length} 字节`);
      }

      // 防御性检查：限制请求体大小
      if (buffer.length > 10 * 1024 * 1024) { // 10MB
        res.writeHead(413, { 'Content-Type': 'text/plain' });
        res.end('Payload Too Large');
        return;
      }
    } catch (err) {
      res.writeHead(400, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: err.message }));
    }
  }
});
```

### 封装一个通用的 BodyParser

```js
class BodyParser {
  constructor(options = {}) {
    this.maxSize = options.maxSize || 1024 * 1024;   // 默认 1MB
    this.timeout = options.timeout || 10000;          // 默认 10s
  }

  async parse(req) {
    const contentLength = parseInt(req.headers['content-length'] || '0', 10);

    if (contentLength > this.maxSize) {
      throw createError(413, `请求体 ${contentLength} 超过限制 ${this.maxSize}`);
    }

    const body = await this.#readStream(req);

    const type = (req.headers['content-type'] || '').split(';')[0].trim();

    switch (type) {
      case 'application/json':
        try { return JSON.parse(body.toString('utf-8')); }
        catch { throw createError(400, '无效的 JSON'); }

      case 'application/x-www-form-urlencoded':
        return Object.fromEntries(new URLSearchParams(body.toString('utf-8')));

      case 'multipart/form-data':
        return await this.#parseMultipart(body, req.headers['content-type']);

      default:
        return body; // 返回原始 Buffer
    }
  }

  #readStream(req) {
    return new Promise((resolve, reject) => {
      let totalLength = 0;
      const chunks = [];
      let timer;

      const cleanup = () => {
        clearTimeout(timer);
        req.removeListener('data', onData);
        req.removeListener('end', onEnd);
        req.removeListener('error', onError);
        req.removeListener('timeout', onTimeout);
      };

      const onData = (chunk) => {
        totalLength += chunk.length;
        if (totalLength > this.maxSize) {
          cleanup();
          reject(createError(413, '超过最大请求体大小'));
          return;
        }
        chunks.push(chunk);
      };

      const onEnd = () => {
        cleanup();
        resolve(Buffer.concat(chunks));
      };

      const onError = (err) => {
        cleanup();
        reject(err);
      };

      const onTimeout = () => {
        cleanup();
        reject(createError(408, '读取超时'));
        req.destroy();
      };

      timer = setTimeout(onTimeout, this.timeout);
      req.on('data', onData);
      req.on('end', onEnd);
      req.on('error', onError);
      req.on('timeout', onTimeout);
    });
  }

  // multipart 解析需要额外的库如 formidable，这里仅作示意
  async #parseMultipart(body, contentType) {
    throw new Error('Multipart 解析请使用 busboy 或 formidable 库');
  }
}

function createError(status, message) {
  const error = new Error(message);
  error.status = status;
  return error;
}

module.exports = BodyParser;
```

## 响应对象 ServerResponse

### 状态码速查表与使用场景

| 状态码 | 含义 | 使用场景 |
|--------|------|---------|
| **200** | OK | 成功的 GET/PUT/PATCH |
| **201** | Created | 成功的 POST 创建资源 |
| **204** | No Content | 成功的 DELETE，无需返回体 |
| **301** | Moved Permanently | 永久重定向 |
| **302** | Found | 临时重定向 |
| **304** | Not Modified | 缓存有效，条件请求 |
| **400** | Bad Request | 请求参数有误 |
| **401** | Unauthorized | 未认证 |
| **403** | Forbidden | 已认证但无权限 |
| **404** | Not Found | 资源不存在 |
| **405** | Method Not Allowed | HTTP 方法不支持 |
| **409** | Conflict | 资源冲突（如重复创建） |
| **413** | Payload Too Large | 请求体过大 |
| **422** | Unprocessable Entity | 格式正确但语义错误 |
| **429** | Too Many Requests | 频率限制 |
| **500** | Internal Server Error | 服务端未知错误 |
| **502** | Bad Gateway | 上游服务错误 |
| **503** | Service Unavailable | 服务暂时不可用 |

## 路由设计

### 基础路由分发

```js
const routes = {
  GET: {},
  POST: {},
  PUT: {},
  DELETE: {},
};

function registerRoute(method, path, handler) {
  routes[method][path] = handler;
}

// 注册路由
registerRoute('GET', '/', (req, res) => {
  res.end('Home Page');
});

registerRoute('GET', '/api/users', (req, res) => {
  res.json?.({ users: [] }) || res.end(JSON.stringify({ users: [] }));
});

registerRoute('GET', '/api/users/:id', (req, res, params) => {
  res.end(`用户 ID: ${params.id}`);
});

registerRoute('POST', '/api/users', async (req, res) => {
  const body = await bodyParser.parse(req);
  res.end(`创建用户: ${JSON.stringify(body)}`);
});

// 路由匹配函数
function matchRoute(method, path) {
  // 先精确匹配
  if (routes[method][path]) {
    return { handler: routes[method][path], params: {} };
  }

  // 再尝试参数化路径匹配
  for (const pattern in routes[method]) {
    const paramNames = [];
    const regexStr = pattern.replace(/:(\w+)/g, (_, name) => {
      paramNames.push(name);
      return '([^/]+)';
    });
    const regex = new RegExp(`^${regexStr}$`);
    const match = path.match(regex);

    if (match) {
      const params = {};
      paramNames.forEach((name, i) => params[name] = match[i + 1]);
      return { handler: routes[method][pattern], params };
    }
  }

  return null; // 未匹配
}
```

### 更完整的路由系统

```js
// router.js — 轻量级路由框架
const http = require('http');

class Router {
  constructor() {
    this.routes = new Map(); // Map<string[], handler>
  }

  get(path, ...handlers) {
    this.#add('GET', path, handlers);
    return this;
  }

  post(path, ...handlers) {
    this.#add('POST', path, handlers);
    return this;
  }

  put(path, ...handlers) {
    this.#add('PUT', path, handlers);
    return this;
  }

  delete(path, ...handlers) {
    this.#add('DELETE', path, handlers);
    return this;
  }

  use(middleware) {
    this.middlewares = this.middlewares || [];
    this.middlewares.push(middleware);
    return this;
  }

  #add(method, path, handlers) {
    const key = `${method}:${path}`;
    this.routes.set(key, { method, path, handlers });
  }

  handle(req, res) {
    const matched = this.#match(req.method, req.url.split('?')[0]);

    if (!matched) {
      this.#send404(res);
      return;
    }

    const { handlers, params } = matched;
    req.params = params;

    // 组合中间件和处理函数
    const chain = [...(this.middlewares || []), ...handlers];
    this.#executeChain(chain, 0)(req, res);
  }

  #executeChain(handlers, index) {
    return (req, res) => {
      if (index >= handlers.length) return;

      const handler = handlers[index];

      try {
        // 支持 async 函数和 next 回调两种风格
        const result = handler(req, res, (err) => {
          if (err) this.#sendError(res, err);
          else this.#executeChain(handlers, index + 1)(req, res);
        });

        // 如果返回 Promise，自动处理
        if (result && typeof result.then === 'function') {
          result.catch((err) => this.#sendError(res, err));
        }
      } catch (err) {
        this.#sendError(res, err);
      }
    };
  }

  #match(method, pathname) {
    // 精确匹配
    const exactKey = `${method}:${pathname}`;
    if (this.routes.has(exactKey)) {
      return { ...this.routes.get(exactKey), params: {} };
    }

    // 参数匹配
    for (const [key, route] of this.routes) {
      if (!key.startsWith(method)) continue;

      const routePath = key.slice(method.length + 1);
      const paramNames = [];
      const pattern = routePath.replace(/:(\w+)/g, (_, name) => {
        paramNames.push(name);
        return '([^/]+)';
      });

      const regex = new RegExp(`^${pattern}$`);
      const match = pathname.match(regex);

      if (match) {
        const params = {};
        paramNames.forEach((name, i) => params[name] = decodeURIComponent(match[i + 1]));
        return { ...route, params };
      }
    }

    return null;
  }

  #send404(res) {
    res.statusCode = 404;
    res.setHeader('Content-Type', 'application/json');
    res.end(JSON.stringify({ error: 'Not Found' }));
  }

  #sendError(res, err) {
    const status = err.status || err.statusCode || 500;
    res.statusCode = status;
    res.setHeader('Content-Type', 'application/json');
    res.end(JSON.stringify({ error: err.message }));
  }
}

// 使用示例
const router = new Router();

// 全局中间件：日志记录
router.use((req, res, next) => {
  const start = Date.now();
  res.on('finish', () => {
    console.log(`${req.method} ${req.url} ${res.statusCode} ${Date.now() - start}ms`);
  });
  next();
});

// 路由定义
router.get('/', (req, res) => {
  res.end('Home');
});

router.get('/api/users/:id', (req, res) => {
  res.end(JSON.stringify({ userId: req.params.id }));
});

router.post('/api/users', (req, res, next) => {
  readBody(req).then(body => {
    res.end(`Created: ${body.length} bytes`);
  }).catch(next);
});

// 创建服务器
const server = http.createServer((req, res) => router.handle(req, res));
server.listen(3000, () => console.log('Server running on :3000'));
```

## 中间件模式

中间件的核心理念是将请求处理拆分为一系列小的、可组合的处理步骤：

```
Request → [Middleware 1] → [Middleware 2] → [Handler] → Response
                │                  │               │
           (next() 调用)    (next() 调用)    (最终响应)
```

常用中间件功能：

```js
// 1. CORS 跨域中间件
function corsMiddleware(options = {}) {
  const defaults = {
    origin: '*',
    methods: 'GET,HEAD,PUT,PATCH,POST,DELETE',
    allowedHeaders: 'Content-Type,Authorization',
    credentials: true,
    maxAge: 86400,
  };
  const config = { ...defaults, ...options };

  return (req, res, next) => {
    // 预检请求（OPTIONS）
    if (req.method === 'OPTIONS') {
      res.writeHead(204, {
        'Access-Control-Allow-Origin': config.origin,
        'Access-Control-Allow-Methods': config.methods,
        'Access-Control-Allow-Headers': config.allowedHeaders,
        'Access-Control-Max-Age': String(config.maxAge),
      });
      res.end();
      return;
    }

    // 正常请求添加 CORS 头
    res.setHeader('Access-Control-Allow-Origin', config.origin);
    if (config.credentials) {
      res.setHeader('Access-Control-Allow-Credentials', 'true');
    }
    next();
  };
}

// 2. 限流中间件
class RateLimiter {
  constructor(options = {}) {
    this.windowMs = options.windowMs || 60000; // 时间窗口
    this.maxRequests = options.maxRequests || 100; // 最大请求数
    this.clients = new Map();
    this.#cleanupInterval = setInterval(() => this.#cleanup(), this.windowMs);
  }

  middleware() {
    return (req, res, next) => {
      const clientId = req.socket.remoteAddress || 'unknown';
      const now = Date.now();

      if (!this.clients.has(clientId)) {
        this.clients.set(clientId, { count: 1, resetAt: now + this.windowMs });
      } else {
        const client = this.clients.get(clientId);
        if (now > client.resetAt) {
          client.count = 1;
          client.resetAt = now + this.windowMs;
        } else {
          client.count++;
        }
      }

      const client = this.clients.get(clientId);

      // 设置速率限制相关响应头
      res.setHeader('X-RateLimit-Limit', this.maxRequests);
      res.setHeader('X-RateLimit-Remaining', Math.max(0, this.maxRequests - client.count));
      res.setHeader('X-RateLimit-Reset', Math.ceil(client.resetAt / 1000));

      if (client.count > this.maxRequests) {
        res.statusCode = 429;
        res.setHeader('Retry-After', Math.ceil((client.resetAt - now) / 1000));
        res.setHeader('Content-Type', 'application/json');
        res.end(JSON.stringify({ error: 'Too Many Requests' }));
        return;
      }

      next();
    };
  }

  #cleanup() {
    const now = Date.now();
    for (const [id, client] of this.clients) {
      if (now > client.resetAt) this.clients.delete(id);
    }
  }

  destroy() {
    clearInterval(this.#cleanupInterval);
  }
}

// 3. 请求日志中间件
function loggerMiddleware(format = ':method :url :status :time') {
  return (req, res, next) => {
    const start = process.hrtime.bigint();

    res.on('finish', () => {
      const duration = Number(process.hrtime.bigint() - start) / 1e6;
      const logEntry = format
        .replace(':method', req.method)
        .replace(':url', req.url.split('?')[0])
        .replace(':status', String(res.statusCode))
        .replace(':time', `${duration.toFixed(2)}ms`)
        .replace(':ip', req.socket.remoteAddress || '-');
      console.log(logEntry);
    });

    next();
  };
}
```

## 静态文件服务

```js
const fs = require('fs/promises');
const path = require('path');
const mime = require('mime-types'); // npm install mime-types

class StaticFileServer {
  constructor(rootDir, options = {}) {
    this.rootDir = path.resolve(rootDir);
    this.maxAge = options.maxAge || 3600;
    this.dotfiles = options.dotfiles || 'ignore'; // ignore | allow | deny
    this.index = options.index || ['index.html'];
  }

  async handle(req, res) {
    // 安全检查：防止目录穿越
    const decodedPath = decodeURIComponent(req.url.split('?')[0]);
    const filePath = path.join(this.rootDir, decodedPath);

    const resolvedPath = path.resolve(filePath);
    if (!resolvedPath.startsWith(this.rootDir)) {
      this.#sendError(res, 403, 'Forbidden');
      return;
    }

    // 检查隐藏文件
    const basename = path.basename(resolvedPath);
    if (basename.startsWith('.') && this.dotfiles === 'deny') {
      this.#sendError(res, 403, 'Hidden file denied');
      return;
    }
    if (basename.startsWith('.') && this.dotfiles === 'ignore') {
      this.#sendError(res, 404, 'Not Found');
      return;
    }

    try {
      const stats = await fs.stat(resolvedPath);

      if (stats.isDirectory()) {
        // 目录 → 查找 index 文件或列出内容
        await this.#serveDirectory(resolvedPath, res);
        return;
      }

      // 条件请求：If-Modified-Since / If-None-Match
      if (req.headers['if-none-match'] === this.#generateETag(stats)) {
        res.writeHead(304);
        res.end();
        return;
      }

      if (req.headers['if-modified-since']) {
        const ifModifiedSince = new Date(req.headers['if-modified-since']);
        if (stats.mtime <= ifModifiedSince) {
          res.writeHead(304);
          res.end();
          return;
        }
      }

      // Range 请求支持（视频等大文件的分段下载）
      if (req.headers.range && stats.size) {
        const rangeResult = this.#parseRange(req.headers.range, stats.size);
        if (rangeResult) {
          const { start, end, length } = rangeResult;
          const fd = await fs.open(resolvedPath, 'r');
          const buffer = Buffer.alloc(length);
          await fd.read(buffer, 0, length, start);
          await fd.close();

          res.writeHead(206, {
            'Content-Range': `bytes ${start}-${end}/${stats.size}`,
            'Accept-Ranges': 'bytes',
            'Content-Length': length,
            'Content-Type': mime.lookup(resolvedPath) || 'application/octet-stream',
            'Cache-Control': `public, max-age=${this.maxAge}`,
            'Last-Modified': stats.mtime.toUTCString(),
            'ETag': this.#generateETag(stats),
          });
          res.end(buffer);
          return;
        }
      }

      // 正常响应
      const data = await fs.readFile(resolvedPath);
      res.writeHead(200, {
        'Content-Type': mime.lookup(resolvedPath) || 'application/octet-stream',
        'Content-Length': data.length,
        'Cache-Control': `public, max-age=${this.maxAge}`,
        'Last-Modified': stats.mtime.toUTCString(),
        'ETag': this.#generateETag(stats),
        'Accept-Ranges': 'bytes',
      });
      res.end(data);

    } catch (err) {
      if (err.code === 'ENOENT') {
        this.#sendError(res, 404, 'File not found');
      } else if (err.code === 'EACCES') {
        this.#sendError(res, 403, 'Permission denied');
      } else {
        this.#sendError(res, 500, 'Internal server error');
      }
    }
  }

  async #serveDirectory(dirPath, res) {
    for (const indexName of this.index) {
      const indexPath = path.join(dirPath, indexName);
      try {
        const stats = await fs.stat(indexPath);
        if (stats.isFile()) {
          const data = await fs.readFile(indexPath);
          res.writeHead(200, { 'Content-Type': 'text/html' });
          res.end(data);
          return;
        }
      } catch { /* 继续尝试下一个 */ }
    }

    // 无 index 文件 → 列出目录（生产环境通常禁用）
    this.#sendError(res, 403, 'Directory listing disabled');
  }

  #generateETag(stats) {
    return `"${stats.mtime.getTime().toString(16)}-${stats.size.toString(16)}"`;
  }

  #parseRange(rangeHeader, fileSize) {
    const match = rangeHeader.match(/bytes=(\d*)-(\d*)/);
    if (!match) return null;

    const start = match[1] ? parseInt(match[1], 10) : 0;
    const end = match[2] ? parseInt(match[2], 10) : fileSize - 1;

    if (start >= fileSize || end < start) return null;

    return {
      start,
      end: Math.min(end, fileSize - 1),
      length: Math.min(end, fileSize - 1) - start + 1,
    };
  }

  #sendError(res, code, message) {
    res.writeHead(code, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: message }));
  }
}
```

## HTTPS 服务器

在生产环境中，HTTPS 是标配：

```js
const https = require('https');
const fs = require('fs');
const path = require('path');

// 方式一：加载证书文件
const options = {
  key: fs.readFileSync(path.join(__dirname, 'certs', 'private-key.pem')),
  cert: fs.readFileSync(path.join(__dirname, 'certs', 'certificate.pem')),
  // 可选：CA 证书链
  ca: fs.readFileSync(path.join(__dirname, 'certs', 'ca-bundle.crt')),
};

const httpsServer = https.createServer(options, (req, res) => {
  res.end('Secure Hello!');
});

httpsServer.listen(443, () => {
  console.log('HTTPS Server running on :443');
});

// 方式二：HTTP → HTTPS 自动重定向
const http = require('http');
http.createServer((req, res) => {
  const target = `https://${req.headers.host}${req.url}`;
  res.writeHead(301, { Location: target });
  res.end();
}).listen(80);
```

## 哲学思考：服务器即服务的具象化

HTTP 服务器是"服务"这个抽象概念的最具体体现。每一个请求都是一个人在敲门，每一次响应都是你开门后的回答。路由系统像是前台的指引员——根据来客的目的将其引向正确的部门；中间件像是安检流程——每个人进门都要经过身份验证、安全检查、登记记录；静态文件服务则像是一个自助图书馆——你告诉它要什么书，它帮你找出来。

好的服务器设计遵循"单一职责"原则：每个中间件只做一件事，每个路由处理器只负责一种资源。就像一家组织良好的机构——前台不负责安保，保安不负责借书。职责清晰，边界明确，整个系统就能以优雅的方式运转。

而这一切的基础，都建立在那个简单的 `createServer` 回调之上——它提醒我们：无论架构变得多复杂，本质永远是对"请求→处理→响应"这一基本循环的不断精化和扩展。
