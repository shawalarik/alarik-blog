# 全局对象 —— global、process、Buffer

> **核心内容**：`global` 与浏览器 `window` 的差异、`process` 对象的核心属性与方法、`Buffer` 二进制数据处理、`__dirname`/`__filename` 与 ESM 的兼容问题。

## 浏览器的 window vs Node.js 的 global

在浏览器中，顶层作用域就是 `window`——所有全局变量都挂载在上面，`var a = 1` 等价于 `window.a = 1`。Node.js 有自己的顶层对象叫 `global`，但行为上有重要区别：

```js
// 浏览器环境
var name = 'browser';
console.log(window.name); // 'browser' — var 声明挂载到 window

// Node.js 环境
var name = 'node';
console.log(global.name); // undefined！var 不挂载到 global
console.log(name);        // 'node' — 但仍然可访问（模块作用域）
```

**原因**：Node.js 中每个文件都是模块（Module），`var` 声明的变量属于**模块作用域**，而非全局作用域。这是 CommonJS 模块封装的结果：

```js
// Node.js 实际上把你的代码包在这个函数里执行：
(function (exports, require, module, __filename, __dirname) {
  // 你的代码在这里
  var name = 'node'; // 这是局部变量，不是 global 的属性
});
```

### 真正的全局变量

只有以下方式才能创建真正的全局变量：

```js
// 方式一：直接赋值给 global
global.myConfig = { debug: true };

// 方式二：省略 var/let/const（不推荐，严格模式下报错）
undeclaredVar = 'leaked to global';

// 方式三：GLOBAL 别名（不常用）
GLOBAL.appName = 'my-app';
```

**最佳实践**：尽量避免使用全局变量。如果必须使用，至少用命名空间封装：

```js
// 推荐：命名空间模式
global.MyApp = {
  config: { apiBase: 'https://api.example.com' },
  cache: new Map(),
  logger: console,
};
```

## process 对象：进程的神经中枢

`process` 是 Node.js 最核心的全局对象，它提供了当前 Node.js 进程的信息和控制能力。不需要 `require`，任何地方直接可用。

### 进程信息

```js
// 进程基本信息
console.log(process.pid);           // 进程 ID：12345
console.log(process.ppid);          // 父进程 ID：12340
console.log(process.version);       // Node.js 版本：v20.11.0
console.log(process.platform);      // 操作系统平台：win32/darwin/linux
console.log(process.arch);          // CPU 架构：x64/arm64
console.log(process.cwd());         // 当前工作目录：/home/user/project

// 内存使用情况
const mem = process.memoryUsage();
console.log({
  rss: `${(mem.rss / 1024 / 1024).toFixed(1)} MB`,   // 常驻内存集（物理内存）
  heapTotal: `${(mem.heapTotal / 1024 / 1024).toFixed(1)} MB`, // V8 堆总大小
  heapUsed: `${(mem.heapUsed / 1024 / 1024).toFixed(1)} MB`,  // V8 已用堆
  external: `${(mem.external / 1024 / 1024).toFixed(1)} MB`,  // C++ 绑定对象内存
  arrayBuffers: `${(mem.arrayBuffers / 1024 / 1024).toFixed(1)} MB`, // ArrayBuffer
});

// CPU 使用情况（需要异步获取）
process.cpuUsage((usage) => {
  console.log({
    user: usage.user,  // 用户态 CPU 时间（微秒）
    system: usage.system, // 内核态 CPU 时间（微秒）
  });
});
```

一个实用的内存监控工具类：

```js
class MemoryMonitor {
  #history = [];
  #maxSamples = 100;

  sample() {
    const mem = process.memoryUsage();
    const entry = {
      timestamp: Date.now(),
      rss: Math.round(mem.rss / 1024 / 1024),
      heapUsed: Math.round(mem.heapUsed / 1024 / 1024),
      heapTotal: Math.round(mem.heapTotal / 1024 / 1024),
    };
    this.#history.push(entry);
    if (this.#history.length > this.#maxSamples) {
      this.#history.shift();
    }
    return entry;
  }

  getStats() {
    if (this.#history.length < 2) return null;
    const latest = this.#history[this.#history.length - 1];
    const oldest = this.#history[0];
    return {
      currentHeap: latest.heapUsed,
      heapGrowth: latest.heapUsed - oldest.heapUsed,
      trend: latest.heapUsed > oldest.heapUsed * 1.5 ? '⚠️ 增长过快' : '✅ 正常',
      samples: this.#history.length,
    };
  }

  startMonitoring(intervalMs = 5000) {
    this.#interval = setInterval(() => this.sample(), intervalMs);
    return this;
  }

  stop() {
    clearInterval(this.#interval);
    return this.getStats();
  }
}
```

### 标准输入输出流

Node.js 提供了三个标准流，它们是 `Stream` 的实例：

```js
// stdin — 标准输入（默认暂停状态）
process.stdin.setEncoding('utf-8');
process.stdin.on('data', (chunk) => {
  const input = chunk.trim();
  console.log(`你输入了: ${input}`);
  if (input === 'exit') process.exit(0);
});

// stdout — 标准输出
process.stdout.write('这是一条不带换行的消息');

// stderr — 标准错误输出（不会被 stdout 的管道捕获影响）
process.stderr.write('错误信息会独立输出');
```

交互式命令行工具示例：

```js
// interactive-cli.js
const readline = require('readline');

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
  prompt: '> ',
});

rl.prompt();

rl.on('line', (line) => {
  const cmd = line.trim().toLowerCase();
  switch (cmd) {
    case 'status':
      console.log(`PID: ${process.pid}, 内存: ${Math.round(process.memoryUsage().heapUsed / 1024 / 1024)}MB`);
      break;
    case 'env':
      console.log(`NODE_ENV: ${process.env.NODE_ENV || '未设置'}`);
      break;
    case 'exit':
      rl.close();
      break;
    default:
      console.log(`未知命令: ${cmd}。可用命令: status, env, exit`);
  }
  rl.prompt();
});
```

### 环境变量

```js
// 读取环境变量
const dbHost = process.env.DB_HOST || 'localhost';
const dbPort = parseInt(process.env.DB_PORT || '5432', 10);
const isProduction = process.env.NODE_ENV === 'production';

// 设置环境变量（仅当前进程生效）
process.env.DEBUG_MODE = 'true';

// 删除环境变量
delete process.env.SENSITIVE_KEY;
```

**安全原则**：永远不要将敏感信息硬编码或提交到版本控制中。环境变量的正确用法是：

```js
// config.js — 安全的环境变量读取
class ConfigManager {
  static required(name) {
    const value = process.env[name];
    if (!value) {
      throw new Error(`Missing required environment variable: ${name}`);
    }
    return value;
  }

  static optional(name, defaultValue = '') {
    return process.env[name] || defaultValue;
  }

  static number(name, defaultValue) {
    const val = process.env[name];
    return val ? parseInt(val, 10) : defaultValue;
  }

  static boolean(name, defaultValue = false) {
    const val = process.env[name]?.toLowerCase();
    if (val === 'true' || val === '1') return true;
    if (val === 'false' || val === '0') return false;
    return defaultValue;
  }

  static url(name) {
    const val = process.env[name];
    if (!val) throw new Error(`Missing URL: ${name}`);
    try { return new URL(val); } catch {
      throw new Error(`Invalid URL for ${name}: ${val}`);
    }
  }
}

// 使用方式
const config = {
  database: {
    host: ConfigManager.required('DB_HOST'),
    port: ConfigManager.number('DB_PORT', 5432),
  },
  app: {
    port: ConfigManager.number('APP_PORT', 3000),
    debug: ConfigManager.boolean('DEBUG', false),
  },
};
```

### 进程退出与信号处理

```js
// 正常退出
process.exit(0);     // 退出码 0 表示成功
process.exit(1);     // 非 0 表示异常

// 更优雅的退出方式（让事件循环完成清理工作）
process.exitCode = 1; // 设置退出码但不立即退出
// 当前微任务队列清空后自动退出

// 退出前钩子
process.on('beforeExit', (code) => {
  console.log(`即将以退出码 ${code} 退出`);
  // 注意：在这里添加异步操作可以阻止进程退出
});

// 进程即将退出时的最终清理（不可阻止退出）
process.on('exit', (code) => {
  console.log(`进程退出码: ${code}`);
  // 这里只能执行同步操作！异步操作不会被执行
  fs.writeFileSync('/tmp/shutdown.log', `Exited at ${new Date().toISOString()}\n`);
});
```

#### 信号处理

进程可以响应操作系统发送的信号，实现优雅关闭：

```js
// graceful-shutdown.js
const http = require('http');
const server = http.createServer(/* ... */);

let isShuttingDown = false;

function gracefulShutdown(signal) {
  if (isShuttingDown) return;
  isShuttingDown = true;

  console.log(`\n收到 ${signal} 信号，开始优雅关闭...`);

  // 1. 停止接受新连接
  server.close(() => {
    console.log('HTTP 服务器已关闭');
  });

  // 2. 给正在处理的请求一些时间完成
  setTimeout(() => {
    console.log('强制退出超时，立即终止');
    process.exit(1);
  }, 10000);

  // 3. 执行其他清理操作
  cleanupDatabase()
    .then(() => flushLogs())
    .then(() => {
      console.log('清理完成');
      process.exit(0);
    });
}

// 监听常见信号
process.on('SIGINT', () => gracefulShutdown('SIGINT'));   // Ctrl+C
process.on('SIGTERM', () => gracefulShutdown('SIGTERM')); // kill 命令（默认信号）

console.log(`进程 PID: ${process.pid} 运行中...`);
console.log('按 Ctrl+C 或发送 SIGTERM 信号来优雅关闭');
```

| 信号 | 名称 | 触发方式 |
|------|------|---------|
| SIGINT | 中断 | Ctrl + C |
| SIGTERM | 终止 | `kill <pid>` |
| SIGHUP | 挂起 | 终端关闭 |
| SIGUSR1 | 用户自定义 | `kill -USR1 <pid>` |

### 其他实用方法

```js
// 获取命令行参数（跳过 node 和脚本路径）
const args = process.argv.slice(2);
// node script.js --port=3000 name=john
// args = ['--port=3000', 'name=john']

// 解析参数的小工具
function parseArgs(argv) {
  const result = {};
  for (const arg of argv) {
    if (arg.startsWith('--')) {
      const [key, value] = arg.slice(2).split('=');
      result[key] = value ?? true;
    }
  }
  return result;
}

// 改变工作目录
process.chdir('/tmp');
console.log(process.cwd()); // /tmp

// 获取并设置 umask（文件权限掩码）
console.log(process.umask()); // 0o0022（八进制）
process.umask(0o0077);       // 新建文件权限更严格

// 获取运行时长
const uptime = process.uptime(); // 秒数
console.log(`已运行 ${Math.floor(uptime / 60)} 分钟`);

// 获取启动时传入的配置
process.config.variables.nodestring_module_registers; // 内部配置
```

## Buffer：二进制数据的容器

JavaScript 语言本身没有处理二进制数据的能力（直到 ES6 的 `ArrayBuffer`）。Node.js 从诞生之初就引入了 `Buffer` 类来处理 TCP 流、文件系统和网络协议中的二进制数据。

### 创建 Buffer

```js
// 方式一：从字符串创建（推荐使用 Buffer.from）
const buf1 = Buffer.from('Hello World', 'utf-8'); // 默认 utf-8
const buf2 = Buffer.from('48656c6c6f', 'hex');     // 从十六进制
const buf3 = Buffer.from('SGVsbG8=', 'base64');    // 从 Base64

// 方式二：指定大小（初始化为全零）
const emptyBuf = Buffer.alloc(1024); // 安全：填充 0x00
const unsafeBuf = Buffer.allocUnsafe(1024); // 快速但可能包含旧数据

// 方式三：Buffer.from(arrayLike)
const buf4 = Buffer.from([0x48, 0x65, 0x6c, 0x6c, 0x6f]); // [72,101,108,108,111]

// ⚠️ 废弃的方式（不要使用）
// new Buffer(size) — 已废弃，等价于 allocUnsafe
```

### Buffer 的关键特性

**Buffer 是内存中的一段固定区域**，与普通数组有本质区别：

```js
const buf = Buffer.from('你好世界', 'utf-8');

console.log(buf.length);          // 12 — 字节数（每个 UTF-8 中文占 3 字节）
console.log(buf.toString());      // 你好世界
console.log(buf.toString('hex')); // e4bda0e5a5bde4b896e7%95%8c
console.log(buf.toJSON());        // { type: 'Buffer', data: [228,189,160,...] }

// Buffer 不能 resize
// buf.length = 100; // 无效操作

// Buffer 可以切片（共享底层内存）
const slice = buf.slice(0, 6);    // "你好" 的字节
console.log(slice.toString());    // 你好

// 修改 slice 会影响原始 buffer（浅拷贝！）
slice[0] = 0xff;
console.log(buf[0]);              // 255 — 原始 buffer 被修改！

// 如果需要独立副本，使用 subarray 或 copy
const copy = Buffer.alloc(6);
buf.copy(copy, 0, 0, 6);
copy[0] = 0xff;
console.log(buf[0]);              // 228 — 原始 buffer 未受影响
```

### 常见操作场景

#### 场景一：拼接多个 Buffer

```js
// 错误方式：toString() 再 from — 编解码开销大且可能导致乱码
const parts = [Buffer.from('你'), Buffer.from('好')];
const bad = Buffer.from(parts.map(p => p.toString()).join('')); // 低效

// 推荐方式一：Buffer.concat
const good = Buffer.concat(parts);
console.log(good.toString()); // 你好

// 推荐方式二：预计算总长度
const totalLength = parts.reduce((sum, p) => sum + p.length, 0);
const merged = Buffer.concat(parts, totalLength);
```

#### 场景二：流式读取中的 Buffer 处理

```js
// 流式读取时，数据可能被任意切割
// 需要处理"半截字符"问题
const { Transform } = require('stream');

class Utf8Decoder extends Transform {
  #remainder = Buffer.alloc(0);

  _transform(chunk, encoding, callback) {
    // 拼接上次剩余的数据和本次新数据
    const combined = Buffer.concat([this.#remainder, chunk]);
    let end = combined.length;

    // 从末尾向前找完整的 UTF-8 字符边界
    while (end > 0) {
      const byte = combined[end - 1];
      // UTF-8 多字节的后续字节特征：10xxxxxx
      if ((byte & 0xc0) !== 0x80) break;
      end--;
    }

    // 输出完整字符部分
    this.push(combined.slice(0, end));
    // 保留可能被切断的部分
    this.#remainder = combined.slice(end);
    callback();
  }

  _flush(callback) {
    // 流结束时，强制输出剩余数据
    if (this.#remainder.length > 0) {
      this.push(this.#remainder);
    }
    callback();
  }
}
```

#### 场景三：Base64 编解码

```js
// 数据传输中常用的 Base64 编码
function encodeToBase64(str) {
  return Buffer.from(str, 'utf-8').toString('base64');
}

function decodeFromBase64(b64) {
  return Buffer.from(b64, 'base64').toString('utf-8');
}

// URL 安全的 Base64（将 +/ 替换为 -_）
function toUrlSafeBase64(b64) {
  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}

// 实际应用：JWT Token 的各部分解析
function decodeJwtPayload(token) {
  const parts = token.split('.');
  if (parts.length !== 3) throw new Error('Invalid JWT format');

  // JWT 的 payload 是 Base64URL 编码
  const payload = parts[1];
  // 补齐 padding
  const padded = payload + '='.repeat((4 - payload.length % 4) % 4);
  const decoded = Buffer.from(padded, 'base64').toString('utf-8');

  return JSON.parse(decoded);
}
```

## __dirname 和 __filename

这两个不是 `global` 的属性，但在每个模块中都可以直接访问：

```js
console.log(__dirname);  // 当前文件所在目录的绝对路径
console.log(__filename); // 当前文件的绝对路径（含文件名）
```

### 在 ESM 中的兼容性问题

ESM 模块中 **没有 `__dirname` 和 `__filename`**：

```js
// CommonJS — 可用
console.log(__dirname);  // /path/to/project/src

// ESM — 报错！ReferenceError: __dirname is not defined

// ESM 中的替代方案
import { fileURLToPath } from 'url';
import { dirname } from 'path';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

console.log(__dirname);  // /path/to/project/src
console.log(__filename); // /path/to/project/src/module.mjs
```

### 封装一个跨模块系统的路径工具

```js
// path-utils.cjs — CommonJS 版本
const path = require('path');
module.exports = {
  dirname: __dirname,
  filename: __filename,
  resolve: (...args) => path.resolve(__dirname, ...args),
};
```

```js
// path-utils.mjs — ESM 版本
import { fileURLToPath } from 'url';
import { dirname, resolve } from 'path';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

export { __dirname, __filename };

export function resolvePath(...segments) {
  return resolve(__dirname, ...segments);
}
```

## 全局函数与实用工具

除了上述主要对象，还有一些常用的全局函数：

```js
// 定时器（浏览器也有，但 Node.js 不需要最小延迟 4ms）
setTimeout(fn, 0);    // Node.js 中真正可以是 0ms（实际 ~1ms）
setInterval(fn, 1000);
clearTimeout(id);
clearInterval(id);
setImmediate(fn);     // Node.js 特有，见上一篇文章

// 即时执行（比 setTimeout(fn, 0) 性能更好）
setImmediate(() => console.log('下一轮 I/O 后立即执行'));

// 结构化克隆（Node.js 17+）
const original = { date: new Date(), map: new Map([['key', 'value']]) };
const cloned = structuredClone(original);
console.log(cloned.date !== original.date);       // true — 深拷贝
console.log(cloned.map.get('key'));                // 'value'

// AbortController（Node.js 15+）
const controller = new AbortController();
fetch(url, { signal: controller.signal })
  .catch(err => {
    if (err.name === 'AbortError') console.log('请求已取消');
  });

// 取消请求
controller.abort();
```

## 哲学思考：全局对象的边界感

浏览器的 `window` 是一个"什么都往里塞"的大杂烩——DOM API、`document`、`setTimeout`、甚至你的 `var` 变量都混在一起。Node.js 的设计更加克制：`global` 只包含必要的内置对象（`process`、`Buffer`、`console`），其余一切通过模块系统显式导入。

这种克制体现了 Unix 哲学："做一件事，把它做好。" `process` 专注管理进程生命周期，`Buffer` 专注二进制数据，`console` 专注输出。职责清晰，边界分明。当你写代码时也遵循这个原则——每个函数只负责一件事，每个模块只解决一个问题——代码就会像 Node.js 的架构一样清晰而有力。
